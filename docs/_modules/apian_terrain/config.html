# Primary: Copernicus DEM 30m (COP30) — AWS-hosted COGs, no key required.
# Secondary: NASADEM via OpenTopography — requires free API key.
# Fallback: OpenTopoData SRTM batch queries.
def _load_opentopo_key():
    env = os.environ.get("OPENTOPOGRAPHY_API_KEY", "")
    if env:
        return env
    key_file = os.path.join(os.path.dirname(__file__), "..", "data", "open_topo_api_key.txt")
    try:
        with open(key_file) as _f:
            return _f.read().strip()
    except OSError:
        return ""

OPENTOPOGRAPHY_API_KEY = _load_opentopo_key()

# ── Global raw-tile stores (region-folder layout) ────────────────────────────
# The Manager's global "Pipeline Cache" dir (settings.json key 'data_dir'):
# GLOBAL raw tile stores shared across regions -- cop30/nasadem 1-degree tiles
# and naturalearth. The pipeline runner wires this from the request (see
# _apply_request_options); the tkinter manager and tools set it from
# settings.json. Empty/unset = every store falls back to the region's own
# cache/ dir (offline / zip-mode safety).
DATA_DIR = ""


def _settings_path():
    """The Manager's settings.json path -- mirrors gui_settings.py exactly
    (%APPDATA%/apian/settings.json, or $APIAN_CONFIG when set). gui_settings
    lives at the repo root and is not importable from the packaged add-on, so
    the small path logic is duplicated here rather than imported."""
    env = os.environ.get("APIAN_CONFIG")
    if env:
        return Path(env)
    if os.name == "nt":
        base = Path(os.environ.get("APPDATA", Path.home()))
    elif os.name == "posix":
        base = Path(os.environ.get("XDG_CONFIG_HOME", Path.home() / ".config"))
    else:
        base = Path.home()
    return base / "apian" / "settings.json"

def load_settings():
    """The Manager's settings.json as a dict ({} when missing/corrupt)."""
    try:
        data = json.loads(_settings_path().read_text(encoding="utf-8"))
        return data if isinstance(data, dict) else {}
    except (OSError, ValueError):
        return {}

def resolve_store_dir(settings, key, sub, fallback):
    """Global raw-tile store dir for `key`: <settings[key]>/<sub> when the key
    is set, else the per-region fallback dir. `settings` is a plain dict (the
    Manager request or settings.json content); `sub` may be "" for the root."""
    base = str((settings or {}).get(key) or "").strip()
    if base:
        return Path(base) / sub if sub else Path(base)
    return Path(fallback)

def data_store_dir(sub, fallback, settings=None):
    """resolve_store_dir for the 'data_dir' (Pipeline Cache) key: the global
    store when data_dir is known (explicit settings arg, the wired DATA_DIR,
    or settings.json), else the region-cache fallback."""
    if settings is None:
        settings = {"data_dir": DATA_DIR} if DATA_DIR else load_settings()
    return resolve_store_dir(settings, "data_dir", sub, fallback)

# Temporal compositing across passes (panel: Imagery -> Compositing):
# "median"        -- per-pixel middle value; clouds must be in the MINORITY of
#                    passes, else the median is a cloud.
# "darkest"       -- per-pixel single darkest sample (clouds are always bright);
#                    strongest cloud rejection, but a shadowed sample can win a
#                    pixel and the result is noisier (one pass wins each pixel).

#                    rejection of darkest while resisting single-shadow outliers.

# ── Landmarks ─────────────────────────────────────────────────────────────────
# Single language for landmark names. The pipeline reads OSM `name:<lang>` for
# every landmark, falling back to the generic `name` tag when that language is
# absent. "" or "local" = use the generic `name` tag (native language) for all.
def _os_language(default="en"):
    """Two-letter ISO 639-1 language code from the OS locale, e.g. 'en', 'fr'.

    Sources are tried in order and short-circuit: POSIX env vars, the Windows
    user locale (BCP-47 like 'en-AU'), then locale.getdefaultlocale() as a last
    resort. Only a real 2-letter code is accepted -- this rejects the Windows
    display name ('English_Australia') and the 'C' locale. Falls back to
    `default` if nothing usable is found.
    """
    import os

    def _winapi():
        import ctypes
        buf = ctypes.create_unicode_buffer(85)
        if ctypes.windll.kernel32.GetUserDefaultLocaleName(buf, 85):
            return buf.value
        return None

    def _defaultlocale():
        import locale
        return locale.getdefaultlocale()[0]

    sources = [
        lambda: os.environ.get("LC_ALL"),
        lambda: os.environ.get("LC_MESSAGES"),
        lambda: os.environ.get("LANG"),
        _winapi,
        _defaultlocale,
    ]
    for src in sources:
        try:
            loc = src()
        except Exception:
            continue
        if loc:
            code = loc.replace("-", "_").split("_")[0].strip().lower()
            if len(code) == 2 and code.isalpha():
                return code
    return default

LANDMARK_NAME_LANG = _os_language()
