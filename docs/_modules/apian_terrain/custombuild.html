"""custombuild.py -- merge custom data into the unified per-cell format.

The download pipeline (process_area) is the DEFAULT source for every cells.csv
column. This module lets custom data OVERRIDE individual columns: bring a CSV
keyed by lat/lon OR by H3 id (at any resolution), and its values are mapped onto
the area's H3 grid (at the build resolution) and written into the chosen column.

Grid geometry (h3_id, lon, lat, rot_z, radius_eq) is NEVER overridden -- it is
derived from the bbox + resolution. Pure stdlib + h3; no bpy, so the CLI and the
GUI can both use it.
"""
import csv
import json
import math
import re
import statistics
from pathlib import Path

import h3
import numpy as np

# cells.csv columns custom data may override, and which are numeric.
OVERRIDABLE_COLUMNS = ["ele_m", "water_depth_m", "water_type", "veg_type", "r", "g", "b"]
_INT_COLS = {"r", "g", "b"}
_FLOAT_COLS = {"ele_m", "water_depth_m"}
_NUMERIC = _INT_COLS | _FLOAT_COLS

# per-type feature layers a custom GeoJSON may replace wholesale.
OVERRIDABLE_LAYERS = ["structures", "bridges", "streets", "landmarks", "water_bodies", "sea"]


def _coerce(column, value):
    if column in _INT_COLS:
        return int(round(float(value)))
    if column in _FLOAT_COLS:
        return float(value)
    return str(value)


def _aggregate(column, values):
    """Combine multiple custom values landing on one build cell."""
    if column in _INT_COLS:
        return int(round(statistics.fmean(values)))
    if column in _FLOAT_COLS:
        return statistics.fmean(values)
    return statistics.mode(values)            # categorical -> most common

def read_custom_rows(path):
    """Read a custom CSV into a list of dict rows."""
    with open(path, newline="", encoding="utf-8") as f:
        return list(csv.DictReader(f))




def _collapse(column, buckets):
    return {cell: (_aggregate(column, vals) if len(vals) > 1 else vals[0])
            for cell, vals in buckets.items()}


def _remap_to_build(cid, build_res):
    """Map one H3 id to the build-res cell(s) it covers."""
    r = h3.get_resolution(cid)
    if r == build_res:
        return [cid]
    if r < build_res:                      # coarser -> broadcast to children
        return h3.cell_to_children(cid, build_res)
    return [h3.cell_to_parent(cid, build_res)]   # finer -> aggregate to parent

def build_override_map(rows, *, column, value_col, key, build_res,
                       lat_col="lat", lon_col="lon", h3_col="h3_id", native_res=None):
    """Map custom rows -> {build_res h3_id: coerced value}, in two stages.

    1. Bin each row to a NATIVE H3 cell:

       - ``key='latlon'`` -> latlng_to_cell at native_res (default build_res);
         the editor's resolution field is this binning granularity.
       - ``key='h3'`` -> the row's own h3 id (its resolution is intrinsic).

    2. Remap native cells to build_res: coarser broadcast to children, finer
       aggregate to parent, equal pass through.

    Rows sharing a cell at any stage are aggregated (mean for numbers, mode for
    categories).
    """
    native = {}
    for row in rows:
        raw = row.get(value_col)
        if raw is None or raw == "":
            continue
        try:
            val = _coerce(column, raw)
        except (ValueError, TypeError):
            continue
        if key == "latlon":
            try:
                cell = h3.latlng_to_cell(float(row[lat_col]), float(row[lon_col]),
                                         native_res if native_res is not None else build_res)
            except (KeyError, ValueError, TypeError):
                continue
        else:  # key == "h3"
            cell = (row.get(h3_col) or "").strip()
            if not cell:
                continue
        native.setdefault(cell, []).append(val)

    out = {}
    for cid, val in _collapse(column, native).items():
        for t in _remap_to_build(cid, build_res):
            out.setdefault(t, []).append(val)
    return _collapse(column, out)

def apply_overrides(cell_rows, overrides):
    """Apply {column: {h3_id: value}} onto cells.csv rows (list of dict), in place.

    Returns (n_cells_changed, {column: n_changed})."""
    counts = {}
    changed = set()
    for column, cell_map in overrides.items():
        n = 0
        for row in cell_rows:
            v = cell_map.get(row["h3_id"])
            if v is not None:
                row[column] = v
                changed.add(row["h3_id"])
                n += 1
        counts[column] = n
    return len(changed), counts

def apply_feature_override(output_dir, area_id, layer, custom_path):
    """Replace a per-type feature GeoJSON with a custom FeatureCollection.
    Returns the feature count. Raises ValueError if the file is not a FC."""
    data = json.loads(Path(custom_path).read_text(encoding="utf-8"))
    if data.get("type") != "FeatureCollection":
        raise ValueError(f"{custom_path}: custom feature file must be a GeoJSON FeatureCollection")
    dst = Path(output_dir) / f"{area_id}_{layer}.geojson"
    dst.write_text(json.dumps(data, separators=(",", ":")), encoding="utf-8")
    return len(data.get("features", []))

def build_data_field(rows, *, value_col, key, aggregation, build_res,
                     lat_col="lat", lon_col="lon", h3_col="h3_id", native_res=None):
    """Map custom rows -> {build_res h3_id: float} for one data field, using the
    same two-stage binning as build_override_map but combining values by
    `aggregation` ('sum' or 'mean'). A coarser-than-build cell broadcasts to its
    children (mean: same value each; sum: value split evenly, so totals hold);
    finer cells aggregate to their parent."""
    agg = _AGG[aggregation]
    native = {}
    for row in rows:
        raw = row.get(value_col)
        if raw is None or raw == "":
            continue
        try:
            val = float(raw)
        except (ValueError, TypeError):
            continue
        if key == "latlon":
            try:
                cell = h3.latlng_to_cell(float(row[lat_col]), float(row[lon_col]),
                                         native_res if native_res is not None else build_res)
            except (KeyError, ValueError, TypeError):
                continue
        else:  # key == "h3"
            cell = (row.get(h3_col) or "").strip()
            if not cell:
                continue
        native.setdefault(cell, []).append(val)

    out = {}
    for cid, vals in native.items():
        v = agg(vals)
        targets = _remap_to_build(cid, build_res)
        share = v / len(targets) if aggregation == "sum" and len(targets) > 1 else v
        for t in targets:
            out.setdefault(t, []).append(share)
    return {c: agg(vs) for c, vs in out.items()}




# ── Interpolation: fill/spread cells from nearby measured cells ───────────────
# Baked at build time (the importer stays read-only). All three methods treat
# the area as a 2D top-down plane: distances are METRIC (equirectangular meters
# between cell centres), never H3 grid steps -- hex-ring distance produced
# hexagon-shaped reach and hexagonally-biased contours. Every method is
# restricted to a domain (all / land / water) and a `radius_m` reach:
#   idw      -- inverse-distance AVERAGE onto empty cells; sources kept. For
#               INTENSIVE fields (temperature, density); totals NOT conserved.
#   voronoi  -- Voronoi Distribution: spread each source's value across its
#               metric-nearest region weighted by distance-to-region-edge,
#               conserving the total. For EXTENSIVE fields (counts).
#   diffusion-- 2D heat-map fill: the closed-form heat equation on the plane.
#               Each empty cell gets the Gaussian-kernel-weighted average of the
#               sources within radius (sigma = radius/2, so the kernel tapers
#               smoothly to the reach edge). `anchor=True` holds measured cells
#               exact and fills the gaps; `anchor=False` additionally rescales
#               the field to conserve the measured total (melt).
INTERP_METHODS = ("none", "idw", "voronoi", "diffusion")
INTERP_LABELS  = {"none": "none", "idw": "IDW", "voronoi": "Voronoi Distribution",
                  "diffusion": "Diffusion"}
_INTERP_KEY    = {label: key for key, label in INTERP_LABELS.items()}
DATA_DOMAINS   = ("all", "land", "water")

def _is_water(cell):
    """A cells.csv row is water if it carries a positive water column or a
    water_type classification; else land."""
    try:
        wet = float(cell.get("water_depth_m") or 0) > 0
    except (TypeError, ValueError):
        wet = False
    return wet or bool((cell.get("water_type") or "").strip())

def domain_ids(cells, domain):
    """The h3_id set eligible for a given interpolation domain."""
    if domain == "land":
        return {c["h3_id"] for c in cells if not _is_water(c)}
    if domain == "water":
        return {c["h3_id"] for c in cells if _is_water(c)}
    return {c["h3_id"] for c in cells}       # "all"

def metric_coords(cells):
    """cells.csv rows -> {h3_id: (x, y)} in local metric coordinates
    (equirectangular about the centroid, x east / y north, meters). All
    spatial methods measure distance in this plane -- the data is treated
    as a 2D top-down view, never as hex-grid topology."""
    lats = [float(c["lat"]) for c in cells]
    lons = [float(c["lon"]) for c in cells]
    lat0 = sum(lats) / len(lats)
    lon0 = sum(lons) / len(lons)
    cos0 = math.cos(math.radians(lat0))
    return {c["h3_id"]: ((float(c["lon"]) - lon0) * cos0 * _METRES_PER_DEGREE,
                         (float(c["lat"]) - lat0) * _METRES_PER_DEGREE)
            for c in cells}

def grid_spacing_m(coords, sample=500):
    """Mean nearest-neighbour spacing of the grid in meters (legacy
    max_ring -> radius conversion; also a sanity scale for UIs)."""
    ids = sorted(coords)
    if len(ids) > sample:
        ids = ids[:: len(ids) // sample]
    pts = np.array([coords[c] for c in ids], dtype=np.float64)
    if len(pts) < 2:
        return 1.0
    best = np.full(len(pts), np.inf)
    for i in range(len(pts)):
        d = np.hypot(pts[:, 0] - pts[i, 0], pts[:, 1] - pts[i, 1])
        d[i] = np.inf
        best[i] = d.min()
    return float(best.mean())




def _pairs_within(tpts, spts, radius_m):
    """All (target, source) index pairs whose metric distance is within
    radius_m. tpts (n,2), spts (m,2) float64. Returns (ti, si, dist) arrays.
    Pure-numpy uniform bucket grid (bucket size = radius): scipy is not a
    pipeline dependency, and per-(slice, column) pair queries at region scale
    need more than a brute-force O(n*m) matrix."""
    ti = np.empty(0, dtype=np.int64)
    si = np.empty(0, dtype=np.int64)
    dist = np.empty(0, dtype=np.float64)
    if len(tpts) == 0 or len(spts) == 0:
        return ti, si, dist
    r = max(float(radius_m), 1e-6)
    mins = np.minimum(tpts.min(axis=0), spts.min(axis=0))
    tb = np.floor((tpts - mins) / r).astype(np.int64)
    sb = np.floor((spts - mins) / r).astype(np.int64)
    grid = {}
    for i in range(len(spts)):
        grid.setdefault((int(sb[i, 0]), int(sb[i, 1])), []).append(i)
    counts = np.zeros(len(tpts), dtype=np.int64)
    flat_si = []
    for i in range(len(tpts)):
        bx, by = int(tb[i, 0]), int(tb[i, 1])
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                hit = grid.get((bx + dx, by + dy))
                if hit:
                    counts[i] += len(hit)
                    flat_si.extend(hit)
    if not flat_si:
        return ti, si, dist
    ti = np.repeat(np.arange(len(tpts), dtype=np.int64), counts)
    si = np.array(flat_si, dtype=np.int64)
    dist = np.hypot(tpts[ti, 0] - spts[si, 0], tpts[ti, 1] - spts[si, 1])
    keep = dist <= r
    return ti[keep], si[keep], dist[keep]


def _scatter_sources(sources, coords):
    """(ordered ids, (m,2) points, values array) for a source map."""
    ids = sorted(sources)
    pts = np.array([coords[c] for c in ids], dtype=np.float64)
    vals = np.array([float(sources[c]) for c in ids], dtype=np.float64)
    return ids, pts, vals

def interpolate_field(cell_map, domain, *, coords, radius_m, power=2.0,
                      method="idw", anchor=False):
    """Fill/spread in-domain cells from the measured source cells on the 2D
    metric plane, restricted to `domain` (an h3_id set) and `radius_m` of a
    source. `coords` maps h3_id -> (x, y) meters (metric_coords).

    method='idw'      -- inverse-distance^power-weighted AVERAGE of the sources
                         within radius onto each empty cell; sources keep their
                         value. For INTENSIVE fields; the grand total is NOT
                         conserved.
    method='voronoi'  -- Voronoi Distribution: each source's value is spread
                         across the in-domain cells metric-nearest it within
                         radius, weighted by each cell's distance to the region
                         boundary (half the gap to the nearest competing
                         source), normalised so the region sum equals the
                         source value (grand total conserved). The source cell
                         is redistributed too -- it keeps only its share.
    method='diffusion'-- 2D heat-map fill: Gaussian-kernel-weighted average,
                         kernel sigma = radius_m / 2 (smooth taper to the reach
                         edge). anchor=True holds measured cells exact (gaps
                         add mass); anchor=False rescales the whole field to
                         the measured total (melt, sources change).

    Returns (filled_map, interp_ids): filled_map is cell_map with the computed
    cells added/overwritten; interp_ids is the set of previously-empty cells
    that received a value (the importer's `<field>__i` mask)."""
    domain = set(domain)
    sources = {c: v for c, v in cell_map.items() if c in domain}
    if not sources or radius_m < 1.0:
        return dict(cell_map), set()
    if method == "idw":
        return _idw_fill(cell_map, sources, domain, coords, radius_m, power)
    if method == "voronoi":
        return _voronoi_distribute(cell_map, sources, domain, coords, radius_m)
    if method == "diffusion":
        return _heat_fill(cell_map, sources, domain, coords, radius_m,
                          anchor=anchor)
    raise ValueError(f"unknown interpolation method: {method!r}")




def _idw_fill(cell_map, sources, domain, coords, radius_m, power):
    """Inverse-distance-weighted average of the sources within radius onto each
    empty in-domain cell. Metric distances; sources are left untouched."""
    targets = sorted(domain - sources.keys())
    if not targets:
        return dict(cell_map), set()
    tpts = np.array([coords[c] for c in targets], dtype=np.float64)
    _, spts, svals = _scatter_sources(sources, coords)
    ti, si, d = _pairs_within(tpts, spts, radius_m)
    w = d ** (-float(power))               # d > 0: targets exclude sources
    num = np.zeros(len(targets))
    den = np.zeros(len(targets))
    np.add.at(num, ti, w * svals[si])
    np.add.at(den, ti, w)
    filled = dict(cell_map)
    interp_ids = set()
    for j, cell in enumerate(targets):
        if den[j] > 0.0:
            filled[cell] = float(num[j] / den[j])
            interp_ids.add(cell)
    return filled, interp_ids


def _heat_fill(cell_map, sources, domain, coords, radius_m, *, anchor):
    """2D heat-map fill: closed-form heat equation on the metric plane. Each
    empty in-domain cell gets the Gaussian-kernel-weighted average of sources
    within radius (sigma = radius/2). anchor=True keeps sources exact;
    anchor=False rescales the whole field to conserve the measured total."""
    targets = sorted(domain - sources.keys())
    if not targets:
        return dict(cell_map), set()
    tpts = np.array([coords[c] for c in targets], dtype=np.float64)
    _, spts, svals = _scatter_sources(sources, coords)
    ti, si, d = _pairs_within(tpts, spts, radius_m)
    sigma = float(radius_m) / 2.0
    k = np.exp(-(d * d) / (2.0 * sigma * sigma))
    num = np.zeros(len(targets))
    den = np.zeros(len(targets))
    np.add.at(num, ti, k * svals[si])
    np.add.at(den, ti, k)
    filled = dict(cell_map)
    interp_ids = set()
    for j, cell in enumerate(targets):
        if den[j] > 0.0:
            filled[cell] = float(num[j] / den[j])
            interp_ids.add(cell)
    if not anchor:
        # Melt: rescale (sources included) so the domain total matches the
        # measured total -- peaks flatten, mass is conserved.
        total0 = float(sum(sources.values()))
        s = float(sum(v for c, v in filled.items() if c in domain))
        if s > 0.0:
            scale = total0 / s
            for c in list(filled):
                if c in domain:
                    filled[c] = filled[c] * scale
    return filled, interp_ids


def _voronoi_distribute(cell_map, sources, domain, coords, radius_m):
    """Radial Voronoi Distribution: each in-domain cell within radius of a
    source belongs to its metric-NEAREST source; the source's value is spread
    across its region (source cell included) weighted by each cell's distance
    to the region boundary -- half the gap to the nearest competing source --
    normalised so the region sum equals the source value (total conserved)."""
    s_ids, spts, svals = _scatter_sources(sources, coords)
    members = sorted(domain)
    mpts = np.array([coords[c] for c in members], dtype=np.float64)
    ti, si, d = _pairs_within(mpts, spts, radius_m)

    # Per member: nearest source, and the margin to the second-nearest (half
    # the gap to the region boundary; lone-source cells taper to the reach).
    order = np.lexsort((d, ti))
    ti_s, si_s, d_s = ti[order], si[order], d[order]
    owner = {}
    margin = {}
    pos = 0
    while pos < len(ti_s):
        end = pos + 1
        while end < len(ti_s) and ti_s[end] == ti_s[pos]:
            end += 1
        cell = members[ti_s[pos]]
        owner[cell] = si_s[pos]
        if end - pos > 1:
            margin[cell] = float(d_s[pos + 1] - d_s[pos]) / 2.0
        else:
            margin[cell] = float(radius_m - d_s[pos]) / 2.0
        pos = end

    regions = {}
    for cell in members:
        if cell in owner:
            regions.setdefault(owner[cell], []).append(cell)
    filled = dict(cell_map)
    interp_ids = set()
    for oi, region in regions.items():
        weights = np.array([max(margin[c], 1e-9) for c in region])
        total_w = float(weights.sum())
        budget = float(svals[oi])
        for cell, wgt in zip(region, weights):
            filled[cell] = budget * float(wgt) / total_w
            if cell not in sources:
                interp_ids.add(cell)
    return filled, interp_ids

def write_celldata(output_dir, area_id, field_maps, interp_maps=None, measured_maps=None):
    """Write <area>_celldata.csv: h3_id + one value column per field. Sparse -- a
    cell absent from a field's map is left blank (the importer reads blank as
    no-data). A field with interpolated cells also gets a `<field>__i` flag
    column ("1" for interpolated cells) so the importer can tell filled cells from
    measured ones. A field whose measured values were altered by a conserving method
    (Voronoi / diffusion-melt) also gets a `<field>__m` column holding the ORIGINAL
    measured value, so re-interpolation can reset to truth (the importer ignores it).
    Returns the csv filename."""
    interp_maps = interp_maps or {}
    measured_maps = measured_maps or {}
    fields = list(field_maps)
    flag  = {fn for fn in fields if interp_maps.get(fn)}       # fields with interp cells
    meas  = {fn for fn in fields if measured_maps.get(fn)}     # fields with a measured snapshot
    ids = sorted(set().union(*(m.keys() for m in field_maps.values()))) if field_maps else []
    header = ["h3_id"]
    for fn in fields:
        header.append(fn)
        if fn in flag:
            header.append(fn + "__i")
        if fn in meas:
            header.append(fn + "__m")
    path = Path(output_dir) / f"{area_id}_celldata.csv"
    with open(path, "w", newline="", encoding="utf-8") as f:
        w = csv.writer(f)
        w.writerow(header)
        for cid in ids:
            row = [cid]
            for fn in fields:
                row.append(field_maps[fn].get(cid, ""))
                if fn in flag:
                    row.append("1" if cid in interp_maps[fn] else "")
                if fn in meas:
                    row.append(measured_maps[fn].get(cid, ""))
            w.writerow(row)
    return path.name

def apply_data_fields(output_dir, area_id, specs, build_res, native_res=None, cells=None):
    """Bin each data-field spec into build-res cells, optionally interpolate the
    gaps, write <area>_celldata.csv, and return (celldata_filename, [field_meta,
    ...]) for the manifest. `specs` items:
    {name, path, value_col, key, aggregation, units?, lat_col?, lon_col?, h3_col?,
     interp?, radius_km?, domain?}.
    The optional *_col names let a CSV use its own headers (default lat/lon/h3_id).
    Interpolation (`interp` in INTERP_METHODS, `radius_km`>0, `domain` in
    DATA_DOMAINS) needs `cells` -- the full cells.csv rows -- for the grid, the
    land/water split, and the metric coordinate plane. A legacy `max_ring` spec
    key (H3 grid steps) is accepted and converted approximately via the grid's
    mean cell spacing. Field names are slugged so each maps to a unique
    data_<slug> attribute."""
    field_maps, interp_maps, metas, seen = {}, {}, [], set()
    coords = metric_coords(cells) if cells else None
    spacing = grid_spacing_m(coords) if coords else 1.0
    for s in specs:
        slug = slug_field(s["name"])
        while slug in seen:                       # keep every field distinct
            slug += "_"
        seen.add(slug)
        rows = read_custom_rows(s["path"])
        cell_map = build_data_field(
            rows, value_col=(s.get("value_col") or s["name"]), key=s["key"],
            aggregation=s["aggregation"], build_res=build_res, native_res=native_res,
            lat_col=s.get("lat_col") or "lat", lon_col=s.get("lon_col") or "lon",
            h3_col=s.get("h3_col") or "h3_id")

        method    = (s.get("interp") or "none").lower()
        domain    = (s.get("domain") or "all").lower()
        radius_km = float(s.get("radius_km") or 0.0)
        if not radius_km and s.get("max_ring"):
            radius_km = float(s["max_ring"]) * spacing / 1000.0   # legacy ring spec
        interp_ids = set()
        if method != "none" and radius_km > 0.0 and cells:
            cell_map, interp_ids = interpolate_field(
                cell_map, domain_ids(cells, domain), coords=coords,
                radius_m=radius_km * 1000.0, method=method)

        field_maps[slug]  = cell_map
        interp_maps[slug] = interp_ids
        metas.append(_data_field_meta(
            slug, cell_map, key=s["key"], aggregation=s["aggregation"],
            units=s.get("units", ""), interp=method, radius_km=radius_km,
            domain=domain, interp_count=len(interp_ids)))
    return write_celldata(output_dir, area_id, field_maps, interp_maps), metas

def read_celldata(path):
    """Inverse of write_celldata: read <area>_celldata.csv back into
    ({field: {h3_id: float}}, {field: {interpolated h3_id}},
     {field: {h3_id: original measured value}}). Value columns are every header
    except h3_id and the `<field>__i` (interp flag) / `<field>__m` (measured
    snapshot) companion columns."""
    field_maps, interp_maps, measured_maps = {}, {}, {}
    with open(path, newline="", encoding="utf-8") as f:
        r = csv.DictReader(f)
        fields = [c for c in (r.fieldnames or [])
                  if c != "h3_id" and not c.endswith("__i") and not c.endswith("__m")]
        for fn in fields:
            field_maps[fn], interp_maps[fn], measured_maps[fn] = {}, set(), {}
        for row in r:
            hid = row.get("h3_id")
            if not hid:
                continue
            for fn in fields:
                raw = row.get(fn, "")
                if raw != "":
                    try:
                        field_maps[fn][hid] = float(raw)
                        if row.get(fn + "__i", "") == "1":
                            interp_maps[fn].add(hid)
                    except (TypeError, ValueError):
                        pass
                msnap = row.get(fn + "__m", "")
                if msnap != "":
                    try:
                        measured_maps[fn][hid] = float(msnap)
                    except (TypeError, ValueError):
                        pass
    return field_maps, interp_maps, {fn: m for fn, m in measured_maps.items() if m}

def reinterpolate(output_dir, area_id, field, metas, cells, *,
                  method="idw", radius_km=1.0, domain="all", anchor=False):
    """(Re)interpolate one already-written field in <area>_celldata.csv in place.
    Previously interpolated cells are dropped first (so measured cells are the only
    sources and re-runs are idempotent), then in-domain gaps are filled by `method`
    on the 2D metric plane (idw / voronoi / diffusion; `anchor` only affects
    diffusion -- see interpolate_field). `method='none'` (or radius_km<=0) just
    clears any existing fill. Other fields are preserved. `metas` (the manifest
    field-meta list) is updated in place for this field. Returns (celldata_filename,
    interp_count). Raises KeyError if the field is unknown."""
    field_maps, interp_maps, measured_maps = read_celldata(
        Path(output_dir) / f"{area_id}_celldata.csv")
    if field not in field_maps:
        raise KeyError(f"field {field!r} not in celldata (have {list(field_maps)})")
    # Reset to the ORIGINAL measured cells: the `__m` snapshot if a prior conserving
    # run altered the sources, else the non-interpolated cells (sources untouched).
    measured = measured_maps.get(field) or {
        c: v for c, v in field_maps[field].items()
        if c not in interp_maps.get(field, set())}
    measured = dict(measured)
    if method == "none" or radius_km <= 0.0:
        filled, interp_ids = measured, set()
    else:
        filled, interp_ids = interpolate_field(
            measured, domain_ids(cells, domain), coords=metric_coords(cells),
            radius_m=radius_km * 1000.0, method=method, anchor=anchor)
    field_maps[field], interp_maps[field] = filled, interp_ids
    # Snapshot the measured baseline only when this method rewrites source values
    # (so display != measured); otherwise the plain column already holds truth.
    if method == "voronoi" or (method == "diffusion" and not anchor):
        measured_maps[field] = measured
    else:
        measured_maps.pop(field, None)
    fname = write_celldata(output_dir, area_id, field_maps, interp_maps, measured_maps)
    m = next((x for x in metas if x.get("name") == field), None)
    if m is not None:
        vals = list(filled.values())
        m.update(min=round(min(vals), 6) if vals else 0.0,
                 max=round(max(vals), 6) if vals else 0.0, count=len(vals),
                 interp=method, radius_km=radius_km, domain=domain,
                 interp_count=len(interp_ids), anchor=bool(anchor))
    return fname, len(interp_ids)
