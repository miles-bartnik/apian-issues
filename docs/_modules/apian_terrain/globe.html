"""
globe.py -- orthographic H3 globe geometry + index-based area selection.

The companion app's area picker is a rotatable orthographic globe of H3 cells:
you mark hexes at any resolution and expand them to a target resolution. All
selection is pure H3 index traversal (cell_to_children / cell_to_parent) -- no
projection-based hit tests -- so it behaves identically everywhere, poles
included. The projection is only for drawing and for turning a click into a
lat/lon (then an H3 index).

Pure numpy + h3 (no bpy / tkinter), shared by the GUI and the pipeline.
"""

import numpy as np

def basis(lat0, lon0):
    """Orthonormal (east, north, out) for a view centred at (lat0, lon0).
    `out` points at the viewer; `east`/`north` span the screen plane."""
    la, lo = np.radians(lat0), np.radians(lon0)
    cla, sla, clo, slo = np.cos(la), np.sin(la), np.cos(lo), np.sin(lo)
    out   = np.array([cla * clo, cla * slo, sla])
    east  = np.array([-slo, clo, 0.0])
    north = np.array([-sla * clo, -sla * slo, cla])
    return east, north, out




def _xyz(lat, lon):
    la, lo = np.radians(lat), np.radians(lon)
    cl = np.cos(la)
    return np.stack([cl * np.cos(lo), cl * np.sin(lo), np.sin(la)], axis=-1)

def project(lat, lon, lat0, lon0):
    """Orthographic (x, y, front) for point(s) on a globe centred at
    (lat0, lon0). x=east, y=north in [-1, 1]; front is True on the near
    hemisphere (the only part that should be drawn)."""
    east, north, out = basis(lat0, lon0)
    v = _xyz(np.asarray(lat, float), np.asarray(lon, float))
    return v @ east, v @ north, (v @ out) > 0.0

def unproject(x, y, lat0, lon0):
    """Inverse of project on the near hemisphere: screen (x, y) with
    x^2 + y^2 <= 1 -> (lat, lon). Returns None outside the globe disk."""
    r2 = x * x + y * y
    if r2 > 1.0:
        return None
    east, north, out = basis(lat0, lon0)
    v = x * east + y * north + np.sqrt(max(0.0, 1.0 - r2)) * out
    return (float(np.degrees(np.arcsin(np.clip(v[2], -1.0, 1.0)))),
            float(np.degrees(np.arctan2(v[1], v[0]))))




def _slerp(a, b, n):
    """n points along the great-circle arc from unit vector a toward b,
    including a and excluding b (n >= 1). Endpoints excluded on the b side so
    consecutive edges don't double up their shared vertex."""
    omega = np.arccos(np.clip(np.dot(a, b), -1.0, 1.0))
    if omega < 1e-9:
        return [a]
    so = np.sin(omega)
    return [np.sin((1 - t) * omega) / so * a + np.sin(t * omega) / so * b
            for t in np.arange(n) / n]

def densify_boundary(cell, n_per_edge=8):
    """Closed ring of (lat, lon) for an H3 cell, with each edge of
    cell_to_boundary slerp-subdivided into n_per_edge segments so the border
    follows the sphere (great-circle arcs) rather than straight chords. This is
    the true H3 geometry -- accurate on an orthographic globe and on a Mercator
    map alike. Pure h3 + numpy."""
    import h3
    verts = h3.cell_to_boundary(cell)
    v = _xyz(np.array([p[0] for p in verts], float),
             np.array([p[1] for p in verts], float))       # (N, 3) unit vectors
    pts = [p for i in range(len(v))
           for p in _slerp(v[i], v[(i + 1) % len(v)], n_per_edge)]
    xyz = np.array(pts)
    lat = np.degrees(np.arcsin(np.clip(xyz[:, 2], -1.0, 1.0)))
    lon = np.degrees(np.arctan2(xyz[:, 1], xyz[:, 0]))
    return list(zip(lat.tolist(), lon.tolist()))

def polygon_cells(marked, target_res):
    """Trace the union outline of the marked cells into a polygon, then fill it
    with target_res cells (h3shape_to_cells). The result follows the shape of the
    selection -- L-shapes and holes survive -- with a clean target-res tiling
    inside, instead of the ragged/fractal edge you get from expanding a mixed-res
    selection down the (non-nesting) H3 hierarchy.

    cells_to_h3shape needs a single resolution, so a mixed-res selection is first
    normalised to its finest res (children approximate their parent). Handles
    disconnected selections (LatLngMultiPoly). Returns a set of ids."""
    import h3
    if not marked:
        return set()
    res = max(h3.get_resolution(c) for c in marked)
    uniform = set()
    for c in marked:
        if h3.get_resolution(c) == res:
            uniform.add(c)
        else:
            uniform.update(h3.cell_to_children(c, res))
    shape = h3.cells_to_h3shape(list(uniform))
    return set(h3.h3shape_to_cells(shape, target_res))

def estimate_cells(marked, target_res):
    """Rough live count of target_res cells the selection will fill: total
    selected area over the average target-res cell area. Area-based (like the
    old km-bbox estimate), cheap to recompute as the selection changes -- unlike
    materialising the actual polygon fill on every redraw."""
    import h3
    if not marked:
        return 0
    area = sum(h3.cell_area(c, unit="km^2") for c in marked)
    return round(area / h3.average_hexagon_area(target_res, unit="km^2"))




# ── adaptive drill-down mesh ──────────────────────────────────────────────────
# The picker's "mesh" is a set of H3 ids forming a disjoint cover of the globe,
# starting as the 122 res-0 cells. drill/undrill/vertex_cells keep it a disjoint
# cover while refining it at mixed resolutions.

def _is_descendant(cell, ancestor):
    """True if `cell` is `ancestor` or lies inside it (finer or equal res)."""
    import h3
    ra = h3.get_resolution(ancestor)
    return h3.get_resolution(cell) >= ra and h3.cell_to_parent(cell, ra) == ancestor


def _mesh_owner(mesh, cell):
    """The mesh cell covering `cell` -- itself or a coarser ancestor -- or None
    (e.g. that area is drilled finer than `cell`)."""
    import h3
    for r in range(h3.get_resolution(cell), -1, -1):
        anc = h3.cell_to_parent(cell, r)
        if anc in mesh:
            return anc
    return None

def drill(mesh, cell):
    """Adaptive subdivide: replace `cell` with its children one resolution finer.
    Returns a new set; mesh stays a disjoint cover of the globe."""
    import h3
    return (set(mesh) - {cell}) | set(
        h3.cell_to_children(cell, h3.get_resolution(cell) + 1))

def undrill(mesh, cell):
    """Inverse of drill: collapse `cell`'s parent subtree back to that parent.
    No-op at res 0."""
    import h3
    r = h3.get_resolution(cell)
    if r == 0:
        return set(mesh)
    parent = h3.cell_to_parent(cell, r - 1)
    return {c for c in mesh if not _is_descendant(c, parent)} | {parent}

def vertex_cells(mesh, cell, vertex_latlng, eps=1e-5):
    """The mesh cells touching `vertex_latlng` (a corner of `cell`): `cell` plus
    the neighbours sharing that corner, each mapped to the mesh cell that covers
    it. 3 cells at a hexagon vertex, up to 5 at a pentagon vertex."""
    import h3
    vlat, vlon = vertex_latlng
    around = {cell}
    for n in h3.grid_disk(cell, 1):
        if n != cell and any(abs(la - vlat) < eps and abs(lo - vlon) < eps
                             for la, lo in h3.cell_to_boundary(n)):
            around.add(n)
    return {o for o in (_mesh_owner(mesh, c) for c in around) if o is not None}
