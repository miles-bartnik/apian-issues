Source code for apian_terrain.layers

"""layers.py -- the modular data-layer registry.

A data LAYER is anything baked onto the base terrain (aggregate metrics today;
navigation / edges / labels later). Each layer declares a KIND, and the three
stages that touch it -- the exporter, the import-contract validator (here), and
the Blender importer (blender/main.py) -- all DISPATCH on that kind. So adding a
new kind of data = register a validator here + a baker in the importer, with no
change to any core loop.

The manifest carries `layers: [{name, kind, file, spec}]`. Legacy manifests
(`data_fields` / `feature_files` / `celldata_csv`) are normalised into the same
shape by `manifest_layers`, so every reader sees one uniform list and the format
stays append-only: an older importer skips an unknown kind, a newer importer reads
an older manifest. Pure module (no bpy) -- unit-tested.
"""
from pathlib import Path
from datetime import datetime
import csv
import json

# A time-series (cell_series) dataset is keyed on a TIMESTAMP column named exactly this.
# Kept in sync with series.EVENT_TS_COLUMN (this add-on-shipped module can't import that
# desktop-only one).
EVENT_TS_COLUMN = "event_ts"

# kind -> validate(data_dir, layer, cell_ids) -> (ok: bool, messages: [str])
LAYER_VALIDATORS = {}

def manifest_layers(manifest):
    """A uniform list of `{name, kind, file, spec}` for a manifest, whether it uses
    the new `layers` block or the legacy `data_fields` / `feature_files`. This is the
    single place that reconciles old + new, so callers never branch on manifest age.

    The canonical `layers` block and the legacy fields are MERGED: a manifest that
    started life with `feature_files` and later gained a `layers` entry (e.g. a
    time-series layer written by the Output tab) keeps every layer visible. When a
    `layers` entry has the same `(name, kind)` as a legacy layer, the canonical
    `layers` entry wins."""
    out = []
    seen = set()

    # Legacy per-cell fields share one CSV.
    celldata = manifest.get("data_fields_csv") or manifest.get("celldata_csv")
    for meta in manifest.get("data_fields") or []:
        layer = {"name": meta["name"], "kind": "cell_field",
                 "file": celldata, "spec": meta}
        key = (layer["name"], layer["kind"])
        if key not in seen:
            seen.add(key)
            out.append(layer)

    # Legacy GeoJSON feature layers.
    for name, file in (manifest.get("feature_files") or {}).items():
        layer = {"name": name, "kind": "feature", "file": file, "spec": {}}
        key = (layer["name"], layer["kind"])
        if key not in seen:
            seen.add(key)
            out.append(layer)

    # Canonical `layers` block is additive and overrides legacy entries with the
    # same (name, kind) so the manifest can migrate piecemeal.
    for layer in manifest.get("layers") or []:
        layer = dict(layer)
        key = (layer.get("name"), layer.get("kind"))
        if key in seen:
            out = [l for l in out
                   if (l.get("name"), l.get("kind")) != key]
        else:
            seen.add(key)
        out.append(layer)

    return out

def validate_layer(data_dir, layer, cell_ids):
    """Validate one layer via its kind's validator. An UNKNOWN kind is not a failure
    -- it is skipped with a note (the importer ignores it too), so a new-kind dataset
    still validates on an older build."""
    validator = LAYER_VALIDATORS.get(layer.get("kind"))
    if validator is None:
        return True, [f"SKIP unknown kind '{layer.get('kind')}' (importer will ignore)"]
    return validator(data_dir, layer, cell_ids)

def validate_layers(data_dir, manifest, cell_ids):
    """Dispatch every layer to its validator. Returns (ok, [message]) with each
    message prefixed by the layer name + kind."""
    ok, msgs = True, []
    for layer in manifest_layers(manifest):
        lok, lmsgs = validate_layer(data_dir, layer, cell_ids)
        ok = ok and lok
        msgs += [f"[{layer.get('name')} : {layer.get('kind')}] {m}" for m in lmsgs]
    return ok, msgs

def _validate_cell_field(data_dir, layer, cell_ids):
    """A per-cell scalar/vector field: the file has an h3_id column, and every h3_id
    carrying THIS field's value is one of the area's cells (fields share one file, so
    we check this field's own column)."""
    path = Path(data_dir) / (layer.get("file") or "")
    if not path.is_file():
        return False, [f"FAIL missing file {layer.get('file')}"]
    name = layer["name"]
    seen = stray = 0
    with open(path, newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        if "h3_id" not in (reader.fieldnames or []):
            return False, ["FAIL no h3_id column"]
        for row in reader:
            if not (row.get(name) or "").strip():
                continue
            seen += 1
            if row["h3_id"] not in cell_ids:
                stray += 1
    if stray:
        return False, [f"FAIL {stray}/{seen} h3_id(s) not in cells"]
    return True, [f"PASS {seen} value(s), all cells"]


def _validate_feature(data_dir, layer, cell_ids):
    """A GeoJSON feature layer: just that the file is present (geometry is validated
    downstream by the importer's own reader)."""
    path = Path(data_dir) / (layer.get("file") or "")
    if not path.is_file():
        return False, [f"FAIL missing file {layer.get('file')}"]
    return True, ["PASS file present"]

# validator ships in the add-on, which must stay stdlib on its read path).
_DTYPE_BYTES = {"float16": 2, "float32": 4, "float64": 8}

def time_labels_datetime_ok(labels):
    """Is a cell_series time axis 'date-dimensioned correctly' -- every step an ISO
    datetime AND strictly increasing? Returns (ok, reason). Per the time-series contract
    (axis = TIMESTAMP column 'event_ts'), a non-datetime axis is a hard FAIL in the series
    validator, not a warning. Stdlib only (ships in the addon)."""
    prev = None
    for s in labels:
        try:
            d = datetime.fromisoformat(str(s))
        except (ValueError, TypeError):
            return False, f"step {s!r} is not an ISO datetime"
        if prev is not None and d <= prev:
            return False, f"steps not strictly increasing at {s!r}"
        prev = d
    return True, ""




def _validate_cell_series(data_dir, layer, cell_ids):
    """A per-cell TIME SERIES (series.py): a JSON header + a `.cells` order file + a sparse
    CSR triple (rowptr int64[n_steps+1], cols int32[nnz], vals <dtype>[nnz]). Checks,
    stdlib-only: the header parses, its companion files exist with byte sizes matching
    nnz/n_steps/dtype, the cells count matches n_cells, and every series cell is one of the
    area's cells. FAILs if the time axis is not the TIMESTAMP column 'event_ts' (the
    time-series contract). Interior discontinuities are permitted -> WARN only."""
    data_dir = Path(data_dir)
    hdr_path = data_dir / (layer.get("file") or "")
    if not hdr_path.is_file():
        return False, [f"FAIL missing header {layer.get('file')}"]
    try:
        h = json.loads(hdr_path.read_text(encoding="utf-8"))
    except (ValueError, OSError) as e:
        return False, [f"FAIL unreadable header: {e}"]
    n_steps, n_cells, nnz = h.get("n_steps"), h.get("n_cells"), h.get("nnz")
    itemsize = _DTYPE_BYTES.get(h.get("dtype"))
    if not itemsize:
        return False, [f"FAIL unknown dtype {h.get('dtype')!r}"]

    sizes = {                           # file key -> expected byte size
        "rowptr_file": ((n_steps or 0) + 1) * 8,
        "cols_file":   (nnz or 0) * 4,
        "vals_file":   (nnz or 0) * itemsize,
    }
    if h.get("interp_file"):            # optional resample/interpolation mask, uint8[nnz]
        sizes["interp_file"] = (nnz or 0) * 1
    for key, want in sizes.items():
        f = data_dir / (h.get(key) or "")
        if not f.is_file():
            return False, [f"FAIL missing {key} {h.get(key)}"]
        if f.stat().st_size != want:
            return False, [f"FAIL {key} is {f.stat().st_size} B, header implies {want} B"]

    cellf = data_dir / (h.get("cells_file") or "")
    if not cellf.is_file():
        return False, [f"FAIL missing cells file {h.get('cells_file')}"]
    series_cells = [c for c in cellf.read_text(encoding="utf-8").split("\n") if c]
    if len(series_cells) != n_cells:
        return False, [f"FAIL {len(series_cells)} cells listed, header says {n_cells}"]
    stray = sum(1 for c in series_cells if c not in cell_ids)
    if stray:
        return False, [f"FAIL {stray}/{n_cells} cell(s) not in area"]

    ok = True
    msgs = [f"PASS {n_steps} steps x {n_cells} cells, {nnz} records, all in area"]
    synth = h.get("interp_count") or 0
    if synth:                           # disclose that some records are interpolated, not measured
        msgs.append(f"WARN {synth}/{nnz} record(s) synthesised by resampling "
                    "(interior time-fills and/or spatial fills), not measured")
    # CONTRACT: the axis must be the TIMESTAMP column 'event_ts' (hard requirement).
    axis = h.get("time_col")
    if axis != EVENT_TS_COLUMN:
        ok = False
        msgs.append(f"FAIL time axis must be the column '{EVENT_TS_COLUMN}', got {axis!r} "
                    "(rewrite the series from a query with an event_ts TIMESTAMP)")
    dt_ok, why = time_labels_datetime_ok(h.get("time_labels") or [])
    if not dt_ok:
        ok = False
        msgs.append(f"FAIL '{EVENT_TS_COLUMN}' is not a TIMESTAMP ({why})")
    disc = h.get("discontinuity_cells") or 0
    if disc:                            # gaps are permitted; just surface them
        msgs.append(f"WARN {disc} cell(s) have interior gaps "
                    f"({h.get('discontinuity_steps', 0)} gap-steps); they display nothing "
                    "at those steps")
    return ok, msgs
