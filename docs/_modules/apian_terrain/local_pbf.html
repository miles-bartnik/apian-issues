"""
local_pbf.py
------------
Local PBF library: find PBFs in the PBF dir whose coverage covers the area.

The local library is scanned first. When nothing local covers the area, the
missing regions are downloaded from Geofabrik (geofabrik.py) using the
configured email address to identify us as a real user. A file's coverage comes
from its PBF header bbox, or from a bbox embedded in the filename (e.g.
planet_6.939,43.508_7.898,43.961.osm.pbf).

Selection: the smallest single file that fully contains the area bbox wins
(smallest is fastest to clip). If no single file contains it, a greedy
smallest-first union of overlapping files is tried -- the stitch itself
(dedupe by OSM id, newest version wins, cross-file reference completion)
happens in pbf_reader._write_extract.

Completed pipeline extracts (*_extract.osm.pbf) ARE first-class sources: they
carry full reference-complete data for their bbox, and their coverage comes
from the *_extract.bbox.json sidecar the pipeline writes alongside. Reusing a
small local extract beats re-clipping a multi-GB region file for any area
inside it. Only in-progress temp writes (*_tmp.osm.pbf) are skipped. An area
never selects its OWN extract (exclude_area_id): a rebuild must re-clip from
real sources -- and the GUI purges that file right after this scan.
"""

import json
import re
from pathlib import Path

# planet_<minlon>,<minlat>_<maxlon>,<maxlat>.osm.pbf  (BBBike / Protomaps style)
_BBOX_NAME_RE = re.compile(
    r"(-?\d+(?:\.\d+)?),(-?\d+(?:\.\d+)?)_(-?\d+(?:\.\d+)?),(-?\d+(?:\.\d+)?)")

_SKIP_SUFFIXES = ("_extract_tmp.osm.pbf", "_tmp.osm.pbf")


def _sane(lat_min, lon_min, lat_max, lon_max):
    return (-90 <= lat_min < lat_max <= 90
            and -180 <= lon_min < lon_max <= 180)


def _bbox_from_filename(name):
    """(lat_min, lon_min, lat_max, lon_max) parsed from the name, or None."""
    m = _BBOX_NAME_RE.search(name)
    if not m:
        return None
    lon_min, lat_min, lon_max, lat_max = (float(g) for g in m.groups())
    if not _sane(lat_min, lon_min, lat_max, lon_max):
        return None
    return (lat_min, lon_min, lat_max, lon_max)


def _bbox_from_sidecar(path):
    """(lat_min, lon_min, lat_max, lon_max) from the pipeline's extract sidecar
    (<name>_extract.bbox.json, written next to <name>_extract.osm.pbf by
    pbf_reader._write_extract), or None."""
    p = Path(path)
    if not p.name.endswith("_extract.osm.pbf"):
        return None
    side = p.with_name(p.name[:-len(".osm.pbf")] + ".bbox.json")
    try:
        b = json.loads(side.read_text(encoding="utf-8"))
        out = (float(b["lat_min"]), float(b["lon_min"]),
               float(b["lat_max"]), float(b["lon_max"]))
    except Exception:
        return None
    return out if _sane(*out) else None


def _bbox_from_header(path):
    """(lat_min, lon_min, lat_max, lon_max) from the PBF header, or None.

    Pipeline-written extracts carry no header bbox (osmium writers don't set
    one); their coverage comes from the *_extract.bbox.json sidecar instead.
    """
    try:
        import osmium.io  # lazy: keeps the scan's pure logic osmium-free
        reader = osmium.io.Reader(str(path))
        try:
            box = reader.header().box()
        finally:
            reader.close()
        if not box.valid():
            return None
        lat_min, lon_min = box.bottom_left.lat, box.bottom_left.lon
        lat_max, lon_max = box.top_right.lat, box.top_right.lon
        if not _sane(lat_min, lon_min, lat_max, lon_max):
            return None
        return (lat_min, lon_min, lat_max, lon_max)
    except Exception:
        return None

def pbf_coverage(path):
    """Best-known coverage bbox for a PBF file, or None if undeterminable."""
    return (_bbox_from_filename(Path(path).name)
            or _bbox_from_sidecar(path)
            or _bbox_from_header(path))




def _contains(outer, inner):
    """True if coverage bbox `outer` fully contains area bbox `inner`."""
    o_lat_min, o_lon_min, o_lat_max, o_lon_max = outer
    i_lat_min, i_lon_min, i_lat_max, i_lon_max = inner
    return (o_lat_min <= i_lat_min and o_lat_max >= i_lat_max
            and o_lon_min <= i_lon_min and o_lon_max >= i_lon_max)


# --- rectangle math for union coverage --------------------------------------
# Rects are (lat_min, lon_min, lat_max, lon_max) throughout.

def _rect_intersection(a, b):
    la0, lo0 = max(a[0], b[0]), max(a[1], b[1])
    la1, lo1 = min(a[2], b[2]), min(a[3], b[3])
    if la0 < la1 and lo0 < lo1:
        return (la0, lo0, la1, lo1)
    return None


def _rect_subtract(rect, cover):
    """The pieces of `rect` NOT covered by `cover`: 0-4 rectangles."""
    inter = _rect_intersection(rect, cover)
    if inter is None:
        return [rect]
    la0, lo0, la1, lo1 = rect
    ia0, io0, ia1, io1 = inter
    out = []
    if la0 < ia0:
        out.append((la0, lo0, ia0, lo1))      # south band
    if ia1 < la1:
        out.append((ia1, lo0, la1, lo1))      # north band
    if lo0 < io0:
        out.append((ia0, lo0, ia1, io0))      # west slice (between the bands)
    if io1 < lo1:
        out.append((ia0, io1, ia1, lo1))      # east slice
    return out


def _select_covering(candidates, bbox):
    """[path, ...] covering `bbox`, or [] when the candidates can't.

    candidates: [(path, size_bytes, coverage|None)]. The smallest single file
    containing the bbox wins outright; otherwise files are added smallest-first
    until their coverage union covers the bbox (greedy set cover on rects).
    """
    ordered = sorted(candidates, key=lambda c: c[1])
    for path, _size, cov in ordered:
        if cov is not None and _contains(cov, bbox):
            return [path]
    uncovered = [bbox]
    chosen = []
    for path, _size, cov in ordered:
        if cov is None:
            continue
        if not any(_rect_intersection(u, cov) for u in uncovered):
            continue
        chosen.append(path)
        rest = []
        for u in uncovered:
            rest.extend(_rect_subtract(u, cov))
        uncovered = rest
        if not uncovered:
            return chosen
    return []

def find_covering_pbfs(lat_min, lon_min, lat_max, lon_max, pbf_dir,
                       exclude_area_id=None):
    """[Path, ...]: smallest local PBF set covering the bbox ([] = none).

    exclude_area_id: that area's own extract (<id>_extract.osm.pbf) is never
    selected -- a rebuild must re-clip from real sources, and the GUI purges
    that file right after this scan."""
    pbf_dir = Path(pbf_dir)
    own = f"{exclude_area_id}_extract.osm.pbf" if exclude_area_id else None
    candidates = []
    if pbf_dir.is_dir():
        for p in pbf_dir.glob("*.osm.pbf"):
            if p.name.endswith(_SKIP_SUFFIXES) or p.name == own:
                continue
            try:
                size = p.stat().st_size
            except OSError:
                continue
            if size == 0:
                continue
            candidates.append((p, size, pbf_coverage(p)))
    return _select_covering(candidates, (lat_min, lon_min, lat_max, lon_max))

def ensure_pbfs(lat_min, lon_min, lat_max, lon_max, pbf_dir, exclude_area_id=None,
                progress_cb=None, allow_download=True):
    """Return [Path, ...] of the local PBF(s) covering the bbox.

    The local library is scanned first. When no local file (or union of files)
    covers the bbox and allow_download is set, the missing regions are downloaded
    from Geofabrik (the configured email address identifies us as a real user --
    see geofabrik.py). With allow_download off (tests, headless, offline), raises
    FileNotFoundError listing the files seen instead of touching the network.

    exclude_area_id: never select that area's own extract (see find_covering_pbfs).
    """
    found = find_covering_pbfs(lat_min, lon_min, lat_max, lon_max, pbf_dir,
                               exclude_area_id=exclude_area_id)
    if found:
        if len(found) == 1:
            print(f"  [pbf] source: {found[0].name} "
                  f"({found[0].stat().st_size / 1_048_576:.0f} MB)", flush=True)
        else:
            print(f"  [pbf] sources (stitched, smallest-first):", flush=True)
            for p in found:
                print(f"    - {p.name} ({p.stat().st_size / 1_048_576:.0f} MB)",
                      flush=True)
        return found

    if allow_download:
        # No local coverage -> Geofabrik fallback.
        from . import config
        from . import geofabrik
        print("  [pbf] no local PBF covers the area -- trying Geofabrik ...", flush=True)
        return geofabrik.ensure_pbfs(lat_min, lon_min, lat_max, lon_max, pbf_dir,
                                     email=getattr(config, "GEOFABRIK_EMAIL", ""),
                                     progress_cb=progress_cb)

    pbf_dir = Path(pbf_dir)
    own = f"{exclude_area_id}_extract.osm.pbf" if exclude_area_id else None
    seen = []
    if pbf_dir.is_dir():
        for p in sorted(pbf_dir.glob("*.osm.pbf")):
            if p.name.endswith(_SKIP_SUFFIXES) or p.name == own:
                continue
            cov = pbf_coverage(p)
            cov_s = (f"covers lat {cov[0]:.3f}..{cov[2]:.3f}, "
                     f"lon {cov[1]:.3f}..{cov[3]:.3f}") if cov else "coverage unknown"
            seen.append(f"    - {p.name}: {cov_s}")
    listing = "\n".join(seen) if seen else "    (no .osm.pbf files found)"
    raise FileNotFoundError(
        f"No PBF in {pbf_dir} covers the area bbox "
        f"(lat {lat_min:.4f}..{lat_max:.4f}, lon {lon_min:.4f}..{lon_max:.4f}) -- "
        f"neither a single file nor any combination of the files present.\n"
        f"Nothing is downloaded automatically: download a PBF that covers the "
        f"area (e.g. download.geofabrik.de, or a bbox extract from "
        f"bbbike.org / protomaps) and place it in that folder.\n"
        f"Files seen:\n{listing}")
