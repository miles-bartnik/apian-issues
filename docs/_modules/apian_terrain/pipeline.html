def resolve_detail(detail, lon_span, lat_span):
    """Whether to build in MAJOR-features mode. detail 'full' forces OSM detail, 'major'
    forces Natural Earth; 'auto' (default) switches to major once the selection spans more
    than MAJOR_SPAN_DEG in either dimension."""
    d = (detail or "auto").lower()
    if d == "full":
        return False
    if d == "major":
        return True
    return max(lon_span, lat_span) > MAJOR_SPAN_DEG




def _resample_street_coords(coords, spacing_m=10.0):
    """Resample [[lon, lat, z], ...] at fixed metre spacing, interpolating Z."""
    if len(coords) < 2:
        return coords
    DEG_TO_M = 111_320.0
    cos_lat = math.cos(math.radians(coords[0][1]))
    dists = [0.0, *itertools.accumulate(
        math.hypot((coords[i][0]-coords[i-1][0])*cos_lat*DEG_TO_M,
                   (coords[i][1]-coords[i-1][1])*DEG_TO_M)
        for i in range(1, len(coords))
    )]
    total = dists[-1]
    if total == 0:
        return coords
    targets = [i*spacing_m for i in range(int((total-1e-6)//spacing_m)+1)] + [total]
    result, j = [], 0
    for t_target in targets:
        while j < len(dists) - 2 and dists[j+1] < t_target - 1e-9:
            j += 1
        frac = (t_target - dists[j]) / (dists[j+1] - dists[j]) if dists[j+1] > dists[j] else 0.0
        result.append([round(coords[j][k] + frac*(coords[j+1][k]-coords[j][k]), (7,7,3)[k]) for k in range(3)])
    return result


# Tunnel values that run at surface grade, not below ground -- a road through a
# building, or an avalanche gallery. Everything else non-"no" (yes, culvert,
# canal, ...) is sub-grade and gets the portal interpolation below.
_AT_GRADE_TUNNELS = {"building_passage", "avalanche_protector"}


def _is_subgrade_tunnel(tag):
    return tag not in ("", "no", None) and tag not in _AT_GRADE_TUNNELS


def _is_bridge(tag):
    """Any non-'no' bridge value rides a deck (yes, viaduct, boardwalk, ...)."""
    return tag not in ("", "no", None)


def _street_type(street):
    """The street's class: the new `street_type` key, falling back to the legacy
    `highway_tag` from pre-rename caches. Defaults to 'road' when untagged."""
    return street.get("street_type") or street.get("highway_tag") or "road"


def _interp_span_z(coords):
    """Straight grade between the two span endpoints: replace interior-node Z with
    a linear ramp between the endpoint elevations, by cumulative horizontal length.

    Used where OSM gives us no interior elevation and the surface terrain is the
    wrong reference. Split-at-portal/abutment way endpoints sit at grade and are
    the trustworthy anchors:
      - sub-grade tunnels: the interior runs straight below the surface it bores
        through (the terrain mesh occludes the sunk span);
      - bridge=yes roads: the interior rides a straight deck between the abutments,
        clearing the valley/water instead of draping into it. (Where a real
        man_made=bridge deck exists the importer still lifts the road onto it.)"""
    if len(coords) < 3:
        return coords
    DEG_TO_M = 111_320.0
    cos_lat = math.cos(math.radians(coords[0][1]))
    dists = [0.0, *itertools.accumulate(
        math.hypot((coords[i][0]-coords[i-1][0])*cos_lat*DEG_TO_M,
                   (coords[i][1]-coords[i-1][1])*DEG_TO_M)
        for i in range(1, len(coords))
    )]
    total = dists[-1]
    if total == 0:
        return coords
    z0, z1 = coords[0][2], coords[-1][2]
    return [[c[0], c[1], round(z0 + (z1-z0)*(d/total), 3)]
            for c, d in zip(coords, dists)]

def _standing_surface(beds, osm_ele):
    """Standing-water surface level. An OSM `ele` tag wins when present
    (including an explicit 0 = sea level); otherwise the 25th percentile of
    the body cells' raw beds — the mean perched lakes implausibly high when
    a polygon includes dry hillside cells (water towers near sharp elevation
    changes), so the level seeks the basin's low edge instead."""
    if osm_ele is not None:
        return float(osm_ele)
    bs = sorted(beds)
    return float(bs[max(0, min(int(len(bs) * 0.25), len(bs) - 1))])


def _custom_bathy_beds(bathy_tiles, id_to_cell, owner):
    """Below-water layer of the "split" custom terrain source: {h3_id: bed
    elevation} for water-body-owned cells the bathy tiles cover (None samples
    filtered out). Empty when no bathy dir is set or no owned cell is covered.
    """
    if bathy_tiles is None or len(bathy_tiles) == 0 or not owner:
        return {}
    cells = [c for cid, c in id_to_cell.items() if cid in owner]
    sampled = bathy_tiles.sample(cells, mode="bilinear")
    return {cid: v for cid, v in sampled.items() if v is not None}


def _cell_union_polygon(cell_ids, h3_resolution):
    """Return a shapely polygon covering the union of the given H3 cells.

    Falls back to None if h3 or shapely are unavailable."""
    try:
        import h3 as _h3
        from shapely.geometry import shape
    except Exception:
        return None
    try:
        geo = _h3.cells_to_geo(list(cell_ids), tight=False)
        return shape(geo)
    except Exception:
        return None


def _safe_intersection(geom, clip):
    """intersection() that repairs invalid OSM input instead of raising.

    Raw OSM rings occasionally self-intersect, and GEOS then throws
    TopologyException ('side location conflict', ...) from intersection() --
    one bad footprint killed an entire new_york run (relation r16243928,
    2026-09-21). Repair with make_valid and retry; a still-failing repair
    yields an EMPTY result so the caller's normal empty handling applies."""
    try:
        return geom.intersection(clip)
    except Exception:
        pass
    try:
        from shapely import make_valid
        return make_valid(geom).intersection(clip)
    except Exception:
        return geom.__class__()


def _clip_footprint_to_cells(footprint, cell_poly):
    """Clip a footprint ring to the cell-union polygon.

    Returns the single largest surviving ring as a list of [lon, lat], or None
    if nothing survives."""
    if cell_poly is None or len(footprint) < 3:
        return None
    try:
        from shapely.geometry import Polygon
    except Exception:
        return None
    poly = Polygon([(p[0], p[1]) for p in footprint])
    clipped = _safe_intersection(poly, cell_poly)
    if clipped.is_empty:
        return None
    rings = []
    if clipped.geom_type == "Polygon":
        rings.append(clipped.exterior.coords[:-1])
    elif clipped.geom_type == "MultiPolygon":
        for p in clipped.geoms:
            rings.append(p.exterior.coords[:-1])
    if not rings:
        return None
    # Keep the largest surviving ring by area.
    biggest = max(rings, key=lambda r: Polygon(r).area)
    return [[float(c[0]), float(c[1])] for c in biggest]


def _clip_street_to_cells(nodes, cell_poly):
    """Clip a street line to the cell-union polygon.

    Returns a list of coordinate sequences [[lon, lat], ...], each fully inside
    the cell union."""
    if cell_poly is None or len(nodes) < 2:
        return []
    try:
        from shapely.geometry import LineString
    except Exception:
        return []
    line = LineString([(n[0], n[1]) for n in nodes])
    clipped = _safe_intersection(line, cell_poly)
    if clipped.is_empty:
        return []
    sequences = []
    if clipped.geom_type == "LineString":
        sequences.append([list(c) for c in clipped.coords])
    elif clipped.geom_type == "MultiLineString":
        for ls in clipped.geoms:
            sequences.append([list(c) for c in ls.coords])
    return sequences

def choose_resolution(lat_min, lon_min, lat_max, lon_max,
                      cell_cap=None, max_res=8):
    """Return resolution and its cell count."""
    import h3 as _h3
    ring = [(lat_min, lon_min), (lat_min, lon_max),
            (lat_max, lon_max), (lat_max, lon_min)]
    poly  = _h3.LatLngPoly(ring)
    count = len(_h3.h3shape_to_cells(poly, max_res))
    print(f"  [h3 res] resolution {max_res}  ({count} cells)")
    return max_res, count

# ── Cell array layout ─────────────────────────────────────────────────────
# Terrain cell: [lon, lat, ele_m, rot_z, radius_eq, h3_id, water_type, r, g, b]
# Water cell:   [lon, lat, terrain_ele_m, rot_z, radius_eq, h3_id, surface_ele_m]
C_LON, C_LAT, C_ELE, C_ROT, C_RAD, C_ID, C_WATER = range(7)


# ── Water-body precedence ───────────────────────────────────────────────────
# When two water polygons overlap a cell, the cell belongs to exactly ONE body,
# chosen here. Lower rank wins; among equal ranks the SMALLER polygon wins, so a
# specific feature beats the big background sheet it sits inside. The sea/lagoon
# are the background fill; anything more specific governs its own cells. Edit the
# tiers to retune -- the rest of the pipeline keys off _water_rank() only.
_WATER_PRECEDENCE = [
    {"dock", "lock"},                                    # engineered, deep
    {"canal"},
    {"harbour"},
    {"basin", "reservoir", "moat", "pond"},
    {"river", "stream", "riverbank", "ditch", "drain"},  # flowing
    {"lake"},
    {"wetland"},                                         # marsh beats the background sheets
    {"lagoon"},
    {"sea", "ocean"},                                    # background sheets
]
_WATER_RANK = {tag: i for i, tier in enumerate(_WATER_PRECEDENCE) for tag in tier}
_WATER_RANK_DEFAULT = next(i for i, t in enumerate(_WATER_PRECEDENCE) if "lake" in t)


def _water_rank(tag):
    """Precedence rank for a water tag (lower wins). Unknown tags rank as lake."""
    return _WATER_RANK.get(tag, _WATER_RANK_DEFAULT)


def _ring_area(ring):
    """Unsigned shoelace area of a [lon, lat] ring (deg**2) -- overlap tie-break."""
    a = 0.0
    for (x0, y0), (x1, y1) in zip(ring, ring[1:]):
        a += x0 * y1 - x1 * y0
    return abs(a) / 2.0

def _structures_feature(area, terrain_data, id_to_cell, h3_resolution):
    """Build the structures Feature, or None if there are no structures.

    Structures are clipped to the selected H3 cell union so any part of a
    building that falls outside the processed region is removed before the
    Blender import. This keeps partial buildings that cross the boundary while
    preventing out-of-region vertices from missing the terrain surface during
    projection."""
    structures = terrain_data.get("structures")
    if not structures:
        return None
    cell_poly = _cell_union_polygon(id_to_cell.keys(), h3_resolution)
    kept = []
    for s in structures:
        fp = s.get("footprint")
        if fp and cell_poly is not None:
            fp = _clip_footprint_to_cells(fp, cell_poly)
            if fp is None:
                continue
            s = {**s, "footprint": fp}
        elif fp:
            # shapely/h3 unavailable -- fall back to centroid containment.
            try:
                import h3 as _h3
                clon = sum(p[0] for p in fp) / len(fp)
                clat = sum(p[1] for p in fp) / len(fp)
                if _h3.latlng_to_cell(clat, clon, h3_resolution) not in id_to_cell:
                    continue
            except Exception:
                continue
        kept.append({**s, "structure_id": s.get("structure_id") or f"osm_{s.get('osm_id', 'unknown')}"})
    if not kept:
        return None
    print(f"  [structures]  {len(kept)}/{len(structures)} inside the hex boundary")
    return {
        "type": "Feature",
        "properties": {
            "Name":         f"{area.area_id} - Structures",
            "area_id":      area.area_id,
            "feature_type": "structures",
            "structures":   kept,
        },
        "geometry": None,
    }


def _street_features(area, terrain_data, id_to_cell, h3_resolution):
    """Build street LineString Features with terrain-elevation-enriched nodes.

    Street ways are first clipped to the H3 cell union so no line segment leaves
    the processed region; this prevents mid-segment vertices from missing the
    terrain surface during projection when the straight line between two in-cell
    nodes cuts through a gap outside the cell set."""
    try:
        import h3 as _h3_st
    except ImportError:
        _h3_st = None

    from .progress import bar as _bar
    _n_ways = len(terrain_data.get("streets", []))
    if _n_ways:
        print(f"  [streets] draping {_n_ways} cached street ways over the terrain: "
              f"re-baking every node's Z from the current cell elevations "
              f"(geometry clipped to the cell union; tunnels/bridges graded straight)")
    street_features = []
    cell_poly = _cell_union_polygon(id_to_cell.keys(), h3_resolution)
    for street in _bar(terrain_data.get("streets", []), label="streets"):
        nodes = street.get("nodes", [])
        if len(nodes) < 2:
            continue

        # Clip the way to the cell union, then enrich each surviving node with
        # the elevation of the cell that contains it.
        if cell_poly is not None:
            segments = _clip_street_to_cells(nodes, cell_poly)
        else:
            segments = [nodes] if _h3_st is None else None
            if segments is None:
                segments = []
                for nd in nodes:
                    lon, lat = nd[0], nd[1]
                    if _h3_st.latlng_to_cell(lat, lon, h3_resolution) in id_to_cell:
                        segments.append([nd])
                segments = [s for s in segments if len(s) >= 2]
        if not segments:
            continue

        for seg in segments:
            enriched = []
            for lon, lat in seg:
                if _h3_st is not None:
                    _c = id_to_cell.get(_h3_st.latlng_to_cell(lat, lon, h3_resolution))
                    if _c is None:
                        continue
                    z = float(_c[C_ELE])
                else:
                    z = 0.0
                enriched.append([lon, lat, z])
            if len(enriched) < 2:
                continue
            enriched = _resample_street_coords(enriched)
            # Sub-grade tunnels and bridge decks: run a straight grade between the span
            # endpoints instead of draping the interior over/under the surface terrain
            # (tunnels bore under it; bridges span above it). A real man_made=bridge
            # deck still lifts the road further in the importer.
            if (_is_subgrade_tunnel(street.get("tunnel", "no"))
                    or _is_bridge(street.get("bridge", "no"))):
                enriched = _interp_span_z(enriched)

            street_features.append({
                "type": "Feature",
                "properties": {
                    "area_id":      area.area_id,
                    "feature_type": "street",
                    "street_type":  _street_type(street),
                    "width_m":      street.get("width_m"),
                    "tunnel":       _is_subgrade_tunnel(street.get("tunnel", "no")),
                    "bridge":       _is_bridge(street.get("bridge", "no")),
                    "osm_id":       street.get("osm_id"),
                },
                "geometry": {
                    "type":        "LineString",
                    "coordinates": enriched,
                },
            })
    return street_features


def _landmarks_feature(area, terrain_data, id_to_cell, h3_resolution):
    """Build the landmarks Feature from all named landmarks, or None.

    Both major and minor landmarks are passed through, each tagged with an
    integer ``landmark_type`` (0 = major, 1 = minor) so the add-on can filter or
    style them per-class. Trimmed to the selected H3 cells (point containment)
    so landmarks respect the same hex boundary as everything else, not the
    wider fetch bbox."""
    try:
        import h3 as _h3
    except ImportError:
        _h3 = None
    lms = []
    for lm in terrain_data.get("landmarks", []):
        if not lm.get("name"):
            continue
        if _h3 is not None and \
                _h3.latlng_to_cell(lm["lat"], lm["lon"], h3_resolution) not in id_to_cell:
            continue
        lms.append(dict(lm))   # don't mutate the cached records
    if not lms:
        return None
    _levels = [lm.get("landmark_level") for lm in lms
               if lm.get("landmark_level") is not None]
    if _levels:
        max_level = max(_levels)
        min_level = min(_levels)
        _level_range = f"{min_level}..{max_level}"
    else:
        _level_range = "none"
    print(f"  [landmarks]  {len(lms)} inside the hex boundary "
          f"(level range {_level_range})")
    return {
        "type": "Feature",
        "properties": {
            "Name":         f"{area.area_id} - Landmarks",
            "area_id":      area.area_id,
            "feature_type": "landmarks",
            "landmarks":    lms,
        },
        "geometry": None,
    }


def _floor_baked_min(ele_floor_m, ele_max_m, pad_factor, floor_m):
    """ele_min_baked_m -- the base plane below the lowest terrain. The pad is a
    FACTOR of the elevation range (ele_max_m - ele_floor_m), not absolute metres:
    e.g. factor 0.1 over a 150 m range (min -50, max 100) puts the base 15 m
    below the floor. `floor_m` is the manual floor override when set, else
    min(0, ele_floor_m). Pure; unit-tested."""
    rng = max(0.0, float(ele_max_m) - float(ele_floor_m))
    pad = max(0.0, float(pad_factor)) * rng
    return round(float(floor_m) - pad, 3)

def process_area(area: Area, extra_features=None, progress_cb=None, cancel_event=None) -> dict:
    """
    Run the full terrain preprocessing pipeline for an area.

    Parameters
    ----------
    area            : Area instance defining the geographic area and directories
    extra_features  : optional list of GeoJSON feature dicts to append to the
                      output FeatureCollection (e.g. custom line features)

    Returns
    -------
    dict with keys:
      terrain_data : raw terrain grid dict (elevation points, OSM layers, etc.)
      features     : list of GeoJSON feature dicts
      area         : the Area instance used
    """
    def _progress(fraction, message):
        if cancel_event is not None and cancel_event.is_set():
            raise PipelineCancelled("Import cancelled by user")
        if progress_cb is not None:
            progress_cb(fraction, message)

    area.region_dir.mkdir(parents=True, exist_ok=True)
    area.raw_data_dir.mkdir(parents=True, exist_ok=True)
    area.output_dir.mkdir(parents=True, exist_ok=True)
    (area.raw_data_dir / "osm").mkdir(parents=True, exist_ok=True)
    (area.raw_data_dir / "cop30").mkdir(parents=True, exist_ok=True)
    (area.raw_data_dir / "nasadem").mkdir(parents=True, exist_ok=True)

    lat_min, lon_min, lat_max, lon_max = area.bbox

    print(f"{'='*60}")
    print(f"apian_terrain {__version__}")
    print(f"[{area.area_id}]")
    print(f"  bbox: ({lat_min:.4f}, {lon_min:.4f}) -> ({lat_max:.4f}, {lon_max:.4f})")

    if area.h3_cells:                        # explicit selection (globe picker)
        h3_resolution = area.h3_resolution   # honour the chosen final resolution
    else:
        h3_resolution, _ = choose_resolution(lat_min, lon_min, lat_max, lon_max,
                                             max_res=area.h3_resolution)

    # ── Detail mode ──────────────────────────────────────────────────────────
    # Above MAJOR_SPAN_DEG (or when forced) the OSM building/road/landmark firehose is
    # replaced by Natural Earth cities/roads/landmarks -- see naturalearth.py.
    major = resolve_detail(getattr(area, "detail", "auto"),
                           lon_max - lon_min, lat_max - lat_min)
    print(f"  [detail] {'MAJOR (Natural Earth)' if major else 'full (OSM)'}  "
          f"(mode '{getattr(area, 'detail', 'auto')}', "
          f"span {max(lon_max - lon_min, lat_max - lat_min):.2f} deg)")

    # ── OSM fetch ────────────────────────────────────────────────────────────
    _progress(0.02, "Fetching OSM data...")
    terrain_data = fetch_osm_data(
        area_id      = area.area_id,
        lat_min      = lat_min,
        lon_min      = lon_min,
        lat_max      = lat_max,
        lon_max      = lon_max,
        lat_centre   = area.lat_centre,
        lon_centre   = area.lon_centre,
        raw_data_dir = area.raw_data_dir,
        out_dir      = area.output_dir,
        progress_cb  = progress_cb,
        obm_dir      = area.obm_tile_dir or None,
        pbf_paths    = getattr(area, "pbf_paths", []),
        pbf_dir      = getattr(area, "_pbf_dir", None),
        major_mode   = major,
    )
    if terrain_data is None:
        return {"terrain_data": None, "features": [], "area": area}

    if major:                                # drop any cached OSM firehose layers; the
        for _k in ("structures", "streets", "landmarks"):   # feature builders then no-op
            terrain_data[_k] = []

    # ── H3 cells + COP30 elevations ──────────────────────────────────────────
    _progress(0.25, "Building H3 cells...")
    if area.h3_cells:
        cell_centres = cell_centres_from_ids(area.h3_cells, h3_resolution)
        print(f"  [h3 terrain centres] {len(cell_centres)} cells from selection "
              f"at resolution {h3_resolution}")
    else:
        cell_centres = build_h3_cell_centres(lat_min, lon_min, lat_max, lon_max, h3_resolution)
    _progress(0.35, "Fetching elevation...")
    h3_elevations = fetch_h3_elevations(area.area_id, cell_centres, area.raw_data_dir,
                                        gebco_tile_dir=area.gebco_tile_dir or None,
                                        dem_tile_dir=area.dem_tile_dir or None,
                                        topo_tile_dir=area.topo_tile_dir or None)
    apply_elevations(cell_centres, h3_elevations)

    _id_to_cell = {c[5]: c for c in cell_centres if len(c) > 5}

    valid_eles = [c[2] for c in cell_centres if c[2] is not None]
    ele_min, ele_max = (min(valid_eles), max(valid_eles)) if valid_eles else (0.0, 0.0)
    terrain_data["ele_min_m"] = round(ele_min, 2)
    terrain_data["ele_max_m"] = round(ele_max, 2)

    # ── Sea polygons ─────────────────────────────────────────────────────────
    _progress(0.55, "Processing coastline...")
    sea_rings = []
    _sea_max_depth = 50.0

    sea_chain = terrain_data.get("sea")
    if sea_chain and not isinstance(sea_chain[0][0], list):
        sea_chain = [sea_chain]
    if sea_chain:
        had_rev = terrain_data.get("sea_had_reversals")
        sea_rings = build_sea_polygons(
            sea_chain,
            lat_min, lon_min, lat_max, lon_max,
            had_reversals=(had_rev if had_rev is not None else True),
            structures=terrain_data.get("structures", []),
        )

    # ── Bathymetry: masked GMRT (real, wins) + unmasked GMRT base + GEBCO ─────
    # Two GMRT grids per area. The unmasked ('topo') grid is the seam-free offshore
    # base (multibeam blended into GEBCO), min'd with the EBP nearshore ramp. The
    # masked ('topo-mask') grid is real multibeam only and OVERRIDES that base
    # where present, so measured deep features (dredged holes, scour) survive the
    # ramp. GEBCO is the base fallback if the unmasked fetch failed. See
    # water.apply_water_depth_profile.
    _gmrt_base_path = fetch_gmrt_grid(area.area_id, lat_min, lon_min, lat_max, lon_max,
                                      area.raw_data_dir, layer="topo")
    _gmrt_mask_path = fetch_gmrt_grid(area.area_id, lat_min, lon_min, lat_max, lon_max,
                                      area.raw_data_dir, layer="topo-mask")
    _gebco_dir  = area.gebco_tile_dir or (area.raw_data_dir / "gebco" / "gebco_2026_geotiff")

    _bathy_tiles = RasterTiles(area.bathy_tile_dir) if getattr(area, "bathy_tile_dir", "") else None

    def _water_depths(cells):
        """(custom, gmrt_real, gmrt_base, gebco) positive-metre depth maps for water
        cells. custom = user-supplied bathymetry raster (wins outright);
        gmrt_real = masked GMRT; gmrt_base = unmasked GMRT (offshore base);
        gebco = GEBCO fallback (elevations <0 become positive depths)."""
        custom = {}
        if _bathy_tiles is not None and len(_bathy_tiles) > 0:
            _elev = _bathy_tiles.sample(cells, mode="bilinear")
            custom = {cid: (-e if e is not None and e < 0 else None)
                      for cid, e in _elev.items()}
        gmrt_real = _gmrt_sample_depths(cells, _gmrt_mask_path)
        gmrt_base = _gmrt_sample_depths(cells, _gmrt_base_path)
        _elev = _gebco_sample_elevations(cells, _gebco_dir)
        gebco = {cid: (-e if e is not None and e < 0 else None)
                 for cid, e in _elev.items()}
        return custom, gmrt_real, gmrt_base, gebco

    # ── H3 mesh generation ────────────────────────────────────────────────────
    features = []

    if terrain_data:
        features.append(_terrain_feature(area, terrain_data, cell_centres, h3_resolution))

    # ── Structures ────────────────────────────────────────────────────────────
    if terrain_data and (structures_feat := _structures_feature(area, terrain_data,
                                                                _id_to_cell, h3_resolution)):
        features.append(structures_feat)

    # ── Water bodies ──────────────────────────────────────────────────────────
    _progress(0.60, "Processing water bodies...")
    if terrain_data:
        water_override_path = area.output_dir / f"{area.area_id} - water - manual override.geojson"
        water_elevations = {}
        if water_override_path.exists():
            try:
                water_elevations = json.loads(water_override_path.read_text(encoding="utf-8"))["features"][0]["properties"].get("water_elevations", {})
            except Exception:
                pass

        _water_body_cell_ids = set()
        _cell_water_type     = {}
        _FLOWING_TAGS        = {"river", "stream", "canal", "drain", "ditch"}

        # The one general land rule: every water-body hole + every place=island is
        # land, subtracted from ALL water (bodies below + sea further down). An
        # island can't be flooded regardless of how the surrounding water is tagged.
        _land_geom = build_land_geometry(terrain_data.get("water", []),
                                         terrain_data.get("islands", []))
        _land_prep = None
        if _land_geom is not None and not _land_geom.is_empty:
            try:
                from shapely.prepared import prep as _prep
                from shapely.geometry import Point as _LPoint
                _land_prep = _prep(_land_geom)
            except Exception:
                _land_prep = None

        from .progress import bar as _bar
        from shapely.geometry import Polygon as _ShapelyPolyWB

        # Pass 1 -- candidate cells + overlap resolution. Each body polyfills its
        # cells, but NOTHING is carved yet: carving is deferred to pass 2 so every
        # body derives its surface from the raw COP30 snapshot below, never from an
        # already-carved neighbour. Cells covered by >1 body are assigned to a
        # single owner by _water_rank (smaller polygon breaks ties), so no cell is
        # ever carved twice -- the "dropped twice" sink (wetland inside the lagoon)
        # is structurally impossible now.
        _orig_ele = {cid: c[2] for cid, c in _id_to_cell.items()}   # raw COP30

        _pending = []          # per-body data carried into pass 2
        _owner   = {}          # h3_id -> index into _pending of the winning body
        _n_bodies = len(terrain_data.get("water", []))
        if _n_bodies:
            print(f"  [water] assigning cells to {_n_bodies} water bodies, then "
                  f"resolving each body's surface + bed depth (EBP ramp; "
                  f"GMRT/GEBCO soundings where present)")
        for body in _bar(terrain_data.get("water", []), label="water bodies"):
            fp = body.get("footprint", [])
            if len(fp) < 3:
                continue
            ring  = fp if fp[0] == fp[-1] else fp + [fp[0]]
            holes = body.get("holes", [])
            tag   = body.get("water_tag", "water")

            cand = build_water_cell_centres(
                _id_to_cell, ring, body.get("ele_m") or 0.0, h3_resolution,
                holes=holes, elevation_overrides=water_elevations,
            )
            # Globally enforce land: drop any cell on another body's island hole or
            # a place=island (overlapping/no-hole polygons, e.g. Venice's duplicate
            # lagoon outline that lacks the island holes).
            if _land_prep is not None and cand:
                cand = [wc for wc in cand
                        if not _land_prep.contains(_LPoint(wc[0], wc[1]))]

            idx  = len(_pending)
            rank = _water_rank(tag)
            area_deg2 = _ring_area(ring)
            _pending.append({
                "body": body, "ring": ring, "holes": holes, "tag": tag,
                "cand": cand,
                "flowing": tag in _FLOWING_TAGS, "rank": rank, "area": area_deg2,
            })
            key = (rank, area_deg2)
            for wc in cand:
                cur = _owner.get(wc[5])
                if cur is None or key < (_pending[cur]["rank"], _pending[cur]["area"]):
                    _owner[wc[5]] = idx

        # Pass 2 -- each body keeps only the cells it WON, derives its surface from
        # the raw COP30 snapshot, and carves each owned cell exactly once.
        # Custom bathymetry (bathy_tile_dir) is the "split" source's below-water
        # layer: where it has a valid sample for an owned body cell, that sample
        # IS the bed. Sampled once for all owned cells up front; the per-body
        # reset to the raw snapshot below still runs first so a body never
        # inherits a carved (or bathy) value from a neighbour.
        _body_bathy = _custom_bathy_beds(_bathy_tiles, _id_to_cell, _owner)
        if _body_bathy:
            print(f"  [water] custom bathymetry: {len(_body_bathy)} body cells "
                  f"from {area.bathy_tile_dir}", flush=True)

        # Decision D4: a body whose cells ALL lose to the clip/ownership
        # resolution sims nothing; its old "corpse" row (empty
        # cell_beds_json) sat in output.water_bodies looking mapped. Count
        # the drops per tag and report once after the loop.
        _dropped_zero_cell = {}

        for idx, pb in _bar(list(enumerate(_pending)), label="water depth"):
            body, tag       = pb["body"], pb["tag"]
            ring, holes     = pb["ring"], pb["holes"]
            water_cells     = [wc for wc in pb["cand"] if _owner.get(wc[5]) == idx]
            if not water_cells:
                _dropped_zero_cell[tag] = _dropped_zero_cell.get(tag, 0) + 1
                continue
            for wc in water_cells:                          # carve from raw COP30, not a carved neighbour
                wc[2] = _orig_ele.get(wc[5], wc[2])

            # Surface: flowing water follows the terrain gradient (per-cell,
            # never flattened). Standing water needs ONE level; without
            # bathymetry the estimator is a modeling choice — the mean of the
            # polygon's beds perches lakes high when the polygon includes dry
            # hillside cells (water towers near sharp elevation changes), so
            # the level seeks the basin's LOW edge: an OSM 'ele' tag wins when
            # present, else the 25th percentile of the body cells' raw beds.
            # Estimated from the RAW (topography) beds — never the custom
            # bathy beds: the bathy override lands after this block.
            _osm_ele = body.get("ele_m")
            if water_cells:
                surface_ele = sum(wc[2] for wc in water_cells) / len(water_cells)
                if pb["flowing"]:
                    for wc in water_cells:
                        wc[6] = wc[2]
                else:
                    surface_ele = _standing_surface(
                        [wc[2] for wc in water_cells], _osm_ele)
                    for wc in water_cells:
                        wc[6] = round(surface_ele, 4)
            else:
                surface_ele = _osm_ele or 0.0

            # Custom bathymetry (bathy_tile_dir) is the "split" source's
            # below-water layer: where it has a valid sample for an owned body
            # cell, that sample IS the bed — applied at BOTH levels the write
            # path reads: the body's feature cells (wc[2], bookkeeping) and the
            # BASE GRID (_id_to_cell, the DB's raw ele — the single bed
            # authority derive/export/sim consult; a feature-only value would
            # die at that boundary). Surface above was already fixed from the
            # topography. Without coverage the bed IS the raw terrain; bad beds
            # are fixed with the terrain editor, not fabricated here.
            # cell_beds_json carries the resulting bed; depth is derived at
            # write time (surface - bed).
            _depth_source = "custom" if any(
                wc[5] in _body_bathy for wc in water_cells) else "none"
            _max_depth_m  = None
            for wc in water_cells:
                _b = _body_bathy.get(wc[5])
                if _b is not None:
                    wc[2] = _b
                    _id_to_cell[wc[5]][2] = _b

            for wc in water_cells:
                _water_body_cell_ids.add(wc[5])
                _cell_water_type[wc[5]] = tag

            geom_rings = [ring] + [
                (h if h[0] == h[-1] else h + [h[0]]) for h in holes
            ]
            features.append({
                "type": "Feature",
                "properties": {
                    "Name":           f"{area.area_id} - {tag.title()}",
                    "area_id":        area.area_id,
                    "feature_type":   "water_body",
                    "water_tag":      tag,
                    "ele_m":          round(surface_ele, 4),
                    "depth_m":        _max_depth_m,
                    "depth_source":   _depth_source,
                    "osm_id":         body.get("osm_id"),
                    "osm_type":       body.get("osm_type"),
                    "touches_bbox":   body.get("touches_bbox", False),
                    "water_cells":    water_cells,
                    # membership + RAW terrain base elevation per cell -- NOT
                    # bathymetry (no auto-generated beds; depth is derived at
                    # write time as surface - bed).
                    "cell_beds_json": {str(wc[5]): wc[2]
                                        for wc in water_cells if len(wc) > 5},
                },
                "geometry": {
                    "type":        "Polygon",
                    "coordinates": geom_rings,
                },
            })

        if _dropped_zero_cell:
            _n_dropped = sum(_dropped_zero_cell.values())
            _per_tag   = ", ".join(f"{t}: {n}"
                                   for t, n in sorted(_dropped_zero_cell.items()))
            print(f"  [water] dropped {_n_dropped} body(ies) mapping to zero "
                  f"cells after clipping ({_per_tag})")

    # ── Sea ───────────────────────────────────────────────────────────────────
    # Union all sea polygons into one Shapely geometry, then check which terrain
    # cell centroids fall inside it. Islands are not part of any sea polygon so
    # they are automatically excluded -- no elevation heuristics needed.
    #
    # Sea level (D3): the ocean's surface for THIS area. Defaults to
    # hydro_sim.SEA_LEVEL_M (0 m); an off-datum coastal DEM (e.g. NYC's COP30
    # running 0-10 m over water) overrides it via area.sea_level_m. The value
    # is baked into the sea feature's ele_m (the fixed head preprocessing
    # turns into per sea cell) AND into the manifest's area block, which is
    # the persisted per-area record every manifest-driven caller reads back.
    _sea_level_m = float(getattr(area, "sea_level_m", 0.0) or 0.0)
    _progress(0.70, "Processing sea depth...")
    if sea_rings:
        try:
            from shapely.geometry import Polygon as _ShapelyPoly
            from shapely.ops import unary_union as _unary_union

            _sea_shapes    = []
            _primary_ring  = None
            _primary_holes = []
            _sea_skipped   = 0
            for sea_ring, sea_holes in sea_rings:
                if not sea_ring or len(sea_ring) < 3:
                    continue
                geo_ring   = sea_ring if sea_ring[0] == sea_ring[-1] else sea_ring + [sea_ring[0]]
                hole_rings = [[(p[0], p[1]) for p in h] for h in sea_holes] if sea_holes else []
                try:
                    poly = _ShapelyPoly([(p[0], p[1]) for p in geo_ring], hole_rings)
                    if not poly.is_valid:
                        poly = poly.buffer(0)
                    if not poly.is_empty:
                        _sea_shapes.append(poly)
                        if _primary_ring is None or poly.area > _ShapelyPoly(
                                [(p[0], p[1]) for p in _primary_ring]).area:
                            _primary_ring  = geo_ring
                            _primary_holes = sea_holes
                except Exception:
                    _sea_skipped += 1
            if _sea_skipped:
                print(f"  [sea] skipped {_sea_skipped} invalid ring(s)")

            sea_union  = _unary_union(_sea_shapes) if _sea_shapes else None

            # Subtract the one general land geometry (all water-body holes +
            # place=island) from the sea, so every island stays land -- a lagoon's
            # island holes (Venice) and a harbour's island holes (Sydney) are
            # handled by the same rule, no special cases.
            if _land_geom is not None and sea_union is not None and not sea_union.is_empty:
                try:
                    sea_union = sea_union.difference(_land_geom)
                except Exception:
                    pass

            sea_name   = f"{area.area_id} - Sea"
            sea_cells_all = []

            if sea_union and not sea_union.is_empty:
                # H3 polyfill is centroid-based -- exactly our rule (a cell is sea
                # iff its centre is in the sea polygon) -- and runs in C with an
                # internal hierarchical fill, so the deep-ocean/deep-land interiors
                # are never tested cell-by-cell. Only enumerate the cells the
                # polygon actually covers instead of probing all the grid's centres.
                import h3 as _h3f
                print("  [sea] polyfilling sea polygon at h3 ...", flush=True)
                _geoms = (sea_union.geoms if sea_union.geom_type == "MultiPolygon"
                          else [sea_union])
                _fill_ids = set()
                for _poly in _geoms:
                    if _poly.is_empty:
                        continue
                    _ext   = [(lat, lon) for lon, lat in _poly.exterior.coords]
                    _holes = [[(lat, lon) for lon, lat in _r.coords]
                              for _r in _poly.interiors]
                    try:
                        _fill_ids |= set(_h3f.h3shape_to_cells(
                            _h3f.LatLngPoly(_ext, *_holes), h3_resolution))
                    except Exception:
                        pass
                for cell_id in _fill_ids - _water_body_cell_ids:
                    c = _id_to_cell.get(cell_id)
                    if c is None:
                        continue
                    sea_cells_all.append([c[0], c[1], c[2], c[3], c[4], cell_id, 0.0])
                    _cell_water_type[cell_id] = "sea"

            # Water bodies (lagoons, harbours, lakes) stay water bodies with their
            # own depth -- no merge-into-sea. Land/water is decided purely by the
            # geometry above (sea region minus land; bodies minus land).
            sea_cell_ids       = {wc[5] for wc in sea_cells_all}
            _open_sea_cell_ids = sea_cell_ids - _water_body_cell_ids
            print(f"  [sea] {len(sea_cells_all)} sea cells classified", flush=True)

            # Custom bathymetry at the bed authority: where the bathy tiles
            # cover a sea cell, their value IS the seabed — carve the base
            # grid (the DB's raw ele, the single bed the write-time model
            # reads) and mirror it into the feature rows built above. The
            # resolved depth profile below stays the no-coverage fallback.
            _sea_bathy_ids = set()
            if _bathy_tiles is not None and len(_bathy_tiles) > 0 and sea_cells_all:
                _sc = [_id_to_cell[cid] for cid in sea_cell_ids
                       if cid in _id_to_cell]
                for cid, v in _bathy_tiles.sample(_sc, mode="bilinear").items():
                    if v is not None:
                        _id_to_cell[cid][2] = v
                        _sea_bathy_ids.add(cid)
                if _sea_bathy_ids:
                    for wc in sea_cells_all:
                        if wc[5] in _sea_bathy_ids:
                            wc[2] = _id_to_cell[wc[5]][2]
                    print(f"  [sea] custom bathymetry: {len(_sea_bathy_ids)} sea cells "
                          f"re-bedded from {area.bathy_tile_dir}", flush=True)

            # Per-cell depth (real GMRT wins; else min(EBP ramp, unmasked GMRT /
            # GEBCO)) for open sea cells only; merged water-body cells (harbours,
            # docks) keep their own depth. Distance-to-shore is from the real
            # coastline (bbox edges excluded) inside apply_water_depth_profile.
            # Cells already re-bedded from custom bathymetry are skipped — the
            # tile wins outright over the profile.
            _sea_cells = [wc for wc in sea_cells_all if wc[5] in _open_sea_cell_ids]
            _sea_ids   = [wc[5] for wc in _sea_cells]
            _custom_map, _gmrt_map, _base_map, _gebco_map = _water_depths(_sea_cells)
            _custom_n = sum(1 for v in _custom_map.values() if v is not None)
            # The profile's surface IS this area's sea level: seabeds derive as
            # sea_level_m - depth, so a cell's depth stays exactly what the
            # bathymetry sources measured (GEBCO/GMRT/EBP are depth-below-sea-
            # surface; the offset moves the surface, not the depth).
            _depth_map = apply_water_depth_profile(
                _sea_ids, sea_union, _sea_level_m,
                ebp_a=_config.EBP_PROFILE_A["sea"],
                depth_cap_m=_config.SEA_MAX_DEPTH_M,
                custom_depth=_custom_map, gmrt_depth=_gmrt_map,
                base_depth=_base_map, gebco_depth=_gebco_map,
                bbox=(lon_min, lat_min, lon_max, lat_max))
            for wc in sea_cells_all:
                if wc[5] in _depth_map and wc[5] not in _sea_bathy_ids:
                    wc[2] = _depth_map[wc[5]]
            # NOTE: _id_to_cell is deliberately NOT carved by the resolved
            # depth profile (raw terrain stays in the DB; that profile is
            # write-path bookkeeping now that layer 1 is the bed authority).
            # Custom bathymetry IS carved into _id_to_cell above — it is the
            # user's below-water ground truth. See the water-bodies note.
            # Max depth is surface-relative: depth = _sea_level_m - bed (the
            # profile's beds derive from the offset surface, so plain -bed
            # would under-read by the offset; identical at the 0 m default).
            _sea_max_depth = round(max((_sea_level_m - e for e in _depth_map.values()),
                                       default=0.0), 3)
            _sea_src = ("custom" if any(v is not None for v in _custom_map.values())
                        else "gmrt" if any(v is not None for v in _gmrt_map.values())
                        else "gmrt-base" if any(v is not None for v in _base_map.values())
                        else "gebco+ebp" if any(v is not None for v in _gebco_map.values())
                        else "ebp")
            if _custom_n:
                print(f"  [sea] custom bathymetry: {_custom_n}/{len(_sea_cells)} wet cells from {area.bathy_tile_dir}",
                      flush=True)
            print(f"  [sea] depth resolved via {_sea_src}; max {_sea_max_depth:.1f} m", flush=True)

            features.append({
                "type": "Feature",
                "properties": {
                    "Name":           sea_name,
                    "area_id":        area.area_id,
                    "feature_type":   "sea",
                    "sea_index":      0,
                    "water_tag":      "sea",
                    # The sea surface IS this area's sea level (_sea_level_m,
                    # 0 m unless the area overrides it): preprocess_water_params
                    # turns this into a fixed head per sea cell so derive exports
                    # the sea wet. Without it the surface falls back to each
                    # cell's own bed, every head is <= its bed, and the sea
                    # exports DRY (Austevoll: 0 wet cells).
                    "ele_m":          _sea_level_m,
                    "depth_m":        _sea_max_depth,
                    "water_cells":    sea_cells_all,
                    "cell_beds_json": {str(wc[5]): wc[2]
                                        for wc in sea_cells_all if len(wc) > 5},
                },
                "geometry": {
                    "type":        "Polygon",
                    "coordinates": ([_primary_ring] + _primary_holes) if _primary_ring else [],
                },
            })
            print(f"  Sea: {len(sea_cells_all)} H3 cells "
                  f"(union of {len(_sea_shapes)} polygon(s))")

        except ImportError:
            print("  [sea] shapely not available -- skipping sea cell generation")

    # ── Vegetation ────────────────────────────────────────────────────────────
    # Vegetation is now a PER-CELL attribute (veg_type), but the assignment is
    # DERIVED at write time from the vegetation table (so edits are observable).
    # Here we only stage the polygons as a feature so they reach the DB table.
    _veg_polys = terrain_data.get("vegetation", []) if terrain_data else []
    if _veg_polys:
        features.append({
            "type": "Feature", "geometry": None,
            "properties": {
                "Name":         f"{area.area_id} - Vegetation",
                "area_id":      area.area_id,
                "feature_type": "vegetation",
                "vegetation":   [{"veg_tag": v.get("veg_tag", "vegetation"),
                                  "footprint": v.get("footprint", [])}
                                 for v in _veg_polys
                                 if len(v.get("footprint", [])) >= 3],
            },
        })

    # ── Streets ───────────────────────────────────────────────────────────────
    _progress(0.87, "Processing streets...")
    if terrain_data:
        features.extend(_street_features(area, terrain_data, _id_to_cell, h3_resolution))

    # ── Landmarks ────────────────────────────────────────────────────────────
    if terrain_data and (landmarks_feat := _landmarks_feature(area, terrain_data,
                                                              _id_to_cell, h3_resolution)):
        features.append(landmarks_feat)

    # ── Major features (Natural Earth) ────────────────────────────────────────
    # In major mode the OSM structures/streets/landmarks above are empty (stripped); the
    # country-scale content comes from Natural Earth instead: city/town footprints +
    # labels, motorway/trunk/primary roads, and notable landmark points. Clipped to the
    # bbox (not the exact hex outline -- fine at this scale).
    if major:
        from . import naturalearth
        _progress(0.90, "Fetching major features (Natural Earth)...")
        ne_dir = _config.data_store_dir("naturalearth",
                                        area.raw_data_dir / "naturalearth")
        ne_feats, ne_stats = naturalearth.major_features(
            ne_dir, (lat_min, lon_min, lat_max, lon_max), area.area_id)
        features.extend(ne_feats)
        print(f"  [naturalearth] {sum(ne_stats.values())} major features -- "
              + ", ".join(f"{k} {v}" for k, v in ne_stats.items()))

    # ── Extra features (e.g. custom line features) ───────────────────────────
    if extra_features:
        features.extend(extra_features)

    # ── Append water_type attribute and derive true elevation floor ───────────
    if terrain_data:
        for c in cell_centres:
            if len(c) == 6:
                c.append(_cell_water_type.get(c[5], ""))

        # The grid keeps RAW terrain elevation now (the bed carve moved to the
        # write-time derive). The true floor must still reach below the deepest
        # bed, so fold the water features' per-cell bed depths into the minimum.
        _raw_min = min((c[2] for c in cell_centres), default=0.0)
        _bed_min = min(
            (wc[2] for f in features
             if (f.get("properties") or {}).get("feature_type") in ("water_body", "sea")
             for wc in (f["properties"].get("water_cells") or []) if len(wc) > 2),
            default=_raw_min)
        ele_floor_m = min(_raw_min, _bed_min)
        ele_max_m   = max((c[2] for c in cell_centres), default=0.0)
        pad_factor  = float(getattr(area, "pad_factor", 0.1))
        if getattr(area, "floor_elevation", None) is not None:
            if area.floor_elevation > ele_floor_m:
                raise ValueError(
                    f"Floor elevation {area.floor_elevation:.1f} m is above terrain "
                    f"minimum {ele_floor_m:.1f} m"
                )
            _floor = area.floor_elevation
        else:
            _floor = min(0.0, ele_floor_m)
        _baked_min = _floor_baked_min(ele_floor_m, ele_max_m, pad_factor, _floor)
        for feat in features:
            if feat.get("properties", {}).get("feature_type") == "terrain":
                feat["properties"]["ele_floor_m"] = round(ele_floor_m, 3)
                feat["properties"]["ele_min_baked_m"] = _baked_min
                break
        terrain_data["ele_min_baked_m"] = _baked_min
        print(f"  [cells] ele_floor_m = {ele_floor_m:.3f} m  ele_max_m = {ele_max_m:.3f} m  "
              f"pad = {pad_factor} x range -> {_baked_min - _floor:.3f} m  "
              f"ele_min_baked_m = {_baked_min:.3f} m")

    # ── Cell RGB: water always takes the regolith palette (imagery shows the
    # surface, not the bed). Land takes either sampled satellite colour or the
    # cartographic OSM-landcover palette, per config.COLOR_SOURCE. The OSM tags
    # (vegetation > natural > landuse) are the satellite fetch-failure fallback.
    _source = getattr(_config, "COLOR_SOURCE", "osm")
    _progress(0.90, "Colouring cells from satellite imagery..."
              if _source == "satellite"
              else "Re-sampling cell colours from the existing texture (no download)..."
              if _source == "texture"
              else "Colouring cells plain white (imagery off)..."
              if _source == "white"
              else "Colouring cells from OSM tags...")
    from . import unified
    _cell_ids     = {c[5] for c in cell_centres if len(c) > 5}
    veg_map       = unified.veg_cell_map(terrain_data.get("vegetation", []),
                                         _cell_ids, h3_resolution)
    landcover_map = unified.landcover_cell_map(terrain_data.get("landuse", []),
                                               terrain_data.get("natural_areas", []),
                                               _cell_ids, h3_resolution)
    _regolith  = _config.WATER_REGOLITH_COLOR
    _land       = _config.LANDCOVER_COLOR
    _factor   = float(getattr(_config, "FACE_COLOR_FACTOR", 0.5))
    _upscale  = bool(getattr(_config, "UPSCALE_TERRAIN_COLOR", False))
    sat_map = {}
    face_points = None
    if _source == "satellite":
        from . import satellite
        _passes = max(1, int(getattr(_config, "SATELLITE_PASSES", 1)))
        try:
            sat_map = satellite.fetch_cell_rgb(
                area.area_id, lat_min, lon_min, lat_max, lon_max,
                cell_centres, area.raw_data_dir, _passes, _factor, _upscale,
                cancel_event=cancel_event,
                progress_cb=lambda f, m: _progress(0.90 + 0.04 * f, m))
        except Exception as e:
            print(f"  [colour] satellite sampling failed ({e}) -- "
                  f"falling back to OSM landcover palette")
    elif _source == "texture":
        # Imagery stage unticked but the area already has an Image Texture:
        # re-sample per-cell colours from it (no download). Cells the texture
        # misses stay out of sat_map and take the white default below.
        from . import satellite as _sat_tex
        from PIL import Image as _PILImage
        import numpy as _np
        _tex = getattr(_config, "TEXTURE_SAMPLE", None) or {}
        _ext = _tex.get("extent") or {}
        _tex_path = Path(area.output_dir) / (_tex.get("file") or "")
        if _tex_path.is_file() and _ext:
            try:
                _arr = _np.asarray(_PILImage.open(_tex_path).convert("RGB"))
                face_points = _sat_tex.cell_face_points(cell_centres, _factor)
                sat_map = _sat_tex.sample_mosaic_points(
                    _arr, _ext["zoom"], _ext["tx0"], _ext["ty0"], face_points)
                print(f"  [colour] {sum(len(v) for v in sat_map.values())} face "
                      f"colours re-sampled from {_tex_path.name} (no download)")
            except Exception as e:
                print(f"  [colour] texture re-sampling failed ({e}) -- "
                      f"land cells fall back to white")
        else:
            print("  [colour] no usable on-disk texture -- "
                  "land cells plain white")
    # 7 colours per cell (centre + 6 ring). For EACH face we store BOTH answers:
    # the imagery colour (satellite sample, or the OSM landcover/veg palette when
    # satellite is unavailable) AND the regolith colour (the cell's predicted
    # water appearance from the source-time water assignment). The write-time join
    # (tyrbuild._derive_cells_from_output) picks one per cell by wet/dry, so water
    # table edits re-colour without re-fetching imagery. face_points give the sample
    # geometry for the colour geojson regardless of source.
    from . import satellite as _sat
    if face_points is None:
        face_points = _sat.cell_face_points(cell_centres, _factor)
    face_colors = {}
    _n_regolith = _n_sat = _n_white = 0
    for c in cell_centres:
        hid = c[5]
        nfaces = len(face_points.get(hid) or [None])
        water_type = c[C_WATER] if len(c) > C_WATER else ""
        # imagery answer (dry appearance)
        if hid in sat_map:
            s = sat_map[hid]                      # 7 samples (upscale) or 1 (repeat)
            img_faces = s if len(s) == nfaces else [s[0]] * nfaces
            _n_sat += 1
        elif _source in ("white", "texture"):
            img_faces = [[255, 255, 255]] * nfaces
            _n_white += 1
        else:
            tag = veg_map.get(hid) or landcover_map.get(hid)
            img_faces = [list(_land.get(tag, _land["default"]))] * nfaces
        # regolith answer (wet appearance) -- predicted from the source assignment
        if water_type:
            sub_faces = [list(_regolith.get(water_type, _regolith["default"]))] * nfaces
            _n_regolith += 1
        else:
            sub_faces = img_faces                 # dry cell: regolith == imagery
        face_colors[hid] = list(zip(img_faces, sub_faces))
    _n_palette = len(cell_centres) - _n_sat - _n_white
    _sampled_label = ("the existing texture" if _source == "texture"
                      else "satellite")
    _white_note = (f", {_n_white} land cells plain white (imagery off)"
                   if _n_white else "")
    print(f"  [colour] {_n_regolith} cells with a regolith (water) colour, "
          f"{_n_sat} land cells from {_sampled_label}, "
          f"{_n_palette} land cells from OSM landcover tags"
          f"{_white_note}")

    # Structure wall/roof appearance: resolve the OSM tags (direct, centroid-
    # grafted, or block/radius-inherited in openbuildingmap) into the form the
    # Blender importer reads. Colours -> wall_colour / roof_colour as #rrggbb
    # hex strings (default when no tag). Materials, roof shape and building type
    # stay as normalized strings; the Blender importer maps them to config enum
    # indices at load time.
    from . import structure_inherit as _si
    from . import roof_generators as _rg
    _wall_def = tuple(_config.DEFAULT_WALL_COLOR)
    _roof_def = tuple(_config.DEFAULT_ROOF_COLOR)
    for _feat in features:
        if (_feat.get("properties") or {}).get("feature_type") != "structures":
            continue
        for s in _feat["properties"].get("structures", []):
            _wall = _si.parse_colour(s.get("wall_colour")) or _wall_def
            _roof = _si.parse_colour(s.get("roof_colour")) or _roof_def
            s["wall_colour"] = "#{:02x}{:02x}{:02x}".format(*_wall)
            s["roof_colour"] = "#{:02x}{:02x}{:02x}".format(*_roof)

            # Roof geometry classification: pipeline decides, Blender follows.
            s["roof_type"] = _rg.classify_roof_type(s)
            if s["roof_type"] == "generated":
                s["roof_height_m"] = _rg.infer_roof_height_m(s)
                s["roof_direction_offset"] = _rg.infer_roof_direction_offset(s)
            else:
                s["roof_height_m"] = 0.0
                s["roof_direction_offset"] = None

    # ── Write output via the region DB: staging -> output -> export ───────────
    # The DB cells table is the BARE grid (raw terrain ele, no water/veg/colour --
    # those derive from the other tables). The export (roundtrip_dataset ->
    # export_dataset_from_output) re-derives them FROM the current output tables,
    # so a re-export after editing output.water_bodies / vegetation is reflected in
    # the written cells.csv. The derived (carved) rows come back for the counts.
    _progress(0.94, "Writing output (region DB round-trip)...")
    print(f"  Building bare cell grid ({len(cell_centres)} cells) ...", flush=True)
    cell_rows  = unified.build_cell_rows(cell_centres)   # bare grid
    from . import tyrbuild as _tb
    written, cell_rows, features = _tb.roundtrip_dataset(
        area, features, cell_rows,
        face_points=face_points, face_colors=face_colors,
        # The DB write is the longest silent phase on large areas: map its
        # 0..1 report into the 0.94..0.98 slice so the bar + ETA move.
        progress_cb=lambda f, m: _progress(0.94 + 0.04 * f, m))
    _n_wet = sum(1 for r in cell_rows if (r.get("water_depth_m") or 0) > 0)
    _n_veg = sum(1 for r in cell_rows if r.get("veg_type"))
    print(f"  cells: {len(cell_rows)} ({_n_wet} wet, {_n_veg} vegetated)")

    _area_meta = {
        "lon_centre":      area.lon_centre,
        "lat_centre":      area.lat_centre,
        "lon_min":         lon_min, "lat_min": lat_min,
        "lon_max":         lon_max, "lat_max": lat_max,
        "ele_min_m":       terrain_data.get("ele_min_m"),
        "ele_max_m":       terrain_data.get("ele_max_m"),
        "ele_min_baked_m": terrain_data.get("ele_min_baked_m"),
        # Persisted per-area sea level (D3): hydro_sim.sea_level_from_manifest
        # reads this back on every manifest-driven water run.
        "sea_level_m":     _sea_level_m,
    }
    _manifest_path = area.region_dir / f"{area.area_id}_manifest.json"
    _feature_counts = {n: c for n, (_, c) in written.items()}
    _progress(0.98, "Writing manifest and source metadata...")
    # Write the manifest NOW, before the optional (and potentially very long)
    # surface-texture download. If the user stops the pipeline during the texture
    # step, the Data Manager can still adopt the completed cells + feature files.
    unified.write_manifest(_manifest_path, area.area_id, h3_resolution, len(cell_rows),
                           _feature_counts, area_meta=_area_meta)
    # Source metadata for the Data Manager is generated from the manifest, so do it
    # here while the state is complete (texture has its own optional block below).
    try:
        _fm_path, _cm_path = _tb.generate_source_metadata(
            area.region_dir, area.area_id, log=print,
            mode=getattr(area, "metadata_mode", "all"),
            feature_dir=area.output_dir)
        unified.set_source_pipelines(_manifest_path, _fm_path, _cm_path)
    except Exception as e:
        print(f"[source metadata] generation failed: {e}")
    # Materialise source.* from those TSVs so the schema ALWAYS exists after a
    # build (T2/D2): the Viewer (read_source_cells), the Data Manager pipeline
    # and interpolation all read source.<dataset>, and nothing else produces it.
    # Transactional drop+create -- a failure leaves any previous source intact.
    try:
        _tb.materialize_source(area.region_dir,
                               str(area.region_dir / f"{area.area_id}.duckdb"),
                               log=print)
    except Exception as e:
        print(f"[source] materialisation failed: {e}")

    # Catalog snapshot: record in the area DB, read-only, what existed at
    # source-creation time (the tyr TSVs + the asset ledger below). Staging /
    # output rebuilds never drop the catalog schema. Failure never fails the build.
    try:
        _cat_fm, _cat_cm = _tb.source_metadata_paths(area.region_dir, area.area_id)
        _cat_con = _tb.connect(area.region_dir / f"{area.area_id}.duckdb")
        try:
            _tb.ensure_catalog_schema(_cat_con)
            _tb.snapshot_catalog_metadata(_cat_con, _cat_fm, _cat_cm)
        finally:
            _cat_con.close()
    except Exception as e:
        print(f"[catalog] metadata snapshot skipped ({e})")

    # Image Texture (optional): stitch the raw Sentinel-2/OSM tiles over
    # the area into one PNG and record its extent, so the add-on can UV-map it onto
    # the Terrain Surface. Reuses the satellite tile math + time-window passes;
    # independent of the per-face colour source above (both can be on at once).
    image_texture = None
    if bool(getattr(_config, "IMAGE_TEXTURE", False)):
        from . import satellite as _sat
        _tsrc  = getattr(_config, "IMAGE_TEXTURE_SOURCE", "s2")
        _tzoom = _sat.texture_zoom(h3_resolution, getattr(_config, "IMAGE_TEXTURE_ZOOM", 0))
        _tpass = max(1, int(getattr(_config, "SATELLITE_PASSES", 1)))
        _texpath = area.output_dir / f"{area.area_id}_image_texture.png"
        _progress(0.98, "Building image texture...")
        try:
            _ext = _sat.build_texture(area.area_id, lat_min, lon_min, lat_max, lon_max,
                                      area.raw_data_dir, _texpath, _tsrc, _tzoom, _tpass,
                                      cancel_event=cancel_event,
                                      progress_cb=lambda f, m: _progress(0.98 + 0.01 * f, m))
            image_texture = {"file": _texpath.name, "extent": _ext,
                             "source": _tsrc, "passes": _tpass}
            print(f"  Saved -> {_texpath.name}  ({_ext['width']}x{_ext['height']}px, "
                  f"{_tsrc} z{_ext['zoom']})")
        except Exception as e:
            print(f"  [texture] image texture skipped ({e})")

    if image_texture:
        _m = unified.read_manifest(_manifest_path)
        _m["image_texture"] = image_texture
        unified._atomic_json_dump(_manifest_path, _m, indent=2)

    # Bitmap colour editor source mosaic — ALWAYS, whatever the colour
    # source (the asset renders from IMAGE_TEXTURE_SOURCE, so an OSM-palette
    # run gets an OSM-rendered basis, not a wasted satellite mosaic). Written
    # to the region's output/ dir (legacy: next to the manifest) via the same
    # tile math as the Image Texture, as
    # color_source.png + sidecar and recorded in catalog.assets
    # (origin='pipeline'). Raw tiles are already cached in data/raw from the
    # per-cell colour sampling above, so the re-stitch is cheap. Failure never    # fails the build, but it is NEVER silent: it lands in the returned
    # warnings (surfaced by the run status) as well as stderr — a missing
    # source asset leaves the Colour Editor empty, which reads exactly like
    # "my data is gone" to the user.
    _warnings = []
    _progress(0.99, "Building colour source mosaic...")
    try:
        _ca = _tb.build_color_source_asset(
            _manifest_path, origin="pipeline",
            progress_cb=lambda f, m: _progress(0.99 + 0.005 * f, m))
        print(f"  Saved -> color_source.png  ({_ca['width']}x{_ca['height']}px)")
    except Exception as e:
        _warnings.append(f"color asset rebuild failed: {e}")
        print(f"  [color asset] FAILED ({e})", file=sys.stderr)

    _progress(1.0, "Area build complete")

    return {
        "terrain_data": terrain_data,
        "features":     features,
        "area":         area,
        "warnings":     _warnings,
    }
