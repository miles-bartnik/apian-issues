Source code for apian_terrain.series

"""series.py -- the `cell_series` data layer: a per-cell value that changes over a
large, discrete set of timesteps, baked for FRAME-DRIVEN STEP playback in Blender.

Where a `cell_field` is one scalar per cell, a `cell_series` is one scalar per cell
PER timestep. Storage is **sparse** (CSR) so the data load is proportional to the
records that actually exist, not to (cells x steps): a cell with no record at a step
simply isn't stored. Gaps are permitted and mean NULL -- the cell displays nothing at
that frame (NOT 0). There is no carry-forward: a missing step and an explicit-null
record are identical (_present = False).

On disk, per field. The bulky CSR blobs live in a per-region `<area>_handlers/` folder
(the header stores their names folder-prefixed); only the `.json` header sits at the
region root, where the manifest points to it. Blob names carry a per-write `<stamp>`:
a reader memmapping the previous blobs (the Blender series cache) makes an in-place
truncate-write fail on Windows, so every write gets fresh names and the header flips
the reference (superseded blobs are cleaned best-effort):
  <area>_handlers/<area>_<field>_series.<stamp>.rowptr
                                   int64[n_steps+1] -- CSR row pointer; the records for
                                   step s are the slice [rowptr[s] : rowptr[s+1]]. This is
                                   the only index playback needs: O(1) random access to any
                                   step, so scrubbing is a seek.
  <area>_<field>_series.cols       int32[nnz] -- the cell (column into the .cells order) of
                                   each record, grouped by step then cell.
  <area>_<field>_series.vals.<dt>  the value of each record (dtype default float32).
  <area>_<field>_series.interp     uint8[nnz] -- OPTIONAL; 1 marks a record synthesised by
                                   resample_series (time-axis interpolation) rather than
                                   measured. Absent when every record is measured.
  <area>_<field>_series.cells      the cell order, one h3_id per line (col index -> h3_id).
  <area>_<field>_series.json       header: dtype, n_steps, n_cells, nnz, min/max, gap policy,
                                   the file names, per-step time_labels, discontinuity stats.

All arrays are mmap-friendly: the OS pages in only the current step's slice, so a huge
series plays back and scrubs in a few MB resident. dtype defaults to float32 (float16
overflows counts to inf + only holds integers exactly to 2048; it is an opt-in halving,
REFUSED when the range won't fit rather than silently corrupting -- see _fits_float16).

Desktop-only (needs numpy); NEVER shipped in the add-on zip. The Blender side opens the
mmaps itself (numpy is bundled there) -- it does not import this module. Pure apart from
numpy, so it is unit-tested without bpy/duckdb.
"""
from pathlib import Path
from collections import defaultdict
from datetime import datetime, timedelta
from time import time_ns
import json
import math

import numpy as np

_EPOCH = datetime(1970, 1, 1)           # naive-datetime ordinal origin (deterministic, tz-free)

# The one column a time-series (cell_series) dataset is keyed on: a TIMESTAMP named
# exactly this. Recorded in every series header so the validator can enforce it. Keep in
# sync with the copy in layers.py (that module can't import this desktop-only one).
EVENT_TS_COLUMN = "event_ts"

# Guardrail: refuse a series whose on-disk footprint exceeds this. With sparse storage a
# huge (cells x steps) grid no longer costs anything when empty, so this now bounds the
# ACTUAL records; its job is to catch a runaway export before it fills a disk.
MAX_SERIES_BYTES = 32 * 1024**3

DTYPE = "float32"                       # default value dtype; float16 is an opt-in
GAP_POLICY = "null"                     # a missing/absent step = NULL (display nothing)


def _fits_float16(vmin, vmax):
    """True if a value range is safe to store as float16: within the finite range AND
    small enough that integer-ish values keep exact/near-exact precision (float16 loses
    integer exactness above 2048). Below that bar we refuse float16 rather than silently
    round counts."""
    return max(abs(vmin), abs(vmax)) <= 2048.0

def build_series(columns, rows, *, value_col, time_col, key="h3_id"):
    """Pivot a tidy query result -- one row per (cell, timestep) -- into the pieces
    write_series needs. Returns (cell_ids, time_labels, triples) where:
      * cell_ids    -- sorted distinct h3 ids that carry a REAL value at some step;
                       index = the CSR column for that cell.
      * time_labels -- sorted distinct timestep values as strings; index = the step
                       (= Blender frame - 1).
      * triples     -- [(step_idx, cell_idx, float_value), ...] for the real values only.

    NULL semantics: a row whose value is null (None or NaN) is NOT a record, but its
    timestep still registers -- so a (cell, t, NULL) makes `t` a real frame at which that
    cell is simply absent (displays nothing), distinct from a real 0. A cell that never
    carries a value (all null, or absent from the query) has no column and shows nothing at
    every step. Rows with a null key or time are dropped. Raises ValueError on a missing
    column."""
    for need in (key, value_col, time_col):
        if need not in columns:
            raise ValueError(f"cell_series needs a '{need}' column (got {columns})")
    ki, ti, vi = columns.index(key), columns.index(time_col), columns.index(value_col)

    times, cells, clean = set(), set(), []
    for row in rows:
        hid, t, v = row[ki], row[ti], row[vi]
        if hid is None or t is None:
            continue
        times.add(t)                        # a timestep registers even if THIS value is null
        if v is None:
            continue
        try:
            fv = float(v)
        except (TypeError, ValueError):
            continue
        if math.isnan(fv):                  # a NaN value is treated as NULL (no record)
            continue
        cells.add(hid)
        clean.append((t, hid, fv))

    time_order = sorted(times)
    cell_order = sorted(cells)
    step_idx = {t: i for i, t in enumerate(time_order)}
    cell_idx = {c: i for i, c in enumerate(cell_order)}
    triples = [(step_idx[t], cell_idx[hid], fv) for t, hid, fv in clean]
    return cell_order, [str(t) for t in time_order], triples




def _discontinuity_stats(records_by_cell):
    """Count cells with INTERIOR gaps and the total gap-steps. An interior gap is a step
    between a cell's first and last record where it has no record -- the value winks out
    then reappears (a flicker), which is worth warning about. A cell that simply starts
    late or ends early is not a discontinuity."""
    disc_cells = disc_steps = 0
    for steps in records_by_cell.values():
        lo, hi, present = min(steps), max(steps), len(steps)
        gaps = (hi - lo + 1) - present
        if gaps > 0:
            disc_cells += 1
            disc_steps += gaps
    return disc_cells, disc_steps

def write_series(output_dir, area_id, field, cell_ids, time_labels, triples, *,
                 dtype=DTYPE, interp_flags=None, time_col=EVENT_TS_COLUMN):
    """Write the sparse CSR series files for one field and return the manifest layer dict
    {name, kind, file, spec}. Duplicate (step, cell) records collapse to the last value.
    Refuses to write if the on-disk footprint exceeds MAX_SERIES_BYTES, or if `dtype` is
    float16 but the data would overflow / lose precision (raises ValueError -- never a
    silent cast). min/max are over the real values; discontinuity stats are recorded in
    the header so the validator can WARN.

    `interp_flags` (optional) is a {(step, cell): True} set/dict marking records whose
    value was SYNTHESISED by resampling/interpolation rather than measured. When given, a
    parallel uint8 `.interp` handler is written and the header records interp_file +
    interp_count, so a reload (or Blender) can tell filled records from real ones -- the
    honest analogue of the celldata `__i` flag. Absent -> no handler, format unchanged."""
    out = Path(output_dir)
    n_steps, n_cells = len(time_labels), len(cell_ids)
    dt = np.dtype(dtype)

    cell_map = {(s, c): v for s, c, v in triples}     # dedupe (step, cell): last wins
    items = sorted(cell_map.items())                  # by (step, cell) -> CSR order
    nnz = len(items)

    nbytes = nnz * (dt.itemsize + 4) + (n_steps + 1) * 8
    if nbytes > MAX_SERIES_BYTES:
        raise ValueError(
            f"cell_series '{field}' would be {nbytes / 1024**3:.1f} GiB "
            f"({nnz} records) -- over the {MAX_SERIES_BYTES / 1024**3:.0f} GiB cap. "
            f"Filter the query or raise series.MAX_SERIES_BYTES.")

    vals_list = [v for _, v in items]
    vmin = min(vals_list) if vals_list else 0.0
    vmax = max(vals_list) if vals_list else 0.0
    if dt == np.float16 and not _fits_float16(vmin, vmax):
        raise ValueError(
            f"cell_series '{field}' range [{vmin:g}, {vmax:g}] does not fit float16 "
            f"safely (max ~2048 for exact integers, 65504 finite) -- use dtype='float32'.")

    steps = np.fromiter((s for (s, _), _ in items), dtype=np.int64, count=nnz)
    cols  = np.fromiter((c for (_, c), _ in items), dtype=np.int32, count=nnz)
    vals  = np.array(vals_list, dtype=dt)
    rowptr = np.zeros(n_steps + 1, dtype=np.int64)
    if nnz:                                            # counts per step -> cumulative offsets
        rowptr[1:] = np.cumsum(np.bincount(steps, minlength=n_steps))

    # The bulky CSR blobs live in a per-region <area>_handlers/ folder for tidiness; the
    # .json header stays at the region root (it is the layer descriptor the manifest points
    # to). Handler names are stored folder-prefixed (posix slash, portable), and EVERY
    # reader resolves them as `<header-dir> / header[name]`, so the move needs no reader
    # change. `out / "a/b"` writes into the subfolder on any OS.
    # Blob names are STAMPED per write, never rewritten in place: a reader holding a
    # memmap on the previous blobs (Blender's open-series cache) makes a truncate-write
    # fail on Windows (OSError 22) -- fresh names sidestep the lock entirely. The header
    # (never memory-mapped) flips the reference; superseded blobs are removed below,
    # best-effort (a still-mapped one survives until a later write).
    handlers = f"{area_id}_handlers"
    (out / handlers).mkdir(parents=True, exist_ok=True)
    json_file = f"{area_id}_{field}_series.json"
    stamp = str(time_ns() // 1_000_000)
    sfx = _dt_suffix(dt)
    files = {
        "rowptr_file": f"{handlers}/{area_id}_{field}_series.{stamp}.rowptr",
        "cols_file":   f"{handlers}/{area_id}_{field}_series.{stamp}.cols",
        "vals_file":   f"{handlers}/{area_id}_{field}_series.{stamp}.vals.{sfx}",
        "cells_file":  f"{handlers}/{area_id}_{field}_series.cells",
    }
    rowptr.tofile(out / files["rowptr_file"])
    cols.tofile(out / files["cols_file"])
    vals.tofile(out / files["vals_file"])
    (out / files["cells_file"]).write_text("\n".join(cell_ids), encoding="utf-8")

    interp_count = 0
    if interp_flags:                                   # parallel uint8 handler in CSR order
        flags = np.fromiter((1 if (s, c) in interp_flags else 0 for (s, c), _ in items),
                            dtype=np.uint8, count=nnz)
        interp_count = int(flags.sum())
        files["interp_file"] = f"{handlers}/{area_id}_{field}_series.{stamp}.interp"
        flags.tofile(out / files["interp_file"])

    records_by_cell = defaultdict(list)
    for (s, c) in cell_map:
        records_by_cell[c].append(s)
    disc_cells, disc_steps = _discontinuity_stats(records_by_cell)

    header = {
        "format":      "csr",
        "dtype":       dt.name,
        "n_steps":     n_steps,
        "n_cells":     n_cells,
        "nnz":         nnz,
        "min":         float(vmin),
        "max":         float(vmax),
        "gap":         GAP_POLICY,        # missing/absent step = NULL (display nothing)
        "time_col":    time_col,          # the TIMESTAMP axis column (contract: 'event_ts')
        "time_labels": time_labels,
        "discontinuity_cells": disc_cells,
        "discontinuity_steps": disc_steps,
        "interp_count": interp_count,     # records synthesised by resampling (0 = all measured)
        **files,
    }
    (out / json_file).write_text(json.dumps(header), encoding="utf-8")
    # Sweep stale blobs for THIS (area, field): anything matching the series prefix
    # that the new header doesn't name is superseded. Best-effort -- a blob still
    # memmapped by a reader can't be unlinked on Windows and survives until the next
    # write sweeps again.
    keep = {v for v in files.values()}
    for stale in (out / handlers).glob(f"{area_id}_{field}_series.*"):
        if f"{handlers}/{stale.name}" not in keep:
            try:
                stale.unlink()
            except OSError:
                pass
    return {
        "name": field,
        "kind": "cell_series",
        "file": json_file,
        "spec": {"n_steps": n_steps, "n_cells": n_cells, "nnz": nnz, "dtype": dt.name,
                 "min": float(vmin), "max": float(vmax), "gap": GAP_POLICY,
                 "discontinuity_cells": disc_cells, "interp_count": interp_count},
    }




def _dt_suffix(dt):
    """Short file suffix for the value array (float32 -> 'f32')."""
    return {np.float16: "f16", np.float32: "f32", np.float64: "f64"}.get(dt.type, dt.name)


RESAMPLE_METHODS = ("linear", "nearest")


def _fmt_num(x):
    """Format a numeric ordinal back to a compact label ('2011' not '2011.0')."""
    r = round(float(x), 6)
    return str(int(r)) if r == int(r) else repr(r)


def _time_coder(labels):
    """Map time-label strings <-> float ordinals. If every label parses as an ISO-8601
    datetime, ordinals are seconds since 1970 (UTC-naive) and to_label formats back to
    ISO; otherwise labels must be plain numbers and ordinals are those numbers. Returns
    (ordinals[list], to_label(ordinal)->str, is_datetime)."""
    try:
        ords = [(datetime.fromisoformat(x) - _EPOCH).total_seconds() for x in labels]
        return ords, (lambda o: (_EPOCH + timedelta(seconds=o)).isoformat()), True
    except ValueError:
        return [float(x) for x in labels], _fmt_num, False


def _parse_bound(x, is_datetime):
    """Parse a start/end bound in the same coordinate as the labels: an ISO datetime ->
    epoch seconds when the axis is datetime, else a plain float."""
    if is_datetime and isinstance(x, str):
        return (datetime.fromisoformat(x) - _EPOCH).total_seconds()
    return float(x)

def suggest_grid(time_labels):
    """Sensible resample-grid defaults for an existing series' labels: its first and last
    labels as ``start`` / ``end``, and the spacing between its first two steps as
    ``interval`` (seconds for a datetime axis, else the numeric step). Returns
    ``(start, end, interval)`` -- start/end as label strings, interval as a float -- or
    ``(None, None, None)`` when there are fewer than two labels to infer a step from."""
    if len(time_labels) < 2:
        return None, None, None
    ords, _to_label, _is_dt = _time_coder(time_labels)
    interval = ords[1] - ords[0]
    return time_labels[0], time_labels[-1], (interval if interval > 0 else None)

def resample_series(cell_ids, time_labels, triples, *, start=None, end=None, interval=None,
                    method="linear", grid_ords=None):
    """Resample a built cell_series (see build_series) onto a regular time grid
    [start, end] stepped by `interval`, filling each cell's INTERIOR gaps -- grid steps
    between that cell's own first and last real observation -- by `method`:

      'linear'  -- straight-line between the two bracketing observations (np.interp).
      'nearest' -- the value of the nearest observation in time.

    Rules (agreed): a cell is NEVER extrapolated before its first or after its last
    observation (those grid steps stay NULL -- the cell shows nothing there); a grid step
    that coincides with a real observation keeps the MEASURED value (flag False); a cell
    with a single observation is never interpolated (it contributes at most the one grid
    step that lands exactly on it). Opting into resampling thus overrides the display-
    nothing gap policy only for interior steps that get a synthesised value.

    Times may be ISO-8601 datetimes or plain numbers; `start`/`end` must match (ISO
    strings for a datetime axis, else numbers) and `interval` is a number in the axis'
    unit (seconds for a datetime axis). Alternatively pass `grid_ords` -- an explicit,
    possibly NON-uniform grid of ordinals (in the labels' coordinate) -- instead of
    start/end/interval; this is how a SQL calendar interval ('1 month 1 day', expanded to
    timestamps by DuckDB upstream, see tyrbuild.parse_interval_grid) drives the grid.

    Returns (new_time_labels, new_triples, interp_flags, warnings):
      * new_time_labels -- the grid as label strings (index = step); every grid step is a
                           frame even if no cell carries a value there (an all-NULL frame).
      * new_triples      -- [(step, cell_idx, value), ...]; cell_idx indexes the SAME
                           cell_ids order passed in.
      * interp_flags     -- {(step, cell_idx)} for the SYNTHESISED records (pass straight
                           to write_series(interp_flags=...)); measured/coincident records
                           are omitted.
      * warnings         -- human-readable strings, chiefly an nnz blow-up / cap estimate
                           (resampling densifies a sparse series -- widen the interval if
                           the projection is large). write_series still hard-caps.
    Raises ValueError on a bad method, a non-positive interval, or end < start."""
    if method not in RESAMPLE_METHODS:
        raise ValueError(f"resample method must be one of {RESAMPLE_METHODS}, got {method!r}")
    ords, to_label, is_dt = _time_coder(time_labels)
    if grid_ords is not None:
        # An explicit, possibly NON-uniform grid (e.g. a SQL calendar interval expanded to
        # timestamps upstream). Ordinals must be in the labels' coordinate and strictly rise.
        grid = np.asarray(grid_ords, dtype=float)
        if grid.size == 0:
            raise ValueError("resample grid is empty")
        if np.any(np.diff(grid) <= 0):
            raise ValueError("resample grid_ords must be strictly increasing")
        n = int(grid.size)
        step_ref = float(np.diff(grid).min()) if grid.size > 1 else 1.0
    else:
        lo_b = _parse_bound(start, is_dt)
        hi_b = _parse_bound(end, is_dt)
        step_ref = float(interval)
        if step_ref <= 0:
            raise ValueError(f"interval must be positive, got {interval!r}")
        if hi_b < lo_b:
            raise ValueError(f"end ({end!r}) is before start ({start!r})")
        n = int(math.floor((hi_b - lo_b) / step_ref + 1e-9)) + 1
        grid = lo_b + step_ref * np.arange(n, dtype=float)
    tol = max(abs(step_ref), 1.0) * 1e-9

    by_cell = defaultdict(list)                       # cell_idx -> [(ordinal, value), ...]
    for s, c, v in triples:
        by_cell[c].append((ords[s], float(v)))

    new_triples, interp_flags = [], set()
    for c, obs in by_cell.items():
        obs.sort()
        ot = np.array([t for t, _ in obs], dtype=float)
        ov = np.array([v for _, v in obs], dtype=float)
        sel = np.nonzero((grid >= ot[0] - tol) & (grid <= ot[-1] + tol))[0]  # interior span
        if sel.size == 0:
            continue
        g = grid[sel]
        if method == "linear" or ot.size == 1:
            gv = np.interp(g, ot, ov)                 # exact at observations, line between
        else:
            pos = np.clip(np.searchsorted(ot, g), 1, ot.size - 1)
            left, right = ot[pos - 1], ot[pos]
            idx = np.where(g - left <= right - g, pos - 1, pos)
            gv = ov[idx]
        # a grid step within tol of an observation is the measured value (not synthesised)
        near = np.searchsorted(ot, g)
        near = np.clip(near, 1, ot.size - 1)
        dist = np.minimum(np.abs(g - ot[near - 1]), np.abs(g - ot[near]))
        real = dist <= tol
        for step_idx, val, is_real in zip(sel.tolist(), gv.tolist(), real.tolist()):
            new_triples.append((step_idx, c, val))
            if not is_real:
                interp_flags.add((step_idx, c))

    warnings = []
    grew, base = len(new_triples), len(triples)
    if grew > base:
        warnings.append(
            f"resample densified {base} -> {grew} records ({grew / max(base, 1):.1f}x) "
            f"across {n} grid steps")
    est = grew * 8 + (n + 1) * 8                       # float32 vals + int32 cols + rowptr
    if est > MAX_SERIES_BYTES:
        warnings.append(
            f"resample: ~{est / 1024**3:.1f} GiB projected exceeds the "
            f"{MAX_SERIES_BYTES / 1024**3:.0f} GiB cap -- write_series will refuse; "
            f"widen the interval")
    elif est > MAX_SERIES_BYTES // 2:
        warnings.append(
            f"resample: ~{est / 1024**3:.1f} GiB projected, nearing the "
            f"{MAX_SERIES_BYTES / 1024**3:.0f} GiB cap")

    return [to_label(g) for g in grid], new_triples, interp_flags, warnings

def open_series(data_dir, header, mode="r"):
    """Open a CSR series as (rowptr, cols, vals) arrays from its header. rowptr is always
    a real array (n_steps+1); cols/vals are empty when nnz == 0 (an all-null series)."""
    d = Path(data_dir)
    dt = np.dtype(header["dtype"])
    nnz = header["nnz"]
    rowptr = np.memmap(d / header["rowptr_file"], dtype=np.int64, mode=mode,
                       shape=(header["n_steps"] + 1,))
    if not nnz:
        return rowptr, np.empty(0, np.int32), np.empty(0, dt)
    cols = np.memmap(d / header["cols_file"], dtype=np.int32, mode=mode, shape=(nnz,))
    vals = np.memmap(d / header["vals_file"], dtype=dt, mode=mode, shape=(nnz,))
    return rowptr, cols, vals

def read_interp(data_dir, header):
    """The uint8 interp mask (1 = synthesised record, parallel to cols/vals), or None if
    this series has no `.interp` handler (every record measured)."""
    name = header.get("interp_file")
    if not name:
        return None
    return np.memmap(Path(data_dir) / name, dtype=np.uint8, mode="r",
                     shape=(header["nnz"],))

def series_row(handles, step):
    """The (cols, vals) records for one step -- the cells present at that step and their
    values. Everything else at that step is NULL (displays nothing)."""
    rowptr, cols, vals = handles
    lo, hi = int(rowptr[step]), int(rowptr[step + 1])
    return cols[lo:hi], vals[lo:hi]

def read_cells(data_dir, header):
    """The series' cell order (col index -> h3_id) from its .cells handler."""
    text = (Path(data_dir) / header["cells_file"]).read_text(encoding="utf-8")
    return text.split("\n") if text else []
