"""
terrain.py
----------
Terrain grid fetching and caching. Fetches elevation data from Copernicus
DEM 30m (COP30) Cloud Optimised GeoTIFFs on AWS S3 (anonymous public bucket),
falling back to GEBCO bathymetry for all-ocean tiles and to NASADEM via
OpenTopography (API-key-gated) for cells still missing -- including GLO-30
restricted territories, where the S3 bucket answers 403 AccessDenied. Ocean
0-fill is healed from GMRT/GEBCO bathymetry.
"""

import json
import math
import time

import numpy as np

from .config import (
    OVERWRITE_STRUCTURES, OVERWRITE_WATER,
    OVERWRITE_VEGETATION, OVERWRITE_STREETS,
    OVERWRITE_LANDCOVER, OVERWRITE_LANDMARKS,
)
from . import config as _config
from .openbuildingmap import fetch_obm_structures
from .bridges     import process_osm_bridges
from .water      import (process_osm_water, process_osm_coastline, build_island_rings,
                          extract_water_inner_rings)
from .vegetation import process_osm_vegetation
from .streets    import process_osm_streets
from .landcover  import process_osm_landcover
from .landmarks  import process_osm_landmarks

R_EARTH              = 6_371_000.0
_COP30_TILE_RETRIES   = 4      # remote S3 tiles flake out ("Read failed"); retry before giving up
_COP30_TILE_BACKOFF_S = 2.0    # linear backoff between attempts
_NASADEM_TILE_RETRIES   = 3    # OpenTopography globaldem requests, per missing 1deg tile
_NASADEM_TILE_BACKOFF_S = 2.0  # linear backoff between attempts
_NASADEM_BBOX_MARGIN    = 0.05 # degrees of bbox margin so edge cells interpolate

_OPENTOPOGRAPHY_BASE = "https://portal.opentopography.org/API/globaldem"
_COP30_BUCKET        = "copernicus-dem-30m"   # public AWS bucket (anonymous read)
_COP30_REGION        = "eu-central-1"
_COP30_BASE          = f"https://{_COP30_BUCKET}.s3.amazonaws.com"

def _cop30_tile_key(tile_lat, tile_lon):
    """S3 object key for the COP30 DEM tile covering (tile_lat, tile_lon)."""
    lat_pfx = "N" if tile_lat >= 0 else "S"
    lon_pfx = "E" if tile_lon >= 0 else "W"
    name    = (f"Copernicus_DSM_COG_10_{lat_pfx}{abs(tile_lat):02d}_00_"
               f"{lon_pfx}{abs(tile_lon):03d}_00_DEM")
    return f"{name}/{name}.tif"


def _cop30_tile_url(tile_lat, tile_lon):
    return f"{_COP30_BASE}/{_cop30_tile_key(tile_lat, tile_lon)}"


def _cop30_s3_client():
    """Lazily build a cached ANONYMOUS S3 client for the public Copernicus DEM
    bucket (UNSIGNED so it works with no AWS credentials, adaptive retries with
    exponential backoff -- the transient 'Read failed' fix). None if boto3 isn't
    installed, so the caller falls back to a plain HTTPS GET."""
    if not hasattr(_cop30_s3_client, "_client"):
        try:
            import boto3
            from botocore import UNSIGNED
            from botocore.config import Config
            _cop30_s3_client._client = boto3.client(
                "s3", region_name=_COP30_REGION,
                config=Config(signature_version=UNSIGNED,
                              retries={"max_attempts": 5, "mode": "adaptive"}))
        except Exception:
            _cop30_s3_client._client = None
    return _cop30_s3_client._client


def _is_missing_tile(exc):
    """True if `exc` means the tile does not exist (Copernicus omits all-ocean
    tiles). Handles both botocore ClientError and requests HTTPError."""
    resp = getattr(exc, "response", None)
    if isinstance(resp, dict):               # botocore ClientError
        code = resp.get("Error", {}).get("Code")
        status = resp.get("ResponseMetadata", {}).get("HTTPStatusCode")
        return code in ("404", "NoSuchKey", "NoSuchBucket") or status == 404
    return resp is not None and getattr(resp, "status_code", None) == 404  # requests


def _is_denied_tile(exc):
    """True if `exc` means the COP30 tile EXISTS but access is denied (403
    AccessDenied -- GLO-30 is restricted over ~two dozen territories, so the
    anonymous bucket refuses the read). Definitive, not transient: no retry."""
    resp = getattr(exc, "response", None)
    if isinstance(resp, dict):               # botocore ClientError
        code = resp.get("Error", {}).get("Code")
        status = resp.get("ResponseMetadata", {}).get("HTTPStatusCode")
        return code == "AccessDenied" or status == 403
    return resp is not None and getattr(resp, "status_code", None) == 403  # requests


def _download_nasadem_tile(url, dest):
    """Download a NASADEM patch from OpenTopography globaldem to `dest`,
    atomically (via a .part temp). 401 = rejected API key -> raise immediately
    (no point retrying a bad key); other HTTP errors retry with linear backoff
    then re-raise -- fail loud, same philosophy as the COP30 tile fetch."""
    import requests
    tmp = dest.with_name(dest.name + ".part")
    for _attempt in range(_NASADEM_TILE_RETRIES):
        r = requests.get(url, timeout=300)
        if r.status_code == 401:
            raise RuntimeError(
                "OpenTopography API key was rejected (401) -- check the key in Settings")
        try:
            r.raise_for_status()
        except Exception as _e:
            if _attempt + 1 < _NASADEM_TILE_RETRIES:
                # Print status only -- str(HTTPError) embeds the request URL,
                # which carries API_Key; the key must never reach the logs.
                print(f"\n  [terrain] NASADEM fetch failed (attempt "
                      f"{_attempt + 1}/{_NASADEM_TILE_RETRIES}): "
                      f"HTTP {r.status_code} {r.reason} -- retrying",
                      flush=True)
                time.sleep(_NASADEM_TILE_BACKOFF_S * (_attempt + 1))
                continue
            raise
        tmp.write_bytes(r.content)
        tmp.replace(dest)
        return

def cell_centres_from_ids(cells, resolution):
    """Build [clon, clat, 0.0, rotation_z_rad, radius_eq, cell_id] rows for an
    explicit iterable of H3 cell ids (all at `resolution`). Elevation (index 2)
    is 0.0 and must be filled in by fetch_h3_elevations."""
    import h3 as _h3

    cells = list(cells)
    if not cells:
        return []

    clats = np.array([_h3.cell_to_latlng(c)[0] for c in cells])
    clons = np.array([_h3.cell_to_latlng(c)[1] for c in cells])

    cos_lats = np.cos(np.radians(clats))
    b0_lats  = np.array([_h3.cell_to_boundary(c)[0][0] for c in cells])
    b0_lons  = np.array([_h3.cell_to_boundary(c)[0][1] for c in cells])
    dx = (b0_lons - clons) * cos_lats
    dy =  b0_lats - clats
    rotations = np.arctan2(dy, dx)
    radius_eq = np.sqrt(dx**2 + dy**2)

    return [[round(float(lon), 8), round(float(lat), 8), 0.0,
             round(float(rot), 6), round(float(rad), 8), c]
            for c, lon, lat, rot, rad in zip(cells, clons, clats, rotations, radius_eq)]

def build_h3_cell_centres(lat_min, lon_min, lat_max, lon_max, resolution):
    """
    Generate H3 cell centres covering the bbox.
    Returns list of [clon, clat, 0.0, rotation_z_rad, radius_eq, cell_id].
    Elevation (index 2) is 0.0 and must be filled in by fetch_h3_elevations.
    """
    import h3 as _h3

    bbox_poly = _h3.LatLngPoly([
        (lat_min, lon_min), (lat_min, lon_max),
        (lat_max, lon_max), (lat_max, lon_min),
    ])
    # Use overlap containment so the cell union covers the whole bbox. The default
    # center containment only keeps cells whose centre lies inside the bbox, leaving
    # gaps near the edges where a cell overlaps the bbox but its centre does not.
    # Those gaps caused structures/streets clipped to the bbox to miss the terrain
    # surface during projection.
    if hasattr(_h3, "h3shape_to_cells_experimental"):
        cells = list(_h3.h3shape_to_cells_experimental(bbox_poly, resolution,
                                                       contain="overlap"))
    else:
        cells = list(_h3.h3shape_to_cells(bbox_poly, resolution))
    result = cell_centres_from_ids(cells, resolution)
    print(f"  [h3 terrain centres] {len(result)} cells at resolution {resolution}")
    return result

def apply_elevations(cell_centres, h3_elevations):
    """Stamp fetched elevations into cell_centres' ele slot (c[2]).

    A cell with NO elevation (all-ocean COP30 tile + no GEBCO tile covering it)
    is a hard failure, never a silent 0 m: a flattened seabed corrupts every
    downstream consumer (staging, output, editor, exports) and looks exactly
    like real data. Matches fetch_h3_elevations' own fail-loud policy.
    """
    missing = [c[5] for c in cell_centres if h3_elevations.get(c[5]) is None]
    if missing:
        raise RuntimeError(
            f"{len(missing)}/{len(cell_centres)} cells have no elevation — "
            f"Copernicus DEM does not cover them and no GEBCO bathymetry tile "
            f"was found (first missing: {missing[0]}). The seabed cannot be "
            f"invented: point area.gebco_tile_dir at a local GEBCO 2026 tile "
            f"directory and re-run. Aborting rather than emitting 0 m cells.")
    for c in cell_centres:
        c[2] = h3_elevations[c[5]]

def fetch_h3_elevations(area_id, cell_centres, raw_data_dir,
                        gebco_tile_dir=None, dem_tile_dir=None,
                        topo_tile_dir=None):
    """
    Fetch elevation for each H3 cell.

    Default source is COP30 (plus GEBCO bathymetry for all-ocean tiles, and
    NASADEM via OpenTopography -- key-gated -- for cells still missing, e.g.
    GLO-30 restricted territories where the bucket answers 403; see
    _nasadem_fallback). Custom GeoTIFF overrides, most general last-resort first in the precedence:

    - `dem_tile_dir` ("unified" custom terrain): a valid pixel wins for ANY
      cell it covers, land or bed — the whole ground in one file.
    - `topo_tile_dir` ("split" custom terrain): a valid pixel applies where the
      unified dir has no coverage. Water-map cells may still be re-bedded from
      `bathy_tile_dir` later in the pipeline (see pipeline's water section).
    COP30/GEBCO are only fetched for cells not covered by the custom tiles or
    where the custom sample is nodata, saving downloads for fully-covered areas.

    Adaptive sampling based on the relationship between H3 cell size and COP30
    pixel size (~30m N-S, ~18-30m E-W depending on latitude):

    - res >= 12 (cells ~4-11m, SMALLER than COP30 pixels):
        Bilinear interpolation at the exact sub-pixel cell-centre position.
        Multiple adjacent cells share the same COP30 pixel but get unique
        interpolated values from the surrounding 4-pixel neighbourhood.

    - res <= 11 (cells ~25-174m, LARGER than or equal to COP30 pixels):
        Adaptive window mean. Window radius is scaled so the sampled area
        approximates the cell inradius. For res9 (~174m) this is roughly an
        11x11 pixel window; for res11 (~25m) it falls to 3x3 or 1x1.

    One HTTP fetch per tile regardless of cell count. Results cached on disk.

    ponytail: do not change this sampling strategy without explicit approval.
    """
    try:
        import rasterio
    except ImportError:
        raise ImportError("rasterio required -- pip install rasterio")

    from collections import defaultdict

    # H3 average edge lengths in metres (h3geo.org/docs/core-library/restable)
    _H3_EDGE_M = {
        7: 1220.6, 8: 461.4, 9: 174.4, 10: 65.9,
        11: 24.9,  12: 9.4,  13: 3.6,  14: 1.3,
    }

    _res = None
    if cell_centres:
        try:
            import h3 as _h3r
            _res = _h3r.get_resolution(cell_centres[0][5])
        except Exception:
            pass
    _res_suffix = f"_res{_res}" if _res is not None else ""
    _dem_suffix = ""
    if dem_tile_dir or topo_tile_dir:
        import hashlib
        # Key on both dirs so adding/editing either re-samples instead of
        # reusing a cache built without it. dem-only keeps the historical key.
        _key_src = (str(dem_tile_dir) if not topo_tile_dir
                    else f"{dem_tile_dir}|{topo_tile_dir}")
        _dem_suffix = f"_dem{hashlib.sha1(_key_src.encode()).hexdigest()[:8]}"
    # v2 cache: new adaptive sampling strategy replaces old 3x3 fixed window
    cache_path = raw_data_dir / "cop30" / f"{area_id}_h3v2{_res_suffix}{_dem_suffix}.json"

    cells_all = cell_centres   # the missing-cell filter below rebinds cell_centres;
                               # the ocean-fill heal must see every cell

    def _heal_zero_elevations(elevations, custom):
        """Re-sample exactly-0 cells from bathymetry, in place. Returns healed count.

        Copernicus GLO-30 is a LAND surface model: its ocean pixels are 0-fill,
        not nodata, so a 0 sample over water is missing data in disguise -- it
        silently flattens the seabed to 0 m in staging/source/output (Austevoll:
        9,526 sea cells at exactly 0, "0 wet" at derive time). Honour the
        sea-depth hierarchy: real GMRT multibeam soundings (topo-mask) win, the
        blended GMRT base (topo) is next, local GEBCO tiles are the fallback.
        Land (>= 0) is never overridden: only true ocean depths heal a 0.
        Custom-DEM cells are user-supplied data: a 0 there is real, untouched.
        """
        zero_cells = [c for c in cells_all
                      if len(c) > 5 and elevations.get(c[5]) == 0.0
                      and custom.get(c[5]) is None]
        if not zero_cells:
            return 0
        from .gmrt import fetch_gmrt_grid, sample_depths
        _bbox = (min(c[1] for c in cells_all), min(c[0] for c in cells_all),
                 max(c[1] for c in cells_all), max(c[0] for c in cells_all))
        _bathy = {}   # h3_id -> positive ocean depth
        for _layer in ("topo-mask", "topo"):
            _grid = fetch_gmrt_grid(area_id, *_bbox, raw_data_dir, layer=_layer)
            if _grid is None:
                continue
            for _hid, _depth in sample_depths(zero_cells, _grid).items():
                if _depth is not None:
                    _bathy.setdefault(_hid, _depth)
        _gmrt_n = len(_bathy)
        _rest = [c for c in zero_cells if c[5] not in _bathy]
        if _rest and gebco_tile_dir:
            from .gebco import sample_elevations
            for _hid, _ele in sample_elevations(_rest, gebco_tile_dir).items():
                if _ele is not None and _ele < 0 and _hid not in _bathy:
                    _bathy[_hid] = -_ele   # GEBCO returns signed elevation
        _healed = 0
        for c in zero_cells:
            _depth = _bathy.get(c[5])
            if _depth is not None:
                elevations[c[5]] = round(-float(_depth), 2)   # elevation = -depth
                _healed += 1
        _flat = len(zero_cells) - _healed
        print(f"  [terrain] ocean-fill: {_healed}/{len(zero_cells)} zero cells "
              f"re-sampled from bathymetry ({_gmrt_n} GMRT"
              + (f", {_healed - _gmrt_n} GEBCO" if _healed > _gmrt_n else "")
              + ")"
              + (f"  [WARN: {_flat} still 0 m -- no bathymetry coverage]" if _flat else ""),
              flush=True)
        return _healed

    # ── Custom terrain (coverage-aware) ──────────────────────────────────────
    # Two cooperative GeoTIFF sources, unified most-general-first in precedence:
    # dem_tile_dir ("unified") wins over topo_tile_dir ("split" topography) on
    # overlap. Water-map cells may STILL be re-bedded from bathy_tile_dir later
    # in the pipeline's water section (most specific of all, where it applies).
    _custom_dirs = [d for d in (dem_tile_dir, topo_tile_dir) if d]

    def _sample_custom(points):
        out = {}
        for d in _custom_dirs:
            from .localdem import RasterTiles
            rt = RasterTiles(d)
            if len(rt) == 0:
                print(f"  [terrain] custom terrain dir has no .tif files: {d}")
                continue
            h3_edge_m = _H3_EDGE_M.get(_res, 30.0)
            h3_inrad_m = h3_edge_m * math.sqrt(3) / 2
            for cid, v in rt.sample(points, mode="mean", radius_m=h3_inrad_m).items():
                if v is not None and out.get(cid) is None:
                    out[cid] = v
        return out

    cached = {}
    from . import config as _config
    if cache_path.exists() and not _config.SOURCE_FRESH:
        cached = json.loads(cache_path.read_text())
        want = {c[5] for c in cell_centres if len(c) > 5}
        # A cell cached as None is a FAILED fetch (e.g. a transient S3 tile read
        # error), not a real value -- treat it as missing so a re-run retries it
        # instead of baking 0 m in forever.
        have = {k for k, v in cached.items() if v is not None}
        missing = want - have
        if not missing:
            # A cached 0 may be a user-supplied custom-terrain value (real data)
            # -- sample the custom tiles so the heal leaves those cells alone.
            _custom = {}
            if _custom_dirs and any(v == 0.0 for v in cached.values()):
                _custom = _sample_custom(cell_centres)
            _healed = _heal_zero_elevations(cached, _custom)
            if _healed:
                cache_path.write_text(json.dumps(cached, separators=(",", ":")))
            print(f"  [terrain] elevation (cached) ... OK ({len(cached)} cells)")
            return cached
        # Bbox grew since the cache was written: keep the cached elevations and
        # fetch ONLY the cells we don't have yet, then merge. Without this the
        # whole-file cache was returned as-is and new cells defaulted to 0 m.
        print(f"  [terrain] elevation: cached {len(cached)}, fetching {len(missing)} new cell(s)")
        cell_centres = [c for c in cell_centres if len(c) > 5 and c[5] in missing]

    # Sample the custom tiles first (local, fast). Cells with valid custom values
    # skip the network fetch entirely; the rest fall through to COP30/GEBCO.
    custom_elevations = _sample_custom(cell_centres) if _custom_dirs else {}
    if _custom_dirs:
        _custom_n = sum(1 for v in custom_elevations.values() if v is not None)
        print(f"  [terrain] custom terrain: {_custom_n}/{len(cell_centres)} cells "
              f"from {' + '.join(_custom_dirs)}", flush=True)

    tile_cells = defaultdict(list)
    for c in cell_centres:
        if custom_elevations.get(c[5]) is None:
            tile_cells[(math.floor(c[1]), math.floor(c[0]))].append(c)
        # else: cell already has a valid custom value, no COP30 fetch needed

    n_tiles = len(tile_cells)
    n_cells_default = len(tile_cells)
    n_cells_custom = len(cell_centres) - n_cells_default
    use_bilinear = _res is not None and _res >= 12
    strategy = "bilinear" if use_bilinear else "window-mean"
    if _custom_dirs:
        print(f"  [terrain] default fetch ({n_cells_default} cells, {n_tiles} tile{'s' if n_tiles > 1 else ''}, {strategy})",
              flush=True)
    else:
        print(f"  [terrain] COP30 h3 ({len(cell_centres)} cells, {n_tiles} tile{'s' if n_tiles > 1 else ''}, {strategy})",
              flush=True)
    from .progress import bar as _bar

    env_opts = dict(
        GDAL_DISABLE_READDIR_ON_OPEN="EMPTY_DIR",
        CPL_VSIL_CURL_ALLOWED_EXTENSIONS="tif",
        GDAL_HTTP_TIMEOUT="60",
        # Retry individual S3 range requests (503/timeout/reset) inside GDAL before
        # the whole-tile retry below ever fires -- catches most transient failures.
        GDAL_HTTP_MAX_RETRY="3",
        GDAL_HTTP_RETRY_DELAY="2",
    )

    # COP30 tiles are read from a LOCAL copy, never streamed over /vsicurl/. GDAL's
    # ranged S3 reads intermittently return a truncated chunk ("TIFFFillTile: got X
    # bytes, expected Y") and then CACHE the bad chunk, so retrying the read fails
    # identically -- that's why an in-place read retry can't fix it. A single-GET
    # download of the whole tile can't truncate silently; a bad one fails cleanly,
    # so we delete it and re-fetch. Tiles are cached on disk (raw source data, NOT
    # area-keyed -- kept across runs like the OBM/GEBCO tiles).
    # Global raw 1-degree tile store when the Pipeline Cache dir is configured
    # (config.DATA_DIR / settings 'data_dir'), else the region's own cache dir.
    tile_dir = _config.data_store_dir("cop30/tiles", raw_data_dir / "cop30" / "tiles")
    tile_dir.mkdir(parents=True, exist_ok=True)
    _s3 = _cop30_s3_client()

    def _download_tile(tile_lat, tile_lon, dest):
        """Download the tile to dest atomically (via a .part temp), size-validated.
        Prefers boto3/S3 (managed transfer: built-in retries, backoff, multipart)
        and falls back to a plain HTTPS GET when boto3 is unavailable. Raises on
        failure; a missing tile surfaces as a 404/NoSuchKey (see _is_missing_tile)."""
        tmp = dest.with_name(dest.name + ".part")
        if _s3 is not None:
            _s3.download_file(_COP30_BUCKET, _cop30_tile_key(tile_lat, tile_lon), str(tmp))
        else:
            import requests
            with requests.get(_cop30_tile_url(tile_lat, tile_lon), stream=True,
                              timeout=120, headers={"User-Agent": "apian-terrain"}) as r:
                r.raise_for_status()
                expected = int(r.headers.get("Content-Length", 0))
                with open(tmp, "wb") as f:
                    for chunk in r.iter_content(1 << 20):
                        if chunk:
                            f.write(chunk)
                if expected and tmp.stat().st_size != expected:
                    tmp.unlink(missing_ok=True)
                    raise IOError(f"short download {tmp.stat().st_size}/{expected} bytes")
        tmp.replace(dest)

    result = dict(cached)  # start from cache; fill in the missing cells below
    # Seed result with valid custom elevations so the COP30 loop skips them.
    for hid, val in custom_elevations.items():
        if val is not None:
            result[hid] = val

    def _sample_group(group, band, t, W, H):
        """Sample a DEM band at the cell centres of `group`: bilinear at the
        sub-pixel position for res>=12, adaptive window mean for res<=11. The
        SAME strategy is used for COP30 and NASADEM bands (both are ~30m DEMs).
        Returns {h3_id: elevation or None} (NaN pixels -> None)."""
        lons = np.array([c[0] for c in group])
        lats = np.array([c[1] for c in group])

        # Floating-point pixel position of each cell centre
        col_f = (lons - t.c) / t.a - 0.5
        row_f = (lats - t.f) / t.e - 0.5

        if use_bilinear:
            # --- Bilinear interpolation ---
            # Each cell is smaller than a DEM pixel; interpolate between
            # the 4 surrounding pixel values at the exact sub-pixel position.
            col0 = np.clip(np.floor(col_f).astype(int), 0, W - 2)
            row0 = np.clip(np.floor(row_f).astype(int), 0, H - 2)
            fc = np.clip(col_f - col0, 0.0, 1.0)
            fr = np.clip(row_f - row0, 0.0, 1.0)

            vals = (band[row0,     col0    ] * (1 - fc) * (1 - fr) +
                    band[row0,     col0 + 1] * fc       * (1 - fr) +
                    band[row0 + 1, col0    ] * (1 - fc) * fr       +
                    band[row0 + 1, col0 + 1] * fc       * fr)
        else:
            # --- Adaptive window mean ---
            # Compute window radius so the sampled square approximates the
            # cell inradius (= edge * sqrt(3)/2).
            h3_edge_m    = _H3_EDGE_M.get(_res, 30.0)
            h3_inrad_m   = h3_edge_m * math.sqrt(3) / 2
            lat_centre   = float(np.mean(lats))
            # DEM pixel width in metres at this latitude
            dem_m_ew     = abs(t.a) * 111_320 * math.cos(math.radians(lat_centre))
            win_rad      = max(1, round(h3_inrad_m / dem_m_ew))

            ci = np.clip(np.round(col_f).astype(int), win_rad, W - win_rad - 1)
            ri = np.clip(np.round(row_f).astype(int), win_rad, H - win_rad - 1)

            offsets = range(-win_rad, win_rad + 1)
            patches = np.stack([band[ri + dr, ci + dc]
                                for dr in offsets
                                for dc in offsets], axis=1)
            vals = np.nanmean(patches, axis=1)

        out = {}
        for c, val in zip(group, vals):
            out[c[5]] = round(float(val), 2) if not np.isnan(val) else None
        return out

    def _nasadem_fallback(tile_lat, tile_lon, group, denied):
        """Fill cells still elevation-less after GEBCO from NASADEM via
        OpenTopography: one globaldem request per missing 1deg tile (bbox = the
        tile + a small margin so edge cells interpolate), GTiff cached on disk
        in the nasadem/tiles store (global Pipeline Cache dir when configured,
        else raw_data_dir/nasadem/tiles) and reused across runs. Key-gated:
        without a key, warn loudly and leave the cells None (downstream
        validate_output already flags missing heights)."""
        rest = [c for c in group if result.get(c[5]) is None]
        if not rest:
            return
        key = _config.OPENTOPOGRAPHY_API_KEY
        if not key:
            why = ("Copernicus DEM access is RESTRICTED in this territory "
                   "(403 AccessDenied)" if denied
                   else "Copernicus DEM has no coverage for this tile")
            print(f"\n  [terrain] tile ({tile_lat},{tile_lon}): {why}; "
                  f"{len(rest)}/{len(group)} cells have NO elevation. Set an "
                  f"OpenTopography API key in Settings to enable the NASADEM "
                  f"fallback (OpenTopography globaldem, demtype=NASADEM).",
                  flush=True)
            return
        nasadem_dir = _config.data_store_dir("nasadem/tiles",
                                             raw_data_dir / "nasadem" / "tiles")
        nasadem_dir.mkdir(parents=True, exist_ok=True)
        lat_pfx = "N" if tile_lat >= 0 else "S"
        lon_pfx = "E" if tile_lon >= 0 else "W"
        dest = nasadem_dir / (f"nasadem_{lat_pfx}{abs(tile_lat):02d}_"
                              f"{lon_pfx}{abs(tile_lon):03d}.tif")
        m   = _NASADEM_BBOX_MARGIN
        url = (f"{_OPENTOPOGRAPHY_BASE}?demtype=NASADEM"
               f"&south={tile_lat - m}&north={tile_lat + 1 + m}"
               f"&west={tile_lon - m}&east={tile_lon + 1 + m}"
               f"&outputFormat=GTiff&API_Key={key}")
        # Ensure a valid LOCAL patch, then read it. A failed open means a
        # corrupt/partial file -> delete and re-download, once.
        band = t = None
        W = H = 0
        for _attempt in range(2):
            if not (dest.exists() and dest.stat().st_size > 0):
                _download_nasadem_tile(url, dest)
            try:
                with rasterio.open(dest) as ds:
                    nodata = ds.nodata
                    band = ds.read(1).astype(np.float64)
                    H, W = band.shape
                    t = ds.transform
                if nodata is not None:
                    band[band == nodata] = np.nan
                break
            except Exception:
                band = t = None
                dest.unlink(missing_ok=True)
        if band is None:
            raise IOError(f"NASADEM patch for tile ({tile_lat},{tile_lon}) is "
                          f"corrupt after re-download: {dest}")
        n = 0
        for hid, val in _sample_group(rest, band, t, W, H).items():
            if val is not None:
                result[hid] = val
                n += 1
        print(f"\n  [terrain] tile ({tile_lat},{tile_lon}): COP30 unavailable -- "
              f"{n}/{len(rest)} cells from NASADEM (OpenTopography)", flush=True)

    with rasterio.Env(**env_opts):
        for (tile_lat, tile_lon), group in _bar(tile_cells.items(), n_tiles, "COP30 tiles"):
            dest = tile_dir / _cop30_tile_key(tile_lat, tile_lon).rsplit("/", 1)[-1]
            try:
                # Ensure a valid LOCAL tile, then read it. A failed read means a
                # corrupt/partial local file -> delete and re-download, up to N tries.
                _err = None
                _denied = False
                for _attempt in range(_COP30_TILE_RETRIES):
                    try:
                        if not (dest.exists() and dest.stat().st_size > 0):
                            _download_tile(tile_lat, tile_lon, dest)
                        with rasterio.open(dest) as ds:
                            nodata = ds.nodata
                            band = ds.read(1).astype(np.float64)
                            H, W = band.shape
                            t = ds.transform
                        if nodata is not None:
                            band[band == nodata] = np.nan
                        _err = None
                        break
                    except Exception as _e:
                        # A missing tile is DEFINITIVE, not transient: Copernicus DEM
                        # does not publish all-ocean tiles, so this tile genuinely has
                        # no land DEM. A denied tile (403 AccessDenied) is definitive
                        # too: GLO-30 is access-restricted over ~two dozen territories.
                        # Both fall back to GEBCO below, then NASADEM for what's still
                        # missing. Don't retry/crash.
                        if _is_missing_tile(_e) or _is_denied_tile(_e):
                            _denied = _is_denied_tile(_e)
                            # Fall back to GEBCO bathymetry -- the REAL seabed
                            # elevation -- not 0 m.
                            from .gebco import sample_elevations
                            _g = sample_elevations(group, gebco_tile_dir)
                            for c in group:
                                result.setdefault(c[5], _g.get(c[5]))
                            _ocean_n = sum(1 for c in group if _g.get(c[5]) is not None)
                            _err = "OCEAN"
                            break
                        _err = _e
                        dest.unlink(missing_ok=True)   # drop corrupt/partial -> re-download
                        if _attempt + 1 < _COP30_TILE_RETRIES:
                            print(f"\n  [terrain] tile ({tile_lat},{tile_lon}) fetch/read failed "
                                  f"(attempt {_attempt + 1}/{_COP30_TILE_RETRIES}): {_e} -- retrying",
                                  flush=True)
                            time.sleep(_COP30_TILE_BACKOFF_S * (_attempt + 1))
                if _err == "OCEAN":
                    _tag = ("Copernicus DEM access restricted (403 AccessDenied)"
                            if _denied else "no Copernicus DEM (all ocean)")
                    print(f"\n  [terrain] tile ({tile_lat},{tile_lon}): {_tag} -- "
                          f"{_ocean_n}/{len(group)} cells from GEBCO bathymetry"
                          + ("" if _ocean_n == len(group)
                             else "  [WARN: rest have no GEBCO tile]"), flush=True)
                    _nasadem_fallback(tile_lat, tile_lon, group, denied=_denied)
                    continue
                if _err is not None:
                    raise _err

                for _hid, _val in _sample_group(group, band, t, W, H).items():
                    result[_hid] = _val

            except Exception as e:
                # Fail LOUD. A tile we can't read means real elevation is missing;
                # never silently leave those cells at 0 m. Persist the cells we did
                # fetch so a re-run resumes from here, then abort the whole run.
                cache_path.parent.mkdir(parents=True, exist_ok=True)
                cache_path.write_text(json.dumps(result, separators=(",", ":")))
                if isinstance(e, RuntimeError) and "OpenTopography" in str(e):
                    raise   # already explicit (rejected API key) -- don't re-wrap
                raise RuntimeError(
                    f"COP30 elevation tile ({tile_lat},{tile_lon}) failed after "
                    f"{_COP30_TILE_RETRIES} attempts -- aborting so we don't emit "
                    f"0 m cells. Re-run to resume (cached tiles are kept).\n"
                    f"  url: {_cop30_tile_url(tile_lat, tile_lon)}\n  cause: {e}"
                ) from e

    # ── Ocean-fill correction ────────────────────────────────────────────────
    # Copernicus GLO-30 fills ocean pixels with 0 (not nodata); a 0 sample over
    # water silently flattens the seabed. Re-sample exactly-0 cells from
    # GMRT/GEBCO bathymetry (see _heal_zero_elevations for the full story).
    _heal_zero_elevations(result, custom_elevations)

    valid = [v for v in result.values() if v is not None]
    if valid:
        _msg = f"OK  ({min(valid):.0f}m - {max(valid):.0f}m)"
        if dem_tile_dir and n_cells_custom:
            _msg += f"  [{n_cells_custom} custom, {n_cells_default} default]"
        print(_msg)
    else:
        print("OK  (no valid elevations)")

    cache_path.parent.mkdir(parents=True, exist_ok=True)
    cache_path.write_text(json.dumps(result, separators=(",", ":")))

    return result

def fetch_osm_data(area_id, lat_min, lon_min, lat_max, lon_max,
                   lat_centre, lon_centre, raw_data_dir, out_dir, progress_cb=None,
                   pbf_paths=None, pbf_dir=None, obm_dir=None, major_mode=False):
    """
    Read all OSM layers from PBF and cache the results.
    Cache stored as raw_data_dir/{area_id}_osm.json (per-region fetch cache).
    Returns a terrain_data-compatible dict with OSM fields and bbox metadata.

    major_mode (country scale) SKIPS the buildings/roads/landmarks layers -- the
    firehose that Natural Earth replaces -- so their (potentially huge) PBF parse is
    never run; the pipeline supplies those features from Natural Earth instead.
    """
    cache_path = raw_data_dir / f"{area_id}_osm.json"

    # Reuse the cache only if it covers the requested bbox. If the bbox grew, the
    # cached OSM is for a smaller area -- refetch every layer, else the new cells
    # get no structures/water/etc.
    _cached = None
    if cache_path.exists():
        try:
            _cached = json.loads(cache_path.read_text())
        except Exception as e:
            print(f"  [osm] cache load failed: {e}")
    _eps = 1e-6
    bbox_stale = _cached is not None and (
        lon_min < _cached.get("lon_min", lon_min) - _eps or
        lat_min < _cached.get("lat_min", lat_min) - _eps or
        lon_max > _cached.get("lon_max", lon_max) + _eps or
        lat_max > _cached.get("lat_max", lat_max) + _eps)
    _osm_dir = raw_data_dir / "osm"
    if bbox_stale:
        print("  [osm] requested bbox exceeds cached bbox -- refetching all layers")
        # process_osm_water/coastline keep their OWN per-layer caches and would
        # short-circuit the refetch, leaving the new area empty -- drop them too.
        for _stale in (_osm_dir / f"{area_id}_water.json",
                       _osm_dir / f"{area_id}_coastline.json"):
            try:
                _stale.unlink()
            except OSError:
                pass
    _cache_ok = _cached is not None and not bbox_stale

    need_structures  = not _cache_ok or OVERWRITE_STRUCTURES
    need_water       = not _cache_ok or OVERWRITE_WATER
    need_vegetation  = not _cache_ok or OVERWRITE_VEGETATION
    need_streets     = not _cache_ok or OVERWRITE_STREETS
    need_landcover   = not _cache_ok or OVERWRITE_LANDCOVER or \
                       not (_osm_dir / f"{area_id}_landcover.json").exists()
    need_landmarks   = not _cache_ok or OVERWRITE_LANDMARKS or \
                       not (_osm_dir / f"{area_id}_landmarks.json").exists()

    if major_mode:                    # country scale: skip the OSM building/road/landmark
        need_structures = need_streets = need_landmarks = False   # firehose entirely

    existing = {}
    if _cache_ok and not all([need_structures, need_water, need_vegetation,
                              need_streets, need_landcover, need_landmarks]):
        existing = _cached

    if not any([need_structures, need_water, need_vegetation, need_streets,
                need_landcover, need_landmarks]):
        print("  [osm] already cached -- loading")
        return existing

    structures    = existing.get("structures", [])
    water         = existing.get("water", [])
    vegetation    = existing.get("vegetation", [])
    streets       = existing.get("streets", [])
    sea           = existing.get("sea")
    sea_had_reversals = existing.get("sea_had_reversals")
    landuse       = existing.get("landuse", [])
    natural_areas = existing.get("natural_areas", [])
    landmarks     = existing.get("landmarks", [])

    def _p(frac, msg):
        if progress_cb:
            progress_cb(frac, msg)

    # Single PBF pass for all layers that need refreshing
    elements = {}
    if pbf_paths:
        from .pbf_reader import read_osm_elements
        _p(0.02, "Reading PBF data...")
        elements = read_osm_elements(
            pbf_paths, lat_min, lon_min, lat_max, lon_max,
            area_id=area_id, pbf_dir=pbf_dir,
        )

    if need_structures:
        _p(0.03, "OBM: fetching structures...")
        print("  [structures] fetching ...", end=" ", flush=True)
        structures = fetch_obm_structures(
            lat_min, lon_min, lat_max, lon_max, area_id, raw_data_dir,
            osm_elements=elements.get("structures", []),
            bridge_elements=elements.get("bridges", []),
            street_elements=elements.get("streets", []),
            obm_dir=obm_dir)
        print(f"OK ({len(structures)} structures)")
        bridges = process_osm_bridges(
            elements.get("bridges", []), lat_min, lon_min, lat_max, lon_max)
        if bridges:
            structures = structures + bridges
            print(f"  [bridges] {len(bridges)} man_made=bridge added")

    if need_water:
        _p(0.07, "OSM: processing water...")
        print("  [water] processing ...", end=" ", flush=True)
        water = process_osm_water(
            elements.get("water", []), area_id, lat_min, lon_min, lat_max, lon_max, raw_data_dir,
            island_elements=elements.get("islands", []))
        print(f"OK ({len(water)} bodies)")

    if need_vegetation:
        _p(0.10, "OSM: processing vegetation...")
        print("  [vegetation] processing ...", end=" ", flush=True)
        vegetation = process_osm_vegetation(
            elements.get("vegetation", []), area_id, lat_min, lon_min, lat_max, lon_max, raw_data_dir)
        print(f"OK ({len(vegetation)} polygons)")

    if need_streets:
        _p(0.13, "OSM: processing streets...")
        print("  [streets] processing ...", end=" ", flush=True)
        streets = process_osm_streets(
            elements.get("streets", []), area_id, lat_min, lon_min, lat_max, lon_max, raw_data_dir)
        print(f"OK ({len(streets)} ways)")

    if need_water:
        _p(0.17, "OSM: processing coastline...")
        print("  [sea] processing coastline ...", end=" ", flush=True)
        sea, sea_had_reversals = process_osm_coastline(
            elements.get("coastline", []), area_id, lat_min, lon_min, lat_max, lon_max, raw_data_dir)
        if sea:
            n_pts = sum(len(c) for c in (sea if isinstance(sea[0][0], list) else [sea]))
            print(f"OK ({n_pts} pts)")
        else:
            print("none (landlocked)")

    # NOTE: tidal water bodies (harbours, lagoons) are NOT merged into the sea.
    # They stay as water bodies; their island holes feed build_land_geometry so
    # the islands stay land. (Old extract_tidal_chains merge dropped those holes
    # and drowned lagoon/harbour islands -- removed 2026-06-25.)

    if need_landcover:
        _p(0.20, "OSM: processing landcover...")
        print("  [landcover] processing ...", end=" ", flush=True)
        lc = process_osm_landcover(
            elements.get("landcover", []), area_id, raw_data_dir)
        landuse       = lc.get("landuse", [])
        natural_areas = lc.get("natural_areas", [])
        print(f"OK ({len(landuse)} landuse, {len(natural_areas)} natural)")

    if need_landmarks:
        _p(0.23, "OSM: processing landmarks...")
        print("  [landmarks] processing ...", end=" ", flush=True)
        landmarks = process_osm_landmarks(
            elements.get("landmarks", []), area_id, raw_data_dir)
        print(f"OK ({len(landmarks)} landmarks)")

    data = {
        "area_id":            area_id,
        "extend_km":          _config.TERRAIN_EXTEND_KM,
        "lon_centre":         lon_centre,
        "lat_centre":         lat_centre,
        "lon_min":            lon_min,
        "lat_min":            lat_min,
        "lon_max":            lon_max,
        "lat_max":            lat_max,
        "ele_min_m":          0.0,
        "ele_max_m":          0.0,
        "structures":         structures,
        "water":              water,
        "vegetation":         vegetation,
        "streets":            streets,
        "sea":                sea,
        "sea_had_reversals":  sea_had_reversals,
        "islands":            (build_island_rings(elements.get("islands", []))
                               + extract_water_inner_rings(elements.get("water", []))) if need_water else [],
        "landuse":            landuse,
        "natural_areas":      natural_areas,
        "landmarks":          landmarks,
    }

    cache_path.parent.mkdir(parents=True, exist_ok=True)
    cache_path.write_text(json.dumps(data, separators=(",", ":")))

    return data

def build_water_cell_centres(id_to_cell, water_ring, water_ele_m, resolution,
                             holes=None, elevation_overrides=None):
    """
    Find terrain cells inside a water body polygon and return water instancing data.
    `id_to_cell` maps h3_id -> the grid's cell array. Returns list of
    [clon, clat, terrain_ele_m, rotation_z_rad, radius_eq, cell_id, water_ele_m].
    """
    try:
        import h3
    except ImportError:
        print("  [h3] not installed -- skipping water cell centre generation")
        return []

    if not water_ring or len(water_ring) < 3:
        return []

    latlng_ring = [(p[1], p[0]) for p in water_ring]
    if latlng_ring[0] == latlng_ring[-1]:
        latlng_ring = latlng_ring[:-1]

    latlng_holes = []
    for hole in (holes or []):
        if not hole or len(hole) < 3:
            continue
        lr = [(p[1], p[0]) for p in hole]
        if lr[0] == lr[-1]:
            lr = lr[:-1]
        if len(lr) >= 3:
            latlng_holes.append(lr)

    try:
        poly          = h3.LatLngPoly(latlng_ring, *latlng_holes)
        water_cell_ids = set(h3.h3shape_to_cells(poly, resolution))
    except Exception as e:
        print(f"  [h3 water] polyfill failed: {e}")
        return []

    if not water_cell_ids:
        return []

    overrides = elevation_overrides or {}
    result = []
    for cell_id in water_cell_ids:               # only the polyfilled cells, not the whole grid
        c = id_to_cell.get(cell_id)
        if c is None or len(c) <= 5:
            continue
        w_ele = overrides.get(cell_id, water_ele_m)
        result.append([c[0], c[1], c[2], c[3], c[4],
                       cell_id, round(float(w_ele), 4)])

    return result
