def _reconcile_column_rows(cm_path, fresh_rows, log):
    """Delta-mode column reconciliation. `fresh_rows` carry the file's current
    columns (sniffed order + detected defaults). Rows already in the TSV are
    PRESERVED for columns that still exist -- hand-set aliases, var types,
    null/default policies and link columns survive a source run -- rows for
    columns the file no longer has are dropped, and new columns get the
    detected defaults. Datasets absent from `fresh_rows` lose all their rows.
    """
    import csv
    existing = {}
    with open(cm_path, encoding="utf-8", newline="") as f:
        for row in csv.DictReader(f, delimiter="\t"):
            ds = (row.get("dataset") or "").strip()
            name = (row.get("column_name") or "").strip()
            if ds and name:
                existing.setdefault(ds, {})[name] = row

    fresh_by_ds = {}
    for r in fresh_rows:
        fresh_by_ds.setdefault(r["dataset"], []).append(r)

    kept = added = dropped = 0
    reconciled = []
    for ds, fresh in fresh_by_ds.items():
        have = existing.pop(ds, {})
        for r in fresh:
            name = r["column_name"]
            if name in have:
                row = have.pop(name)
                old = {c: (row.get(c) or "") for c in COLUMN_META_COLUMNS}
                old["ordinal_position"] = r["ordinal_position"]  # file's current order
                reconciled.append(old)
                kept += 1
            else:
                reconciled.append(r)
                added += 1
        dropped += len(have)                       # columns the file no longer has
    dropped += sum(len(v) for v in existing.values())   # datasets gone from the manifest

    log(f"[source metadata] column delta: kept {kept}, added {added}, dropped {dropped}")
    return reconciled

def generate_source_metadata(output_dir, area_id, log=print, force=False, mode=None,
                             feature_dir=None):
    """Create file_metadata.tsv and column_metadata.tsv for a region from its manifest.

    Generates rows for the cells CSV and every tabular layer declared in the
    manifest (feature GeoJSONs, celldata CSVs, and any future layer kind with a
    tabular file), sniffing columns for each.

    `output_dir` is the REGION dir: the manifest and the written TSVs live there.
    `feature_dir` (defaults to `output_dir`) is where the layer data files live
    -- the region-folder layout keeps them in the region's output/ subdir.

    mode:
      "all"     (default, and the legacy ``force=True`` spelling): rewrite both
                TSVs from scratch so they always match the files the pipeline
                just wrote.
      "delta":   reconcile instead. File rows are rebuilt from the manifest
                (structural). Column rows are preserved where the column still
                exists in the dataset's file -- hand-maintained aliases, var
                types, null/default policies and link columns survive the run --
                rows for vanished columns are dropped and new columns get
                detected defaults. Datasets gone from the manifest lose their
                rows. Absent TSVs are created as in "all".
      "unlocked": file rows are rebuilt structurally with each dataset's
                ``lock_column_metadata`` flag carried over by name (default
                False for new datasets). Column metadata is then regenerated
                from scratch ONLY for unlocked datasets -- locked datasets keep
                their existing column rows verbatim, so hand-maintained
                metadata survives the run untouched. Absent TSVs fall back to
                "all"; datasets gone from the manifest lose their rows either
                way.

    Returns (file_meta_path, column_meta_path).
    """
    mode = mode or ("all" if force else "delta")
    if mode not in ("all", "delta", "unlocked"):
        raise ValueError(f"unknown metadata mode {mode!r}")

    def _truthy(v):
        return str(v or "").strip().lower() in {"true", "1", "yes", "t"}

    out = Path(output_dir)
    feat = Path(feature_dir) if feature_dir else out
    manifest_path = out / f"{area_id}_manifest.json"
    manifest = unified.read_manifest(manifest_path)
    fm_path, cm_path = source_metadata_paths(out, area_id)
    if mode in ("delta", "unlocked") and not (fm_path.exists() and cm_path.exists()):
        mode = "all"  # nothing to reconcile against -- create fresh
    # Lock flags from the PREVIOUS file metadata, by dataset: the flags live on
    # file rows, which this function rebuilds each run -- carry them over or
    # every run would wipe the user's locks.
    locks = {}
    old_cm_by_dataset = {}
    if mode == "unlocked":
        for r in read_meta(fm_path, FILE_META_COLUMNS):
            ds = (r.get("dataset") or "").strip()
            if ds:
                locks[ds] = _truthy(r.get("lock_column_metadata"))
        for r in read_meta(cm_path, COLUMN_META_COLUMNS):
            ds = (r.get("dataset") or "").strip()
            if ds:
                old_cm_by_dataset.setdefault(ds, []).append(r)
    if mode == "all":
        for p in (fm_path, cm_path):
            try:
                if p.exists():
                    p.unlink()
            except OSError as e:
                log(f"[source metadata] {area_id}: could not remove {p}: {e}")

    file_rows, col_rows = [], []

    def _add_source(dataset, src_path, delim=","):
        if not src_path.is_file():
            log(f"[source metadata] {dataset}: {src_path.name} not found, skipping")
            return
        locked = mode == "unlocked" and locks.get(dataset, False)
        if locked:
            # Locked: keep the existing column rows verbatim -- no re-detect.
            file_rows.append({"schema": "staging", "dataset": dataset,
                              "file_regex": str(src_path), "delim": "c",
                              "distinct": "False", "required": "True",
                              "lock_column_metadata": "True"})
            col_rows.extend(old_cm_by_dataset.get(dataset, []))
            log(f"[source metadata] {dataset}: locked -- column metadata preserved")
            return
        # Sniff FIRST: a file with no columns at all (e.g. an empty
        # FeatureCollection) declares NOTHING -- tyr requires every dataset to
        # have declared columns (lineage), so an undeclarable dataset is
        # skipped rather than invented around.
        try:
            sniffed = detect_columns(str(src_path), dataset, delim=delim)
        except Exception as e:
            log(f"[source metadata] {dataset} detection failed: {e}")
            return
        if not sniffed:
            log(f"[source metadata] {dataset}: no columns sniffed from "
                f"{src_path.name} (empty file?) -- dataset not declared")
            return
        file_rows.append({"schema": "staging", "dataset": dataset,
                          "file_regex": str(src_path), "delim": "c", "distinct": "False",
                          "required": "True",
                          "lock_column_metadata": "False"})
        col_rows.extend(sniffed)

    cells_csv = manifest.get("cells_csv")
    if cells_csv:
        _add_source("cells", out / cells_csv, delim=",")

    # Walk the normalised layer list so new `layers` blocks are handled alongside
    # legacy `feature_files` / `data_fields`. Skip binary cell_series blobs.
    seen = set()
    area_id = manifest.get("area_id") or area_id
    prefix = f"{area_id}_"
    for layer in layers.manifest_layers(manifest):
        if layer.get("kind") == "cell_series":
            continue
        fname = layer.get("file")
        if not fname:
            continue
        p = feat / fname
        if p.suffix.lower() not in (".csv", ".tsv", ".geojson", ".json") or p in seen:
            continue
        seen.add(p)
        stem = p.stem
        dataset = stem[len(prefix):] if stem.startswith(prefix) else stem
        delim = "\t" if p.suffix.lower() == ".tsv" else ","
        _add_source(dataset, p, delim=delim)

    # DB-native OPTIONAL datasets (T5/D5): CHELSA is ingested straight into the
    # DB by ingest_chelsa -- it has no unified file, so the manifest walk never
    # declares it. When the area already holds the dataset, declare it here with
    # required=false: the staging/output contract gate and the Data Manager
    # pipeline consult that flag, so a tyr staging rebuild (which cannot recreate
    # a fileless dataset) never wedges the next Source export.
    _db_path = out / f"{area_id}.duckdb"
    if _db_path.is_file():
        try:
            _c = connect(str(_db_path))
            try:
                _has_chelsa = _c.execute(
                    "SELECT 1 FROM information_schema.tables "
                    "WHERE table_schema = 'project' AND table_name = 'chelsa'"
                ).fetchone() is not None
            finally:
                _c.close()
            if _has_chelsa:
                from . import climate as _climate
                file_rows.append({"schema": "staging", "dataset": "chelsa",
                                  "file_regex": "", "delim": "c", "distinct": "False",
                                  "required": "False",
                                  "lock_column_metadata": "True"
                                  if mode == "unlocked" and locks.get("chelsa", False)
                                  else "False"})
                if mode == "unlocked" and locks.get("chelsa", False):
                    col_rows.extend(old_cm_by_dataset.get("chelsa", []))
                else:
                    col_rows.extend(
                        _col_row("staging", "chelsa", c, c,
                                 "categorical" if c == "h3_id" else "numeric",
                                 "VARCHAR" if c == "h3_id" else "DOUBLE", i)
                        for i, c in enumerate(_climate.CHELSA_DATASET_COLUMNS))
                log(f"[source metadata] {area_id}: declared optional dataset 'chelsa'")
        except Exception as e:
            log(f"[source metadata] optional dataset declaration failed: {e}")

    if mode == "delta":
        col_rows = _reconcile_column_rows(cm_path, col_rows, log)
    write_meta(fm_path, FILE_META_COLUMNS, file_rows)
    write_meta(cm_path, COLUMN_META_COLUMNS, col_rows)
    log(f"[source metadata] {area_id}: wrote {len(file_rows)} file(s), {len(col_rows)} column(s) [{mode}]")
    return fm_path, cm_path




def _resolve_meta_paths(config_dir, db_path=None):
    """Pick the metadata TSVs to use for a pipeline run.

    Prefer manifest-backed source/pipelines paths (looked up from the DB path's
    parent). Fall back to legacy <config_dir>/file_metadata.tsv when no manifest
    paths are declared, so existing tests and pre-manifest areas keep working.
    """
    config_dir = Path(config_dir)
    db_path = Path(db_path) if db_path else None
    if db_path:
        manifest_path = db_path.parent / f"{db_path.stem}_manifest.json"
        if manifest_path.is_file():
            pipelines = (unified.read_manifest(manifest_path).get("source") or {}).get("pipelines") or {}
            fm_rel = pipelines.get("file_metadata")
            cm_rel = pipelines.get("column_metadata")
            if fm_rel and cm_rel:
                return db_path.parent / fm_rel, db_path.parent / cm_rel
    return file_meta_path(config_dir), column_meta_path(config_dir)

def read_meta(path, columns):
    """Read a metadata TSV into a list of dict rows (empty list if absent)."""
    path = Path(path)
    if not path.is_file():
        return []
    with open(path, newline="", encoding="utf-8") as f:
        return [{c: r.get(c, "") for c in columns} for r in csv.DictReader(f, delimiter="\t")]

def write_meta(path, columns, rows):
    """Write dict rows to a metadata TSV (only the known columns, in order).
    Atomic: a reader always sees the previous or the new file, never a partial write."""
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    with open(tmp, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=columns, delimiter="\t", extrasaction="ignore")
        w.writeheader()
        for row in rows:
            w.writerow({c: row.get(c, "") for c in columns})
    tmp.replace(path)




def _vtype_for(dtype):
    """var_type for a DuckDB column type string."""
    d = dtype.upper()
    if any(k in d for k in ("INT", "DOUBLE", "DECIMAL", "FLOAT", "REAL", "BIGINT")):
        return "numeric"
    if "TIMESTAMP" in d or "DATE" in d:
        return "timestamp"
    return "categorical"

def detect_columns(file_regex, dataset, schema="staging", delim=None):
    """Infer column_metadata rows for a source file by letting DuckDB sniff its
    header + types (`read_csv_auto` for CSV/TSV, ST_Read's inferred schema for
    GeoJSON). Returns dict rows pre-filled with a sensible var_type/data_type the
    user can then edit. A file with nothing to sniff (e.g. an empty
    FeatureCollection) returns ZERO rows -- an empty file is empty, never a
    phantom wildcard. Note ST_Read scans the whole file to union the feature
    properties (~2 s per 100 MB), so big layers take a moment."""
    ext = Path(file_regex).suffix.lower().lstrip(".")
    con = duckdb.connect()
    try:
        if ext in ("json", "geojson"):
            return _geojson_columns(con, file_regex, schema, dataset)
        reader = f"read_csv_auto('{file_regex}'" + (f", delim='{delim}'" if delim else "") + ")"
        info = con.execute(f"DESCRIBE SELECT * FROM {reader}").fetchall()
    finally:
        con.close()
    return [_col_row(schema, dataset, name, name, _vtype_for(dtype), dtype, i)
            for i, (name, dtype, *_rest) in enumerate(info)]




def _scan_geojson(path):
    """Parse a GeoJSON file once: returns (property_keys, feature_count) --
    every property key across all features, plus how many features the file
    holds. A full-file parse, so very large files cost memory, but it
    guarantees no column is silently dropped because a sampler missed it."""
    path = Path(path)
    if not path.is_file():
        return [], 0
    data = json.loads(path.read_text(encoding="utf-8"))
    feats = data.get("features") or []
    keys = set()
    for feat in feats:
        keys.update((feat.get("properties") or {}).keys())
    return sorted(keys), len(feats)


def _geojson_columns(con, file_regex, schema, dataset):
    """Sniffed column rows for a GeoJSON.

    Uses DuckDB ST_Read for type inference where possible, but supplements it
    with a Python scan of all property keys. This prevents sparse or empty JSON
    properties (e.g. cell_beds_json on water bodies with no cells) from being
    silently dropped from the inferred schema.
    """
    st_read_types = {}
    try:
        con.execute("INSTALL spatial; LOAD spatial;")   # INSTALL caches the binary
        safe = str(file_regex).replace("'", "''")
        for name, dtype, *_rest in con.execute(
                f"DESCRIBE SELECT * FROM ST_Read('{safe}')").fetchall():
            st_read_types[name] = dtype
    except Exception:
        pass

    try:
        keys, n_features = _scan_geojson(file_regex)
    except Exception:
        keys, n_features = [], 0

    if n_features == 0:
        # Empty FeatureCollection: the file HAS no columns. Per tyr's design a
        # dataset MUST have declared columns, so callers treat "no columns" as
        # "do not declare this dataset" (an empty file has nothing to stage).
        # Never invent ST_Read's phantom OGC_FID/geom bookkeeping schema, and
        # never invent a wildcard '*'.
        return []

    # Merge ST_Read's inferred columns (keeps geom / OGC_FID) with the Python
    # scan of all property keys (keeps sparse/empty JSON properties).
    merged = list(st_read_types.keys()) + [k for k in keys if k not in st_read_types]
    return [_col_row(schema, dataset, k, k,
                     _vtype_for(st_read_types.get(k, "VARCHAR")),
                     st_read_types.get(k, "VARCHAR"), i)
            for i, k in enumerate(merged)]


def _col_row(schema, dataset, name, alias, var_type, data_type, ordinal):
    row = {c: "" for c in COLUMN_META_COLUMNS}

               var_type=var_type, data_type=data_type, on_null="PASS", on_filter="PASS",
               is_primary_key="False", is_event_time="False", ordinal_position=ordinal)
    return row

def classify_dataset(column_meta_rows):
    """Derive a dataset's data-layer shape from its tyr column-metadata rows (as read by
    :py:func:`read_meta` -- one dict per column, ``is_primary_key`` / ``is_event_time`` as
    'True'/'False' strings, ``var_type`` from :py:data:`VAR_TYPES`). Pass the rows for a
    SINGLE dataset. Names reported are OUTPUT names (alias, else source name).

    Returns ``{kind, time_col, entity_keys, value_cols, problems}``:

    * ``kind`` -- ``'cell_series'`` when the dataset declares an ``is_event_time`` column
      (a time axis => animate over it), else ``'cell_field'`` (a static per-cell snapshot).
      The presence/absence of an event time is what routes the two layer kinds.
    * ``time_col`` -- the event-time column's name, or ``None``. The axis may be named
      anything -- :py:func:`export_cell_series` accepts the declared column under its own
      name and writes the series file with the canonical ``'event_ts'`` header.
    * ``entity_keys`` -- the ``is_primary_key`` columns MINUS the event-time column: the
      static grain rows group by to form one timeline per entity. For a compound
      ``(entity, time)`` key this keeps only the entity half.
    * ``value_cols`` -- ``numeric`` columns that are neither key nor event-time: the
      measures that can be baked/animated (candidates; the exporter or GUI picks which).
    * ``problems`` -- actionable notes about the event-time setup only (empty => the
      time axis is unambiguous). A dataset with no numeric measures is a legitimate
      static dataset and is NOT a problem -- tyr stages whatever the metadata TSVs
      declare, and the Data Model grid already shows each column's var type.

    Null handling is inherited from tyr's ``on_null`` (a fill is an explicit user choice);
    this classifier does not touch it."""
    rows = list(column_meta_rows)
    event_times = [n for r in rows if _truthy(r.get("is_event_time")) and (n := _out_name(r))]
    entity_keys = [n for r in rows if _truthy(r.get("is_primary_key"))
                   and not _truthy(r.get("is_event_time")) and (n := _out_name(r))]
    value_cols = [n for r in rows
                  if (r.get("var_type") or "").strip().lower() == "numeric"
                  and not _truthy(r.get("is_primary_key"))
                  and not _truthy(r.get("is_event_time")) and (n := _out_name(r))]

    time_col = event_times[0] if event_times else None
    kind = "cell_series" if time_col else "cell_field"
    ds = (rows[0].get("dataset") or "").strip() if rows else ""
    prefix = f"{ds}: " if ds else ""
    problems = []
    if kind == "cell_series":
        if not entity_keys:
            problems.append(
                f"{prefix}column '{time_col}' is marked as the event time, but no column "
                "is marked as a primary key -- each row is treated as its own timeline")
        if len(event_times) > 1:
            problems.append(f"{prefix}{len(event_times)} columns are marked as the event "
                            f"time; using '{time_col}' as the time axis")
        et_row = next((r for r in rows if _out_name(r) == time_col
                       and _truthy(r.get("is_event_time"))), None)
        if et_row and (et_row.get("var_type") or "").strip().lower() != "timestamp":
            problems.append(
                f"{prefix}column '{time_col}' is marked as the event time but its var "
                "type is not 'timestamp' -- set the var type to timestamp (or CAST it) "
                "so it can be the time axis")
    return {"kind": kind, "time_col": time_col, "entity_keys": entity_keys,
            "value_cols": value_cols, "problems": problems}

def plan_series_export(result_columns, event_time_names, key="h3_id"):
    """Decide how a query result should be written, from its column names alone. If the
    result carries the ``key`` (h3_id), exactly one event-time column, and at least one
    other column to use as the value, it is a ``cell_series`` -- returns
    ``('cell_series', time_col, value_col)`` using the first such value column. Otherwise
    ``('cell_field', None, None)`` (the static-field path). The single-event-time rule
    keeps the routing unambiguous; a result with two event-time columns falls back to
    fields rather than guessing the axis."""
    if key not in result_columns:
        return ("cell_field", None, None)
    times = [c for c in result_columns if c in event_time_names]
    if len(times) != 1:
        return ("cell_field", None, None)
    time_col = times[0]
    value_col = next((c for c in result_columns if c not in (key, time_col)), None)
    if value_col is None:
        return ("cell_field", None, None)
    return ("cell_series", time_col, value_col)




def _load_extensions(con):
    """LOAD every extension (best effort -- a missing optional extension logs but
    doesn't block). Shared by connect() and connect_wait().

    A copy shipped under apian_terrain/_extensions/ is LOADed by explicit path
    first: a fresh customer machine has no ~/.duckdb extension cache, and the
    community h3 binary is not always reachable/installable from the extension
    repo (clean-container smoke 2026-09-23: INSTALL left no file behind and
    every connection printed "extension 'h3' unavailable"). Only when no
    bundled copy exists do we fall back to INSTALL (which hits the network)."""
    for name, origin in EXTENSIONS:
        bundled = Path(__file__).parent / "_extensions" / f"{name}.duckdb_extension"
        if bundled.is_file():
            try:
                # SQL-double any single quote: a Windows profile path may
                # legally contain an apostrophe (O'Brien), which flows in via
                # Path(__file__), and would break the string literal.
                path = bundled.as_posix().replace("'", "''")
                con.execute(f"LOAD '{path}'")
                continue
            except Exception as e:
                print(f"[tyrbuild] bundled extension '{name}' failed ({e}); "
                      f"falling back to INSTALL", file=sys.stderr)
        stmt = (f"INSTALL {name} FROM community; LOAD {name};" if origin == "community"
                else f"INSTALL {name}; LOAD {name};")
        try:
            con.execute(stmt)
        except Exception as e:                 # optional ext (e.g. st_read_multi) absent
            print(f"[tyrbuild] extension '{name}' unavailable: {e}", file=sys.stderr)

def connect(db_path, read_only=False, max_retries=5, retry_delay=0.2):
    """Open the per-area DuckDB and LOAD every extension (best effort -- a missing
    optional extension logs but doesn't block). Callers close it.

    Retries on Windows file-lock races when another process has just released
    (or is about to release) the database file.
    """
    path = str(db_path)
    last_error = None
    for attempt in range(max_retries):
        try:
            con = duckdb.connect(path, read_only=read_only)
            break
        except Exception as e:
            last_error = e
            if attempt < max_retries - 1:
                time.sleep(retry_delay * (attempt + 1))
    else:
        raise last_error

    _load_extensions(con)
    return con




_LOCK_PHRASES = ("being used by another process", "could not set lock",
                 "conflicting lock", "database is locked")


def _is_lock_error(e):
    msg = str(e).lower()
    return any(p in msg for p in _LOCK_PHRASES)

def connect_wait(db_path, read_only=False, timeout_s=300.0, poll_s=2.0):
    """Like connect(), but waits out Windows file-lock contention.

    Editor-facing scripts use this so a load/save initiated while a writer
    (e.g. write_output_schema.py) holds the DuckDB lock waits politely
    instead of failing with a raw duckdb IOException. Prints a one-line
    status to stderr once per poll interval while waiting. Raises
    RuntimeError with a clean message after timeout_s.
    """
    path = str(db_path)
    deadline = time.monotonic() + timeout_s
    last_print = 0.0
    while True:
        try:
            con = duckdb.connect(path, read_only=read_only)
            break
        except Exception as e:
            if not _is_lock_error(e):
                raise
            if time.monotonic() >= deadline:
                raise RuntimeError(
                    f"Database {path} stayed locked for more than {timeout_s:.0f}s "
                    f"(another Apian process is holding it). Close other Apian "
                    f"operations and try again.") from e
            now = time.monotonic()
            if now - last_print >= poll_s:
                last_print = now
                print(f"[tyrbuild] waiting for database lock on {Path(path).name} "
                      f"(another Apian process is writing)...", file=sys.stderr)
            time.sleep(poll_s)

    _load_extensions(con)
    return con




def _materialize(con, schema, log, skip=None):
    """Create a tyr schema in the DB from each table's tyr-authored SQL. tyr's
    settings.sql is empty in this build, so we DROP+CREATE the schema and
    CREATE TABLE per table (source before staging -- staging derives from source).

    `skip` (optional) is a predicate on the table name: skipped tables are not
    built (used for DB-native optional datasets tyr cannot rebuild from files,
    T5/D5). The drop+create runs in a transaction: if any table fails, the whole
    schema rebuild rolls back and the previous schema is left intact (a half-built
    schema is worse than none -- see the austevoll structures incident)."""
    con.execute("BEGIN TRANSACTION")
    try:
        con.execute(f"DROP SCHEMA IF EXISTS {schema.name} CASCADE")
        con.execute(f"CREATE SCHEMA {schema.name}")
        for table in schema.tables.list_tables():
            if skip is not None and skip(table.name):
                log(f"  [{schema.name}] {table.name}: skipped (optional, no source file)")
                continue
            con.execute(f"CREATE TABLE {schema.name}.{table.name} AS {table.sql}")
            n = con.execute(f"SELECT count(*) FROM {schema.name}.{table.name}").fetchone()[0]
            log(f"  [{schema.name}] {table.name}: {n} rows")
        con.execute("COMMIT")
    except Exception:
        con.execute("ROLLBACK")
        raise


def _db_native_optional(fm_rows):
    """Datasets declared required=false (DB-native datasets like CHELSA with no
    source file, ingested straight into the DB, plus optional file-backed
    datasets whose declared file is simply absent). Tyr cannot rebuild these
    from the unified files, so source/staging materialisation skips them
    (T5/D5). The optional set keys off the file_metadata `required` flag ALONE
    -- an optional dataset stays optional even when it declares a file_regex;
    otherwise an absent declared file would wedge the whole run with a
    RuntimeError despite the dataset opting out."""
    return {(r.get("dataset") or "").strip() for r in fm_rows
            if not _meta_required(r)}


def _materialize_source_preserving_db_native(con, source, db_native, log):
    """_materialize(source) that leaves DB-native optional datasets (CHELSA:
    required=false, no unified file) intact -- the schema DROP would take them
    down and tyr cannot rebuild them from a file."""
    keep = {t: con.execute(f"SELECT * FROM source.{t}").df()
            for t in db_native if _table_exists(con, "source", t)}
    _materialize(con, source, log, skip=lambda n: n in db_native)
    for t, df in keep.items():
        con.register(f"_preserve_{t}", df)
        try:
            con.execute(f"CREATE OR REPLACE TABLE source.{t} AS "
                        f"SELECT * FROM _preserve_{t}")
        finally:
            con.unregister(f"_preserve_{t}")

def build_schemas(config_dir, db_path=None):
    """Build the tyr Source + Staging objects from the metadata TSVs (no DB yet).

    If `db_path` is supplied, prefer manifest-backed source/pipelines metadata paths
    declared in the region manifest; otherwise fall back to the legacy TSVs in
    `config_dir`.
    """
    fm_path, cm_path = _resolve_meta_paths(config_dir, db_path)
    src_mod = tyr.lineage.schema.source
    _file_meta = src_mod.read_file_metadata(str(fm_path))
    _col_meta = src_mod.read_column_metadata(str(cm_path), schema="staging")
    # tyr requires every staged dataset to have at least one declared column
    # (its lineage model). A dataset whose metadata has zero column rows
    # (e.g. its source file is empty) or still carries a legacy '*' wildcard
    # row can never be staged -- SKIP it with a loud warning instead of
    # failing the run: an empty file is empty, not an error. Regenerate File
    # Metadata drops such datasets; Regenerate Column Metadata drops stale
    # rows -- either cleans the TSVs for the next run.
    _skipped = {}   # dataset -> reason
    for _ds in list(_file_meta):
        _rows = _col_meta.get(_ds) or {}
        if not _rows:
            _skipped[_ds] = "no column metadata -- empty source file?"
        elif any(str(c.column_name).strip() == "*" for c in _rows.values()):
            _skipped[_ds] = "legacy '*' wildcard column row"
    if _skipped:
        print("[tyrbuild] WARNING: skipping undeclarable dataset(s): "
              + "; ".join(f"{ds} ({why})" for ds, why in _skipped.items()),
              file=sys.stderr)
        _file_meta = {ds: f for ds, f in _file_meta.items()
                      if ds not in _skipped}
        _col_meta = {ds: rows for ds, rows in _col_meta.items()
                     if ds in _file_meta}
    source = src_mod.Source(settings=src_mod.SourceSettings(
        file_metadata=_file_meta,
        # The staging build consumes these rows (StagingSettings name="staging";
        # every production area tags its rows "staging") -- pass that schema so
        # rows tagged for a foreign schema cannot drive the staging transform.
        expected_column_metadata=_col_meta,
    ))
    staging = tyr.lineage.schema.staging.Staging(
        source=source,
        settings=tyr.lineage.schema.staging.StagingSettings(name="staging"))
    return source, staging

def run_pipeline(config_dir, db_path, log=print):
    """Materialise tyr Staging from the EXISTING source schema, then carry
    staging -> project so project is ALWAYS constructed from the freshly
    materialised staging (previously tyr left project stale, and a
    value edited/defaulted in staging never reached project). Water sim
    parameters are re-preprocessed from the new project.water_bodies.

    A source schema that already exists is left untouched -- run_pipeline
    never drops or rebuilds one. When (and only when) source tables are
    ABSENT, it materialises them from the area's metadata TSVs first
    (previously this hard-failed with "run Source for this area first",
    advice no production UI could follow -- T2/D2). The explicit error
    survives only for the case where that materialisation itself cannot
    produce the tables. OPTIONAL datasets (declared required=false, e.g.
    CHELSA -- T5/D5) never count toward the missing set and are skipped by
    the staging rebuild, so an absent optional table cannot wedge the
    pipeline. Staging's rebuild is transactional, so a failure leaves the
    previous staging intact.
    Returns {'staging': [table...], 'carried': [...]}."""
    source, staging = build_schemas(config_dir, db_path)
    fm_rows = read_meta(_resolve_meta_paths(config_dir, db_path)[0], FILE_META_COLUMNS)
    db_native = _db_native_optional(fm_rows)
    con = connect(db_path)
    try:
        # Every REQUIRED dataset the metadata declares must exist as a source
        # table -- staging SQL reads source.<dataset>, never the files.
        # Optional (required=false) datasets are DB-native; they never count
        # toward the missing set (T5/D5).
        def _missing():
            return [d for d in source.tables.list_names()
                    if d not in db_native
                    and con.execute(
                        "SELECT 1 FROM information_schema.tables WHERE table_schema = ? "
                        "AND table_name = ?", ["source", d]).fetchone() is None]
        missing = _missing()
        if missing:
            log(f"[tyrbuild] source schema is missing table(s): {', '.join(missing)}"
                " -- materialising source from the area's metadata TSVs ...")
            try:
                _materialize_source_preserving_db_native(con, source, db_native, log)
            except Exception as e:
                raise RuntimeError(
                    f"source schema could not be materialised: {e}") from e
            still_missing = _missing()
            if still_missing:
                raise RuntimeError(
                    f"source schema could not be materialised; table(s) still "
                    f"missing: {', '.join(still_missing)}")
        log("[tyrbuild] materialising staging (existing source schema left untouched) ...")
        _materialize(con, staging, log, skip=lambda n: n in db_native)
        log("[tyrbuild] carrying staging -> project ...")
        carried = carry_tyr_staging_to_project(con, log=log)
        # Water sim initial conditions against the just-carried bodies (only
        # when the area has a cell grid -- ad-hoc tyr datasets may not).
        # Lazy import: hydro_sim imports tyrbuild at module level (circular).
        from . import hydro_sim
        if con.execute("SELECT 1 FROM information_schema.tables "
                       "WHERE table_schema='project' AND table_name='cells'").fetchone():
            _manifest = Path(db_path).with_name(
                f"{Path(db_path).stem}_manifest.json")
            pre = hydro_sim.preprocess_water_params(
                con, session_path=hydro_sim.session_water_path(_manifest)
                if _manifest.exists() else None)
            if pre["n_rate_cells"] or pre["n_head_cells"]:
                log(f"  [water] preprocessed {pre['n_rate_cells']} rate + "
                    f"{pre['n_head_cells']} head cell(s)")
        # Publish project + edits into output. output is ALWAYS rewritten --
        # never preserved -- so no edit can bake into it (DEFCON-1 fix).
        rewrite_output(con, log=log)
    finally:
        con.close()
    return {"staging": staging.tables.list_names(),
            "carried": carried}

def validate_output(output_dir, area_id, region_dir=None):
    """Importer-contract gate on the produced files: (1) cells.csv header ==
    unified.CELL_COLUMNS, (2) every declared layer passes its KIND's validator
    (layers.validate_layers -- dispatched, so a new data kind is checked by its own
    rule), (3) the REAL importer read path (load_for_import) parses them. Returns
    (ok, [messages]). ASCII-only messages (Windows console safe).

    `output_dir` is where the layer data files (feature geojsons, color geojson)
    live; `region_dir` (defaults to `output_dir`) is the region folder holding
    the manifest + cells csv -- the region-folder layout keeps identity in the
    region root and derived files in output/."""
    out = Path(output_dir)
    region = Path(region_dir) if region_dir else out
    msgs, ok = [], True

    def check(label, cond):
        nonlocal ok
        ok = ok and bool(cond)
        msgs.append(("PASS " if cond else "FAIL ") + label)

    try:
        manifest = unified.read_manifest(region / f"{area_id}_manifest.json")
        with open(region / manifest["cells_csv"], newline="", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            check("cells.csv header == CELL_COLUMNS",
                  (reader.fieldnames or []) == unified.CELL_COLUMNS)
            cell_ids = {row["h3_id"] for row in reader}
        # per-layer, kind-dispatched checks (the modular part)
        lok, lmsgs = layers.validate_layers(out, manifest, cell_ids)
        ok = ok and lok
        msgs += lmsgs
        # the ultimate gate: the real importer read path parses them. A
        # provisional manifest (first-build pipeline path, pre-overwrite) is
        # exempt: load_for_import rightly refuses it, so rejecting here would
        # resurrect the bogus-FAIL bug -- WARN instead (91a4c9d).
        if manifest.get("provisional"):
            msgs.append("WARN provisional manifest (area build incomplete)")
        else:
            fbt = unified.load_for_import(region, area_id, data_dir=out)
            check("load_for_import parsed the dataset", bool(fbt.get("terrain")))
            # Height coverage: the importer's extruder silently skips structures
            # with no height_m. Not a contract failure — filling via a column
            # metadata on_null=DEFAULT policy is a deliberate per-area choice --
            # but a forgotten default quietly loses the whole town, so surface
            # the proportion with the fix spelled out.
            structs = (fbt.get("structures") or [{}])[0].get("properties", {}).get("structures", [])
            if structs:
                n_no_h = sum(1 for s in structs if s.get("height_m") is None)
                if n_no_h:
                    msgs.append(
                        f"WARN structures: {n_no_h}/{len(structs)} "
                        f"({100.0 * n_no_h / len(structs):.1f}%) have no height_m and will "
                        "NOT import -- set on_null=DEFAULT with a default_value on "
                        "staging.structures.height_m in column metadata to fill them")
    except Exception as e:
        check("load_for_import raised: " + str(e), False)
    return ok, msgs

def export_data_fields(output_dir, area_id, columns, rows, key="h3_id", interp_col_prefix="__interp_",
                       region_dir=None):
    """Write query results (an `h3_id` column + >=1 numeric value column) as
    data_<field> celldata onto the area's unified files, then validate them.
    Boolean columns named `<interp_col_prefix><value col>` (the Output interpolation
    convention) are not exported as values -- they mark synthesised cells, written as
    the celldata `<field>__i` flags. Returns (metas, (ok, messages)). Raises
    ValueError on a malformed result. `region_dir` (defaults to `output_dir`)
    locates the manifest for the data-layer merge."""
    mdir = Path(region_dir) if region_dir is not None else Path(output_dir)
    if key not in columns:
        raise ValueError(f"query result needs an '{key}' column (got {columns})")
    ki = columns.index(key)
    interp_cols = {c[len(interp_col_prefix):]: c for c in columns
                   if c.startswith(interp_col_prefix)}
    value_cols = {c: cb.slug_field(c) for c in columns
                  if c != key and c not in interp_cols.values()}
    if not value_cols:
        raise ValueError(f"query result has no value columns besides '{key}'")
    col_idx = {c: columns.index(c) for c in value_cols}
    field_maps = {slug: {} for slug in value_cols.values()}
    interp_maps = {slug: set() for slug in value_cols.values()}
    for row in rows:
        hid = row[ki]
        if hid is None:
            continue
        for col, slug in value_cols.items():
            v = row[col_idx[col]]
            if v is None:
                continue
            try:
                field_maps[slug][hid] = float(v)
            except (TypeError, ValueError):
                pass
            icol = interp_cols.get(col)
            if icol and row[columns.index(icol)]:
                interp_maps[slug].add(hid)
    fname = cb.write_celldata(output_dir, area_id, field_maps,
                              interp_maps={k: v for k, v in interp_maps.items() if v})
    metas = []
    for slug, m in field_maps.items():
        vals = list(m.values())
        metas.append({"name": slug, "key": "query", "aggregation": "sum", "units": "",
                      "min": min(vals) if vals else 0.0, "max": max(vals) if vals else 0.0,
                      "count": len(vals), "interp": "none", "radius_km": 0.0,
                      "domain": "all", "interp_count": 0})
    unified.add_data_fields(mdir / f"{area_id}_manifest.json", fname, metas)
    return metas, validate_output(output_dir, area_id, region_dir=region_dir)

def export_cell_series(output_dir, area_id, columns, rows, *, value_col, time_col,
                       key="h3_id", field=None, dtype=sr.DTYPE, interp_col=None,
                       region_dir=None):
    """Write a query result -- one row per (cell, timestep) -- as a `cell_series` layer
    (series.py) onto the area's unified files, then validate them. The result needs a
    `key` (h3_id) column, a `time_col` (the timestep), and a `value_col`. `field` names
    the baked attribute (defaults to the slugged value_col). `interp_col`, when given,
    names a boolean column marking rows SYNTHESISED upstream (interpolation/resampling);
    those records get the series' `.interp` handler so they stay distinct from measured
    data. Returns (layer, (ok, msgs)).

    CONTRACT: the time axis is whatever column the caller declares as `time_col`
    (its own name -- no fixed name is demanded), but its values must be real
    TIMESTAMPs (Python datetime, i.e. a DuckDB TIMESTAMP -- a string that merely
    looks like a date is refused). The written series file always carries the
    canonical `event_ts` header (sr.EVENT_TS_COLUMN), whatever the source column
    was called. Raises ValueError on non-timestamp values, or on a malformed /
    oversized / precision-unsafe grid."""
    from datetime import datetime as _dt
    tvals = [r[columns.index(time_col)] for r in rows]
    if not tvals or not all(isinstance(v, _dt) for v in tvals):
        raise ValueError(
            f"column '{time_col}' is the time axis but holds non-timestamp values -- "
            "CAST it to TIMESTAMP (a string that looks like a date is not a TIMESTAMP "
            "object and is refused)")
    field = cb.slug_field(field or value_col)
    cell_ids, time_labels, triples = sr.build_series(
        columns, rows, value_col=value_col, time_col=time_col, key=key)
    if not cell_ids or not time_labels:
        raise ValueError("query result has no (cell, timestep) rows to write")
    interp_flags = None
    if interp_col is not None:
        # flags key on (step, cell) in build_series' index space: its orders are
        # sorted-distinct over the same rows, so rebuild them identically here
        ki, ti, vi, ii = (columns.index(c) for c in (key, time_col, value_col, interp_col))
        step_of = {str(t): i for i, t in enumerate(time_labels)}
        col_of = {c: i for i, c in enumerate(cell_ids)}
        interp_flags = set()
        for row in rows:
            hid, t, v = row[ki], row[ti], row[vi]
            if hid is None or t is None or v is None or not row[ii]:
                continue
            try:
                fv = float(v)
            except (TypeError, ValueError):
                continue
            if math.isnan(fv):
                continue
            interp_flags.add((step_of[str(t)], col_of[hid]))
    layer = sr.write_series(output_dir, area_id, field, cell_ids, time_labels, triples,
                            dtype=dtype, time_col=sr.EVENT_TS_COLUMN,
                            interp_flags=interp_flags)
    unified.add_layer((Path(region_dir) if region_dir is not None
                       else Path(output_dir)) / f"{area_id}_manifest.json", layer)
    return layer, validate_output(output_dir, area_id, region_dir=region_dir)

def parse_interval_grid(start, end, interval):
    """Expand a SQL INTERVAL into the resample grid it defines over [start, end].

    `interval` is DuckDB interval text ('1 day', '1 month 1 day', '90 minutes', ...) and
    `start`/`end` are ISO datetime strings. Runs `CAST(<interval> AS INTERVAL)` first so a
    bad interval fails fast -- ValueError('invalid interval ...') -- then expands it with
    `generate_series`, which is calendar-correct ('1 month' steps by months, not 30 days).
    Values are bound as parameters (never string-formatted into SQL). Returns the grid as
    float ordinals (seconds since 1970-01-01, the coordinate series.resample_series' datetime
    axis uses) to hand straight to `resample_series(grid_ords=...)`. Raises ValueError on a
    bad interval or an empty grid (end before start)."""
    con = duckdb.connect()             # in-memory: interval parsing needs no extensions
    try:
        try:
            con.execute("SELECT CAST(? AS INTERVAL)", [interval])
        except Exception as e:
            raise ValueError(f"invalid interval {interval!r}: {e}")
        rows = con.execute(
            "SELECT unnest(generate_series(CAST(? AS TIMESTAMP), CAST(? AS TIMESTAMP), "
            "CAST(? AS INTERVAL)))", [str(start), str(end), interval]).fetchall()
    finally:
        con.close()
    if not rows:
        raise ValueError(f"interval {interval!r} yields an empty grid over "
                         f"[{start}, {end}] (is end before start?)")
    return [(r[0] - _EPOCH).total_seconds() for r in rows]

def detect_key_resolution(db_path, dataset, column):
    """The modal H3 resolution of a staging column's cells (plus how many distinct
    resolutions occur) -- the Output tab needs it to reconcile a keyed dataset to
    the grid. Raises ValueError if the column holds no parseable cells."""
    con = connect(db_path)
    try:
        row = con.execute(
            f"SELECT mode(h3_get_resolution({_q(column)})), "
            f"       count(DISTINCT h3_get_resolution({_q(column)})) "
            f"FROM staging.{_q(dataset)} WHERE {_q(column)} IS NOT NULL").fetchone()
    finally:
        con.close()
    if row is None or row[0] is None:
        raise ValueError(f"no parseable H3 cells in staging.{dataset}.{column}")
    return int(row[0]), int(row[1])

def build_output_sql(spec, grid_res, data_res=None):
    """Generate the `output.data` SELECT from a builder spec:

      {"dataset": <staging table>,
       "key":   {"mode": "latlon", "lat": c, "lon": c}
              | {"mode": "column", "column": c},
       "time":  {"mode": "column", "column": c, "trunc": <TRUNC_UNITS>}
              | {"mode": "constant", "value": "YYYY-MM-DD[ HH:MM:SS]"},
       "values": [{"func": <AGGREGATES>|"CUSTOM", "column": c|"", "alias": a,
                   "sql": raw}, ...]}

    `grid_res` is the manifest's h3_resolution -- the ONLY resolution the grid takes;
    lat/lon binning lands there directly. A keyed column is reconciled to it
    (`data_res`, see detect_key_resolution): finer data rolls UP via
    h3_cell_to_parent; coarser data spreads DOWN over h3_cell_to_children -- where
    SUM/COUNT auto-split by the row's child count so extensive totals conserve, and
    AVG/MIN/MAX/MEDIAN broadcast (the custombuild sum-splits/mean-broadcasts
    convention). Raises ValueError on a malformed spec."""
    ds = (spec.get("dataset") or "").strip()
    if not ds:
        raise ValueError("no dataset selected")
    grid_res = int(grid_res)

    key = spec.get("key") or {}
    spread = False
    if key.get("mode") == "column":
        col = (key.get("column") or "").strip()
        if not col:
            raise ValueError("key: no column selected")
        if data_res is None or data_res == grid_res:
            key_sql = f"{_q(col)}" + ("" if col == "h3_id" else f" AS {_q('h3_id')}")
        elif data_res > grid_res:                # finer data -> roll up to the parent
            key_sql = f"h3_cell_to_parent({_q(col)}, {grid_res}) AS h3_id"
        else:                                    # coarser data -> spread over children
            spread = True
            key_sql = "u.child AS h3_id"
    else:                                        # default: bin lat/lon at the grid res
        lat, lon = (key.get("lat") or "").strip(), (key.get("lon") or "").strip()
        if not (lat and lon):
            raise ValueError("key: need lat + lon columns")
        key_sql = (f"h3_latlng_to_cell_string({_q(lat)}, {_q(lon)}, {grid_res}) "
                   f"AS h3_id")

    tm = spec.get("time") or {}
    if tm.get("mode") == "constant":
        value = (tm.get("value") or "").strip()
        try:
            datetime.fromisoformat(value)
        except ValueError:
            raise ValueError(f"time: constant {value!r} is not an ISO datetime")
        time_sql = f"CAST({_l(value)} AS TIMESTAMP) AS {sr.EVENT_TS_COLUMN}"
    else:                                        # default: a timestamp column
        col = (tm.get("column") or "").strip()
        trunc = (tm.get("trunc") or "").strip()
        if not col:
            raise ValueError("time: no timestamp column selected")
        if trunc and trunc not in TRUNC_UNITS:
            raise ValueError(f"time: trunc must be one of {TRUNC_UNITS}, got {trunc!r}")
        inner = f"date_trunc({_l(trunc)}, {_q(col)})" if trunc else _q(col)
        time_sql = f"{inner} AS {sr.EVENT_TS_COLUMN}" if col != sr.EVENT_TS_COLUMN or trunc \
            else _q(sr.EVENT_TS_COLUMN)

    values = spec.get("values") or []
    if not values:
        raise ValueError("add at least one value column")
    val_sql = []
    for v in values:
        func = (v.get("func") or "").strip().upper()
        alias = (v.get("alias") or "").strip()
        if not alias:
            raise ValueError(f"value: {func or '?'} needs an alias")
        if func == "CUSTOM":
            # The escape hatch: a raw SQL expression spliced in verbatim (it lives
            # inside the generated GROUP BY query, so it must still be an aggregate
            # or group constant -- DuckDB will say so if not). Same trust level as
            # the SQL console: the user's own database, their own SQL.
            raw = (v.get("sql") or "").strip()
            if not raw:
                raise ValueError("value: Custom SQL needs an expression")
            val_sql.append(f"{raw} AS {_q(alias)}")
            continue
        if func not in AGGREGATES:
            raise ValueError(f"value: func must be one of {AGGREGATES} or 'CUSTOM', "
                             f"got {func!r}")
        col = (v.get("column") or "").strip()
        if spread and func in ("SUM", "COUNT"):
            # Coarser data spread over children: split each record's contribution by
            # its child count so extensive totals conserve (the celldata sum-splits
            # convention). AVG/MIN/MAX/MEDIAN broadcast (exact for intensives);
            # COUNT_DISTINCT broadcasts too (distincts aren't additive anyway).
            # CUSTOM is verbatim either way; in spread mode `child` + `t._n_kids`
            # are in scope.
            if func == "SUM":
                val_sql.append(f"sum({_q(col)} / t._n_kids) AS {_q(alias)}")
            else:
                val_sql.append((f"sum(1.0 / t._n_kids) AS {_q(alias)}" if not col else
                                f"sum(({_q(col)} IS NOT NULL)::DOUBLE / t._n_kids) "
                                f"AS {_q(alias)}"))
            continue
        if func == "COUNT" and not col:
            val_sql.append(f"count(*) AS {_q(alias)}")
        else:
            if not col and func != "CUSTOM":
                raise ValueError(f"value: {func} needs a column")
            inner = f"DISTINCT {_q(col)}" if func == "COUNT_DISTINCT" else _q(col)
            val_sql.append(f"{_AGG_SQL[func]}({inner}) AS {_q(alias)}")

    select = ",\n       ".join([key_sql, time_sql] + val_sql)
    if not spread:
        return f"SELECT {select}\nFROM staging.{_q(ds)}\nGROUP BY 1, 2"
    col = (key.get("column") or "").strip()
    return (f"SELECT {select}\n"
            f"FROM (SELECT *, array_length(h3_cell_to_children({_q(col)}, {grid_res})) "
            f"AS _n_kids FROM staging.{_q(ds)}) t,\n"
            f"     UNNEST(h3_cell_to_children(t.{_q(col)}, {grid_res})) AS u(child)\n"
            f"GROUP BY 1, 2")




def _check_output_data(con, table=OUTPUT_TABLE, require_ts=True, schema="output"):
    """The output contract, checked against the DB: the table exists; h3_id column;
    event_ts column typed TIMESTAMP; >=1 further value column (`__interp_*` flag
    columns don't count). Returns (ok, msgs, info) with info = {columns, value_cols,
    interp_cols, rows, n_cells, n_steps, min_ts, max_ts}. require_ts=False (the
    interpolator's source-table contract) tolerates a missing event_ts -- a STATIC
    source, where time resample is impossible and n_steps is 0 / the ts fields None.
    `schema` selects the checked schema (default output, the built contract table;
    the interpolator checks its project.<source_table> with schema="project")."""
    cols = [r[0] for r in con.execute(
        "SELECT column_name FROM information_schema.columns WHERE table_schema=? "
        "AND table_name=? ORDER BY ordinal_position", [schema, table]).fetchall()]
    if not cols:
        return False, [f"{schema}.{table} does not exist -- Build it first"], {}
    types = {r[0]: r[1] for r in con.execute(
        "SELECT column_name, data_type FROM information_schema.columns "
        "WHERE table_schema=? AND table_name=?", [schema, table]).fetchall()}
    ok, msgs = True, []
    if "h3_id" not in cols:
        ok, msgs = False, msgs + [f"FAIL {schema}.{table} needs an h3_id column"]
    elif (types.get("h3_id") or "").upper() not in ("VARCHAR", "CHAR", "TEXT"):
        # The grid and every fill domain key on the canonical H3 STRING; a
        # UBIGINT h3_id (easy to produce with the wrong h3 function, or raw
        # arithmetic) matches NOTHING downstream -- the space fill sees zero
        # sources and silently fills nothing. Fail HERE, naming the cause and
        # the fix, instead of emitting an empty-looking interpolation.
        ok, msgs = False, msgs + [
            f"FAIL {schema}.{table}.h3_id must be the canonical H3 STRING, got "
            f"{types['h3_id']} -- the region grid and the spatial fill key on "
            f"string cells, so {types['h3_id']} values match nothing and the "
            f"fill would silently produce no rows. Fix the SOURCE table, e.g. "
            f"recreate it with h3_latlng_to_cell_string(...) (or convert: "
            f"h3_id = lower(to_base(h3_id, 16))), then re-run"]
    if sr.EVENT_TS_COLUMN not in cols:
        if require_ts:
            ok, msgs = False, msgs + [f"FAIL {schema}.{table} needs an "
                                      f"{sr.EVENT_TS_COLUMN} column"]
    elif "TIMESTAMP" not in (types.get(sr.EVENT_TS_COLUMN) or "").upper():
        ok, msgs = False, msgs + [f"FAIL {sr.EVENT_TS_COLUMN} must be TIMESTAMP, got "
                                  f"{types.get(sr.EVENT_TS_COLUMN)}"]
    value_cols = [c for c in cols
                  if c not in ("h3_id", sr.EVENT_TS_COLUMN)
                  and not c.startswith(INTERP_PREFIX)]
    if not value_cols:
        ok, msgs = False, msgs + [f"FAIL {schema}.{table} needs >= 1 value column"]
    if not ok:
        return False, msgs, {}
    if sr.EVENT_TS_COLUMN not in cols:
        info = dict(zip(("rows", "n_cells"),
                        con.execute(f"SELECT count(*), count(DISTINCT h3_id) "
                                    f"FROM {schema}.{_q(table)}").fetchone()))
        info.update({"n_steps": 0, "min_ts": None, "max_ts": None})
    else:
        info = dict(zip(("rows", "n_cells", "n_steps", "min_ts", "max_ts"),
                        con.execute(f"SELECT count(*), count(DISTINCT h3_id), "
                                    f"count(DISTINCT {sr.EVENT_TS_COLUMN}), "
                                    f"min({sr.EVENT_TS_COLUMN}), max({sr.EVENT_TS_COLUMN}) "
                                    f"FROM {schema}.{_q(table)}").fetchone()))
    info["columns"], info["value_cols"] = cols, value_cols
    info["interp_cols"] = [c for c in cols if c.startswith(INTERP_PREFIX)]
    msgs.append(f"PASS {schema}.{table}: {info['rows']} rows, {info['n_cells']} cells, "
                f"{info['n_steps']} step(s), values {value_cols}")
    return True, msgs, info

def check_data_table_contract(con, table, schema="output"):
    """The data-table contract, checked against the DB -- the EXPORT path's
    gate (export_output_data / write_output_schema --table) and the custom-data
    REGISTRY's gate (registered tables must conform BEFORE registration).
    Enforced at the output level by default: ANY output.<table> may carry data,
    not just output.data. `schema` selects the checked schema -- the registry
    and the data-carrier build check project.<table> with schema="project"
    (Model B: registered custom data is a PROJECT-level input).

      * h3_id column REQUIRED -- missing fails the export with a clear message.
      * event_ts OPTIONAL: ABSENT means the data is STATIC -- exported as
        cell_field data fields (no time axis; the manifest layer is the static
        kind and the Blender carrier binds pure vertex data). PRESENT must be
        TIMESTAMP, else the export fails.
      * every other column is freeform with TWO restrictions: numeric or
        BOOLEAN DuckDB types only (the carrier is FLOAT-typed), and the name
        must not start with '_' (reserved for system attributes on the
        Blender Data Carrier). Offenders are named in the failure message.
        `__interp_<col>` BOOLEAN flag columns (the interpolation convention)
        are not values -- they ride along as the series .interp / celldata
        __i flags.

    Returns (ok, msgs, info) with info = {columns, value_cols, interp_cols,
    static, rows, n_cells, n_steps, min_ts, max_ts} (n_steps 0 / ts None when
    static)."""
    cols = [r[0] for r in con.execute(
        "SELECT column_name FROM information_schema.columns WHERE table_schema=? "
        "AND table_name=? ORDER BY ordinal_position", [schema, table]).fetchall()]
    if not cols:
        return False, [f"{schema}.{table} does not exist -- Build it first"], {}
    types = {r[0]: r[1] for r in con.execute(
        "SELECT column_name, data_type FROM information_schema.columns "
        "WHERE table_schema=? AND table_name=?", [schema, table]).fetchall()}
    ok, msgs = True, []
    if "h3_id" not in cols:
        ok, msgs = False, msgs + [f"FAIL {schema}.{table} needs an h3_id column"]
    elif (types.get("h3_id") or "").upper() not in ("VARCHAR", "CHAR", "TEXT"):
        # Same rule as _check_output_data: the grid keys on the canonical H3
        # STRING. Without this type pin the registry gate passed UBIGINT-keyed
        # tables that the interpolator then rejected -- the two gates must
        # agree or registration succeeds and the run still fails.
        ok, msgs = False, msgs + [
            f"FAIL {schema}.{table}.h3_id must be the canonical H3 STRING, got "
            f"{types['h3_id']} -- convert: h3_id = lower(to_base(h3_id, 16)), "
            f"or recreate with h3_latlng_to_cell_string(...), then re-run"]
    reserved = [c for c in cols
                if c.startswith("_") and not c.startswith(INTERP_PREFIX)
                and c not in ("h3_id", sr.EVENT_TS_COLUMN)]
    if reserved:
        ok, msgs = False, msgs + [
            "FAIL value columns may not start with '_' -- the prefix is "
            "reserved for system attributes on the Blender Data Carrier: "
            + ", ".join(reserved)]
    bad = [f"{c} ({types[c]})" for c in cols
           if c not in ("h3_id", sr.EVENT_TS_COLUMN)
           and (types[c] or "").upper() not in _DATA_VALUE_TYPES]
    if bad:
        ok, msgs = False, msgs + [
            "FAIL the data carrier is FLOAT-typed: value columns must be "
            "numeric or BOOLEAN, offending column(s): " + ", ".join(bad)]
    static = sr.EVENT_TS_COLUMN not in cols
    if not static and "TIMESTAMP" not in (types.get(sr.EVENT_TS_COLUMN) or "").upper():
        ok, msgs = False, msgs + [f"FAIL {sr.EVENT_TS_COLUMN} must be TIMESTAMP, got "
                                  f"{types.get(sr.EVENT_TS_COLUMN)} (drop the column for "
                                  "a static, cell_field export)"]
    value_cols = [c for c in cols
                  if c not in ("h3_id", sr.EVENT_TS_COLUMN)
                  and not c.startswith(INTERP_PREFIX)]
    if not value_cols:
        ok, msgs = False, msgs + [f"FAIL {schema}.{table} needs >= 1 value column"]
    if not ok:
        return False, msgs, {}
    if static:
        info = dict(zip(("rows", "n_cells"),
                        con.execute(f"SELECT count(*), count(DISTINCT h3_id) "
                                    f"FROM {schema}.{_q(table)}").fetchone()))
        info.update({"n_steps": 0, "min_ts": None, "max_ts": None})
    else:
        info = dict(zip(("rows", "n_cells", "n_steps", "min_ts", "max_ts"),
                        con.execute(f"SELECT count(*), count(DISTINCT h3_id), "
                                    f"count(DISTINCT {sr.EVENT_TS_COLUMN}), "
                                    f"min({sr.EVENT_TS_COLUMN}), max({sr.EVENT_TS_COLUMN}) "
                                    f"FROM {schema}.{_q(table)}").fetchone()))
    info["columns"], info["value_cols"] = cols, value_cols
    info["interp_cols"] = [c for c in cols if c.startswith(INTERP_PREFIX)]
    info["static"] = static
    msgs.append(f"PASS {schema}.{table}: {'STATIC ' if static else ''}"
                f"{info['rows']} rows, {info['n_cells']} cells, "
                f"{info['n_steps']} step(s), values {value_cols}")
    return True, msgs, info

def build_data_carrier_table(db_path, source_table, log=print):
    """Build the dedicated data-carrier table: materialise project.<source_table>
    into output.data_carrier (CREATE OR REPLACE), the table the export path's
    data layer resolves automatically (write_output_schema).

    CONTRACT-FIRST: check_data_table_contract runs BEFORE anything is written,
    so a missing/breached source raises ValueError with the contract messages
    and the pre-existing output.data_carrier (plus its sidecar) survives the
    failed build untouched. `source_table` must not be data_carrier itself,
    and must be REGISTERED in registry.custom_data (the custom-data registry
    is the binding contract for carrier sources -- register with
    scripts/register_custom_data.py); the gate runs before the contract check
    so an unregistered source names the register path, not just the breach.
    Model B: registered custom data is a PROJECT-level input -- the contract
    is checked against project.<source_table> and the source is read from
    project (the write still lands in output).

    Records a sidecar next to the manifest (<area>_data_carrier.json) with the
    source table, the build time (ISO 8601) and a cheap fingerprint
    {row_count, columns (sorted), min_ts, max_ts (null when static)} -- the
    Manager's Custom Data panel compares the source's CURRENT fingerprint
    against the recorded one to flag "source changed since last build".

    Returns {source_table, static, n_rows, value_cols, built_at}."""
    if source_table == DATA_CARRIER_TABLE:
        raise ValueError(f"source_table must not be '{DATA_CARRIER_TABLE}' itself")
    con = connect(db_path)
    try:
        creg.ensure_registry(con)
        if not creg.is_registered(con, source_table):
            raise ValueError(
                f"project.{source_table} is not registered in "
                f"registry.custom_data -- register it first: python "
                f"{creg.REGISTER_SCRIPT} \"<dbPath>\" {source_table}")
        ok, msgs, info = check_data_table_contract(con, source_table, schema="project")
        for m in msgs:
            log(f"  {m}")
        if not ok:
            raise ValueError(f"project.{source_table} fails the data-carrier contract:\n"
                             + "\n".join(msgs))
        con.execute("CREATE SCHEMA IF NOT EXISTS output")
        con.execute(f"CREATE OR REPLACE TABLE output.{_q(DATA_CARRIER_TABLE)} "
                    f"AS SELECT * FROM project.{_q(source_table)}")
        built_at = datetime.now(timezone.utc).isoformat()
        fingerprint = {
            "row_count": info["rows"],
            "columns": sorted(info["columns"]),
            "min_ts": info["min_ts"].isoformat() if info["min_ts"] is not None else None,
            "max_ts": info["max_ts"].isoformat() if info["max_ts"] is not None else None,
        }
        unified._atomic_json_dump(data_carrier_sidecar_path(db_path), {
            "source_table": source_table,
            "built_at": built_at,
            "fingerprint": fingerprint,
        }, indent=2)
        log(f"  [data-carrier] output.{DATA_CARRIER_TABLE} <- project.{source_table}: "
            f"{info['rows']} rows ({'STATIC' if info['static'] else 'series'})")
        return {"source_table": source_table, "static": info["static"],
                "n_rows": info["rows"], "value_cols": info["value_cols"],
                "built_at": built_at}
    finally:
        con.close()

def materialize_output(db_path, spec, grid_res, log=print):
    """(Re)build output.data from a builder spec and verify the contract. `grid_res`
    is the manifest's h3_resolution -- the only resolution the grid takes; a keyed
    column's own resolution is detected here and reconciled by the generator (see
    build_output_sql). Returns the info dict (rows/cells/steps/span); raises
    ValueError on a bad spec or a contract breach."""
    grid_res = int(grid_res)
    key = spec.get("key") or {}
    data_res = None
    if key.get("mode") == "column":
        data_res, n_res = detect_key_resolution(db_path, spec.get("dataset"),
                                                key.get("column") or "")
        if n_res > 1:
            log(f"  WARNING: the key column mixes {n_res} resolutions; reconciling "
                f"from the modal {data_res}")
        if data_res != grid_res:
            how = ("rolls up to parent cells" if data_res > grid_res else
                   "spreads over child cells (SUM/COUNT split to conserve totals)")
            log(f"  key column is res {data_res}, grid is res {grid_res}: {how}")
    sql = build_output_sql(spec, grid_res, data_res)
    con = connect(db_path)
    try:
        con.execute("CREATE SCHEMA IF NOT EXISTS output")
        con.execute(f"CREATE OR REPLACE TABLE output.{OUTPUT_TABLE} AS {sql}")
        ok, msgs, info = _check_output_data(con)
        for m in msgs:
            log(f"  {m}")
        if not ok:
            raise ValueError("output.data failed its contract:\n" + "\n".join(msgs))
        return info
    finally:
        con.close()

def export_output_data(output_dir, area_id, db_path, table=OUTPUT_TABLE,
                       region_dir=None):
    """Write an output.<table> to the unified files, routed by its own data
    (any output.<table> may carry data -- see check_data_table_contract, the
    gate this call enforces): no event_ts column, or a single distinct
    event_ts -> static data field(s) (the constant time axis is dropped);
    multiple -> one cell_series PER value column. `__interp_<col>` BOOLEAN
    columns (the interpolation convention) ride along as the celldata `__i` /
    series `.interp` flags, so synthesised records stay distinguishable from
    measured ones. Returns (written_names, (ok, msgs)) combining each write's
    validation report. `region_dir` (defaults to `output_dir`) locates the
    manifest for the data-layer merge -- the region layout keeps it at the
    region root, not in output/."""
    con = connect(db_path)
    try:
        ok, msgs, info = check_data_table_contract(con, table)
        if not ok:
            raise ValueError(f"output.{table} fails the data-carrier contract:\n"
                             + "\n".join(msgs))
        interp = {c[len(INTERP_PREFIX):]: c for c in info["interp_cols"]}
        if info["static"] or info["n_steps"] <= 1:

                    + [interp[c] for c in info["value_cols"] if c in interp])

                                            region_dir=region_dir)
            return [m["name"] for m in metas], rep
        written, all_ok, all_msgs = [], True, []

# Sourced from project.<source_table> (default project.data) -- h3_id the key,

# output at publish time) -- the result stays a consumer table in output only.
# Every synthesised record is flagged in a parallel __interp_<col>
# BOOLEAN column, which export carries into the series .interp handler / celldata

# A source table must be a bare identifier (it lands in project.<source_table>

_SOURCE_TABLE_IDENT = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*$")
# The single time-slice label for a STATIC source (no event_ts): space fill
# groups by label, so every record shares this one.
_STATIC_LABEL = "static"


def _check_source_table_ident(source_table):
    if not _SOURCE_TABLE_IDENT.match(source_table or ""):
        raise ValueError("source_table must be a simple identifier "
                         f"(letters, digits, underscore): {source_table!r}")


def _write_output_table(con, table, coldefs, rows):
    """(Re)create output.<table> with coldefs [(name, sqltype)] and insert rows
    (params, never string-formatted). Rows go in as a REGISTERED PANDAS
    DATAFRAME, not executemany: DuckDB's Python-object executemany runs at
    ~1k rows/s (measured 186 s for 215k rows -- minutes of dead time in a
    region-scale interpolation), a DataFrame insert ~100x faster."""
    import pandas as pd                     # hard dep already (tyr metadata)
    con.execute("CREATE SCHEMA IF NOT EXISTS output")
    con.execute(f'DROP TABLE IF EXISTS output."{table}"')
    con.execute(f'CREATE TABLE output."{table}" (' +
                ", ".join(f'"{n}" {t}' for n, t in coldefs) + ")")
    if not rows:
        return
    names = [n for n, _ in coldefs]
    df = pd.DataFrame(rows, columns=names)
    for (n, t) in coldefs:
        u = t.upper()
        if u == "DOUBLE":
            df[n] = pd.to_numeric(df[n], dtype_backend="numpy_nullable").astype("float64")
        elif u.startswith("TIMESTAMP") or u.startswith("DATE"):
            df[n] = pd.to_datetime(df[n])     # None -> NaT -> SQL NULL
        elif u == "BOOLEAN":
            df[n] = df[n].astype("boolean")   # nullable; None stays NULL
    con.register("_write_rows", df)
    try:
        con.execute(f'INSERT INTO output."{table}" SELECT * FROM _write_rows')
    finally:
        con.unregister("_write_rows")

def estimate_interpolate_output_data(db_path, *, cells, time=None, space=None,
                                     source_table=OUTPUT_TABLE,
                                     sample_cells=500, sample_slices=3):
    """Rough wall-clock estimate for the matching interpolate_output_data call, on THIS
    data and machine -- backs the GUI's Proceed? confirmation. Per phase:

      * contract check + COUNT aggregates -- DuckDB does the counting (fast);
      * full-table load, per-column dict build, triple build, merge + write -- linear
        phases, modelled with conservative per-op constants;
      * the time resample and the per-slice spatial fill -- MEASURED by running the
        real functions on a small sample (sampled cells / evenly spaced slices) and
        scaling by cells-with-data / grid steps.

    Spatial is sampled on SOURCE slices; post-resample slices are denser, so with both
    axes on the space figure is an upper bound. All estimates are rough -- treat as
    2-3x either way. Returns {"rows", "n_cells", "n_cols", "src_steps", "grid_steps",
    "out_rows", "seconds", "breakdown"}. Raises ValueError on the same bad specs the
    run would hit (no axes, broken contract, bad interval, time spec against a
    STATIC source) -- the confirmation doubles as instant validation."""
    if not time and not space:
        raise ValueError("nothing to do: give a time and/or a space spec")
    _check_source_table_ident(source_table)
    con = connect(db_path)
    try:
        ok, msgs, info = _check_output_data(con, source_table, require_ts=False,
                                            schema="project")
        if not ok:
            raise ValueError(f"project.{source_table} fails the contract:\n"
                             + "\n".join(msgs))
        series = sr.EVENT_TS_COLUMN in info["columns"]
        if time and not series:
            raise ValueError(f"project.{source_table} has no {sr.EVENT_TS_COLUMN} "
                             "column -- time resample needs a time axis")
        value_cols = info["value_cols"]
        ncols = len(value_cols)
        rows, n_cells = info["rows"], info["n_cells"]
        src_steps = info["n_steps"]
        grid_ords = None
        grid_steps = src_steps or 1          # a STATIC source has a single slice
        if time:
            grid_ords = parse_interval_grid(time["start"], time["end"],
                                            time["interval"])
            grid_steps = len(grid_ords)

        # per-column data shape: (non-null records, cells with data), one pass
        agg = ", ".join(f"count({_q(c)}), count(DISTINCT CASE WHEN {_q(c)} IS NOT NULL"
                        f" THEN h3_id END)" for c in value_cols)
        cs = con.execute(f"SELECT {agg} FROM project.{_q(source_table)}").fetchone()
        colstats = [(int(cs[2 * i]), int(cs[2 * i + 1])) for i in range(ncols)]

        K_ROW, K_DICT, K_MERGE = 1.5e-6, 6e-7, 1.0e-6    # per-op seconds, conservative
        load_s = rows * (2 + 2 * ncols) * K_ROW
        build_s = sum(r for r, _ in colstats) * K_DICT

        resample_s = 0.0
        if time:
            # measure the resampler on a cell sample; scale by cells-with-data
            samp = [r[0] for r in con.execute(
                f"SELECT DISTINCT h3_id FROM project.{_q(source_table)} LIMIT ?",
                [sample_cells]).fetchall()]
            sel = ", ".join(_q(c) for c in value_cols)
            marks = ", ".join(["?"] * len(samp))
            srows = con.execute(
                f"SELECT h3_id, {sr.EVENT_TS_COLUMN}, {sel} FROM "
                f"project.{_q(source_table)} WHERE h3_id IN ({marks})", samp).fetchall()
            for i, _vc in enumerate(value_cols):
                recs = {}
                for row in srows:
                    v = row[2 + i]
                    if v is not None:
                        recs[(str(row[1]), row[0])] = float(v)
                s_cells = sorted({h for _, h in recs})
                if not s_cells:
                    continue                         # sample has none of this column
                s_labels = sorted({lb for lb, _ in recs})
                li = {lb: k for k, lb in enumerate(s_labels)}
                ci = {h: k for k, h in enumerate(s_cells)}
                triples = [(li[lb], ci[h], v) for (lb, h), v in recs.items()]
                t0 = perf_counter()
                sr.resample_series(s_cells, s_labels, triples,
                                   method=time.get("method", "linear"),
                                   grid_ords=grid_ords)
                resample_s += (perf_counter() - t0) * (colstats[i][1] / len(s_cells))

        space_s = 0.0
        dom = None
        radius_m = 0.0
        if space:
            dom = cb.domain_ids(cells, (space.get("domain") or "all").lower())
            coords = cb.metric_coords(cells)
            radius_km = float(space.get("radius_km") or 0.0)
            if not radius_km and space.get("max_ring"):
                # legacy ring-step spec: approximate via the grid's mean spacing
                radius_km = (float(space["max_ring"])
                             * cb.grid_spacing_m(coords) / 1000.0)
            radius_m = radius_km * 1000.0
            labels_all = [r[0] for r in con.execute(
                f"SELECT DISTINCT {sr.EVENT_TS_COLUMN} FROM project.{_q(source_table)} "
                f"ORDER BY 1").fetchall()] if series else [None]
            if labels_all:
                step = max(1, len(labels_all) - 1) / max(1, sample_slices - 1)
                picks = sorted({labels_all[round(i * step)]
                                for i in range(sample_slices)})
                # time the fill on the DENSEST column (most work per slice)
                dense = value_cols[max(range(ncols), key=lambda i: colstats[i][0])]
                tsum = 0.0
                for lb in picks:
                    where = (f"{sr.EVENT_TS_COLUMN} = ? AND " if series else "") \
                        + f"{_q(dense)} IS NOT NULL"
                    params = [lb] if series else []
                    cm = {h: float(v) for h, v in con.execute(
                        f"SELECT h3_id, {_q(dense)} FROM project.{_q(source_table)} "
                        f"WHERE {where}", params).fetchall()}
                    t0 = perf_counter()
                    cb.interpolate_field(cm, dom, coords=coords, radius_m=radius_m,
                                         method=space.get("method", "idw"),
                                         anchor=bool(space.get("anchor", False)))
                    tsum += perf_counter() - t0
                space_s = (tsum / len(picks)) * grid_steps * ncols

        if space:
            # Radius-bounded methods (all three) fill only within radius_km of
            # a source: the whole grid is the pre-trim domain, NOT the output
            # size. The old model predicted grid_steps x len(dom) -- 41M rows
            # and a ~5-minute 'merge' for Sydney when the real write is ~2.6M.
            # Estimate the metric reach on subsamples: the fraction of a domain
            # subsample within radius of a source subsample, scaled up.
            src_cells = [r[0] for r in con.execute(
                f"SELECT DISTINCT h3_id FROM project.{_q(source_table)} LIMIT 300"
                ).fetchall()]
            if radius_m > 0 and src_cells and dom:
                dsample = sorted(dom)[:: max(1, len(dom) // 20000)]
                tpts = np.array([coords[c] for c in dsample], dtype=np.float64)
                spts = np.array([coords[c] for c in src_cells if c in coords],
                                dtype=np.float64)
                ti, _, _ = cb._pairs_within(tpts, spts, radius_m)
                frac = len(np.unique(ti)) / max(1, len(dsample))
                out_rows = grid_steps * int(len(dom) * frac)
            else:
                out_rows = 0
        elif time:
            out_rows = sum(c for _, c in colstats) * grid_steps
        else:
            out_rows = rows
        merge_s = out_rows * (1 + 2 * ncols) * K_MERGE

        breakdown = {"load": load_s, "build": build_s,
                     "resample": resample_s, "space": space_s, "merge": merge_s}
        return {"rows": rows, "n_cells": n_cells, "n_cols": ncols,
                "src_steps": src_steps, "grid_steps": grid_steps,
                "out_rows": out_rows, "seconds": sum(breakdown.values()),
                "breakdown": breakdown}
    finally:
        con.close()

def interpolate_output_data(db_path, *, cells, time=None, space=None,
                            out_table="data_interp", source_table=OUTPUT_TABLE,
                            add_is_interpolated=False, log=print):
    """Interpolate project.<source_table> along the opted-in axes and write
    output.<out_table> (same contract shape + __interp_<col> flags), leaving the
    source untouched.

    The source defaults to the enforced contract table project.data -- h3_id the
    cell key, event_ts the time axis, every further column a value column to
    interpolate (__interp_<col> flags ride along, they are not values). ANY
    project.<source_table> may be interpolated instead (Model B: registered
    custom data is a project-level input); a source with NO
    event_ts column is STATIC -- the time axis is impossible (a time spec
    raises) and the space fill treats the whole table as one slice.

      time  -- {"start","end","interval","method"}: resample every value column onto
               the SHARED grid (parse_interval_grid; calendar-correct), filling each
               cell's INTERIOR gaps by method ('linear'|'nearest') -- never its ends.
      space -- {"method","radius_km","domain","anchor"}: per (step, column) slice, fill
               the grid's empty in-domain cells from the slice's present cells on the
               2D metric plane (cb.interpolate_field). `radius_km` bounds the fill
               reach in kilometers (legacy `max_ring` grid steps are converted via
               the grid's mean cell spacing).
      add_is_interpolated -- also write a single BOOLEAN `is_interpolated` column,
               the OR of the row's __interp_<col> flags (FALSE when nothing in the
               row was synthesised). Off by default: the per-column flags already
               carry the information; the row-level column is a convenience for
               consumers that filter on "any derived value".

    `cells` = the region's cells.csv rows (the grid + land/water domains). Returns
    (out_table, {rows, n_cells, n_steps, synth}). Raises ValueError with no axes, a
    missing/contract-breaking project.<source_table>, a time spec against a STATIC
    source, or a bad spec. The result is a consumer table: it lands in output
    only and is NEVER registered (registry membership means "project input")."""
    if not time and not space:
        raise ValueError("nothing to do: give a time and/or a space spec")
    _check_source_table_ident(source_table)
    con = connect(db_path)
    try:
        ok, msgs, info = _check_output_data(con, source_table, require_ts=False,
                                            schema="project")
        if not ok:
            raise ValueError(f"project.{source_table} fails the contract:\n"
                             + "\n".join(msgs))
        series = sr.EVENT_TS_COLUMN in info["columns"]
        if time and not series:
            raise ValueError(f"project.{source_table} has no {sr.EVENT_TS_COLUMN} "
                             "column -- time resample needs a time axis")
        value_cols = info["value_cols"]
        interp_cols = {c[len(INTERP_PREFIX):]: c for c in info["interp_cols"]}
        sel = (["h3_id", sr.EVENT_TS_COLUMN] if series else ["h3_id"]) \
            + value_cols + list(interp_cols.values())
        rows = con.execute("SELECT " + ", ".join(_q(c) for c in sel)
                           + f" FROM project.{_q(source_table)}").fetchall()
        # per-column record sets, keyed (label, h3) -> [value, flagged]
        grid_ords = None
        if time:
            grid_ords = parse_interval_grid(time["start"], time["end"], time["interval"])
        dom = cb.domain_ids(cells, (space or {}).get("domain", "all").lower()) \
            if space else None
        coords = cb.metric_coords(cells) if space else None
        radius_km = 0.0
        if space:
            radius_km = float(space.get("radius_km") or 0.0)
            if not radius_km and space.get("max_ring"):
                # legacy ring-step spec: approximate via the grid's mean spacing
                radius_km = (float(space["max_ring"])
                             * cb.grid_spacing_m(coords) / 1000.0)
        base = 2 if series else 1             # h3_id + event_ts prefix width
        per_col = {}
        for vc in value_cols:
            flag_name = interp_cols.get(vc)
            # column positions in `rows`: h3_id, [event_ts,] value cols..., flags...
            vi = base + value_cols.index(vc)
            fi = (base + len(value_cols)
                  + list(interp_cols.values()).index(flag_name)) if flag_name else None
            recs = {}
            for row in rows:
                v = row[vi]
                if v is None:
                    continue
                lb = str(row[1]) if series else _STATIC_LABEL
                recs[(lb, row[0])] = [float(v), bool(row[fi]) if fi else False]
            per_col[vc] = recs

        out_recs = {}
        for vc, recs in per_col.items():
            cell_ids = sorted({h for _, h in recs})
            if time:
                src_flags = {k: f for k, (v, f) in recs.items() if f}
                labels = sorted({lb for lb, _ in recs})
                # dict lookups, not per-record list .index() scans -- the scans made
                # this O(R x (L + C)) and dominated big tables
                li = {lb: i for i, lb in enumerate(labels)}
                ci = {h: i for i, h in enumerate(cell_ids)}
                triples = [(li[lb], ci[h], v)
                           for (lb, h), (v, _f) in recs.items()]
                labels, triples, flags, warns = sr.resample_series(
                    cell_ids, labels, triples, method=time.get("method", "linear"),
                    grid_ords=grid_ords)
                for w in warns:
                    log(f"  [interpolate:{vc}] {w}")
                flag_map = {(labels[s], cell_ids[c]): True for s, c in flags}
                recs = {}
                for s, c, v in triples:
                    lb, h = labels[s], cell_ids[c]
                    # synthesised by this resample, OR already flagged upstream
                    # (chained interpolation keeps the honest mark)
                    recs[(lb, h)] = [v, flag_map.get((lb, h), False)
                                     or src_flags.get((lb, h), False)]
            if space:
                by_step = defaultdict(dict)
                for (lb, h), (v, f) in recs.items():
                    by_step[lb][h] = v
                for lb, cell_map in by_step.items():
                    filled, interp_ids = cb.interpolate_field(
                        cell_map, dom, coords=coords, radius_m=radius_km * 1000.0,
                        method=space.get("method", "idw"),
                        anchor=bool(space.get("anchor", False)))
                    for h in interp_ids:
                        key = (lb, h)
                        if key in recs:                      # reinterpolation revisits:
                            recs[key][1] = True              # mark, don't re-fill
                        else:
                            recs[key] = [float(filled[h]), True]
            out_recs[vc] = recs

        # merge to wide rows: one row per (label, h3) present in any column
        keys = sorted({k for recs in out_recs.values() for k in recs},
                      key=lambda kh: (kh[0], kh[1]))
        coldefs = [("h3_id", "VARCHAR"), (sr.EVENT_TS_COLUMN, "TIMESTAMP")]
        for vc in value_cols:
            coldefs += [(vc, "DOUBLE"), (INTERP_PREFIX + vc, "BOOLEAN")]
        if add_is_interpolated:
            coldefs += [(IS_INTERPOLATED_COLUMN, "BOOLEAN")]
        out_rows = []
        synth = 0
        for lb, h in keys:
            row = [h, datetime.fromisoformat(lb) if series else None]
            row_interp = False
            for vc in value_cols:
                got = out_recs[vc].get((lb, h))
                row += ([got[0], got[1]] if got else [None, None])
                if got and got[1]:
                    synth += 1
                    row_interp = True
            if add_is_interpolated:
                row += [row_interp]
            out_rows.append(row)
        _write_output_table(con, out_table, coldefs, out_rows)
        info2 = dict(zip(("rows", "n_cells", "n_steps"),
                         con.execute(f"SELECT count(*), count(DISTINCT h3_id), "
                                     f"count(DISTINCT {sr.EVENT_TS_COLUMN}) "
                                     f"FROM output.{_q(out_table)}").fetchone()))
        info2["synth"] = synth
        log(f"  [interpolate] output.{out_table}: {info2['rows']} rows, "
            f"{info2['n_cells']} cells, {info2['n_steps']} steps, {synth} synthesised")
        # Model B: NO self-registration. The result is a consumer table that
        # lives in output only; registry membership means "project-level
        # input", and an output result is never an input.
        return out_table, info2
    finally:
        con.close()

def output_schema_reference():
    """Return a dict describing every output.* dataset for the GUI reference panel.

    {dataset: {required: bool, description: str,
               columns: {name: {data_type: str, required: bool,
                                description: str,
                                valid_values: str}, ...}}, ...}
    """
    ref = {}
    for table, cols in _TABLE_COLS.items():
        required_cols = _OUTPUT_REQUIRED_COLUMNS.get(table, set())
        valid_values = _OUTPUT_VALID_VALUES.get(table, {})
        columns = {
            "seq": {"data_type": "BIGINT",
                    "required": True,
                    "description": "Row order / feature index. Internal bookkeeping "
                                   "that preserves pipeline feature order and groups "
                                   "multi-row expansions back into a single feature.",
                    "valid_values": ""},
        }
        for c, t in cols:
            col_name = c.strip('"')
            col_valid = valid_values.get(col_name)
            columns[col_name] = {
                "data_type": t,
                "required": col_name in required_cols,
                "description": _OUTPUT_COLUMN_DESCRIPTIONS
                .get(table, {}).get(col_name, ""),
                "valid_values": ", ".join(col_valid) if col_valid else "",
            }
        ref[table] = {
            "required": False,
            "description": _OUTPUT_DATASET_DESCRIPTIONS.get(table, ""),
            "columns": columns,
        }
    data_required = _OUTPUT_REQUIRED_COLUMNS.get("data", set())
    ref["data"] = {
        "required": False,
        "description": _OUTPUT_DATASET_DESCRIPTIONS["data"],
        "columns": {
            name: {"data_type": "",
                   "required": name in data_required,
                   "description": desc,
                   "valid_values": ""}
            for name, desc in _OUTPUT_COLUMN_DESCRIPTIONS["data"].items()
        },
    }
    return ref

def _bulk_insert(con, schema, table, columns):
    """Vectorised bulk insert: `columns` is a {col_name: array-like} mapping covering
    [seq] + _TABLE_COLS[table], loaded via a pandas DataFrame -> DuckDB relation.
    executemany() on millions of rows is the regression this fixes (measured ~640x
    slower: 2.2k rows/s vs 1.4M rows/s on a 210k-row slice). Row order == the seq
    column order; the export reads ORDER BY seq, so order is preserved."""
    import pandas as pd
    cols = ["seq"] + [c for c, _t in _TABLE_COLS[table]]
    df = pd.DataFrame({c: columns[c] for c in cols})
    rel_name = f"_bulk_{table}"
    con.register(rel_name, df)
    try:
        con.execute(f"INSERT INTO {schema}.{table} SELECT * FROM {rel_name}")
    finally:
        con.unregister(rel_name)


def _insert_rows(con, table, rws):
    """Bulk-insert prepared staging rows (row[0] = seq, then _TABLE_COLS[table]
    order). Vectorised through _bulk_insert (pandas) with an executemany
    fallback only when pandas/numpy is unavailable -- executemany is ~640x
    slower on millions of rows (the 0.13.0 regression), so it is the fallback,
    not the default. NYC-scale structures (~1.8M rows) staged in tens of
    seconds vectorised vs ~15 minutes via executemany."""
    cols = ["seq"] + [c for c, _ in _TABLE_COLS[table]]
    try:
        import numpy as _np
    except ImportError:
        n = len(_TABLE_COLS[table]) + 1
        col_sql = ", ".join(cols)
        con.executemany(f"INSERT INTO staging.{table} ({col_sql}) VALUES "
                        f"({', '.join('?' * n)})", rws)
        return
    columns = {"seq": _np.arange(len(rws), dtype=_np.int64)}
    for i, (c, ty) in enumerate(_TABLE_COLS[table]):
        if ty == "DOUBLE":
            columns[c] = _np.array([r[i + 1] for r in rws], dtype=_np.float64)
        elif ty == "INTEGER":
            columns[c] = _np.array(

                dtype=object)

def stage_dataset(con, features, cell_rows, face_points=None, face_colors=None,
                  progress_cb=None):
    """Decompose the assembled dataset into the staging tables (one row per cell /
    street / structure / body / landmark / face-colour / NE feature). Unknown
    feature types are dropped. Replaces the tables wholesale -- a rebuild re-stages.
    Callers: process_area, then carry_staging_to_project + export.

    progress_cb(fraction, message), when given, reports 0..1 across the staging
    work (the phase used to be one silent print followed by tens of minutes of
    bulk inserts on large areas)."""
    def _tick(frac, msg):
        if progress_cb is not None:
            progress_cb(frac, msg)

    con.execute("CREATE SCHEMA IF NOT EXISTS staging")
    con.execute("CREATE SCHEMA IF NOT EXISTS output")
    for t in DATASET_TABLES:
        _create_dataset_table(con, "staging", t)
    _tick(0.02, "staging: tables ready")

    if cell_rows:
        import numpy as _np
        # Bare grid + the propagated-layer columns the rows may carry (the wide
        # cells table; see _TABLE_COLS note). Rows missing a layer key stage
        # NULL -- propagation at pipeline time fills output. int keys:
        # is_pentagon; text keys: h3_id + the propagated TEXT layers.
        _INT_KEYS = {"is_pentagon"}
        _TEXT_KEYS = {"h3_id", "vegetation_type", "provenance"}
        columns = {"seq": _np.arange(len(cell_rows), dtype=_np.int64)}
        for k in _CELL_ROW_KEYS:
            if k in _INT_KEYS:
                columns[k] = _np.array([_i(r.get(k)) or 0 for r in cell_rows],
                                       dtype=_np.int32)
            elif k in _TEXT_KEYS:
                columns[k] = _np.array([_s(r.get(k)) for r in cell_rows], dtype=object)
            else:
                # float, None -> NaN (pandas maps NaN -> NULL on insert)
                columns[k] = _np.array(
                    [_f(r.get(k)) if _f(r.get(k)) is not None else _np.nan
                     for r in cell_rows], dtype=_np.float64)
        _bulk_insert(con, "staging", "cells", columns)
        _tick(0.10, f"staged cells ({len(cell_rows):,})")

    rows = {t: [] for t in ("streets", "structures", "water_bodies",
                            "landmarks", "vegetation", "ne_features")}
    _dropped_types = set()
    _n_feat = len(features)
    for seq, feat in enumerate(features):
        if progress_cb is not None and seq and seq % 25000 == 0:
            _tick(0.10 + 0.45 * seq / max(_n_feat, 1),
                  f"decomposing features ({seq:,}/{_n_feat:,})")
        props = feat.get("properties") or {}
        ftype = props.get("feature_type")
        if ftype == "street":
            rows["streets"].append(
                [seq, _s(props.get("osm_id")),
                 _s(props.get("street_type") or props.get("highway_tag")),
                 _f(props.get("width_m")),
                 bool(props.get("tunnel")), bool(props.get("bridge")),
                 json.dumps(((feat.get("geometry") or {}).get("coordinates") or []))])
        elif ftype == "structures":
            for s in props.get("structures", []):
                roof_shape = _s(s.get("roof_shape"))
                roof_type = _s(s.get("roof_type"))
                roof_height_m = _f(s.get("roof_height_m"))
                roof_direction_offset = _f(s.get("roof_direction_offset"))
                # Default output has no roof generation unless explicitly specified.
                if not roof_shape:
                    roof_shape = "flat"
                if not roof_type:
                    roof_type = "flat"
                if roof_direction_offset is None:
                    roof_direction_offset = 90.0
                rows["structures"].append(
                    [seq, _s(s.get("structure_id")), _s(s.get("osm_id")),
                     _s(s.get("source")),
                     _s(s.get("kind")), bool(s.get("notable")),
                     _f(s.get("height_m")),
                     _s(s.get("object_name")),
                     _s(s.get("borough")), _s(s.get("suburb")), _s(s.get("usage")),
                     _s(s.get("primary_osm_tag")), _s(s.get("secondary_osm_tag")),
                     _s(s.get("name")),
                     _s(s.get("historic")), _s(s.get("tourism")),
                     _s(s.get("amenity")), _s(s.get("man_made")),
                     _s(s.get("leisure")),
                     _s(s.get("building_type")),
                     json.dumps(s.get("footprint") or []),
                     _s(s.get("wall_colour")), _s(s.get("roof_colour")),
                     _s(s.get("wall_material")), _s(s.get("roof_material")),
                     roof_shape,
                     roof_type, roof_height_m, roof_direction_offset])
        elif ftype in ("water_body", "sea"):
            # cell_beds_json is built during source processing and arrives as a
            # regular property in the source water_bodies GeoJSON. Fall back to
            # deriving it from legacy water_cells for older datasets/tests.
            _beds = props.get("cell_beds_json") or {}
            if isinstance(_beds, str):
                try:
                    _beds = json.loads(_beds)
                except Exception:
                    _beds = {}
            if not _beds and props.get("water_cells"):
                _beds = {str(wc[5]): wc[2]
                         for wc in props["water_cells"] if len(wc) > 5}
            rows["water_bodies"].append(
                [seq, ftype, _s(props.get("Name")), _s(props.get("water_tag")),
                 _f(props.get("ele_m")), _f(props.get("depth_m")),
                 _s(props.get("depth_source")), _s(props.get("osm_id")),
                 _s(props.get("osm_type")), bool(props.get("touches_bbox", False)),
                 json.dumps((feat.get("geometry") or {}).get("coordinates")),
                 json.dumps({k: v for k, v in props.items()
                             if k not in ("Name", "water_tag", "ele_m", "depth_m",
                                          "depth_source", "osm_id", "osm_type",
                                          "touches_bbox", "feature_type",
                                          "water_cells", "cell_beds_json")}),
                 json.dumps(_beds)])
        elif ftype == "vegetation":
            for v in props.get("vegetation", []):
                rows["vegetation"].append(
                    [seq, _s(v.get("veg_tag")),
                     json.dumps(v.get("footprint") or [])])
        elif ftype == "landmarks":
            for lm in props.get("landmarks", []):
                rest = {k: v for k, v in lm.items()
                        if k not in ("name", "tag", "category", "cell_id",
                                     "lon", "lat")}
                rows["landmarks"].append(
                    [seq, _s(lm.get("name")), _s(lm.get("tag")),
                     _s(lm.get("category")), _s(lm.get("cell_id")),
                     _f(lm.get("lon")), _f(lm.get("lat")), json.dumps(rest)])
        elif ftype in ("ne_urban_area", "ne_place", "ne_major_road", "ne_landmark"):
            rest = {k: v for k, v in props.items()
                    if k not in ("area_id", "feature_type", "Name")}
            rows["ne_features"].append(
                [seq, ftype, _s(props.get("Name")),
                 json.dumps(feat.get("geometry")),
                 json.dumps(rest)])
        elif ftype == "terrain":
            # Terrain is not a misc feature. The importer reconstructs it from
            # cells.csv + the manifest, so it does not need to round-trip through
            # the output tables or the combined GeoJSON.
            continue
        else:
            _dropped_types.add(ftype)
    if _dropped_types:
        print(f"  [stage] dropped unsupported feature types: {sorted(_dropped_types)}",
              flush=True)

    for t, rws in rows.items():
        if not rws:
            continue
        _insert_rows(con, t, rws)
        _tick(0.55 + 0.25 * len(rws) / max(
                  sum(len(x) for x in rows.values()), 1),
              f"staged {t} ({len(rws):,})")

    # vegetation_params: seed the density lookup with the built-in defaults
    # (the carry moves them to output; ensure_layer_columns backfills any type
    # a customised output table is missing). seq is feature-order bookkeeping --
    # meaningless for a parameter table, left NULL.
    con.executemany("INSERT INTO staging.vegetation_params "
                    "(vegetation_type, region, density) VALUES (?, ?, ?)",
                    VEGETATION_PARAM_DEFAULTS)

    if face_points and face_colors:
        import numpy as _np
        # 7 faces per cell -> ~5.76M rows on a large area; vectorised bulk insert
        # (the per-row executemany was the ~50-minute regression at 0.13.0).
        # face_colors[hid] = [(imagery_rgb, regolith_rgb), ...] per face; the
        # regolith half is a leftover of the removed automatic assignment and
        # is NOT persisted (color-layers epic #1) — only imagery r/g/b.
        total = sum(len(pts) for pts in face_points.values())
        print(f"  [DB] staging face colours ({total:,} faces) ...", flush=True)
        hid_a = _np.empty(total, dtype=object)
        face_a = _np.empty(total, dtype=_np.int32)
        lon_a = _np.empty(total, dtype=_np.float64)
        lat_a = _np.empty(total, dtype=_np.float64)
        r_a = _np.empty(total, dtype=_np.int32)
        g_a = _np.empty(total, dtype=_np.int32)
        b_a = _np.empty(total, dtype=_np.int32)
        i = 0
        for hid, pts in face_points.items():
            cols_ = face_colors.get(hid) or []
            k = len(pts)
            hid_a[i:i + k] = hid
            for fi, (lon, lat) in enumerate(pts):
                j = i + fi
                face_a[j] = fi
                lon_a[j] = lon
                lat_a[j] = lat
                entry = cols_[fi] if fi < len(cols_) else None
                # entry is (imagery_rgb, regolith_rgb); tolerate a bare rgb
                try:
                    img, _sub = entry
                except (TypeError, ValueError):
                    img = entry
                img = img or (128, 128, 128)
                r_a[j], g_a[j], b_a[j] = int(img[0]), int(img[1]), int(img[2])
            i += k
        _bulk_insert(con, "staging", "face_colors", {
            "seq": _np.arange(total, dtype=_np.int64),
            "h3_id": hid_a, "face": face_a, "lon": lon_a, "lat": lat_a,
            "r": r_a, "g": g_a, "b": b_a,
        })
        _tick(0.97, f"staged face colours ({total:,})")

    _tick(1.0, "staging complete")
    return {t: con.execute(f"SELECT count(*) FROM staging.{t}").fetchone()[0]
            for t in DATASET_TABLES}

def _read_column_metadata(path):
    """Read a tyr column_metadata TSV into dict rows. Returns [] on missing/empty.
    A present-but-unreadable file RAISES (path + original error) rather than
    silently disabling defaults -- the schema build should fail anyway when
    its metadata cannot be read."""
    if path is None or not Path(path).is_file():
        return []
    rows = []
    try:
        with open(path, "r", encoding="utf-8", newline="") as fp:
            reader = csv.DictReader(fp, delimiter="\t")
            for row in reader:
                rows.append({k: (row.get(k) or "").strip() for k in row})
    except Exception as e:
        raise RuntimeError(
            f"column_metadata TSV is unreadable: {path}: {e}") from e
    return rows


def _default_select_expr(con, table, col_meta_rows):
    """Build a SELECT expression for staging.<table> that applies column_metadata
    DEFAULT fills. Columns without a DEFAULT policy are passed through unchanged.
    """
    info = con.execute(f"PRAGMA table_info('staging.{table}')").fetchall()
    staging_cols = [r[1] for r in info]
    # The CAST target must be the staging column's REAL type. column_metadata
    # data_type is sniffed from the source geojson (often VARCHAR) and pre-dates
    # tyr's type propagation, so trusting it produced
    # COALESCE(<DOUBLE>, CAST('6.0' AS VARCHAR)) binder errors (D6 repro).
    staging_types = {r[1]: r[2] for r in info}
    defaults = {}
    for row in col_meta_rows:
        if row.get("schema") != "staging":
            continue
        if row.get("dataset") != table:
            continue
        if row.get("on_null") != "DEFAULT":
            continue
        col = row.get("column_name", "")
        default = row.get("default_value", "")
        if col and default:
            defaults[col] = row

    exprs = []
    for col in staging_cols:
        if col in defaults:
            row = defaults[col]
            dtype = staging_types.get(col) or row.get("data_type") or "VARCHAR"
            safe_default = str(row["default_value"]).replace("'", "''")
            exprs.append(f'COALESCE("{col}", CAST(\'{safe_default}\' AS {dtype})) AS "{col}"')
        else:
            exprs.append(f'"{col}"')
    return ", ".join(exprs)

def carry_staging_to_project(con, tables=DATASET_TABLES):
    """The default carry-over: project.<t> = staging.<t>, with column_metadata
    DEFAULT policies applied to null values. THIS is the edit surface -- replace
    it per table to shape the dataset in SQL en route to the files the importer
    reads."""
    con.execute("CREATE SCHEMA IF NOT EXISTS project")
    cm_path = _area_column_metadata_path(con)
    col_meta = _read_column_metadata(cm_path)
    for t in tables:
        # tyr-staged areas may simply lack an optional table (austevoll has no
        # staging.vegetation) -- carry what exists instead of catalog-erroring.
        if not _table_exists(con, "staging", t):
            continue
        if col_meta:
            select_expr = _default_select_expr(con, t, col_meta)
        else:
            select_expr = "*"
        con.execute(f"CREATE OR REPLACE TABLE project.{t} AS SELECT {select_expr} FROM staging.{t}")
    for t in OPTIONAL_DATASET_TABLES:
        if not _table_exists(con, "staging", t):
            continue
        if col_meta:
            select_expr = _default_select_expr(con, t, col_meta)
        else:
            select_expr = "*"
        con.execute(f"CREATE OR REPLACE TABLE project.{t} AS SELECT {select_expr} FROM staging.{t}")

def carry_tyr_staging_to_project(con, log=print):
    """(Re)build the file-round-trippable project tables from tyr staging,
    casting every column to the canonical _TABLE_COLS type (tyr types geojson
    properties VARCHAR, so a naive pass-through would land strings in project).
    on_null=DEFAULT is deliberately NOT applied here: tyr's staging SQL
    already applied it (macros/columns.py), and re-applying it cast the
    default literal to the physical staging column type instead of the
    metadata data_type. Tables with no staging counterpart keep their current
    project table. Returns the carried table names.

    project holds the NON-EDITED state: edits are NEVER folded here (the old
    unfold hack is gone -- with output rewritten from project+edits there is
    no fixed point to protect)."""
    con.execute("CREATE SCHEMA IF NOT EXISTS project")

    def _table_exists(schema, name):
        return con.execute(
            "SELECT 1 FROM information_schema.tables "
            "WHERE table_schema = ? AND table_name = ?",
            [schema, name]).fetchone() is not None

    carried = []
    for table, sources in _TYR_CARRY.items():
        present = [s for s in sources if _table_exists("staging", s)]
        if not present:
            log(f"  [tyr-carry] no staging source for {table} -- output.{table} retained")
            continue
        first = True
        for s in present:
            cols = {r[1].lower(): r[1]
                    for r in con.execute(f"PRAGMA table_info('staging.{s}')").fetchall()}
            has_geom = "geom" in cols
            exprs = []
            for name, typ in _TABLE_COLS[table]:
                lname = name.lower()
                if lname in cols:
                    src = f'"{cols[lname]}"'
                    exprs.append(f"CAST({src} AS {typ}) AS \"{name}\"")
                elif lname in ("coords_json", "footprint_json", "geom_json") and has_geom:
                    exprs.append(f"ST_AsGeoJSON(\"{cols['geom']}\") AS \"{name}\"")
                else:
                    exprs.append(f"NULL::{typ} AS \"{name}\"")
            select = f"SELECT {', '.join(exprs)} FROM staging.{s}"
            if first:
                con.execute(f"CREATE OR REPLACE TABLE project.{table} AS {select}")
                first = False
            else:
                con.execute(f"INSERT INTO project.{table} {select}")
        carried.append(table)
        log(f"  [tyr-carry] project.{table} <- staging.{' + '.join(present)}")
    return carried

def rewrite_output(con, log=lambda *_: None):
    """Publish project + edits into output. output is ALWAYS rewritten from
    project (never preserved), so purge-edits + one rewrite is clean by
    construction -- the DEFCON-1 bake (edits leaking into a long-lived
    output) cannot recur. Project-carried tables only; consumer RESULT
    tables in output (data_carrier, data_interp) are left untouched.
    Model B: every table registered in registry.custom_data is a
    project-level input and is ALSO published verbatim (project -> output)
    here, so output stays the single read surface for consumers. Idempotent.

    The cells overlay reuses _derive_cells_from_output (the effective-state
    math: ele offsets, regolith, water carving bed/depth/type, water-wins
    vegetation, winning face colour); wide project.columns pass through
    untouched via EXCLUDE, overlaid columns come from the derived rows."""
    import pandas as _pd
    con.execute("CREATE SCHEMA IF NOT EXISTS output")

    if _table_exists(con, "project", "cells"):
        (cell_rows, _fp, _fc, _rfc, _nw, _nv) = _derive_cells_from_output(
            con, area_id="")
        if not cell_rows:
            con.execute("CREATE OR REPLACE TABLE output.cells AS "
                        "SELECT * FROM project.cells")
            log("  [rewrite] output.cells <- project.cells (empty)")
        else:
            frame = _pd.DataFrame(cell_rows)
            con.register("_eff_cells", frame)
            try:
                proj_types = {r[0]: r[1] for r in con.execute(
                    "SELECT column_name, data_type FROM "
                    "information_schema.columns WHERE table_schema='project' "
                    "AND table_name='cells'").fetchall()}
                overlaid = [c for c in frame.columns
                            if c != "h3_id" and c in proj_types]
                exclude = ", ".join(f'"{c}"' for c in overlaid)
                eff_cols = ", ".join(
                    f'CAST(e."{c}" AS {proj_types[c]}) AS "{c}"'
                    for c in overlaid)
                con.execute(
                    f"CREATE OR REPLACE TABLE output.cells AS "
                    f"SELECT p.* EXCLUDE ({exclude}), {eff_cols} "
                    f"FROM project.cells p JOIN _eff_cells e "
                    f"ON e.h3_id = p.h3_id")
            finally:
                con.unregister("_eff_cells")
            log("  [rewrite] output.cells <- project.cells + edits")

    if _table_exists(con, "project", "face_colors"):
        con.execute("CREATE OR REPLACE TABLE output.face_colors AS "
                    "SELECT * FROM project.face_colors")
        con.execute(
            "UPDATE output.face_colors f SET r = e.r, g = e.g, b = e.b "
            "FROM edits.face_colors e "
            "WHERE e.h3_id = f.h3_id AND e.face = f.face")
        log("  [rewrite] output.face_colors <- project.face_colors + edits")

    if _table_exists(con, "project", "water_cell_params"):
        # edits.water_cell_params carries only the user's pins (rate_lps,
        # head_m, src, kind); water_type / water_source_infinite are pipeline
        # state and pass through verbatim.
        con.execute(
            "CREATE OR REPLACE TABLE output.water_cell_params AS "
            "SELECT p.h3_id, "
            "COALESCE(e.rate_lps, p.rate_lps) AS rate_lps, "
            "COALESCE(e.head_m, p.head_m) AS head_m, "
            "p.water_type AS water_type, "
            "p.water_source_infinite AS water_source_infinite "
            "FROM project.water_cell_params p "
            "LEFT JOIN edits.water_cell_params e ON e.h3_id = p.h3_id")
        log("  [rewrite] output.water_cell_params <- project + edits")

    if _table_exists(con, "project", "cell_water"):
        con.execute("CREATE OR REPLACE TABLE output.cell_water AS "
                    "SELECT * FROM project.cell_water")
        con.execute(
            "UPDATE output.cell_water c SET water_depth_m = e.water_depth_m "
            "FROM edits.cell_water e WHERE e.h3_id = c.h3_id")
        log("  [rewrite] output.cell_water <- project.cell_water + edits")

    for t in PROJECT_CARRIED:
        if t in ("cells", "face_colors", "water_cell_params", "cell_water"):
            continue
        if not _table_exists(con, "project", t):
            continue
        con.execute(f"CREATE OR REPLACE TABLE output.{t} AS "
                    f"SELECT * FROM project.{t}")
        log(f"  [rewrite] output.{t} <- project.{t}")

    # Model B publish: registered custom data tables are PROJECT-level inputs.
    # Carry each registered project table into output verbatim, so consumers
    # reading output.* (export, Data Viewer) see the same data. A registered
    # name with no project table is reported, not silently published. In the
    # Lite bundle custom data does not exist (custom_registry is stripped and
    # `creg` raises) -- there is simply nothing to publish.
    try:
        creg.ensure_registry(con)
        registered = [r[0] for r in con.execute(
            f"SELECT table_name FROM {creg.REGISTRY_SCHEMA}.{creg.REGISTRY_TABLE} "
            "ORDER BY table_name").fetchall()]
    except RuntimeError:
        registered = []
    for t in registered:
        if not _table_exists(con, "project", t):
            log(f"  [rewrite] WARNING registry table {t} has no "
                f"project.{t} -- skipped (register/migrate it, or unregister)")
            continue
        con.execute(f"CREATE OR REPLACE TABLE output.{_q(t)} AS "
                    f"SELECT * FROM project.{_q(t)}")
        log(f"  [rewrite] output.{t} <- project.{t} (registered)")

    optional = set(OPTIONAL_DATASET_TABLES)

        fm_path, _cm = source_metadata_paths(db_path.parent, db_path.stem)

def ensure_layer_columns(con):
    """Idempotently add the propagated layer columns to output.cells and
    backfill the pipeline defaults (humidity_factor=1.0, ice_thickness_m=0.0,
    vegetation_type='', vegetation_density=0.0, provenance='{}'). Only NULL
    cells are filled, so values propagation already wrote survive. Also
    ensures output.vegetation_params exists and seeds the built-in defaults
    for any missing vegetation type (custom per-region rows are kept).
    Returns True when at least one column was newly added (a fresh migration);
    the caller then re-propagates from the current layer tables."""
    if not _table_exists(con, "project", "cells"):
        return False
    added = False
    for name, typ, _default in _LAYER_COLUMN_DEFAULTS:
        have = con.execute(
            "SELECT 1 FROM information_schema.columns "
            "WHERE table_schema = 'project' AND table_name = 'cells' "
            "AND column_name = ?", [name]).fetchone()
        if have is None:
            con.execute(f"ALTER TABLE project.cells ADD COLUMN IF NOT EXISTS "
                        f"{name} {typ}")
            added = True
    for name, _typ, default in _LAYER_COLUMN_DEFAULTS:
        con.execute(f"UPDATE project.cells SET {name} = {default} "
                    f"WHERE {name} IS NULL")
    # The density lookup table: create + seed missing types on any DB that
    # lacks it (pre-change area DB, tyr-only round trips, SQL-customised).
    con.execute("CREATE TABLE IF NOT EXISTS project.vegetation_params ("
                "vegetation_type TEXT PRIMARY KEY, region TEXT, density DOUBLE)")
    for veg_type, region, density in VEGETATION_PARAM_DEFAULTS:
        con.execute("INSERT INTO project.vegetation_params "
                    "(vegetation_type, region, density) "
                    "SELECT ?, ?, ? WHERE NOT EXISTS (SELECT 1 FROM "
                    "project.vegetation_params WHERE vegetation_type = ?)",
                    [veg_type, region, density, veg_type])
    return added




def _annualized_climate_factors(clim):
    """Annualized (humidity_factor, ice_thickness_m) from one cell's monthly
    climate block (climate.read_climate shape). Documented simple rules (see
    the constants above): humidity = annual P / annual PET clamped to
    [0, 2] (1.0 when PET ~ 0); ice = 1 m per deg C the mean annual
    temperature sits below 0 C, capped at ICE_MAX_M, else 0.0."""
    days = (31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31)
    annual_p = sum(clim["precip_mm_monthly"])
    annual_pet = sum(p * d for p, d in zip(clim["pet_mm_day_monthly"], days))
    if annual_pet > 1e-9:
        humidity = min(HUMIDITY_FACTOR_MAX,
                       max(HUMIDITY_FACTOR_MIN, annual_p / annual_pet))
    else:
        humidity = 1.0
    mean_t = sum((a + b) / 2.0 for a, b in zip(clim["tmin_c_monthly"],
                                               clim["tmax_c_monthly"])) / 12.0
    if mean_t >= ICE_THRESHOLD_C:
        ice = 0.0
    else:
        ice = min(ICE_MAX_M, (ICE_THRESHOLD_C - mean_t) * ICE_MM_PER_C / 1000.0)
    return humidity, ice


def _climate_provenance_token(source):
    """Compact provenance token for an output.climate source string."""
    s = (source or "").lower()
    if "chelsa" in s:
        return "chelsa"
    if "analytic" in s:
        return "analytic"
    return source or "none"

def propagate_layer_columns(con, log=None):
    """Pipeline-time propagation of the layer parameters onto the wide cells
    row (docs/layers.rst): re-derives humidity_factor + ice_thickness_m from
    output.climate (absent -> the defaults 1.0 / 0.0), stamps vegetation_type
    from the output.vegetation footprint overlap (unified._VEG_PRIORITY,
    denser type wins), looks up vegetation_density from
    output.vegetation_params (0.0 when no vegetation or unknown type), and
    rewrites the per-cell provenance map for the layers it manages
    (regolith/climate/vegetation/ice). Idempotent: a full re-derivation from
    the current layer tables. Called from the pipeline hooks
    (stage_dataset_to_output / roundtrip_dataset) and after build_climate.
    Returns a small report dict."""
    report = {"n_cells": 0, "n_climate": 0, "n_vegetated": 0, "migrated": False}
    if not _table_exists(con, "project", "cells"):
        return report
    report["migrated"] = ensure_layer_columns(con)

    cells = con.execute(
        "SELECT h3_id, regolith_material FROM project.cells").fetchall()
    by_id = {r[0]: r[1] for r in cells}
    report["n_cells"] = len(by_id)
    if not by_id:
        return report

    # ── Climate factors (annualized; monthly seasonality stays in the
    # output.climate block for the Blender-side climate model).
    climate_vals = {}
    from . import climate as _climate
    if _climate.climate_available(con):
        src_by_hid = dict(con.execute(
            "SELECT h3_id, source FROM project.climate").fetchall())
        for hid, clim in _climate.read_climate(con).items():
            if hid not in by_id:
                continue
            humidity, ice = _annualized_climate_factors(clim)
            climate_vals[hid] = (humidity, ice, src_by_hid.get(hid))

    # ── Vegetation stamping: the denser-type-wins footprint overlap moves to
    # pipeline time (unified.veg_cell_map; water exclusion stays a write-time
    # overlay, as does the claiming-body overlay for water columns).
    veg_vals = {}
    if _table_exists(con, "project", "vegetation"):
        veg_polys = []
        for r in con.execute(
                "SELECT veg_tag, footprint_json FROM project.vegetation").fetchall():
            try:
                fp = json.loads(r[1]) if r[1] else []
            except Exception:
                fp = []
            if len(fp) >= 3:
                veg_polys.append({"veg_tag": r[0] or "vegetation",
                                  "footprint": fp})
        if veg_polys:
            res = _grid_resolution(by_id)
            veg_vals = unified.veg_cell_map(veg_polys, set(by_id), res)
    density_by_type = {}
    for t, d in con.execute(
            "SELECT vegetation_type, density FROM project.vegetation_params") \
            .fetchall():
        density_by_type[t] = d if d is not None else 0.0
    report["n_vegetated"] = len(veg_vals)
    report["n_climate"] = len(climate_vals)

    # ── Per-cell provenance for the propagated layers (the vocabulary the UI
    # shows: "this cell's regolith: default; climate: analytic; ...").
    provenance = {}
    for hid in by_id:
        if hid in climate_vals:
            _h, ice, source = climate_vals[hid]
            provenance[hid] = {
                "regolith": "default",       # the pipeline default layer
                "climate": _climate_provenance_token(source),
                "ice": "climate-derived" if ice > 0.0 else "none",
            }
        else:
            provenance[hid] = {"regolith": "default",
                               "climate": "none", "ice": "none"}
    for hid in by_id:
        if veg_vals.get(hid):
            provenance[hid]["vegetation"] = "osm"
        else:
            provenance[hid]["vegetation"] = "none"

    # ── One wholesale write via a temp table (per-cell executemany UPDATEs on
    # a 200k-cell grid is the regression this avoids).
    import numpy as _np
    import pandas as _pd
    hids = sorted(by_id)
    frame = _pd.DataFrame({
        "h3_id": _np.array(hids, dtype=object),
        "humidity_factor": [climate_vals[h][0] if h in climate_vals else 1.0
                            for h in hids],
        "ice_thickness_m": [climate_vals[h][1] if h in climate_vals else 0.0
                            for h in hids],
        "vegetation_type": [veg_vals.get(h, "") for h in hids],
        "vegetation_density": [density_by_type.get(veg_vals.get(h, ""), 0.0)
                               for h in hids],
        "provenance": [json.dumps(provenance[h], separators=(",", ":"))
                       for h in hids],
    })
    con.register("_prop_layer", frame)
    try:
        con.execute(
            "UPDATE project.cells c SET "
            "humidity_factor = p.humidity_factor, "
            "ice_thickness_m = p.ice_thickness_m, "
            "vegetation_type = p.vegetation_type, "
            "vegetation_density = p.vegetation_density, "
            "provenance = p.provenance "
            "FROM _prop_layer p WHERE c.h3_id = p.h3_id")
    finally:
        con.unregister("_prop_layer")
    if log:
        log(f"  [layers] propagated {report['n_cells']} cells "
            f"({report['n_climate']} climate-derived, "
            f"{report['n_vegetated']} vegetated)")
    return report

def validate_staging_project(con, tables=DATASET_TABLES):
    """The contract gate: output.<t> carries exactly staging.<t>'s columns (the
    output format == the staging format, so the exporter can rely on the shape).

    REQUIRED tables must exist in BOTH schemas with identical columns -- an
    absent required table fails loudly. OPTIONAL datasets (declared
    required=false in the area's file metadata, e.g. CHELSA -- T5/D5) may be
    absent from EITHER side: a tyr staging rebuild cannot recreate DB-native
    datasets (no unified file), so an output-only optional table must not stop
    the pipeline. A present optional table on both sides must still match.
    Returns (ok, [messages])."""
    optional = _declared_optional_datasets(con)
    msgs, ok = [], True

    def _check_optional(t):
        """Optional dataset: absence from EITHER side is acceptable (a tyr
        staging rebuild cannot recreate DB-native datasets); when both sides
        exist their columns must still match."""
        nonlocal ok
        in_staging = _table_exists(con, "staging", t)
        in_output = _table_exists(con, "project", t)
        if not in_staging and not in_output:
            return                            # absent entirely: nothing to check
        if not in_staging or not in_output:
            side = "output-only" if in_output else "staging-only"
            msgs.append(f"PASS project.{t} optional, {side} "
                        f"(required=false) -- accepted")
            return
        s = [r[1] for r in con.execute(f"PRAGMA table_info('staging.{t}')").fetchall()]
        o = [r[1] for r in con.execute(f"PRAGMA table_info('project.{t}')").fetchall()]
        good = s == o
        ok = ok and good
        msgs.append(("PASS " if good else "FAIL ") + f"project.{t} columns == staging")

    for t in tables:
        if t in optional:
            _check_optional(t)
            continue
        in_staging = _table_exists(con, "staging", t)
        in_output = _table_exists(con, "project", t)
        good = in_staging and in_output
        if good:
            s = [r[1] for r in con.execute(f"PRAGMA table_info('staging.{t}')").fetchall()]
            o = [r[1] for r in con.execute(f"PRAGMA table_info('project.{t}')").fetchall()]
            good = s == o
        ok = ok and good
        msgs.append(("PASS " if good else "FAIL ") + f"project.{t} columns == staging")
    for t in optional - set(tables):
        _check_optional(t)
    return ok, msgs




def _seq_expr(con, schema, table):
    """Return a SELECT-able expression for the row order column: use `seq` if the
    table has one, otherwise synthesise one with row_number(). Keeps reads robust
    when output tables pre-date the seq column (e.g. older DBs / tyr-created
    staging tables carried over without it)."""
    has = con.execute(
        "SELECT 1 FROM information_schema.columns "
        "WHERE table_schema = ? AND table_name = ? AND column_name = 'seq'",
        (schema, table)).fetchone()
    return "seq" if has else "row_number() OVER () - 1"


def _has_col(con, schema, table, col):
    return con.execute(
        "SELECT 1 FROM information_schema.columns "
        "WHERE table_schema = ? AND table_name = ? AND column_name = ?",
        (schema, table, col)).fetchone() is not None


def _geom_json_expr(con, schema, table, json_col, geom_col):
    """Return a SELECT expression that yields geometry as JSON text.

    Prefer the hand-rolled JSON text column (e.g. geom_json / coords_json /
    footprint_json) if present; otherwise convert DuckDB's ST_Read geometry
    column to GeoJSON text. This lets _rebuild_features tolerate both the
    Python feature path and tyr's ST_Read path."""
    if _has_col(con, schema, table, json_col):
        return f'"{json_col}"'
    if _has_col(con, schema, table, geom_col):
        return f'ST_AsGeoJSON("{geom_col}")'
    return "NULL"


def _geom_coords_from_text(text):
    """Normalise a geometry string to a GeoJSON coordinate array.

    Input may be a bare coordinate array (our geom_json / coords_json /
    footprint_json columns) or a full GeoJSON geometry object (from
    ST_AsGeoJSON). Returns None for empty/malformed input."""
    if not text:
        return None
    try:
        data = json.loads(text)
    except Exception:
        return None
    if isinstance(data, dict):
        return data.get("coordinates")
    return data


def _geom_from_text(text):
    """Normalise a geometry string to a full GeoJSON geometry dict.

    Input may be a full GeoJSON geometry object (our ne_features.geom_json
    column) or a bare coordinate array. Returns None for empty/malformed input."""
    if not text:
        return None
    try:
        data = json.loads(text)
    except Exception:
        return None
    if isinstance(data, dict) and "type" in data:
        return data
    # Bare coordinate array -- infer type from shape (best-effort).
    coords = data
    if coords is None:
        return None
    if isinstance(coords, list) and coords:
        first = coords[0]
        if isinstance(first, (list, tuple)):
            # Polygon (closed ring) or LineString.
            if len(coords) >= 1 and coords[0] == coords[-1]:
                return {"type": "Polygon", "coordinates": coords}
            return {"type": "LineString", "coordinates": coords}
        return {"type": "Point", "coordinates": coords}
    return None


def _col_or_null(con, schema, table, col, sqltype):
    """Return a quoted column reference or a typed NULL if the column is absent.
    Lets _rebuild_features survive output tables that pre-date a column or were
    created from a narrower ST_Read schema."""
    if _has_col(con, schema, table, col):
        return f'"{col}"'
    return f"NULL::{sqltype}"

# Lowest zoom the colour-source mosaic is built at: below this the image is too
# coarse to edit meaningfully, so an area that still overflows the pixel cap at
# this zoom must be split rather than stitched any blurrier.
_COLOR_ASSET_MIN_ZOOM = 8


def _fit_color_asset_zoom(zoom, lat_min, lon_min, lat_max, lon_max):
    """Preflight the colour-source mosaic against the bitmap editor's pixel cap
    BEFORE any tile download (a too-large mosaic would otherwise only surface
    after the whole tile budget is spent). Lowers the zoom until the stitched
    mosaic fits satellite.MAX_EDIT_DIMENSION, logging each step. Raises
    ValueError -- the exact check_edit_dimensions wording -- if even the minimum
    zoom cannot fit, or if the tile count at the fitted zoom still exceeds the
    configured IMAGE_TEXTURE_MAX_TILES budget."""
    from . import satellite
    cap = satellite.MAX_EDIT_DIMENSION
    while True:
        tx0, ty0 = satellite._deg2tile(lat_max, lon_min, zoom)
        tx1, ty1 = satellite._deg2tile(lat_min, lon_max, zoom)
        if (tx1 - tx0 + 1) * 256 <= cap and (ty1 - ty0 + 1) * 256 <= cap:
            break
        if zoom <= _COLOR_ASSET_MIN_ZOOM:
            satellite.check_edit_dimensions((tx1 - tx0 + 1) * 256,
                                            (ty1 - ty0 + 1) * 256)
        print(f"color asset: zoom {zoom} -> {zoom - 1} to fit {cap}px cap "
              f"({tx1 - tx0 + 1}x{ty1 - ty0 + 1} tiles)", file=sys.stderr)
        zoom -= 1
    n_tiles = (tx1 - tx0 + 1) * (ty1 - ty0 + 1)
    budget = getattr(config, "IMAGE_TEXTURE_MAX_TILES",
                     satellite.MAX_TEX_TILES)
    if n_tiles > budget:
        raise ValueError(
            f"color asset at zoom {zoom} needs {n_tiles} tiles "
            f"(> the {budget} tile budget); raise IMAGE_TEXTURE_MAX_TILES "
            f"or lower the Texture Detail zoom")
    return zoom


def _color_area_bounds(manifest, output_dir, db_path):
    """(lat_min, lon_min, lat_max, lon_max): manifest area meta if complete,
    else the bbox of output.cells in the area DB."""
    area = manifest.get("area") or {}
    keys = ("lon_min", "lat_min", "lon_max", "lat_max")
    if all(area.get(k) is not None for k in keys):
        return (float(area["lat_min"]), float(area["lon_min"]),
                float(area["lat_max"]), float(area["lon_max"]))
    if not db_path.exists():
        raise ValueError("manifest has no area bounds and the area DB is missing")
    con = connect_wait(db_path)
    try:
        row = con.execute("SELECT min(lon), min(lat), max(lon), max(lat) "
                          "FROM project.cells").fetchone()
    finally:
        con.close()
    if not row or row[0] is None:
        raise ValueError("cannot derive bounds: project.cells is empty")
    return (float(row[1]), float(row[0]), float(row[3]), float(row[2]))


def _invalidate_derived_color_edits(output_dir, db_path):
    """Delete colour edits derived from a PREVIOUS color_source basis.

    The painted canvas (color_edited.png) and the resampled per-face rows
    (edits.face_colors, which win the export join) are both derivatives of a
    specific source mosaic. When the source is rebuilt they are stale by
    definition: if they survive, the editor and the export silently show the
    old edited picture over the fresh source. Best-effort on files; DB rows
    only when the table exists. Returns (edited_png_removed, face_colors_rows).
    """
    removed = False
    try:
        (Path(output_dir) / "color_edited.png").unlink(missing_ok=True)
        removed = True
    except OSError:
        pass
    rows = 0
    if Path(db_path).exists():
        con = connect_wait(db_path)
        try:
            if con.execute("SELECT COUNT(*) FROM information_schema.tables "
                           "WHERE table_schema = 'edits' AND table_name = 'face_colors'"
                           ).fetchone()[0]:
                rows = con.execute("SELECT COUNT(*) FROM edits.face_colors").fetchone()[0]
                con.execute("DELETE FROM edits.face_colors")
        finally:
            con.close()
    return removed, rows

def build_color_source_asset(manifest_path, origin, progress_cb=None):
    """Build the bitmap Colour Editor's color_source.png mosaic + sidecar and
    record it in catalog.assets.

    Shared by the pipeline (origin='pipeline') and the editor's repair path
    (scripts/build_color_asset.py, origin='editor'). Bounds come from the
    manifest area bbox, else output.cells in the area DB; source/zoom/passes
    mirror the pipeline's surface-texture block (config overrides included).
    Rebuilding the source INVALIDATES every colour edit derived from the
    previous basis (see _invalidate_derived_color_edits) — a stale edited
    image must never override the fresh source.
    `progress_cb`, when given, is forwarded to satellite.build_texture so the
    caller can map the mosaic's 0..1 report into its own progress slice.
    Returns the sidecar summary plus 'origin'."""
    from . import satellite

    manifest_path = Path(manifest_path)
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    area_id = manifest["area_id"]
    region_dir = manifest_path.parent
    # Region layout: derived colour assets live in <region>/output. Legacy
    # pre-layout areas (no output/ dir) keep them next to the manifest; the
    # C# Colour Editor read side tries output/ first, then the root.
    output_dir = (region_dir / "output"
                  if (region_dir / "output").is_dir() else region_dir)
    db_path = region_dir / f"{area_id}.duckdb"

    lat_min, lon_min, lat_max, lon_max = _color_area_bounds(manifest, output_dir, db_path)

    raw = (manifest.get("raw_data_dir")
           or (manifest.get("area") or {}).get("raw_data_dir"))
    if raw:
        raw_data_dir = Path(raw)
        if not raw_data_dir.is_absolute():
            raw_data_dir = Path(__file__).resolve().parent.parent / raw_data_dir
    elif (manifest_path.parent / "cache" / "s2_tiles").is_dir():
        # Manager layout: the manifest sits in the region root and raw
        # downloads live in the region's cache/ subdir (the runner sets
        # raw_data_dir = <region>/cache), and manifests don't record
        # raw_data_dir -- without this the cache lookup fell through to repo
        # data/raw and re-downloaded everything.
        raw_data_dir = manifest_path.parent / "cache"
    elif (manifest_path.parent / "s2_tiles").is_dir():
        # Legacy pre-region-layout areas: raw downloads co-located with the
        # manifest (the runner used to set raw = output = the manifest folder).
        raw_data_dir = manifest_path.parent
    else:
        raw_data_dir = Path(__file__).resolve().parent.parent / "data/raw"

    # Mirror pipeline.py's image-texture block (config overrides included).
    h3_resolution = int(manifest.get("h3_resolution") or 12)
    source = getattr(config, "IMAGE_TEXTURE_SOURCE", "s2")
    zoom = satellite.texture_zoom(h3_resolution,
                                  getattr(config, "IMAGE_TEXTURE_ZOOM", 0))
    passes = max(1, int(getattr(config, "SATELLITE_PASSES", 1)))
    zoom = _fit_color_asset_zoom(zoom, lat_min, lon_min, lat_max, lon_max)

    out_path = output_dir / "color_source.png"
    extent = satellite.build_texture(area_id, lat_min, lon_min, lat_max, lon_max,
                                     raw_data_dir, out_path, source, zoom, passes,
                                     progress_cb=progress_cb)
    satellite.check_edit_dimensions(extent["width"], extent["height"])

    sidecar = dict(extent)
    sidecar.update({"area_id": area_id, "source": source, "passes": passes,
                    "max_edit_dimension": satellite.MAX_EDIT_DIMENSION})
    unified._atomic_json_dump(output_dir / "color_source.json", sidecar,
                              indent=2)

    # Ledger entry: only when the area DB exists (a manifest-only area has no DB
    # to record in; never create one just for the ledger).
    if db_path.exists():
        con = connect_wait(db_path)
        try:
            ensure_catalog_schema(con)
            record_asset(con, "color_source", out_path.name,
                         extent["width"], extent["height"],
                         json.dumps(sidecar, indent=2), origin)
        finally:
            con.close()
    else:
        print(f"[catalog] {db_path.name} not found, asset not recorded",
              file=sys.stderr)

    removed, stale_rows = _invalidate_derived_color_edits(output_dir, db_path)
    if removed or stale_rows:
        print(f"[color asset] source rebuilt — invalidated derived colour edits "
              f"(color_edited.png removed: {removed}, edits.face_colors rows "
              f"deleted: {stale_rows})", file=sys.stderr)

    return {"ok": True, "path": str(out_path),
            "width": extent["width"], "height": extent["height"],
            "origin": origin}

def base_fingerprint(con):
    """Order-independent fingerprint of the area's base terrain (project.cells
    ele_m at centimetre resolution). The Terrain Editor snapshots this at
    load; write_terrain_edits recomputes it at save and refuses to persist
    when it changed — offsets are raw-relative, so a base that moved under an
    open editor session would silently mis-base every stroke (lake_como

    water over the carved beds). BIT_XOR makes it deterministic regardless of
    scan order; a count guards against row add/remove.

    n, x = con.execute(
        "SELECT COUNT(*), BIT_XOR(COALESCE(CAST(ROUND(ele_m, 2) * 100 AS BIGINT), "
        "-9223372036854775807)) FROM project.cells"
    ).fetchone()
    return f"{n}:{x}"

def effective_water_state(con):
    """The write-time water assignment: {h3_id: (water_type, surface_ele, bed_ele)}.

    Water state is ALWAYS derived. Per-cell boundary parameters (see
    hydro_sim) decide where water is: head params become water at head
    level at their own cells, and infinite (sea-level) heads additionally
    contour-flood every connected cell below their level — the same
    dam-aware geometry the editor preview renders, so output data matches
    what the editor shows even before the sim is run/saved. The saved sim
    output (edits.cell_water) merges on top with highest-surface-wins.
    output.water_bodies polygons are preprocessing inputs only: deleting a
    body row changes nothing until preprocessing re-runs (the next pipeline
    stage, or `manage_water_sources.py preprocess`). Self-heals: a missing
    output.water_cell_params (pre-change area DB) is built on the fly. Pure
    table reads; no rasters, no network.
    """
    from . import hydro_sim

    seq = _seq_expr(con, "project", "cells")
    rows = con.execute(
        f"SELECT {seq}, h3_id, ele_m FROM project.cells "
        f"ORDER BY {seq}").fetchall()
    cells = [{"h3_id": r[1], "ele_m": r[2]} for r in rows]
    by_id = {c["h3_id"]: c for c in cells}
    if not cells:
        return {}

    # The Terrain Editor's elevation offsets apply to the bed so raised
    # terrain also moves the seabed/riverbed.
    ele_offsets = {}
    for r in con.execute("SELECT h3_id, ele_offset_m FROM edits.cells").fetchall():
        ele_offsets[r[0]] = r[1]
    for c in cells:
        c["ele_m"] += ele_offsets.get(c["h3_id"], 0.0)

    # ── Water: head parameters become water at head level at their own
    # cells. A head at or below the bed is dry at that cell, but an
    # INFINITE (sea-level) head still contour-floods connected low ground
    # below its level (the dam-aware pass below — editor-preview parity).
    # The exported type is the param row's real source tag ('sea', a
    # standing body's tag) — a legacy tool head row, a flooded cell with no
    # claiming param row, or untyped head exports "" (D33: agnostic of
    # water type once the sim has built everything).
    _rates, heads, types, _n_default = hydro_sim.effective_water_params(con)
    water_entries = []   # (h3_id, water_type, surface_ele, bed_ele)
    for hid, level in heads.items():
        c = by_id.get(hid)
        if c is None or level <= c["ele_m"]:
            continue
        water_entries.append((hid, types.get(hid) or "", float(level),
                              float(c["ele_m"])))

    # Infinite (sea-level) heads ALSO contour-flood every connected cell
    # below their level — the same dam-aware geometry the editor preview
    # renders (hydro_sim.preview_wet_cells, single shared classifier via
    # infinite_head_ids). Without this the export carried only the head
    # cells themselves plus the SAVED sim output (project.cell_water), so an
    # unsaved or never-run sim left the entire flooded shoreline sheet in
    # the editor but out of every export (the Manhattan mismatch: 1,863
    # cells wet on screen, absent from output data). Flooded cells keep the
    # D33 rule: the claiming body's tag, "" when untyped.
    infinite = hydro_sim.infinite_head_ids(con)
    if infinite:
        beds = {h: c["ele_m"] for h, c in by_id.items()}
        dams = hydro_sim.effective_components(con)[0]
        for hid, d in hydro_sim.flood_from_heads(
                beds, {h: v for h, v in heads.items() if h in infinite},
                dams=dams).items():
            if hid not in by_id or d is None or d <= 0:
                continue
            water_entries.append((hid, types.get(hid) or "",
                                  beds[hid] + float(d), beds[hid]))

    # Simulated water (project.cell_water): the cell-domain flow sim's output.
    # The surface rides on the CURRENT (offset-applied) terrain — sim water
    # sits above the ground like a water column, it does not carve a bed.
    # Merges below with highest-surface-wins, after the head parameters.
    # No fabricated type either: the param-row tag when a real source body
    # claims the cell, "" otherwise (D33).
    if _table_exists(con, "project", "cell_water"):
        for hid, depth in con.execute(
                "SELECT h3_id, water_depth_m FROM project.cell_water").fetchall():
            c = by_id.get(hid)
            if c is None or depth is None or depth <= 0:
                continue
            bed = c["ele_m"]
            water_entries.append((hid, types.get(hid) or "", bed + float(depth), bed))

    # Combine overlapping entries with per-cell max surface.
    water = {}   # h3_id -> (water_type, surface_ele, bed_ele)
    for hid, tag, surface, bed in water_entries:
        if hid not in water or surface > water[hid][1]:
            water[hid] = (tag, float(surface), float(bed))
    return water

    """Join the output tables onto the bare cell grid.

    Returns (cell_rows, face_points_out, face_colors_out, n_wet, n_veg):
    cell_rows are FULL unified rows (all CELL_COLUMNS present, bed carved into

    (the regolith-material palette for wet cells, imagery for dry). Pure
    table reads; no rasters, no network, no footprint polyfill (vegetation
    was stamped at pipeline time -- see propagate_layer_columns).
    """
    from .coast import coast_fields, slope_fields
    from . import config as _cfg

    from . import hydro_sim as _hydro
    # Regolith defaults may be missing on pre-change area DBs — fold them
    # (ADD COLUMN IF NOT EXISTS + NULL fill) before the column list below is
    # selected off output.cells.
    _hydro.fold_regolith_defaults(con)
    # Wide-cells layer columns: migrate pre-change DBs (ADD COLUMN IF NOT
    # EXISTS + default backfill); a fresh migration also re-propagates from
    # the current layer tables so the rows below carry real values, not just
    # the defaults (identical cells.csv either way -- vegetation stamping
    # moved to pipeline time, see propagate_layer_columns).
    if ensure_layer_columns(con):
        propagate_layer_columns(con)

    ensure_edits_schema(con)

    seq = _seq_expr(con, "project", "cells")
    rows = con.execute(
        f"SELECT {seq}, " + ", ".join(_CELL_ROW_KEYS) + " FROM project.cells "
        f"ORDER BY {seq}").fetchall()
    cells = [dict(zip(_CELL_ROW_KEYS, r[1:])) for r in rows]
    if not cells:
        return [], {}, {}, {}, 0, 0

    # ── Elevation offsets from the Terrain Editor
    ele_offsets = {}
    for r in con.execute("SELECT h3_id, ele_offset_m FROM edits.cells").fetchall():
        ele_offsets[r[0]] = r[1]
    for c in cells:
        c["ele_m"] += ele_offsets.get(c["h3_id"], 0.0)

    by_id = {c["h3_id"]: c for c in cells}

    # ── Water: the write-time water assignment (see effective_water_state).
    water = effective_water_state(con)   # h3_id -> (water_type, surface_ele, bed_ele)
    # Effective per-cell source rates (pipeline defaults overlaid by the
    # user's edits) — exported next to water_depth_m in cells.csv.
    _rates, _heads, _types, _n_default = _hydro.effective_water_params(con)

    # ── Regolith: pipeline defaults overlaid by the editor's deltas
    # (per-column; a NULL edit column keeps the default) so cells.csv carries
    # the effective material/seepage/depth the sim and Blender read.
    _sub = {c["h3_id"]: [c["regolith_material"] or "bedrock",
                         c["regolith_seepage_mm_day"]
                         if c["regolith_seepage_mm_day"] is not None else 0.0,
                         c["regolith_depth_m"]
                         if c["regolith_depth_m"] is not None else 0.0]
            for c in cells}
    for r in con.execute(
            "SELECT h3_id, regolith_material, regolith_seepage_mm_day, "
            "regolith_depth_m FROM edits.regolith").fetchall():
        row = _sub.get(r[0])
        if row is None:
            continue
        if r[1] is not None:
            row[0] = r[1]
        if r[2] is not None:
            row[1] = float(r[2])
        if r[3] is not None:
            row[2] = float(r[3])

    # ── Vegetation: stamped at pipeline time into output.cells.vegetation_type
    # (propagate_layer_columns, the _VEG_PRIORITY footprint overlap) -- the
    # write path only applies the water-wins overlay below; it does not
    # re-stamp from the polygons.

    # ── Assemble full rows: carve the bed into ele_m for wet cells (the importer
    # expects ele_m = bed, water_depth_m = surface - bed), apply the water-wins
    # overlay to the propagated vegetation, and pass the propagated layer
    # columns (humidity/ice/provenance) through as-is: they are the wide
    # cells table's pipeline-time values (docs/layers.rst).
    cell_rows = []
    for c in cells:
        hid = c["h3_id"]
        w = water.get(hid)
        if w is not None:
            tag, surface, bed = w
            depth = max(0.0, surface - bed)
            ele, water_type = bed, tag
            veg_type = ""                       # water always wins over veg
        else:
            depth, ele, water_type = 0.0, c["ele_m"], ""
            veg_type = c["vegetation_type"] or ""
        cell_rows.append({
            "h3_id": hid, "lon": c["lon"], "lat": c["lat"],
            "ele_m": ele, "water_depth_m": round(depth, 4),
            "water_type": water_type, "veg_type": veg_type,
            "water_source_flow_lps": round(_rates.get(hid) or 0.0, 4),
            "regolith_material": _sub[hid][0],
            "regolith_seepage_mm_day": round(_sub[hid][1], 4),
            "regolith_depth_m": round(_sub[hid][2], 4),
            "rot_z": c["rot_z"], "radius_eq": c["radius_eq"],
            "is_pentagon": c["is_pentagon"],
            "humidity_factor": round(c["humidity_factor"], 4),
            "ice_thickness_m": round(c["ice_thickness_m"], 4),
            "vegetation_type": veg_type,      # effective (water wins), like veg_type
            "vegetation_density": round(c["vegetation_density"], 4)
            if veg_type else 0.0,
            "provenance": c["provenance"] or "{}",
        })

    # ── Coast + slope: pure functions of the final rows (coast reads the water
    # mask; slope reads the carved ele grid).
    cf = coast_fields(cell_rows)
    sf = slope_fields(cell_rows)
    for r in cell_rows:
        cd, dx, dy = cf.get(r["h3_id"], (0.0, 0.0, 0.0))
        r["coast_dist_m"], r["coast_dir_x"], r["coast_dir_y"] = round(cd, 4), round(dx, 6), round(dy, 6)
        s = sf.get(r["h3_id"])
        if s is None:
            r["slope_x"] = r["slope_y"] = None   # unknown bed -- NULL, not fake 0
        else:
            r["slope_x"], r["slope_y"] = round(s[0], 6), round(s[1], 6)

    # ── Colour join: a face's colour is the painted colour (edits.face_colors,
    # user intent for any face wet or dry) when present, else the stored
    # imagery colour (r/g/b, refreshed by an imagery reprocess). The
    # automatic regolith-material palette rule is REMOVED (color-layers
    # epic #1, 2026-09-18): regolith intent no longer assigns colours
    # anywhere — the color editor's Regolith layer is the only regolith
    # colouring, and it is user-controlled. cells.csv r/g/b = face 0.
    edit_face_colors = {}
    for r in con.execute("SELECT h3_id, face, r, g, b FROM edits.face_colors").fetchall():
        edit_face_colors[(r[0], r[1])] = (r[2], r[3], r[4])

    # Regolith answer for the split colour export (image_color vs
    # regolith_color on the Blender mesh): the canonical palette colour of
    # each cell's EFFECTIVE regolith material (edits win, per
    # regolith_colors). Flat per cell — the accepted loss documented there.
    from . import regolith_colors as _sc
    _sub_rgb = {hid: list(_sc.regolith_palette_rgb(mat))
                for hid, mat in _sc.effective_regolith_material(con).items()}

    face_points_out, face_colors_out, regolith_face_colors_out = {}, {}, {}
    if _table_exists(con, "project", "face_colors"):
        _fc_rows = con.execute("SELECT h3_id, face, lon, lat, r, g, b "
                               "FROM project.face_colors").fetchall()
    else:
        _fc_rows = []
    for r in _fc_rows:
        hid, fi = r[0], r[1]
        if hid not in by_id:
            continue
        face_points_out.setdefault(hid, []).append((r[2], r[3]))
        rgb = edit_face_colors.get((hid, fi), (r[4], r[5], r[6]))
        face_colors_out.setdefault(hid, []).append([rgb[0], rgb[1], rgb[2]])
    for hid, faces in face_colors_out.items():
        sub = _sub_rgb.get(hid, [128, 128, 128])
        regolith_face_colors_out[hid] = [list(sub) for _ in faces]
    for r in cell_rows:
        fc = face_colors_out.get(r["h3_id"])
        rgb = fc[0] if fc else [128, 128, 128]
        r["r"], r["g"], r["b"] = int(rgb[0]), int(rgb[1]), int(rgb[2])

    n_wet = len(water)
    n_veg = sum(1 for r in cell_rows if r["veg_type"])
    return (cell_rows, face_points_out, face_colors_out,
            regolith_face_colors_out, n_wet, n_veg)

def export_dataset_from_output(con, output_dir, area_id, contour_interval_m=0.0,
                               contour_major_interval_m=0.0,
                               provisional_manifest=False, region_dir=None):
    """Write the unified dataset files from the OUTPUT tables only: cells.csv, the
    combined <area>.geojson, the per-type feature files, the face-colour file, and
    (when an interval is set) the contour lines recomputed from the derived cells.
    The cells are DERIVED from the current tables first (water/veg/colour join --
    see _derive_cells_from_output), so table edits land in the written files.
    Returns (written, cell_rows): written = {name: (path, count)} for the manifest.

    `output_dir` receives the derived layer files (geojsons); `region_dir`
    (defaults to `output_dir`) receives the IDENTITY files (cells csv, the
    manifest) -- the region-folder layout keeps those in the region root.

    provisional_manifest=True is for the FIRST-BUILD pipeline path only: the
    caller (roundtrip_dataset -> pipeline.py) overwrites the provisional manifest
    with the complete one (area_meta included) moments after this returns, but
    the importer-contract validation below needs a manifest on disk NOW. Re-export
    callers must leave it False: without a manifest their export is incomplete
    and the loud failure is correct -- a provisional manifest surviving here
    would permanently lack area_meta (origin, bbox, sea level)."""
    output_dir = Path(output_dir)
    region = Path(region_dir) if region_dir else output_dir

    (cell_rows, face_points, face_colors,
     regolith_face_colors, n_wet, n_veg) = _derive_cells_from_output(
        con, area_id)
    print(f"  [output] derived {len(cell_rows)} cells from the current tables "
          f"({n_wet} wet, {n_veg} vegetated)", file=sys.stderr)
    cells_path = region / f"{area_id}_cells.csv"
    unified.write_cell_csv(cells_path, cell_rows)

    features = _rebuild_features(con, area_id)

    out_path = output_dir / f"{area_id}.geojson"
    unified._atomic_json_dump(out_path, {"type": "FeatureCollection", "features": features})
    written = unified.write_feature_files(output_dir, area_id, features)

    if face_points:
        _colpath = output_dir / f"{area_id}_color.geojson"
        written["color"] = (_colpath,
                            unified.write_color_geojson(_colpath, face_points,
                                                        face_colors,
                                                        regolith_face_colors))

    if contour_interval_m > 0:
        from . import contours
        _clines = contours.build_contour_lines(cell_rows, contour_interval_m,
                                               contour_major_interval_m)
        _cpath = output_dir / f"{area_id}_contour_lines.geojson"
        contours.write_contour_geojson(_cpath, _clines)
        written["contour_lines"] = (_cpath, len(_clines))

    # Climate initial-condition block (present once build_climate has run).
    from . import climate
    _climate_path = climate.export_climate_json(con, output_dir, area_id)
    _n = 0
    if _climate_path is not None:
        _n = con.execute("SELECT COUNT(*) FROM project.climate").fetchone()[0]
        written["climate"] = (_climate_path, _n)

    # First build: the pipeline writes the full manifest (with area_meta) right
    # after this export returns, but the importer-contract validation below
    # needs the manifest on disk NOW -- without it load_for_import fails with a
    # raw FileNotFoundError and the run prints a scary, bogus FAIL. Only the
    # pipeline path (provisional_manifest=True) may write this; re-export
    # callers leaving it False get the loud failure, because a provisional
    # manifest surviving a re-export would permanently lack area_meta.
    _mpath = region / f"{area_id}_manifest.json"
    if provisional_manifest and not _mpath.is_file():
        import h3 as _h3
        _res = _h3.get_resolution(cell_rows[0]["h3_id"]) if cell_rows else 0
        unified.write_manifest(_mpath, area_id, _res, len(cell_rows),
                               {n: c for n, (_, c) in written.items()},
                               provisional=True)

    # Repair the manifest's layer declarations to match what THIS export
    # actually produced — the manifest must never claim files that do not
    # exist (validation and the importer trust it):
    #  - climate is written as <area>_climate.json, not the blanket .geojson
    #    the manifest template assumes (write_manifest);
    #  - contour_lines is written only when intervals are set; a stale
    #    declaration from an earlier contour run must not outlive the file.
    if _mpath.is_file():
        _manifest = unified.read_manifest(_mpath)
        _ff = dict(_manifest.get("feature_files") or {})
        _fc = dict(_manifest.get("feature_counts") or {})
        _changed = False
        if (_climate_path is not None
                and _ff.get("climate") != _climate_path.name):
            _ff["climate"] = _climate_path.name
            _fc["climate"] = _n
            _changed = True
        if (contour_interval_m <= 0 and "contour_lines" in _ff
                and not (output_dir / _ff["contour_lines"]).is_file()):
            _ff.pop("contour_lines", None)
            _fc.pop("contour_lines", None)
            _changed = True
        if _changed:
            _manifest["feature_files"] = _ff
            _manifest["feature_counts"] = _fc
            unified._atomic_json_dump(_mpath, _manifest)

    # Importer-contract validation on the just-written files, printed for the
    # caller (WARNs -- e.g. height-less structures -- do not fail the export).
    _vok, _vmsgs = validate_output(output_dir, area_id, region_dir=region)
    for _m in _vmsgs:
        print(f"  [validate] {_m}", file=sys.stderr)

    return written, cell_rows

def stage_dataset_to_output(area, features, cell_rows, face_points=None, face_colors=None,
                            progress_cb=None):
    """Process-side: stage the assembled dataset and carry it to output, then run the
    staging/output contract gate. Writes NO files -- the unified files are produced
    later, on demand, by write_output_schema (the Output tab's export). Returns the
    per-table staged counts (for the manifest's feature_counts).

    progress_cb(fraction, message) reports 0..1 across the write phases."""
    def _tick(frac, msg):
        if progress_cb is not None:
            progress_cb(frac, msg)

    db_path = area_db_path(area)
    con = connect(db_path)
    try:
        counts = stage_dataset(con, features, cell_rows,
                               face_points=face_points, face_colors=face_colors,
                               progress_cb=lambda f, m: _tick(0.6 * f, m))
        print(f"  [output] staged: " + ", ".join(f"{t} {n}" for t, n in counts.items()))
        _tick(0.65, "carrying staging -> project")
        carry_staging_to_project(con)
        _tick(0.8, "validating staging/project contract")
        ok, msgs = validate_staging_project(con)
        for m in msgs:
            print(f"  [output-check] {m}")
        if not ok:
            raise RuntimeError("project/staging contract check failed -- see above")
        # Water sim initial conditions: preprocess the water bodies just
        # written into per-cell parameters (flowing -> upstream rates,
        # standing/sea -> fixed heads). Full replace -- user deltas live in
        # edits.water_cell_params. The area's sea_level_m offset (D3) rides
        # along so the enforced ocean edge matches the baked sea surface.
        # Lazy import: hydro_sim imports tyrbuild at module level (circular
        # otherwise).
        from . import hydro_sim
        pre = hydro_sim.preprocess_water_params(
            con, sea_level_m=float(getattr(area, "sea_level_m", 0.0) or 0.0))
        if pre["n_rate_cells"] or pre["n_head_cells"]:
            print(f"  [water] preprocessed {pre['n_rate_cells']} rate + "
                  f"{pre['n_head_cells']} head cell(s)")
        # Wide-cells propagation: layer parameters onto project.cells (climate
        # factors, vegetation stamping + density, provenance) -- the pipeline-
        # time half of the data-layer model (docs/layers.rst).
        _tick(0.9, "propagating layer columns")
        propagate_layer_columns(con)
        # Publish project + edits into output (output is always rewritten,
        # never preserved -- the DEFCON-1 fix).
        rewrite_output(con)
        _tick(1.0, "write complete")
        return counts
    finally:
        con.close()

def write_output_schema(output_dir, area_id, db_path, contour_interval_m=0.0,

    return written, cell_rows, data_names, data_rep

def read_project_cells(db_path):
    """Read the cell grid from the region DB's PROJECT schema -- the
    interpolator's domain/grid input (owner call 2026-09-30: the
    interpolator operates on project, never the merged output or the
    exported cells.csv). Drop-in replacement for unified.read_cells_csv:
    CSV-shaped dict rows (string values, "" for NULL), keys = column names,
    ordered by rowid. Raises FileNotFoundError when project.cells does not
    exist yet."""

    con = connect(str(db_path))
    try:
        cols = [r[0] for r in con.execute(
            "SELECT column_name FROM information_schema.columns "
            "WHERE table_schema = 'project' AND table_name = 'cells' "
            "ORDER BY ordinal_position").fetchall()]
        if "cells" not in {r[0] for r in con.execute(
                "SELECT table_name FROM information_schema.tables "
                "WHERE table_schema = 'project'").fetchall()}:
            raise FileNotFoundError(
                f"project.cells not found in {db_path} -- run Source for "
                f"this area first")
        q = ", ".join(f'"{c}"' for c in cols)
        rows = con.execute(f"SELECT {q} FROM project.cells ORDER BY rowid") \
                 .fetchall()
    finally:
        con.close()
    out = []
    for r in rows:
        row = {}
        for c, v in zip(cols, r):
            if v is None:
                row[c] = ""
            elif isinstance(v, bool):
                row[c] = "1" if v else "0"
            elif isinstance(v, float) and v.is_integer():
                row[c] = str(int(v))     # "255.0" -> "255" for int()-consumers
            else:
                row[c] = str(v)
        out.append(row)
    return out

def regenerate_contour_layer(manifest_path, contour_interval_m,
                             contour_major_interval_m=0.0):
    """Independently regenerate (or remove) the area's contour-lines layer from
    the CURRENT output tables -- no source re-run needed. Cells are re-derived
    from output.* (water/veg/colour edits reflected), contours recomputed, the
    <area>_contour_lines.geojson rewritten, and the manifest's feature_files /
    feature_counts updated. interval <= 0 REMOVES the layer (file + manifest
    entries). Returns a report dict."""
    from . import contours

    manifest_path = Path(manifest_path)
    manifest = unified.read_manifest(manifest_path)
    area_id = manifest["area_id"]
    region_dir = manifest_path.parent
    db_path = region_dir / f"{area_id}.duckdb"
    if not db_path.is_file():
        raise FileNotFoundError(
            f"no region database at {db_path} -- run the pipeline first")

    # Region-folder layout: contour lines are derived -> the region's output/
    # subdir; legacy flat areas (no output/ dir) keep the file at the root.
    out_dir = region_dir / "output"
    if not out_dir.is_dir():
        out_dir = region_dir
    cpath = out_dir / f"{area_id}_contour_lines.geojson"
    if contour_interval_m <= 0:
        removed = cpath.exists()
        if removed:
            cpath.unlink()
        _register_contour_layer(manifest_path, area_id, None)
        return {"status": "removed" if removed else "absent",
                "path": str(cpath), "interval_m": 0.0}

    con = connect(str(db_path))
    try:
        cell_rows, *_ = _derive_cells_from_output(con, area_id)
    finally:
        con.close()
    clines = contours.build_contour_lines(cell_rows, contour_interval_m,
                                          contour_major_interval_m)
    contours.write_contour_geojson(cpath, clines)
    _register_contour_layer(manifest_path, area_id, len(clines))
    return {"status": "written", "path": str(cpath), "lines": len(clines),
            "cells": len(cell_rows), "interval_m": contour_interval_m,
            "major_interval_m": contour_major_interval_m}




def _register_contour_layer(manifest_path, area_id, count):
    """Add/update (count is not None) or remove (count is None) the manifest's
    contour_lines entries. The importer discovers the file by convention, so
    this is bookkeeping -- but stale entries lie to anything that reads them."""
    manifest = unified.read_manifest(manifest_path)
    if count is None:
        for key in ("feature_files", "feature_counts"):
            (manifest.get(key) or {}).pop("contour_lines", None)
    else:
        manifest.setdefault("feature_files", {})["contour_lines"] = \
            f"{area_id}_contour_lines.geojson"
        manifest.setdefault("feature_counts", {})["contour_lines"] = count

def roundtrip_dataset(area, features, cell_rows, face_points=None, face_colors=None,
                      progress_cb=None):
    """The write path, DB-round-tripped: stage the assembled dataset, carry it to
    output, validate the staging/output contract, then export the unified files
    FROM output. The importer's files are never written ad-hoc. Returns
    (written, cell_rows, features_out).

    progress_cb(fraction, message) reports 0..1 across the write phases (staging
    0-0.6, carry, validate, water/layer prep, export) -- the phase used to be
    silent for tens of minutes on large areas.

    NOTE: superseded by stage_dataset_to_output + write_output_schema (Process stages,
    the Output tab exports). Kept for the CLI/back-compat."""
    def _tick(frac, msg):
        if progress_cb is not None:
            progress_cb(frac, msg)

    db_path = area_db_path(area)
    con = connect(db_path)
    try:
        print(f"  [DB] staging {len(cell_rows):,} cells + {len(features):,} features ...",
              flush=True)
        counts = stage_dataset(con, features, cell_rows,
                               face_points=face_points, face_colors=face_colors,
                               progress_cb=lambda f, m: _tick(0.6 * f, m))
        print(f"  [DB] staged: " + ", ".join(f"{t} {n}" for t, n in counts.items()))
        _tick(0.62, "carrying staging -> project")
        print("  [DB] carrying staging -> project ...", flush=True)
        carry_staging_to_project(con)
        _tick(0.75, "validating staging/project contract")
        print("  [DB] validating staging/project contract ...", flush=True)
        ok, msgs = validate_staging_project(con)
        for m in msgs:
            print(f"  [output-check] {m}")
        if not ok:
            raise RuntimeError("project/staging contract check failed -- see above")
        _tick(0.85, "preparing water + layer columns")
        # Water sim initial conditions (same hook as stage_dataset_to_output,
        # same per-area sea_level_m offset).
        from . import hydro_sim
        _manifest = Path(db_path).with_name(f"{area.area_id}_manifest.json")
        hydro_sim.preprocess_water_params(
            con, sea_level_m=float(getattr(area, "sea_level_m", 0.0) or 0.0),
            session_path=hydro_sim.session_water_path(_manifest)
            if _manifest.exists() else None)
        # Wide-cells propagation (same hook as stage_dataset_to_output).
        propagate_layer_columns(con)
        # Publish project + edits into output (same hook as everywhere else).
        rewrite_output(con)
        _tick(0.95, "exporting unified files")
        print("  [DB] exporting unified files ...", flush=True)
        written, cell_rows_out = export_dataset_from_output(
            con, area.output_dir, area.area_id,
            contour_interval_m=float(getattr(area, "contour_interval_m", 0.0) or 0.0),
            contour_major_interval_m=float(getattr(area, "contour_major_interval_m", 0.0) or 0.0),
            region_dir=getattr(area, "region_dir", None),
            # First-build path: the caller overwrites the provisional manifest
            # with the complete one right after this returns.
            provisional_manifest=True)
        features_out = _rebuild_features(con, area.area_id)
        _tick(1.0, "write complete")
    finally:
        con.close()
    for _name, (_p, _cnt) in written.items():
        print(f"  Saved -> {_p.name}  ({_cnt} features)")
    return written, cell_rows_out, features_out
