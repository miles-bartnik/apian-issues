"""
unified.py
----------
The unified per-cell format: the stable contract between data sources and the
import. The pipeline is Raw Data -> Unified Format -> Import; this module owns
the right-hand seam of that arrow for per-cell attributes.

One row per H3 cell, keyed by h3_id. Any data source (COP30, a custom DEM, GEBCO,
custom water, ...) contributes columns; the import is a pure consumer that knows
nothing about where the data came from.

Water model (see the data dictionary):
  ele_m         terrain/bed elevation, always real ground/seabed
  water_depth_m positive water-column thickness (0 = dry). surface = ele_m + water_depth_m
  water_type    classification only ('' = land), independent of geometry
"""

# Vegetation overlap precedence: when a cell falls under more than one veg
# polygon, the denser type wins (lower number = denser = wins). water always
# wins over vegetation regardless (handled in build_cell_rows).
_VEG_PRIORITY = {
    "forest": 0, "wood": 0,
    "scrub":  1,
    "heath":  2,
    "park":   3, "garden": 3,
    "meadow": 4, "grass":  4,
}
_VEG_DEFAULT_PRIORITY = 5  # unknown tag: least dense, loses to anything known

def veg_cell_map(vegetation, cell_id_set, resolution):
    """Map h3_id -> veg_type by polyfilling the OSM vegetation polygons.

    Restricted to cells in the area grid. On overlap the denser tag wins
    (_VEG_PRIORITY). Water exclusion is applied later in build_cell_rows.
    """
    import h3
    veg = {}
    for v in vegetation or []:
        tag = v.get("veg_tag", "vegetation")
        prio = _VEG_PRIORITY.get(tag, _VEG_DEFAULT_PRIORITY)
        fp = v.get("footprint", [])
        if len(fp) < 3:
            continue
        ring = [(p[1], p[0]) for p in fp]  # h3 wants (lat, lng)
        if ring[0] == ring[-1]:
            ring = ring[:-1]
        try:
            cells = h3.h3shape_to_cells(h3.LatLngPoly(ring), resolution)
        except Exception:
            continue
        for cid in cells:
            if cid not in cell_id_set:
                continue
            cur = veg.get(cid)
            if cur is None or prio < _VEG_PRIORITY.get(cur, _VEG_DEFAULT_PRIORITY):
                veg[cid] = tag
    return veg

def landcover_cell_map(landuse, natural_areas, cell_id_set, resolution):
    """Map h3_id -> landcover tag by polyfilling OSM landuse + natural polygons.

    Restricted to cells in the area grid. natural (physical surface) overrides
    landuse (zoning) on overlap, and within a source the later polygon wins --
    good enough for cartographic cell colour. veg_type takes precedence over
    this map (applied by the caller).
    """
    import h3
    out = {}
    for src in (landuse, natural_areas):   # natural second -> wins on overlap
        for poly in src or []:
            tag = poly.get("tag")
            fp = poly.get("footprint", [])
            if not tag or len(fp) < 3:
                continue
            ring = [(p[1], p[0]) for p in fp]  # h3 wants (lat, lng)
            if ring[0] == ring[-1]:
                ring = ring[:-1]
            try:
                cells = h3.h3shape_to_cells(h3.LatLngPoly(ring), resolution)
            except Exception:
                continue
            for cid in cells:
                if cid in cell_id_set:
                    out[cid] = tag
    return out

def build_cell_rows(cell_centres, features=None, veg_map=None):
    """Assemble the BARE per-cell grid rows for the DB (pure; no I/O).

    Only intrinsic cell facts are produced: h3_id, lon, lat, ele_m (RAW terrain
    elevation -- the bed carve moved to the write-time derive), rot_z, radius_eq,
    is_pentagon. The columns derived from OTHER tables (water_depth_m/water_type
    from water_bodies, veg_type from vegetation, colours from face_colors, and the
    coast/slope fields) are NOT set here -- they are computed at write time by
    tyrbuild._derive_cells_from_output so table edits are reflected in the written
    cells.csv. `features`/`veg_map` are accepted for call-site compatibility and
    ignored.
    """
    import h3
    rows = []
    for c in cell_centres:
        if len(c) <= 5:
            continue
        h3_id = c[5]
        try:
            is_pentagon = int(h3.is_pentagon(h3_id))
        except (ValueError, TypeError):          # malformed id -> treat as an ordinary hexagon
            is_pentagon = 0
        rows.append({
            "h3_id":         h3_id,
            "lon":           c[0],
            "lat":           c[1],
            "ele_m":         c[2],
            "rot_z":         c[3],
            "radius_eq":     c[4],
            "is_pentagon":   is_pentagon,
        })
    return rows

def write_cell_csv(path, rows):
    """Write rows to a CSV keyed by h3_id, with a header row.

    Atomic like _atomic_json_dump: a crash mid-export must not leave a
    truncated CSV that the manifest still names — write a temp file next to
    the target, then os.replace (atomic on Windows for same-volume)."""
    path = Path(path)
    tmp = path.with_suffix(path.suffix + ".tmp")
    with open(tmp, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=CELL_COLUMNS)
        w.writeheader()
        w.writerows(rows)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp, path)




def _atomic_json_dump(path, obj, separators=(",", ":"), indent=None):
    """Write JSON atomically: temp file in the same directory, then rename.
    Rejects NaN/Infinity so invalid floats fail at write time instead of producing
    malformed JSON. `separators` applies to the compact form only; with `indent`
    set, json's default (",", ": ") separators are used."""
    path = Path(path)
    tmp = path.with_suffix(path.suffix + ".tmp")
    with open(tmp, "w", encoding="utf-8") as fp:
        if indent is None:
            json.dump(obj, fp, separators=separators, allow_nan=False)
        else:
            json.dump(obj, fp, indent=indent, allow_nan=False)
        fp.flush()
        os.fsync(fp.fileno())
    tmp.replace(path)


# ── Vector feature files (one FeatureCollection per type) ──────────────────────
# These are the discrete objects (buildings, roads, ...). Each type gets its own
# file so a source can be swapped per type. Vegetation is NOT here -- it is a
# per-cell attribute (veg_type) now.

def _polygon(footprint, holes=None):
    """GeoJSON Polygon from a [[lon,lat],...] ring (+ optional holes), closed."""
    def _close(r):
        return r if r and r[0] == r[-1] else r + [r[0]]
    rings = [_close([[p[0], p[1]] for p in footprint])]
    for h in holes or []:
        if len(h) >= 3:
            rings.append(_close([[p[0], p[1]] for p in h]))
    return {"type": "Polygon", "coordinates": rings}


def _feature(fid, geometry, props):
    f = {"type": "Feature", "geometry": geometry, "properties": props}
    if fid is not None:
        f["id"] = fid
    return f


def _footprint_rings(fp):
    """Normalize a structure footprint to (outer_ring, holes).

    Two shapes arrive: a bare ring [[lon,lat],...] (pipeline-native, from
    in-memory features) or full Polygon coordinates [[[lon,lat],...],...]
    (rebuilt from the DB, where footprint_json may hold a GeoJSON geometry
    object). Rings shorter than 3 points are dropped; None outer means the
    footprint is unusable.
    """
    if not fp:
        return None, None
    try:
        if isinstance(fp[0][0], (list, tuple)):
            rings = [r for r in fp if len(r) >= 3]
            if not rings:
                return None, None
            return rings[0], rings[1:]
        if len(fp) >= 3:
            return fp, None
    except (TypeError, IndexError):
        pass
    return None, None

def build_feature_collections(features):
    """Split the pipeline's feature list into per-type FeatureCollections.

    Returns {type_name: [feature, ...]}. Explodes the list-in-properties shapes
    (structures, landmarks) into one feature per object, splits bridges out of
    structures, drops water_cells from sea/water bodies, and skips vegetation.
    """
    out = {"structures": [], "bridges": [], "streets": [],
           "landmarks": [], "water_bodies": [], "sea": [],
           # Natural Earth "major features" (country-scale detail mode) -- already proper
           # GeoJSON Features (naturalearth.build_features), so they pass straight through.
           "ne_urban_areas": [], "ne_places": [], "ne_major_roads": [], "ne_landmarks": []}

    _NE = {"ne_urban_area": "ne_urban_areas", "ne_place": "ne_places",
           "ne_major_road": "ne_major_roads", "ne_landmark": "ne_landmarks"}

    for f in features:
        props = f.get("properties", {}) or {}
        ftype = props.get("feature_type")

        if ftype in _NE:
            out[_NE[ftype]].append(f)

        elif ftype == "structures":
            for s in props.get("structures", []):
                outer, holes = _footprint_rings(s.get("footprint", []))
                if outer is None:
                    continue
                rest = {k: v for k, v in s.items() if k != "footprint"}
                is_bridge = rest.get("kind") == "bridge" or rest.get("deck")
                rest["feature_type"] = "bridge" if is_bridge else "structure"
                feat = _feature(s.get("structure_id") or s.get("osm_id"),
                                _polygon(outer, holes), rest)
                (out["bridges"] if is_bridge else out["structures"]).append(feat)

        elif ftype == "landmarks":
            for lm in props.get("landmarks", []):
                if lm.get("lat") is None or lm.get("lon") is None:
                    continue
                rest = {k: v for k, v in lm.items() if k not in ("lat", "lon")}
                rest["feature_type"] = "landmark"
                out["landmarks"].append(_feature(
                    None, {"type": "Point", "coordinates": [lm["lon"], lm["lat"]]}, rest))

        elif ftype == "street":
            out["streets"].append(f)

        elif ftype in ("water_body", "sea"):
            slim = {k: v for k, v in props.items() if k != "water_cells"}
            key = "sea" if ftype == "sea" else "water_bodies"
            out[key].append(_feature(props.get("osm_id"), f.get("geometry"), slim))

        # vegetation: intentionally skipped -- now per-cell (veg_type)

    return out

def write_feature_files(output_dir, area_id, features):
    """Write per-type GeoJSON files. Returns {type_name: (path, count)}."""
    collections = build_feature_collections(features)
    written = {}
    for name, feats in collections.items():
        path = output_dir / f"{area_id}_{name}.geojson"
        _atomic_json_dump(path, {"type": "FeatureCollection", "features": feats})
        written[name] = (path, len(feats))
    return written

def write_color_geojson(path, face_points, face_colors, regolith_face_colors=None):
    """Per-face colour samples: one Point feature per (cell, face). face 0 = the cell
    centre, 1..N = ring sectors in edge-midpoint order (see satellite.cell_face_points).
    Geometry is the sample location (debuggable in QGIS); properties carry h3_id, face,
    the image colour r/g/b (0-255) and — when regolith_face_colors is given — the
    regolith colour sub_r/sub_g/sub_b (the canonical palette colour of the cell's
    effective regolith material). The importer bakes both onto the Terrain Centres as
    vectors (image_color_* / regolith_color_*), so geonodes can pick per surface.
    Returns the feature count."""
    path = Path(path)
    regolith_face_colors = regolith_face_colors or {}
    feats = []
    for hid, pts in face_points.items():
        cols = face_colors.get(hid) or []
        sub_cols = regolith_face_colors.get(hid) or []
        for fi, (lon, lat) in enumerate(pts):
            rgb = cols[fi] if fi < len(cols) else (cols[0] if cols else [128, 128, 128])
            sub = sub_cols[fi] if fi < len(sub_cols) else rgb
            feats.append({
                "type": "Feature",
                "geometry": {"type": "Point", "coordinates": [round(lon, 7), round(lat, 7)]},
                "properties": {"h3_id": hid, "face": fi,
                               "r": int(rgb[0]), "g": int(rgb[1]), "b": int(rgb[2]),
                               "sub_r": int(sub[0]), "sub_g": int(sub[1]), "sub_b": int(sub[2])}})
    _atomic_json_dump(path, {"type": "FeatureCollection", "features": feats})
    return len(feats)

def write_manifest(path, area_id, h3_resolution, n_cells, feature_counts,
                   area_meta=None, image_texture=None, provisional=False):
    """One manifest listing every unified artifact + the schema version.

    `area_meta` carries the area-level values the import needs but that are not
    per-cell (origin, bbox, elevation floor) so the import can run from the
    unified files alone. `image_texture` (optional) is {file, extent} for the
    tile raster the add-on UV-maps onto the Terrain Surface.
    `provisional=True` stamps the manifest as an area build that was cut short
    before its full manifest write -- load_for_import refuses it, so a pipeline
    killed mid-build can never be silently imported with (0, 0) area defaults.
    """
    manifest = {
        "schema_version": SCHEMA_VERSION,
        "area_id":        area_id,
        "h3_resolution":  h3_resolution,
        "area":           area_meta or {},
        "cells_csv":      f"{area_id}_cells.csv",
        "n_cells":        n_cells,
        "feature_files":  {name: (f"{area_id}_{name}.json" if name == "climate"
                                  else f"{area_id}_{name}.geojson")
                           for name in feature_counts},
        "feature_counts": feature_counts,
        "attribution":    ATTRIBUTION,
    }
    if provisional:
        manifest["provisional"] = True
    if image_texture:
        manifest["image_texture"] = image_texture
    _atomic_json_dump(path, manifest, indent=2)

def read_subset_manifest(path):
    """Read a slim subset manifest -> (region_id, cell_ids: list, area: dict). Raises

def add_layer(manifest_path, layer):
    """Add (or replace, by name+kind) one modular layer in the manifest's `layers[]`
    block -- the append-only path any KIND uses (see layers.py). A reader that predates
    the kind simply skips it. Returns the updated layer list."""
    m = read_manifest(manifest_path)
    existing = m.get("layers") or []
    existing = [l for l in existing
                if not (l.get("name") == layer.get("name")
                        and l.get("kind") == layer.get("kind"))]
    existing.append(dict(layer))
    m["layers"] = existing
    _atomic_json_dump(manifest_path, m, indent=2)
    return existing

def _cells_to_arrays(rows):
    """CSV dict rows -> positional cell arrays in the layout the import reads:
    [lon, lat, ele_m, rot_z, radius_eq, h3_id, water_type, r, g, b,
     veg_type, water_depth_m, coast_dist_m, coast_dir_x, coast_dir_y, is_pentagon,
     slope_x, slope_y, regolith_material, regolith_seepage_mm_day,
     regolith_depth_m].
    Append-only: new fields go at the end so existing indices never shift."""
    rows = list(rows)
    # An EMPTY slope field in a current-schema CSV means an unknown bed (the
    # writer stores NULL deliberately, coast.slope_fields: "never silently
    # flattened to 0"). The Blender attribute is a dense float array, so 0
    # (flat, no shoreline steering) is the only representable fallback -- but
    # it must not be silent. A CSV that PREDATES the slope column has no such
    # field at all; that case stays quiet (genuinely unknown -> default).
    _n_null_slope = sum(1 for d in rows
                        if "slope_x" in d and not (d.get("slope_x") or "").strip())
    if _n_null_slope:
        print(f"  [import] WARNING: {_n_null_slope:,} cells have NULL slope "
              f"(unknown bed); the Blender terrain_slope falls back to flat "
              f"(0, 0) for them -- no shoreline wave steering on those cells",
              flush=True, file=sys.stderr)
    return [[
        float(d["lon"]), float(d["lat"]), float(d["ele_m"]),
        float(d["rot_z"]), float(d["radius_eq"]), d["h3_id"],
        d["water_type"], int(d["r"]), int(d["g"]), int(d["b"]),
        d["veg_type"], float(d["water_depth_m"]),
        float(d.get("coast_dist_m") or 0.0),
        float(d.get("coast_dir_x") or 0.0),
        float(d.get("coast_dir_y") or 0.0),
        int(d.get("is_pentagon") or 0),
        float(d.get("slope_x") or 0.0),
        float(d.get("slope_y") or 0.0),
        d.get("regolith_material") or "bedrock",
        float(d.get("regolith_seepage_mm_day") or 0.0),
        float(d.get("regolith_depth_m") or 0.0),
    ] for d in rows]


def _signed_area2(ring):
    """Twice the signed area of a [[lon, lat], ...] ring (shoelace). Sign is all
    the callers need: positive = CCW (interior on the left of each edge)."""
    n = len(ring)
    return sum(ring[i][0] * ring[(i + 1) % n][1]
               - ring[(i + 1) % n][0] * ring[i][1] for i in range(n))

def outward_ring(ring):
    """Normalise a footprint ring for extrusion: unclosed, no consecutive
    duplicate points, CCW winding; [] for a degenerate ring (< 3 pts or zero
    area -- collinear rings cannot extrude).

    CCW is the winding the Blender extrude assumes: for CCW input the wall
    quads (base_i, base_i+1, top_i+1, top_i) face OUTWARD and the roof cap
    faces UP. Source rings arrive in either winding (OSM ways are unordered;
    WKB has no winding mandate) and ~45% of real footprints are CW, which
    inverts every normal in the mesh -- and the Structures material maps its
    wall/roof textures by face normal, so inverted normals corrupt the
    texture. lon/lat vs projected XY does not matter here: the equirectangular
    import projection scales both axes positively, preserving orientation.
    """
    pts = [list(p[:2]) for p in ring]
    if len(pts) > 1 and pts[0] == pts[-1]:
        pts.pop()                                    # unclose
    dedup = pts[:1]
    for p in pts[1:]:
        if p != dedup[-1]:
            dedup.append(p)
    if len(dedup) > 1 and dedup[0] == dedup[-1]:
        dedup.pop()                                  # closure hidden by a dup run
    if len(dedup) < 3:
        return []
    a2 = _signed_area2(dedup)
    if abs(a2) < 1e-14:                              # zero-area / collinear
        return []
    if a2 < 0:
        dedup.reverse()
    return dedup




def _ring_from_polygon(geometry):
    coords = (geometry or {}).get("coordinates") or []
    return outward_ring(coords[0] if coords else [])


def _leaf_lonlat(coords):
    """Yield every (lon, lat) leaf pair in a GeoJSON coordinates array, at any nesting
    depth (Point / LineString / Polygon / Multi*)."""
    if not coords:
        return
    if len(coords) >= 2 and isinstance(coords[0], (int, float)) \
            and isinstance(coords[1], (int, float)):
        yield coords[0], coords[1]
        return
    for c in coords:
        yield from _leaf_lonlat(c)


class _Region:
    """The exact outline of the selected H3 cells as one shapely (multi)polygon, used
    to clip OSM features to a subset. `h3.cells_to_geo` builds the union boundary of the
    cells (concavities and holes and all -- the 'cutting plane' the whole feature spans);
    shapely does the actual geometry clipping. A prepared copy makes the inside tests
    fast, and the region's bounding box cheaply rejects the far-away features (streamed
    from the full-area source) before any shapely geometry is built."""

    def __init__(self, cell_ids):
        import h3
        from shapely.geometry import shape
        from shapely.prepared import prep
        from shapely.validation import make_valid
        self._shape = shape
        self._make_valid = make_valid
        geom = shape(h3.cells_to_geo(sorted(cell_ids)))
        self.geom = geom if geom.is_valid else make_valid(geom)
        self.prep = prep(self.geom)
        self.bounds = self.geom.bounds              # (lon_min, lat_min, lon_max, lat_max)

    def near(self, feature):
        """Cheap streaming pre-reject: any vertex inside the region bbox. Prunes the
        millions of far features before shapely touches them."""
        lo0, la0, lo1, la1 = self.bounds
        return any(lo0 <= lo <= lo1 and la0 <= la <= la1
                   for lo, la in _leaf_lonlat((feature.get("geometry") or {}).get("coordinates")))

    def clip(self, geometry):
        """Intersect a GeoJSON geometry dict with the region. A feature that is wholly
        inside skips the (expensive) overlay; only border features are actually cut.
        Returns a shapely geometry, empty when the feature falls outside."""
        g = self._shape(geometry)
        if not g.is_valid:
            g = self._make_valid(g)
        if self.prep.contains(g):
            return g
        return g.intersection(self.geom)            # empty geometry if disjoint

    def contains_point(self, lon, lat):
        from shapely.geometry import Point
        return self.prep.intersects(Point(lon, lat))


def _polys(geom):
    """Yield each non-empty shapely Polygon in geom (Polygon / MultiPolygon / mixed
    GeometryCollection -- an intersection can drop to lines/points, which we skip)."""
    if geom.is_empty:
        return
    if geom.geom_type == "Polygon":
        yield geom
    elif geom.geom_type in ("MultiPolygon", "GeometryCollection"):
        for g in geom.geoms:
            yield from _polys(g)


def _lines(geom):
    """Yield each non-empty line in geom as a list of [lon, lat, (z)] points.
    Z is preserved when the source carries it: shapely 2 keeps Z through the
    intersection and interpolates it at cut vertices on the region boundary,
    so subset-imported roads keep their pipeline-baked elevation (a 2D list
    here meant every subset road imported flat at Z=0)."""
    if geom.is_empty:
        return
    if geom.geom_type == "LineString":
        yield [list(c) for c in geom.coords]
    elif geom.geom_type in ("MultiLineString", "GeometryCollection"):
        for g in geom.geoms:
            yield from _lines(g)


def _footprints(geometry, region):
    """Exterior rings (unclosed, >= 3 pts) of a polygon feature. With no region the
    footprint is kept whole; with one it is clipped to the region outline first, so a
    border building or a kilometres-long bridge deck becomes its in-region part(s) --
    possibly several rings when the cut splits the polygon."""
    if region is None:
        ring = _ring_from_polygon(geometry)
        if len(ring) >= 3:
            yield ring
        return
    for poly in _polys(region.clip(geometry or {})):
        ring = outward_ring(poly.exterior.coords)    # drops Z, uncloses, forces CCW
        if len(ring) >= 3:
            yield ring


def _clip_line_features(features, region):
    """Clip each line feature to the region outline, emitting one LineString Feature per
    surviving segment (a road that leaves and re-enters the region is split, not bridged
    across the gap). Properties are carried through."""
    for ft in features:
        props = ft.get("properties")
        for line in _lines(region.clip(ft.get("geometry") or {})):
            if len(line) >= 2:
                yield {"type": "Feature", "properties": props,
                       "geometry": {"type": "LineString", "coordinates": line}}


def _ticker(iterable, report, lo, hi, label, throttle=0.3):
    """Pass `iterable` through unchanged while driving `report(frac, msg)` at most once
    per `throttle` seconds. Feature totals are unknown (the source layers are streamed,
    and counting first would double the read), so the fraction creeps asymptotically from
    `lo` toward `hi` as the count grows -- always visibly moving, never overshooting the
    phase's band. `report` is a no-op when the caller passed no progress callback."""
    import time
    last = time.monotonic()
    n = 0
    for item in iterable:
        yield item
        n += 1
        now = time.monotonic()
        if now - last >= throttle:
            report(lo + (hi - lo) * (1.0 - 1.0 / (1.0 + n / 3000.0)), f"{label}: {n:,}")
            last = now
    report(hi, f"{label}: {n:,}")


def _iter_features(path, keep):
    """Yield features of a FeatureCollection where keep(feature) is truthy (keep=None ->
    all). Streams with ijson when a NATIVE (C) backend is present, so peak memory stays
    low on the ~500MB source layers. Otherwise -- no ijson (Blender's bundled Python), or
    only ijson's pure-Python backend, which is slower than a C json load -- fall back to a
    full read. `pip install ijson` (with its yajl2 C backend) into Blender's Python to get
    the streaming path there."""
    try:
        import ijson
        if getattr(ijson, "backend", "python") != "python":     # C backend only
            with open(path, "rb") as f:
                for feat in ijson.items(f, "features.item", use_float=True):
                    if keep is None or keep(feat):
                        yield feat
            return
    except ImportError:
        pass
    for feat in read_feature_collection(path):
        if keep is None or keep(feat):
            yield feat

def load_for_import(output_dir, area_id, progress=None, subset=None, data_dir=None):
    """Reconstruct `features_by_type` for the import from the unified files.

    Returns the same dict shape the import builds from the combined GeoJSON:
    a synthesized `terrain` feature (area meta + cells), plus structures (with
    bridges folded back in), streets and landmarks. No sea/water_body/vegetation
    features -- those are per-cell attributes now.

    `output_dir` / `area_id` name a REGION manifest -- the sole owner of the data
    files. `subset`, if given, is a {cell_ids, area} trim (from `read_subset_manifest`):
    only those cells become Terrain Centres and OSM features are clipped to their exact
    outline (see `_Region`); its `area` bbox overrides the region's for OSM trimming.
    A subset manifest passed as the region raises -- a subset is never a standalone
    dataset (data is managed at the region level).

    `data_dir` (optional) is where the layer DATA files (feature geojsons, celldata
    csvs, color geojson) live; it defaults to `output_dir` (the legacy single-folder
    layout). The region-folder layout keeps identity files in the region root and
    derived files in the region's output/ subdir, so callers pass data_dir=region
    output there. The manifest + cells csv are always read from `output_dir`.

    `progress`, if given, is called as `progress(frac, message)` (frac in 0..1) as each
    layer streams, so a caller can drive a progress bar during the (deliberately correct,
    not fast) clip. Exceptions from it are swallowed -- progress never breaks an import.
    """
    def report(frac, msg):
        if progress is not None:
            try:
                progress(frac, msg)
            except Exception:
                pass
    output_dir = Path(output_dir)
    manifest = read_manifest(output_dir / f"{area_id}_manifest.json")
    if manifest.get("provisional"):
        raise ValueError(
            f"{area_id}_manifest.json is a provisional manifest -- area build "
            "incomplete; re-run the pipeline")
    if is_subset_manifest(manifest):
        raise ValueError(f"{area_id}_manifest.json is a subset manifest; pass the "
                         "REGION manifest as the dataset and the subset separately")
    data_dir = Path(data_dir) if data_dir else output_dir   # region owns its data
    src_area = area_id
    cell_set = set(subset["cell_ids"]) if subset and subset.get("cell_ids") else None

    area = manifest.get("area", {})
    if subset and subset.get("area"):        # subset's trimmed bbox for OSM trimming/scale
        area = {**area, **subset["area"]}
    all_rows = read_cells_csv(output_dir / manifest["cells_csv"])
    if cell_set is not None:
        # Trim OSM features to the EXACT outline of the selected cells: the region is the
        # h3 union polygon of cell_set, and every layer is clipped to it (polygons cut,
        # lines split, points tested) -- no bbox, no Voronoi approximation. `keep` is a
        # cheap bbox pre-reject that streams the big source layers feature-by-feature.
        region = _Region(cell_set)
        keep = region.near
        rows = [r for r in all_rows if r["h3_id"] in cell_set]
        # A subset is resolution-specific: rebuilding the region at a different res
        # (or with a shifted selection) orphans every id in it. Zero overlap used to
        # slide through as a terrain feature with no cells -- the import then
        # silently produced no Terrain Centres/Surface while geometry-clipped
        # layers (contour lines) still landed. Fail LOUD instead.
        if not rows:
            raise ValueError(
                f"subset has 0 of {len(cell_set):,} cells in common with the region's "
                f"{len(all_rows):,} -- it was likely created against an older build "
                f"(resolution or selection changed). Re-create the subset.")
        if len(rows) < len(cell_set):
            print(f"  [subset] warning: {len(rows):,} of {len(cell_set):,} subset cells "
                  f"found in the region ({100 * len(rows) / len(cell_set):.0f}%) -- "
                  f"the subset may be stale")
    else:
        region = keep = None                                   # full area: no trimming
        rows = all_rows
    cells = _cells_to_arrays(rows)

    terrain_feat = {"type": "Feature", "geometry": None, "properties": {
        "feature_type":    "terrain",
        "area_id":         area_id,
        "data_dir":        str(data_dir),   # so cell_series bakers can open the CSR handlers
        "lon_centre":      area.get("lon_centre", 0.0),
        "lat_centre":      area.get("lat_centre", 0.0),
        "lon_min":         area.get("lon_min"),
        "lat_min":         area.get("lat_min"),
        "lon_max":         area.get("lon_max"),
        "lat_max":         area.get("lat_max"),
        "ele_min_m":       area.get("ele_min_m", 0.0),
        "ele_max_m":       area.get("ele_max_m", 0.0),
        "ele_min_baked_m": area.get("ele_min_baked_m", 0.0),
        "h3_resolution":   manifest.get("h3_resolution"),
        "n_lat": 0, "n_lon": 0,
        "cells": cells,
    }}

    # Data layer (optional): the manifest's per-field metadata + a
    # {h3_id: {field: value}} map from <area>_celldata.csv, so the import can bake
    # a Data object. Absent for areas built without custom data.
    data_fields = manifest.get("data_fields") or []
    cd_name = manifest.get("celldata_csv")
    celldata = {}
    interpdata = {}                        # {h3_id: {field}} -- fields filled by interpolation
    if data_fields and cd_name and (data_dir / cd_name).exists():
        names = [m["name"] for m in data_fields]
        for row in read_cells_csv(data_dir / cd_name):
            hid = row.get("h3_id")
            if not hid or (cell_set is not None and hid not in cell_set):
                continue
            vals, interp = {}, set()
            for fn in names:
                raw = row.get(fn, "")
                if raw != "":
                    try:
                        vals[fn] = float(raw)
                    except (ValueError, TypeError):
                        continue
                    if row.get(fn + "__i", "") == "1":   # interpolated, not measured
                        interp.add(fn)
            if vals:
                celldata[hid] = vals
            if interp:
                interpdata[hid] = interp
    terrain_feat["properties"]["data_fields"] = data_fields
    terrain_feat["properties"]["celldata"] = celldata
    terrain_feat["properties"]["interpdata"] = interpdata
    # Normalised layer list (new `layers` block, or legacy data_fields/feature_files)
    # so the importer bakes each declared layer dispatched by kind. See layers.py.
    from . import layers
    terrain_feat["properties"]["layers"] = layers.manifest_layers(manifest)

    # Per-face colours (optional): <area>_color.geojson -> {h3_id: [[r,g,b], ...]} ordered
    # by face (0 = centre, 1..6 = ring sectors). The import bakes these as vectors onto the
    # Terrain Centres; the Terrain Surface's hex_face attr picks which per face. The file
    # ALSO carries the regolith answer (sub_r/g/b — the palette colour of the cell's
    # effective regolith material); older files without it fall back to the image colour
    # per cell. Subset-
    # filtered by cell_set. Point features keyed by (h3_id, face).
    face_colors = {}
    regolith_face_colors = {}
    colp = data_dir / f"{src_area}_color.geojson"
    if colp.exists():
        by_cell = {}
        sub_by_cell = {}
        for feat in _iter_features(colp, None):
            p = feat.get("properties") or {}
            hid = p.get("h3_id")
            if hid is None or (cell_set is not None and hid not in cell_set):
                continue
            by_cell.setdefault(hid, {})[int(p.get("face", 0))] = [
                int(p.get("r", 128)), int(p.get("g", 128)), int(p.get("b", 128))]
            if "sub_r" in p:
                sub_by_cell.setdefault(hid, {})[int(p.get("face", 0))] = [
                    int(p.get("sub_r", 128)), int(p.get("sub_g", 128)), int(p.get("sub_b", 128))]
        for hid, fmap in by_cell.items():
            face_colors[hid] = [fmap.get(i, fmap.get(0, [128, 128, 128]))
                                for i in range(max(fmap) + 1)]
            smap = sub_by_cell.get(hid)
            regolith_face_colors[hid] = (
                [smap.get(i, smap.get(0, [128, 128, 128])) for i in range(max(fmap) + 1)]
                if smap else
                [list(face_colors[hid][0]) for _ in face_colors[hid]])
    terrain_feat["properties"]["face_colors"] = face_colors
    terrain_feat["properties"]["regolith_face_colors"] = regolith_face_colors

    # Image Texture (optional): the tile raster + its Web-Mercator extent,
    # for the add-on to UV-map onto the Terrain Surface. The path is resolved against
    # data_dir here so the importer needs no path logic. Absent unless the pipeline
    # built one; a missing PNG is dropped silently. Legacy manifests name the block
    # `surface_texture` -- manifest_image_texture reads both.
    tex = manifest_image_texture(manifest)
    if tex and tex.get("file") and (data_dir / tex["file"]).exists():
        terrain_feat["properties"]["image_texture"] = {
            "path":   str((data_dir / tex["file"]).resolve()),
            "extent": tex.get("extent") or {}}

    fbt = {"terrain": [terrain_feat]}

    # structures + bridges -> one list-in-properties feature (decks fold back in
    # so bridge=yes road projection still finds deck=True targets). Both are polygon
    # layers clipped to the region outline the same way (`region=None` -> kept whole).
    # Phase bands for the progress bar (structures dominate the clip cost). Each layer's
    # stream creeps its band toward the next while the source is read + clipped.
    structs = []
    for (lo, hi), layer in zip(((0.00, 0.40), (0.40, 0.50)),
                               (f"{src_area}_structures.geojson", f"{src_area}_bridges.geojson")):
        pf = data_dir / layer
        if not pf.exists():
            continue
        label = "bridges" if lo else "structures"
        for feat in _ticker(_iter_features(pf, keep), report, lo, hi, label):
            props = dict(feat.get("properties") or {})
            props.pop("feature_type", None)
            for ring in _footprints(feat.get("geometry"), region):
                s = dict(props)
                s["footprint"] = ring
                structs.append(s)
    if structs:
        fbt["structures"] = [{"type": "Feature", "geometry": None, "properties": {
            "feature_type": "structures", "area_id": area_id, "structures": structs}}]

    sp = data_dir / f"{src_area}_streets.geojson"
    if sp.exists():
        stream = _ticker(_iter_features(sp, keep), report, 0.50, 0.75, "streets")
        fbt["street"] = (list(_clip_line_features(stream, region))
                         if region is not None else list(stream))

    lp = data_dir / f"{src_area}_landmarks.geojson"
    if lp.exists():
        lms = []
        for feat in _ticker(_iter_features(lp, keep), report, 0.75, 0.80, "landmarks"):
            coords = (feat.get("geometry") or {}).get("coordinates") or [None, None]
            lon, lat = coords[0], coords[1]
            if region is not None and not (lon is not None and region.contains_point(lon, lat)):
                continue                        # point outside the selected-cell outline
            props = dict(feat.get("properties") or {})
            props.pop("feature_type", None)
            props["lon"], props["lat"] = lon, lat
            lms.append(props)
        if lms:
            fbt["landmarks"] = [{"type": "Feature", "geometry": None, "properties": {
                "feature_type": "landmarks", "area_id": area_id, "landmarks": lms}}]

    cp = data_dir / f"{src_area}_contour_lines.geojson"
    if cp.exists():
        stream = _ticker(_iter_features(cp, keep), report, 0.80, 1.00, "contours")
        clines = (list(_clip_line_features(stream, region))
                  if region is not None else list(stream))
        if clines:
            fbt["contour_lines"] = clines

    report(1.0, "done")
    return fbt
