"""
water.py
--------
OSM water body and coastline fetching, polygon assembly, sea normal
determination, H3 hex mesh generation, and water/sea depression baking.
"""

import math
import json as _json

from . import config as _config
from .osm_geom import split_elements, stitch_rings, endpoints_match
from .pbf_reader import _is_island
from .progress import bar

R_EARTH = 6_371_000.0

_FLOWING_WATERWAYS = {"river", "stream", "canal", "drain", "ditch"}

# Tags that must NEVER be enclosed-carved out of land: the large tidal/sea masses.
# Their OSM outers can trace the land itself (the Venice lagoon stitches into
# outers over the city), and carving those floods the city. Every other tag --
# including generic `natural=water` and `wetland` -- IS carved when it sits
# enclosed in land (Donegal loughs, Arsenale darsene, Austevoll's island ponds),
# so real enclosed bodies fill like the sea does. Two guards keep this safe:
# the body must be >=80% inside the land union, and it must not swallow >=60%
# of any single island/hole piece (the Sydney Goat Island case).
_NO_CARVE_TAGS = {"lagoon", "sea", "ocean", "coastline", "bay"}


def _parse_width_m(tags):
    """Leading numeric value of an OSM `width` tag, in metres, or None."""
    import re
    w = tags.get("width")
    if not w:
        return None
    m = re.match(r"\s*([0-9]+(?:\.[0-9]+)?)", str(w))
    return float(m.group(1)) if m else None


def _parse_ele_m(tags):
    """OSM `ele` tag (metres) as a float, else None when the tag is absent
    or carries no number. An explicit "0" parses to 0.0 — distinct from
    absence — so a genuinely sea-level standing body keeps its real surface
    instead of falling back to the terrain percentile in the pipeline."""
    import re
    e = tags.get("ele")
    if e is None:
        return None
    m = re.match(r"\s*(-?[0-9]+(?:\.[0-9]+)?)", str(e))
    return float(m.group(1)) if m else None

def buffer_centreline(geom, width_m):
    """Buffer a waterway centreline [(lon, lat), ...] by `width_m` into a polygon
    footprint [[lon, lat], ...]. Buffers in a locally isotropic frame (lon scaled
    by cos(lat)) so the width is correct in metres regardless of latitude. Flat
    end caps keep the canal from over-running its endpoints. Returns the exterior
    ring, or None if it can't be built."""
    try:
        from shapely.geometry import LineString
    except ImportError:
        return None
    pts = [(p[0], p[1]) for p in geom]
    if len(pts) < 2 or not width_m or width_m <= 0:
        return None
    lat0 = sum(p[1] for p in pts) / len(pts)
    k = math.cos(math.radians(lat0)) or 1.0
    line = LineString([(x * k, y) for x, y in pts])
    poly = line.buffer((width_m / 2.0) / 111_320.0, cap_style=2, join_style=1)
    if poly.is_empty:
        return None
    if poly.geom_type == "MultiPolygon":
        poly = max(poly.geoms, key=lambda g: g.area)
    return [[x / k, y] for x, y in poly.exterior.coords]




def _osm_cache_path(area_id, data_type, raw_data_dir):
    return raw_data_dir / "osm" / f"{area_id}_{data_type}.json"


def _load_osm_cache(area_id, data_type, raw_data_dir):
    p = _osm_cache_path(area_id, data_type, raw_data_dir)
    return _json.loads(p.read_text()) if p.exists() else None


def _save_osm_cache(area_id, data_type, data, raw_data_dir):
    p = _osm_cache_path(area_id, data_type, raw_data_dir)
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(_json.dumps(data))

def _extended_bbox(lat_min, lon_min, lat_max, lon_max):
    """Compute extended bbox corners at +_config.COASTLINE_EXTEND_KM."""
    ext_m      = _config.COASTLINE_EXTEND_KM * 1000.0
    lat_centre = (lat_min + lat_max) / 2.0
    cos_lat    = math.cos(math.radians(lat_centre))
    d_lat      = math.degrees(ext_m / R_EARTH)
    d_lon      = math.degrees(ext_m / (R_EARTH * cos_lat))
    return (
        lon_min - d_lon, lat_min - d_lat,
        lon_max + d_lon, lat_max + d_lat,
    )

def build_island_rings(island_elements):
    """Stitch island/land polygons (place=island/islet + named '... Island' land)
    into closed rings [[lon,lat],...]. Used to subtract islands from both water
    bodies (process_osm_water) and the sea polygon (pipeline), so harbour islands
    render as land rather than being flooded as water/sea.

    Only GENUINE island elements are stitched. Geometry completion drags plain
    `natural=coastline` mainland ways into the islands bucket (e.g. Donegal: 6886
    open coastline ways vs 1347 real islands); stitching those built the whole
    coastline into bogus 'island' land that, subtracted from the sea, ate the
    coastal cells. The coastline is build_sea_polygons' job -- not land here."""
    if not island_elements:
        return []
    i_geom, i_ways, i_rels = split_elements(island_elements)
    rings = []
    for el in i_ways:
        if not _is_island(el.get("tags") or {}):
            continue
        g = i_geom.get(el["id"], [])
        if len(g) >= 3:
            rings.append([[p[0], p[1]] for p in g])
    for el in i_rels:
        if not _is_island(el.get("tags") or {}):
            continue
        outer = [m["ref"] for m in el.get("members", [])
                 if m.get("type") == "way" and m.get("role") == "outer"]
        for r in stitch_rings(outer, i_geom):
            if len(r) >= 3:
                rings.append([[p[0], p[1]] for p in r])
    return rings

def extract_water_inner_rings(water_elements):
    """Every INNER ring of a water multipolygon is land (an island the water
    encloses). Stitch them straight from the relations -- independent of whether
    the OUTER ring stitches into a usable body. This is what makes harbour/lagoon
    islands (e.g. Goat Island, carried as an inner ring on a relation whose outer
    barely closes) survive as land, generally, with no per-feature special cases."""
    if not water_elements:
        return []
    way_geom, _ways, relations = split_elements(water_elements)
    rings = []
    for el in relations:
        inner = [m["ref"] for m in el.get("members", [])
                 if m.get("type") == "way" and m.get("role") == "inner"]
        if not inner:
            continue
        for r in stitch_rings(inner, way_geom):
            if len(r) >= 3:
                rings.append([[p[0], p[1]] for p in r])
    return rings

def build_land_geometry(water_bodies, island_rings):
    """The single, general definition of land-inside-water: the union of every
    inner-ring HOLE of a water body (an island the water encloses) plus every
    `place=island` ring, MINUS the explicit through-water (canals/rivers/...) that
    cuts across it. Subtracted from ALL water (sea + bodies) so an island is never
    flooded, no matter how the surrounding water is tagged. Replaces the old
    per-relation / peninsula / tidal special-casing. Returns a shapely geometry or
    None.

    The through-water carve fixes Venice: a `place=island` is the whole sestiere,
    its outline is the OUTER coast and does NOT subtract the canals threading
    through it -- so without this, every internal canal sits inside the island
    polygon and the blanket land rule re-floods it to land. Canals are water, so
    carve canal/river/stream/drain/ditch footprints back out. Other standing water
    (lake/pond/wetland/harbour/...) is carved too, but ONLY when it sits (almost)
    entirely inside the land and does not swallow a whole island piece;
    lagoon/sea/ocean/coastline/bay are excluded by tag outright (their coarse
    outers can trace the land itself)."""
    try:
        from shapely.geometry import Polygon
        from shapely.ops import unary_union
    except ImportError:
        return None

    def _poly(ring, holes=None):
        if not ring or len(ring) < 3:
            return None
        try:
            p = Polygon([(x, y) for x, y in ring],
                        [[(x, y) for x, y in h] for h in (holes or []) if len(h) >= 3])
            if not p.is_valid:
                p = p.buffer(0)
            return None if p.is_empty else p
        except Exception:
            return None

    polys = []
    for body in water_bodies or []:
        for hole in body.get("holes", []):
            if (p := _poly(hole)) is not None:
                polys.append(p)
    for ring in island_rings or []:
        if (p := _poly(ring)) is not None:
            polys.append(p)
    if not polys:
        return None
    land = unary_union(polys)

    # Carve explicit water back out of the land where mapped water conflicts with
    # the coarse land geometry. Two cases, both keeping each body's own holes so an
    # islet inside the water stays land:
    #   - flowing water (canal/river/...) ALWAYS cuts through -- a canal threading
    #     a place=island must stay water;
    #   - any other water body (lake/reservoir/pond/basin/harbour/dock/wetland/...)
    #     cuts through ONLY when it sits (almost) entirely inside the land -- an
    #     enclosed body walled in by land (a Donegal lough, an Arsenale darsena) --
    #     AND it does not SWALLOW a whole island. The `harbour` tag is ambiguous: a
    #     small basin (Arsenale) is enclosed BY land and must carve, but a coastal
    #     harbour's outer (Sydney's Port Jackson, rel 1252425) can stitch into a
    #     fragment that closes tightly around an island (Goat Island) -- also "inside
    #     the land", but it IS the island, not a basin in it. So an enclosed body
    #     that covers the bulk of any single island/hole piece is refused: carving
    #     it would delete that island. lagoon/sea are excluded by tag outright.
    cut = []
    for body in water_bodies or []:
        p = _poly(body.get("footprint"), body.get("holes"))
        if p is None:
            continue
        if body.get("water_tag") in _FLOWING_WATERWAYS:
            cut.append(p)
        elif (body.get("water_tag") not in _NO_CARVE_TAGS and p.area > 0
              and p.intersection(land).area >= 0.8 * p.area
              and not any(ip.area > 0 and p.intersection(ip).area >= 0.6 * ip.area
                          for ip in polys)):
            cut.append(p)
    if cut:
        land = land.difference(unary_union(cut))
    return None if land.is_empty else land

def process_osm_water(elements, area_id, lat_min, lon_min, lat_max, lon_max, raw_data_dir,
                      island_elements=None):
    """Process water body elements from PBF into water body dicts.

    Args:
        elements: Overpass-compatible element list from pbf_reader (ways +
            relations).
        island_elements: optional land polygons (place=island/islet + named
            island land such as the Garden Island military base) to subtract
            from water -- for islands/peninsulas the harbour outer overlaps but
            that OSM does not model as inner rings.

    Returns:
        list of {osm_id, osm_type, water_tag, footprint, holes, ele_m}.
    """
    cached = _load_osm_cache(area_id, "water", raw_data_dir)
    if cached is not None:
        return cached

    way_geom, direct_ways, relations = split_elements(elements)

    def primary_tag(el):
        tags = el.get("tags", {})
        for key in ("water", "natural", "landuse", "waterway"):
            if key in tags:
                return tags[key]
        return "water"

    water = []
    seen  = set()

    for el in bar(direct_ways, total=len(direct_ways), label="[water] ways"):
        eid = el["id"]
        if eid in seen:
            continue
        seen.add(eid)
        geom = way_geom.get(eid, [])
        if len(geom) < 2:
            continue
        tag  = primary_tag(el)
        if tag == "bay":
            continue
        tags = el.get("tags", {})
        is_closed = len(geom) >= 3 and endpoints_match(geom[0], geom[-1])

        # Linear waterway centreline (canal/river/stream/...): an open way, so the
        # raw geometry is a line that collapses to a zero-area polygon and yields
        # no cells. Give it a body by buffering to its mapped `width`. No width tag
        # -> skip: we don't assume a default, so Venice's many width-less Rio
        # canals stay out by design.
        if not is_closed and tags.get("waterway") in _FLOWING_WATERWAYS:
            width_m = _parse_width_m(tags)
            if width_m is None:
                continue
            fp = buffer_centreline(geom, width_m)
            if not fp or len(fp) < 3:
                continue
            water.append({"osm_id": eid, "osm_type": "way",
                          "water_tag": tags["waterway"],
                          "footprint": fp, "holes": [],
                          "ele_m": _parse_ele_m(tags)})
            continue

        if len(geom) < 3:
            continue
        fp  = [[p[0], p[1]] for p in geom]
        water.append({"osm_id": eid, "osm_type": "way", "water_tag": tag,
                      "footprint": fp, "holes": [],
                      "ele_m": _parse_ele_m(tags),
                      })

    for el in bar(relations, total=len(relations), label="[water] relations"):
        eid = el["id"]
        if eid in seen:
            continue
        seen.add(eid)
        outer_ids = [m["ref"] for m in el.get("members", [])
                     if m.get("type") == "way" and m.get("role") == "outer"]
        inner_ids = [m["ref"] for m in el.get("members", [])
                     if m.get("type") == "way" and m.get("role") == "inner"]
        if not outer_ids:
            continue
        tag = primary_tag(el)
        if tag == "bay":
            continue

        # Separate pre-closed ways from open ways before stitching.
        # Stitching all 483 outer ways into one ring produces a self-intersecting
        # monster; each independently closed way is already a valid polygon.
        closed_fps = []
        open_ids   = []
        for oid in outer_ids:
            geom = way_geom.get(oid, [])
            if len(geom) >= 3 and endpoints_match(geom[0], geom[-1]):
                closed_fps.append([[p[0], p[1]] for p in geom])
            elif geom:
                open_ids.append(oid)

        stitched_fps = [[[p[0], p[1]] for p in r] for r in stitch_rings(open_ids, way_geom)]
        all_fps = closed_fps + stitched_fps

        if not all_fps:
            continue

        for mid in outer_ids:
            seen.add(mid)

        holes = []
        if inner_ids:
            for ir in stitch_rings(inner_ids, way_geom):
                hole = [[p[0], p[1]] for p in ir]
                if len(hole) >= 3:
                    holes.append(hole)
            for mid in inner_ids:
                seen.add(mid)

        for fp in all_fps:
            if len(fp) < 3:
                continue
            water.append({"osm_id": eid, "osm_type": "relation", "water_tag": tag,
                          "footprint": fp, "holes": holes,
                          "ele_m": _parse_ele_m(el.get("tags") or {}),
                          })

    # Clip to bbox and drop bodies covered by larger bodies
    try:
        from shapely.geometry import Polygon as ShapelyPoly, box
        from shapely.validation import make_valid as _make_valid
        import copy as _copy
        bbox_poly = box(lon_min, lat_min, lon_max, lat_max)
        # Islands are handled generally downstream: every water inner-ring +
        # place=island feeds build_land_geometry, which is subtracted from all
        # water in the pipeline. process_osm_water just keeps each body's own
        # holes; no cross-relation island collection / punching here.
        clipped   = []
        n_trimmed = 0
        n_dropped = 0
        for body in water:
            fp = body.get("footprint", [])
            if len(fp) < 3:
                continue
            try:
                # Build the polygon WITH its holes before clipping, so islands
                # are part of the geometry. A naive footprint-only clip that
                # produced a MultiPolygon (big harbour split at the bbox edge)
                # used to drop every hole -- losing all of Port Jackson's ~25
                # island inner rings. intersection() distributes the interiors
                # to whichever part contains them.
                holes_in = [[(p[0], p[1]) for p in h]
                            for h in body.get("holes", []) if len(h) >= 3]
                poly = ShapelyPoly([(p[0], p[1]) for p in fp], holes_in)
                if not poly.is_valid:
                    poly = _make_valid(poly)
                if bbox_poly.contains(poly):
                    clipped.append(body)
                    continue
                intersect = poly.intersection(bbox_poly)
                if intersect.is_empty:
                    n_dropped += 1
                    continue
                geoms = intersect.geoms if hasattr(intersect, "geoms") else [intersect]
                parts = [g for g in geoms if g.geom_type == "Polygon" and not g.is_empty]
                if not parts:
                    n_dropped += 1
                    continue
                for k, part in enumerate(parts):
                    b = body if k == 0 else _copy.copy(body)
                    b["footprint"] = [[c[0], c[1]] for c in part.exterior.coords]
                    b["holes"] = [[[c[0], c[1]] for c in r.coords]
                                  for r in part.interiors]
                    b["touches_bbox"] = True
                    clipped.append(b)
                n_trimmed += 1
            except Exception:
                clipped.append(body)
        water = clipped
        print(f"  [water clip] {len(water)} kept, {n_trimmed} trimmed, {n_dropped} dropped")

        # NOTE: no coverage dedup. Overlapping water bodies are kept -- dropping a
        # body that is "covered by a larger one" also discarded its island HOLES
        # (e.g. Goat Island's hole rides the Sydney Harbour relation, which the
        # larger hole-less Port Jackson relation would shadow). Keeping every body
        # means build_land_geometry sees every hole. Per-cell water is idempotent
        # under overlap, so this only adds redundant features, not wrong geometry.

    except ImportError:
        print("  [water clip] shapely not available - skipping clip")

    _save_osm_cache(area_id, "water", water, raw_data_dir)
    return water

def process_osm_coastline(elements, area_id, lat_min, lon_min, lat_max, lon_max, raw_data_dir):
    """
    Process coastline way elements from PBF into chain data.

    elements : coastline way elements from pbf_reader (no bbox pre-filter).
    Returns (chains, had_reversals) where chains is a list of [[lon,lat],...] open chains,
    or (None, False) if no coastline found.
    """
    cached = _load_osm_cache(area_id, "coastline", raw_data_dir)
    if cached is not None:
        chains_data, had_rev = cached[0], cached[1]
        if chains_data and not isinstance(chains_data[0][0], list):
            chains_data = [chains_data]
        return chains_data, had_rev

    # Filter to extended bbox so we get coverage beyond the terrain edge
    # without stitching the entire national coastline.
    ext_lon_min, ext_lat_min, ext_lon_max, ext_lat_max = _extended_bbox(
        lat_min, lon_min, lat_max, lon_max)

    segments = []
    for el in elements:
        if el.get("type") != "way":
            continue
        geom = el.get("geometry", [])
        if not geom:
            continue
        if not any(ext_lat_min <= g['lat'] <= ext_lat_max and
                   ext_lon_min <= g['lon'] <= ext_lon_max for g in geom):
            continue
        nodes = [(g["lon"], g["lat"]) for g in geom]
        if len(nodes) >= 2:
            segments.append(nodes)

    if not segments:
        return None, False

    TOL = 1e-5

    def endpoints_match(a, b):
        return abs(a[0] - b[0]) < TOL and abs(a[1] - b[1]) < TOL

    chains        = [list(seg) for seg in segments]
    had_reversals = False
    merged        = True
    while merged:
        merged = False
        for i in range(len(chains)):
            for j in range(len(chains)):
                if i == j:
                    continue
                if endpoints_match(chains[i][-1], chains[j][0]):
                    chains[i].extend(chains[j][1:])
                    chains.pop(j)
                    merged = True
                    break
                if endpoints_match(chains[i][-1], chains[j][-1]):
                    chains[i].extend(reversed(chains[j][:-1]))
                    chains.pop(j)
                    had_reversals = True
                    merged        = True
                    break
            if merged:
                break

    chains_out = [[[p[0], p[1]] for p in ch] for ch in sorted(chains, key=len, reverse=True)]
    _save_osm_cache(area_id, "coastline", [chains_out, had_reversals], raw_data_dir)
    return chains_out, had_reversals

def build_sea_polygons(sea_chains, lat_min, lon_min, lat_max, lon_max,
                       had_reversals=False, structures=None):
    """
    Build sea polygon(s) for the terrain bbox.

    Strategy:
      - Closed chains (islands) are clipped to the bbox and added as land polygons,
        AND join the polygonize line work so the tidal channels around islands
        form faces of their own.
      - Open chains + closed rings are polygonized together with the bbox
        boundary to produce faces.  Each face part is classified by a majority
        vote of its own boundary edges against the original oriented coastline
        segments (geometric match within ~1 cm, immune to unary_union
        re-noding): an edge running WITH a land-on-the-left segment puts a
        CCW-ringed face on the land side.  Faces with no matching boundary
        edge fall back to the representative-point side test.  This is
        deterministic and handles arbitrarily complex coastlines including
        bays, headlands, island straits, and multiple chain passes through
        the bbox.
      - Sea = bbox minus union(all land polygons).

    Returns a list of (exterior_coords, [hole_coords, ...]) tuples, or [].
    """
    try:
        from shapely.geometry import box, LineString, Polygon as _SPoly
        from shapely.ops import unary_union, polygonize
    except ImportError:
        print("  [sea] shapely not installed - cannot build sea polygon")
        return []

    if not sea_chains:
        return []
    if not isinstance(sea_chains[0][0], list):
        sea_chains = [sea_chains]

    _TOL            = 1e-5
    MIN_AREA        = 1e-8
    OUTPUT_MIN_AREA = 2e-7
    margin = max(lon_max - lon_min, lat_max - lat_min) * 0.02

    bbox_poly  = box(lon_min, lat_min, lon_max, lat_max)
    land_polys = []
    n_closed   = 0
    n_open     = 0

    def _overlaps(pts):
        lons = [p[0] for p in pts]; lats = [p[1] for p in pts]
        return not (max(lons) < lon_min or min(lons) > lon_max or
                    max(lats) < lat_min or min(lats) > lat_max)

    def _extend(inner, outer):
        dx = outer[0] - inner[0]; dy = outer[1] - inner[1]
        L = (dx*dx + dy*dy) ** 0.5 or 1.0
        s = (L + margin) / L
        return (inner[0] + dx*s, inner[1] + dy*s)

    # ------------------------------------------------------------------
    # Pass 1: closed chains (islands / enclosed land)
    # ------------------------------------------------------------------
    open_chains = []
    closed_ring_pts = []   # island rings, also fed into the pass-2 line work
    ccw_rings = []         # land interiors (islands)
    cw_rings = []          # WATER interiors per the convention (lagoons / inlets)
    for chain in sea_chains:
        pts = [(p[0], p[1]) for p in chain]
        if len(pts) < 3 or not _overlaps(pts):
            continue
        is_closed = (abs(pts[0][0]-pts[-1][0]) < _TOL and
                     abs(pts[0][1]-pts[-1][1]) < _TOL)
        if is_closed:
            # Land-on-the-left: a CCW ring encloses land (island); a CW ring
            # encloses WATER (lagoon / sea inlet). CW interiors become holes
            # in any CCW land poly that contains them, and every ring joins
            # the pass-2 line work: without it the tidal channels around an
            # island never form faces of their own -- they are absorbed into
            # the adjacent mainland face and inherit its verdict (the Hudson /
            # East River around Manhattan, the straits around Staten Island
            # vanished from the sea this way).
            ring_a2 = sum(pts[i][0] * pts[i + 1][1] - pts[i + 1][0] * pts[i][1]
                          for i in range(len(pts) - 1))
            (ccw_rings if ring_a2 > 0 else cw_rings).append(pts)
            closed_ring_pts.append(pts)
        else:
            extended = [_extend(pts[1], pts[0])] + pts + [_extend(pts[-2], pts[-1])]
            open_chains.append((pts, LineString(extended)))

    cw_polys = []          # (pts, shapely poly) for the water-interior rings
    for pts in cw_rings:
        try:
            p = _SPoly(pts)
            if not p.is_valid:
                p = p.buffer(0)
            if not p.is_empty:
                cw_polys.append((pts, p))
        except Exception:
            pass

    for pts in ccw_rings:
        try:
            poly = _SPoly([(x, y) for x, y in pts])
            if not poly.is_valid:
                poly = poly.buffer(0)
            holes = [cw for cw, cp in cw_polys if poly.covers(cp)]
            if holes:
                poly = _SPoly([(x, y) for x, y in pts],
                              [[(x, y) for x, y in h] for h in holes])
                if not poly.is_valid:
                    poly = poly.buffer(0)
            clipped = poly.intersection(bbox_poly)
            if clipped.is_empty:
                continue
            for g in ([clipped] if clipped.geom_type == 'Polygon'
                      else clipped.geoms if clipped.geom_type == 'MultiPolygon'
                      else []):
                if g.area >= MIN_AREA:
                    land_polys.append(g)
                    n_closed += 1
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Pass 2: open chains via polygonize + segment-direction classification
    # ------------------------------------------------------------------
    print(f"  [sea] {len(open_chains)} open chain(s), had_reversals={had_reversals}")
    for ci, (pts, _) in enumerate(open_chains):
        print(f"    chain {ci}: {len(pts)} pts  "
              f"start=({pts[0][0]:.4f},{pts[0][1]:.4f})  "
              f"end=({pts[-1][0]:.4f},{pts[-1][1]:.4f})")

    if open_chains or closed_ring_pts:
        try:
            from shapely.strtree import STRtree
            from shapely.geometry import Point as _Pt

            # Classify each polygonized face by a MAJORITY VOTE of its own
            # boundary edges, matched back to the original oriented coastline
            # segments. OSM coastline runs land-on-the-left, so an edge that
            # coincides with an original segment tells us directly which side
            # the face's interior is on: same direction as the original (and a
            # CCW ring, i.e. interior on the left) -> the face is land.
            #
            # Why a vote and not the old single representative_point() +
            # nearest-segment side test: on a city-scale coast the faces are
            # huge (New York: one face covered all of mainland New Jersey),
            # and the test point lands kilometres from the nearest segment
            # (measured 6.6 km), where "left of segment" says nothing about
            # the face -- NJ flooded as sea and the Hudson was stamped land
            # in the same run. The boundary vote is local by construction.
            #
            # Matching is geometric (nearest original segment within
            # _EDGE_TOL of the edge midpoint), not exact-equality: unary_union
            # re-nodes the coastline, splitting segments at every
            # near-intersection, which is what killed the old exact-match
            # classifier on Venice's lagoon. Edges that match no coastline
            # segment (bbox boundary, artificial chain-end extensions) abstain.
            # Faces with zero votes fall back to the old side test -- their
            # representative point is then small-sliver-close to a segment.
            seg_lines, seg_dirs = [], []
            for cpts, _ in open_chains:
                for i in range(len(cpts) - 1):
                    a, b = cpts[i], cpts[i + 1]
                    if a != b:
                        seg_lines.append(LineString([a, b]))
                        seg_dirs.append((a, b))
            for rpts in closed_ring_pts:
                for i in range(len(rpts) - 1):
                    a, b = rpts[i], rpts[i + 1]
                    if a != b:
                        seg_lines.append(LineString([a, b]))
                        seg_dirs.append((a, b))
            seg_tree = STRtree(seg_lines) if seg_lines else None

            _EDGE_TOL = 1e-7   # ~1 cm: re-noded edges lie exactly on the
                               # originals bar float dust; real gaps are metres

            def _area2(coords):
                return sum(coords[i][0] * coords[i + 1][1]
                           - coords[i + 1][0] * coords[i][1]
                           for i in range(len(coords) - 1))

            def _classify(poly):
                """True if `poly` (one face part) is land: majority vote of its
                exterior edges against the original oriented coastline segments;
                representative-point side test as the zero-vote fallback."""
                coords = list(poly.exterior.coords)
                ccw = _area2(coords) > 0.0     # CCW ring -> interior left of edges
                land = sea = 0.0
                if seg_tree is not None:
                    for i in range(len(coords) - 1):
                        a, b = coords[i], coords[i + 1]
                        if a == b:
                            continue
                        mid = _Pt((a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0)
                        idx = seg_tree.nearest(mid)
                        if seg_lines[idx].distance(mid) > _EDGE_TOL:
                            continue                       # bbox edge: abstains
                        oa, ob = seg_dirs[idx]
                        dot = ((ob[0] - oa[0]) * (b[0] - a[0])
                               + (ob[1] - oa[1]) * (b[1] - a[1]))
                        if dot == 0:
                            continue
                        w = math.hypot(b[0] - a[0], b[1] - a[1])
                        if (dot > 0) == ccw:
                            land += w
                        else:
                            sea += w
                if land + sea > 0.0:
                    return land >= sea
                if seg_tree is not None:
                    rp = poly.representative_point()     # guaranteed inside
                    a, b = seg_dirs[seg_tree.nearest(rp)]
                    # cross((b-a), (rp-a)) > 0  =>  rp is left of a->b  =>  land
                    cross = (b[0]-a[0])*(rp.y-a[1]) - (b[1]-a[1])*(rp.x-a[0])
                    return cross > 0
                return False

            all_lines = ([bbox_poly.boundary] + [ls for _, ls in open_chains]
                         + [LineString(r) for r in closed_ring_pts])
            merged    = unary_union(all_lines)
            faces     = list(polygonize(merged))
            print(f"  [sea] polygonize produced {len(faces)} face(s)")

            for face in faces:
                clipped = face.intersection(bbox_poly)
                if clipped.is_empty or clipped.area < MIN_AREA:
                    continue
                if clipped.geom_type not in ('Polygon', 'MultiPolygon'):
                    continue

                parts = ([clipped] if clipped.geom_type == 'Polygon'
                         else [g for g in clipped.geoms
                               if g.geom_type == 'Polygon' and g.area >= MIN_AREA])
                for part in parts:
                    is_land = _classify(part)
                    cx, cy = part.centroid.x, part.centroid.y
                    print(f"    face centroid=({cx:.4f},{cy:.4f}) area={part.area:.6f} "
                          f"-> {'LAND' if is_land else 'sea'}")
                    if is_land:
                        land_polys.append(part)
                        n_open += 1

        except Exception as _exc:
            import traceback as _tb
            print(f"  [sea] open-chain polygonize error: {_exc}")
            _tb.print_exc()

    # ------------------------------------------------------------------
    # Sea = bbox minus all land
    # ------------------------------------------------------------------
    if not land_polys and not open_chains and not closed_ring_pts:
        print("  [sea] no chains intersect bbox - skipping sea")
        return []
    print(f"  [sea] {n_closed} island land polygon(s), {n_open} mainland land polygon(s)")
    island_area = sum(p.area for p in land_polys[:n_closed]) if n_closed else 0.0
    main_area   = sum(p.area for p in land_polys[n_closed:]) if n_open else 0.0
    print(f"  [sea] bbox_area={bbox_poly.area:.6f}  "
          f"island_land={island_area:.6f}  mainland_land={main_area:.6f}")

    try:
        land_union = unary_union(land_polys) if land_polys else None
        if land_union is not None:
            print(f"  [sea] union(land_polys).area={land_union.area:.6f}")
        sea_area = bbox_poly.difference(land_union) if land_union is not None else bbox_poly
    except Exception as e:
        print(f"  [sea] land subtraction failed ({e})")
        sea_area = bbox_poly

    if sea_area.is_empty:
        print("  [sea] sea area is empty")
        return []

    sea_shapes = ([sea_area] if sea_area.geom_type == 'Polygon'
                  else list(sea_area.geoms) if sea_area.geom_type == 'MultiPolygon'
                  else [])

    result = []
    for poly in sea_shapes:
        if poly.area < OUTPUT_MIN_AREA:
            continue
        exterior = [[c[0], c[1]] for c in poly.exterior.coords]
        holes    = [[[c[0], c[1]] for c in ring.coords] for ring in poly.interiors]
        result.append((exterior, holes))
    return result

# GEBCO is only a fallback when the GMRT fetch fails. Pure EBP where neither.

_SHORE_EXTEND_M = 20_000.0   # push coast endpoints past the bbox (open-sea cells)

def apply_water_depth_profile(cell_ids, polygon, surface_ele_m, *,
                              ebp_a, depth_cap_m,
                              custom_depth=None, gmrt_depth=None,
                              base_depth=None, gebco_depth=None,
                              bbox=None, cell_surface_eles=None):
    """
    Assign per-cell absolute bed elevations from the unified bathymetry model.

    Two ingredients, resolved per cell:

      1. base = min(A * x**(2/3), grid), where x is the cell's metric distance to
         shore and `grid` is the UNMASKED-GMRT depth (multibeam blended into the
         GEBCO base -- continuous, no seam), or GEBCO if GMRT is unavailable, or
         neither for inland lakes (pure EBP ramp). This gives a plausible, smooth
         nearshore-to-offshore profile.
      2. MASKED GMRT (gmrt_depth) -- a genuine multibeam sounding -- then OVERRIDES
         that base outright where present, so real deep features (dredged holes,
         scour channels) survive rather than being clamped away by the shallow
         nearshore EBP ramp.

    So EBP only ever shapes cells with no real sounding; measured depth always
    wins.

    The result is clamped to depth_cap_m (WATER_DEPTH_BY_TAG / SEA_MAX_DEPTH_M).
    Distance-to-shore uses the natural coastline: bbox-edge segments (both
    endpoints on the bbox boundary) are excluded when bbox is given so the
    profile is driven by real shore, not the map edge.

    Parameters
    ----------
    cell_ids        : iterable of H3 cell-id strings for this water body
    polygon         : Shapely Polygon/MultiPolygon defining the water body
    surface_ele_m   : water surface elevation in metres
    ebp_a           : EBP scale A for this water_type (config.EBP_PROFILE_A)
    depth_cap_m     : maximum depth (positive), clamps the result
    custom_depth    : {cell_id: positive_depth_m or None} from user-supplied raster
    gmrt_depth      : {cell_id: positive_depth_m or None} from GMRT
    gebco_depth     : {cell_id: positive_depth_m or None} from GEBCO
    bbox            : optional (lon_min, lat_min, lon_max, lat_max)
    cell_surface_eles : optional {cell_id: surface_ele_m} (flowing water)

    Returns
    -------
    dict mapping h3_cell_id -> absolute bed elevation_m
    """
    import h3

    cell_list = list(cell_ids)
    if not cell_list:
        return {}

    try:
        from shapely.geometry import LineString as _LS, MultiLineString as _MLS, Point as _Pt
    except ImportError:
        return {c: round(surface_ele_m - max(depth_cap_m, 0.0), 3) for c in cell_list}

    latlngs = [h3.cell_to_latlng(c) for c in cell_list]
    lat_c, lon_c = (sum(ll[k] for ll in latlngs) / len(latlngs) for k in (0, 1))
    m_lat = 111_320.0
    m_lon = m_lat * math.cos(math.radians(lat_c))

    def _to_m(lng, lat):
        return ((lng - lon_c) * m_lon, (lat - lat_c) * m_lat)

    # Build shore reference geometry.
    # For sea (bbox given): exterior ring is filtered to skip pure-bbox segments;
    # interior rings (islands) are included in full.
    # For inland water bodies (no bbox): use the full polygon boundary.
    if bbox is not None:
        _blon_min, _blat_min, _blon_max, _blat_max = bbox
        _BTOL = 1e-4

        def _on_bbox(p):
            return (abs(p[0] - _blon_min) < _BTOL or abs(p[0] - _blon_max) < _BTOL or
                    abs(p[1] - _blat_min) < _BTOL or abs(p[1] - _blat_max) < _BTOL)

        # Extend endpoint past the bbox so that open-sea cells near bbox-crossing
        # points don't measure a false (short) distance to the coast segment's
        # cut end.
        _ext_deg = _SHORE_EXTEND_M / 111_320.0

        def _extend_endpoint(pt, neighbor):
            if not _on_bbox(pt):
                return pt
            dx = pt[0] - neighbor[0]; dy = pt[1] - neighbor[1]
            L = (dx * dx + dy * dy) ** 0.5 or 1.0
            return (pt[0] + dx / L * _ext_deg, pt[1] + dy / L * _ext_deg)

        def _exterior_segs(coords):
            """Coastline segments from the exterior ring, bbox-only segments skipped,
            endpoints extended past the bbox to prevent false taper near exit points."""
            segs, current = [], []
            pts = list(coords)
            for i in range(len(pts) - 1):
                a, b = pts[i], pts[i + 1]
                if _on_bbox(a) and _on_bbox(b):
                    if len(current) >= 2:
                        segs.append(current)
                    current = []
                else:
                    if not current:
                        current = [a, b]
                    else:
                        current.append(b)
            if len(current) >= 2:
                segs.append(current)
            # Extend bbox-touching endpoints outward
            extended = []
            for seg in segs:
                seg = list(seg)
                seg[0]  = _extend_endpoint(seg[0],  seg[1])
                seg[-1] = _extend_endpoint(seg[-1], seg[-2])
                extended.append(seg)
            return extended

        def _geom_rings_filtered(geom, is_interior=False):
            if hasattr(geom, "geoms"):
                for g in geom.geoms:
                    yield from _geom_rings_filtered(g, is_interior)
            else:
                if is_interior:
                    yield list(geom.exterior.coords)
                else:
                    yield from _exterior_segs(geom.exterior.coords)
                for interior in geom.interiors:
                    yield list(interior.coords)

        try:
            raw_segs = list(_geom_rings_filtered(polygon))
            if not raw_segs:
                # No natural coastline found -- fall back to full boundary
                raw_segs = list(_geom_rings_filtered(polygon, is_interior=True))
            rings_m = [[_to_m(c[0], c[1]) for c in seg] for seg in raw_segs]
        except Exception:
            rings_m = []
    else:
        def _geom_rings(geom):
            if hasattr(geom, "geoms"):
                for g in geom.geoms:
                    yield from _geom_rings(g)
            else:
                yield list(geom.exterior.coords)
                for interior in geom.interiors:
                    yield list(interior.coords)
        try:
            rings_m = [[_to_m(c[0], c[1]) for c in ring] for ring in _geom_rings(polygon)]
        except Exception:
            rings_m = []

    if not rings_m:
        return {c: round(surface_ele_m - max(depth_cap_m, 0.0), 3) for c in cell_list}

    try:
        coast_m = _LS(rings_m[0]) if len(rings_m) == 1 else _MLS(rings_m)
    except Exception:
        return {c: round(surface_ele_m - max(depth_cap_m, 0.0), 3) for c in cell_list}

    depth_cap_m  = max(depth_cap_m, 0.0)
    custom_depth = custom_depth or {}                # user data, wins outright
    gmrt_depth   = gmrt_depth or {}                  # masked GMRT (real)
    base_depth   = base_depth or {}                  # unmasked GMRT (offshore base)
    gebco_depth  = gebco_depth or {}

    # Per-cell distance to shore (metres). Vectorised (Shapely 2.0) with a
    # per-cell fallback if the array path is unavailable.
    try:
        import numpy as _np
        import shapely as _shapely
        coords_m = _np.array([_to_m(ll[1], ll[0]) for ll in latlngs])
        dists_m = _shapely.distance(_shapely.points(coords_m), coast_m).tolist()
    except Exception:
        dists_m = [coast_m.distance(_Pt(_to_m(ll[1], ll[0]))) for ll in latlngs]

    result = {}
    for cell_id, x in zip(cell_list, dists_m):
        depth = ebp_a * (x ** (2.0 / 3.0))          # equilibrium beach profile
        # Offshore base: unmasked GMRT (multibeam blended into the GEBCO base;
        # continuous, no seam), else GEBCO. Blended with the EBP ramp via min --
        # nearshore the ramp is shallower and wins, offshore the grid wins.
        base = base_depth.get(cell_id)
        if base is None:
            base = gebco_depth.get(cell_id)
        if base is not None:
            depth = min(depth, base)
        # Real multibeam (masked GMRT) OVERRIDES the base+ramp outright, so a
        # measured deep feature nearshore (e.g. a dredged hole) is not clamped
        # away by the shallow EBP ramp.
        real = gmrt_depth.get(cell_id)
        if real is not None:
            depth = real
        # User-supplied bathymetry outranks everything else.
        custom = custom_depth.get(cell_id)
        if custom is not None:
            depth = custom
        depth = min(depth, depth_cap_m)
        s = cell_surface_eles.get(cell_id, surface_ele_m) if cell_surface_eles else surface_ele_m
        result[cell_id] = round(s - depth, 3)

    return result

def build_water_hexagons(ring, resolution=9, holes=None):
    """
    Convert a closed polygon ring [[lon, lat], ...] into H3 triangles.
    Each H3 cell in the polyfill becomes 6 triangles (centre + boundary vertices).
    Holes (islands) are excluded via h3.LatLngPoly hole rings.
    """
    try:
        import h3
    except ImportError:
        return []

    if not ring or len(ring) < 3:
        return []

    latlng_ring = [(p[1], p[0]) for p in ring]
    if latlng_ring[0] == latlng_ring[-1]:
        latlng_ring = latlng_ring[:-1]

    latlng_holes = [
        ([(p[1], p[0]) for p in h] if h[0] != h[-1] else [(p[1], p[0]) for p in h[:-1]])
        for h in (holes or []) if h and len(h) >= 3
    ]

    try:
        poly  = h3.LatLngPoly(latlng_ring, *latlng_holes)
        cells = h3.h3shape_to_cells(poly, resolution)
    except Exception as e:
        print(f"  [h3] polyfill failed: {e}")
        return []

    triangles = []
    for cell in cells:
        clat, clon = h3.cell_to_latlng(cell)
        centre     = [clon, clat]
        boundary   = h3.cell_to_boundary(cell)
        n          = len(boundary)
        for i in range(n):
            v0 = [boundary[i][1],       boundary[i][0]]
            v1 = [boundary[(i+1)%n][1], boundary[(i+1)%n][0]]
            triangles.append([centre, v0, v1])

    return triangles
